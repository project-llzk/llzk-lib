40 mlir::LogicalResult
41 resolveFamily(component::StructType type, StorageDescription &info, mlir::Operation *site) {
42 for (mlir::Attribute record : info.family) {
43 auto entry = llvm::dyn_cast<mlir::DictionaryAttr>(record);
44 auto origin = entry ? entry.getAs<mlir::TypeAttr>(FAMILY_TYPE_KEY) : mlir::TypeAttr();
45 auto indices = entry ? entry.getAs<mlir::ArrayAttr>(FAMILY_INDICES_KEY) : mlir::ArrayAttr();
46 if (!origin || !indices || origin.getValue() != type ||
47 indices.size() != info.indices.size()) {
48 continue;
49 }
50 bool matches = llvm::all_of(llvm::zip(indices, info.indices), [](auto pair) {
51 auto attr = llvm::dyn_cast<mlir::IntegerAttr>(std::get<0>(pair));
52 return attr && attr.getValue().isSignedIntN(64) && attr.getInt() == std::get<1>(pair);
53 });
54 if (!matches) {
55 continue;
56 }
57 auto id = entry.getAs<mlir::IntegerAttr>(FAMILY_SPECIALIZATION_KEY);
58 if (!id || !id.getValue().isSignedIntN(64) || id.getInt() < 0) {
59 return constraintError(site, "invalid family specialization ID");
60 }
61 auto target =
62 specializations.resolve(tables, type, id.getInt(), site, "constraint flattening");
63 if (mlir::failed(target)) {
64 return mlir::failure();
65 }
66 if (info.definition && info.definition != *target) {
67 return constraintError(site, "ambiguous concrete struct family element");
68 }
69 info.definition = *target;
70 }
71 return mlir::success();
72 }
73
75 mlir::LogicalResult
76 resolveComponent(mlir::Value value, StorageDescription &info, mlir::Operation *site) {
77 auto type = llvm::dyn_cast<component::StructType>(value.getType());
78 if (!type) {
79 return mlir::success();
80 }
81 if (info.family && mlir::failed(resolveFamily(type, info, site))) {
82 return mlir::failure();
83 }
84 if (info.definition) {
85 return mlir::success();
86 }
87 auto target = lookupTopLevelSymbol<component::StructDefOp>(tables, type.getNameRef(), site);
88 if (mlir::failed(target)) {
89 return mlir::failure();
90 }
91 if (isInTemplate(target->get()) || (type.getParams() && !type.getParams().empty())) {
92 return constraintError(site, "requires a concrete struct or matching poly.family element");
93 }
94 info.definition = target->get();
95 return mlir::success();
96 }

99 mlir::LogicalResult
100 memberInfo(component::MemberReadOp read, StorageDescription &info, mlir::Operation *site) {
101 auto resolved = describe(read.getComponent(), site);
102 if (mlir::failed(resolved)) {
103 return mlir::failure();
104 }
105 auto parent = *resolved;
106 if (!parent.definition || read.getTableOffset()) {
107 return constraintError(site, "member read requires concrete non-column component storage");
108 }
109 auto member = parent.definition.getMemberDef(builder.getStringAttr(read.getMemberName()));
110 if (!member || member.getColumn()) {
111 return constraintError(site, "unknown or unsupported column member");
112 }
113 if (parent.definition != main && !member.hasPublicAttr()) {
114 return constraintError(site, "run llzk-expose-child-members before reading child storage");
115 }
116 auto rolled = lookupTopLevelSymbol<component::StructDefOp>(
117 tables, llvm::cast<component::StructType>(read.getComponent().getType()).getNameRef(), site
118 );
119 if (mlir::failed(rolled)) {
120 return mlir::failure();
121 }
122 if (!typesUnify(read.getType(), member.getType())) {
123 return constraintError(site, "member read type does not match the concrete specialization");
124 }
125 auto declared = rolled->get().getMemberDef(builder.getStringAttr(read.getMemberName()));
126 if (rolled->get() != parent.definition &&
127 (!declared || !declared.hasPublicAttr() ||
128 !typesUnify(read.getType(), declared.getType(), rolled->getNamespace()))) {
129 return constraintError(
130 site, "concrete family member cannot be represented through the rolled storage type"
131 );
132 }
133 info.family = member->getAttrOfType<mlir::ArrayAttr>(FAMILY_ATTR);
134 info.signal = bool(member.getSignal()) || (parent.definition == main && member.hasPublicAttr());
135 return mlir::success();
136 }

139 ConstraintStorage(

145 specializations(rootModule) {}

150 mlir::FailureOr<StorageDescription> describe(mlir::Value value, mlir::Operation *site) {
151 StorageDescription info;
152 if (auto argument = llvm::dyn_cast<mlir::BlockArgument>(value)) {
153 info.definition = argument.getArgNumber() == 0 ? main : component::StructDefOp();
154 info.signal = argument.getArgNumber() != 0;
155 } else if (auto member = value.getDefiningOp<component::MemberReadOp>()) {
156 if (mlir::failed(memberInfo(member, info, site))) {
157 return mlir::failure();
158 }
159 } else if (auto element = value.getDefiningOp<array::ReadArrayOp>()) {
160 auto parent = describe(element.getArrRef(), site);
161 if (mlir::failed(parent)) {
162 return mlir::failure();
163 }
164 info = *parent;
165 info.definition = {};
166 if (info.family) {
167 for (mlir::Value index : element.getIndices()) {
168 auto coordinate = facts.integer(index);
169 if (!coordinate) {
170 return constraintError(site, "array path requires compile-time indices");
171 }
172 info.indices.push_back(*coordinate);
173 }
174 }
175 } else if (auto record = value.getDefiningOp<pod::ReadPodOp>()) {
176 auto parent = describe(record.getPodRef(), site);
177 if (mlir::failed(parent)) {
178 return mlir::failure();
179 }
180 info = *parent;
181 info.definition = {};
182 if (llvm::isa<array::ArrayType>(value.getType())) {
183 // Discovery records family coordinates relative to the nearest array.
184 info.indices.clear();
185 }
186 }
187 if (mlir::failed(resolveComponent(value, info, site))) {
188 return mlir::failure();
189 }
190 return info;
191 }
192};
