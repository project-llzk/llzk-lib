27static DynamicAPInt binaryBitOp(
28 const DynamicAPInt &lhs, const DynamicAPInt &rhs,
29 function_ref<APInt(const APInt &, const APInt &)> fn
30) {
31 APSInt a = llzk::toAPSInt(lhs), b = llzk::toAPSInt(rhs);
32 unsigned width = std::max(a.getBitWidth(), b.getBitWidth());
33 return DynamicAPInt(fn(a.sext(width), b.sext(width)));
34}

38DynamicAPInt toSignedDynamicAPInt(ArrayRef<uint64_t> parts) {
39 if (parts.empty()) {
40 return DynamicAPInt(0);

43 return DynamicAPInt(APInt(static_cast<unsigned>(parts.size()) * 64, parts));

48Expected<DynamicAPInt> parseDynamicAPInt(StringRef str) {
49 StringRef digits = str;
50 bool negative = digits.consume_front("-");
51 APInt magnitude;
52 if (digits.getAsInteger(10, magnitude)) {
53 return createStringError(inconvertibleErrorCode(), "expected signed decimal integer");
54 }
55 DynamicAPInt value = toDynamicAPInt(magnitude);
56 return negative ? -value : value;
57}

64Expected<APInt> checkedToAPInt(const DynamicAPInt &value, unsigned bitWidth, bool isSigned) {
65 APSInt bits = toAPSInt(value);
66 bool fits = bitWidth != 0 &&
67 (isSigned ? bits.isSignedIntN(bitWidth) : value >= 0 && bits.isIntN(bitWidth));
68 if (!fits) {
69 return createStringError(inconvertibleErrorCode(), "integer does not fit requested width");
70 }
71 return isSigned ? bits.sextOrTrunc(bitWidth) : bits.zextOrTrunc(bitWidth);
72}

101DynamicAPInt operator<<(const DynamicAPInt &lhs, const DynamicAPInt &rhs) {
102 APSInt bits = toAPSInt(lhs);
103 unsigned width = bits.getSignificantBits();
104 auto maxAmount =
105 toDynamicAPInt(APSInt::getUnsigned(std::numeric_limits<unsigned>::max() - width));
106 if (rhs < 0 || rhs > maxAmount) {
107 llvm::report_fatal_error("invalid or unrepresentable left shift");
108 }
109 if (lhs == 0) {
110 return DynamicAPInt(0);
111 }
112 unsigned amount = checkedCast<unsigned>(cantFail(checkedToUInt64(rhs)));
113 return DynamicAPInt(bits.sextOrTrunc(width + amount).shl(amount));
114}

116DynamicAPInt operator>>(const DynamicAPInt &lhs, const DynamicAPInt &rhs) {
117 if (rhs < 0) {
118 llvm::report_fatal_error("negative right shift");
119 }
120 APSInt bits = toAPSInt(lhs);
121 if (rhs >= toDynamicAPInt(APSInt::getUnsigned(bits.getSignificantBits()))) {
122 return DynamicAPInt(lhs < 0 ? -1 : 0);
123 }
124 return DynamicAPInt(bits.ashr(checkedCast<unsigned>(cantFail(checkedToUInt64(rhs)))));
125}

129DynamicAPInt toDynamicAPInt(const APSInt &i) {
130 // DynamicAPInt interprets APInt (implicit cast from APSInt for the constructor below) as
131 // signed. Extend unsigned APSInts with a 0 sign bit so their positive value is preserved.
132 if (i.isUnsigned() && i.isSignBitSet()) {
133 return DynamicAPInt(i.zext(i.getBitWidth() + 1));
134 } else {
135 return DynamicAPInt(i);
136 }
137}

150static LLVM_ATTRIBUTE_NOINLINE APSInt toAPSIntSlow(const DynamicAPInt &i) {
151 // 254-bit magnitude needs at most 77 decimal digits plus a minus sign; 96 inline characters cover
152 // this with headroom for larger intermediates so the pre-defined primes don't require resizing.
153 SmallString<96> repr;
154 raw_svector_ostream(repr) << i;
155
156 // This is based on `APSInt(StringRef)` and uses its conservative size estimate of 64/19 bits per
157 // decimal digit. However, we take some additional precautions to allocate space for the sign bit
158 // from the start, avoid truncating when a bitwidth change would not reduce the number of words
159 // required, and move the final result into the return value to avoid unnecessary copies.
160 size_t digits = repr.size() - (repr.front() == '-');
161 constexpr uint64_t maxDigits = ((uint64_t(numeric_limits<unsigned>::max()) - 2) * 19) / 64;
162 // Bound the digit count before multiplying or narrowing to APInt's unsigned bit width.
163 if (digits > maxDigits) {
164 llvm::report_fatal_error("decimal integer exceeds APInt's maximum parse width");
165 }
166 unsigned parseWidth = static_cast<unsigned>((uint64_t(digits) * 64) / 19 + 2);
167 APInt bits(parseWidth, repr, /*radix=*/10);
168 // getSignificantBits() includes the sign bit for either sign. Truncation
169 // allocates a new APInt, so only do it when it saves a storage word.
170 unsigned minWidth = bits.getSignificantBits();
171 if (APInt::getNumWords(minWidth) < bits.getNumWords()) {
172 bits = bits.trunc(minWidth);
173 }
174 return APSInt(std::move(bits), /*isUnsigned=*/false);
175}

177APSInt toAPSInt(const DynamicAPInt &i) {
178 if (numeric_limits<int64_t>::min() <= i && i <= numeric_limits<int64_t>::max()) {
179 // Fast path for smaller values, just use the int64_t conversion
180 return APSInt::get(int64_t(i));
181 }
182 return toAPSIntSlow(i);
183}

185DynamicAPInt modExp(const DynamicAPInt &base, const DynamicAPInt &exp, const DynamicAPInt &mod) {
186 if (exp < 0 || mod <= 0) {
187 llvm::report_fatal_error(
188 "modular exponentiation requires a nonnegative exponent and positive modulus"
189 );
190 }
191 DynamicAPInt result = llvm::mod(DynamicAPInt(1), mod);
192 DynamicAPInt b = llvm::mod(base, mod);
193 DynamicAPInt e = exp;
194 DynamicAPInt one(1);
195
196 while (e != 0) {
197 if (e % 2 != 0) {
198 result = (result * b) % mod;
199 }
200
201 b = (b * b) % mod;
202 e = e >> one;
203 }
204 return result;
205}

207DynamicAPInt modInversePrime(const DynamicAPInt &f, const DynamicAPInt &p) {
208 assert(f != 0 && "0 has no inverse");
209 // Fermat: f^(p-2) mod p
210 DynamicAPInt exp = p - 2;
211 DynamicAPInt result = modExp(f, exp, p);
212 assert(llvm::mod(f * result, p) == 1 && "inverse is incorrect");
213 return result;
214}
