41bool isValidFamilyEntry(TypeAttr type, ArrayAttr indices, IntegerAttr id) {
42 if (!type || !isa<llzk::component::StructType>(type.getValue()) || !indices || !id ||
43 !id.getValue().isSignedIntN(64) || id.getInt() < 0) {
44 return false;
45 }
46 return llvm::all_of(indices, [](Attribute attr) {
47 auto index = dyn_cast<IntegerAttr>(attr);
48 return index && index.getType().isIndex() && index.getInt() >= 0;
49 });
50}

54public:

57 FailureOr<llzk::LLZKLayout> build(ModuleOp module) {
58 auto mainType = llzk::getMainInstanceType(module);
59 if (failed(mainType)) {
60 return failure();
61 }
62 if (*mainType && mainType->getParams() && !mainType->getParams().empty()) {
63 return module.emitError("LLZK layout requires a monomorphized llzk.main");
64 }
65 auto main = llzk::getMainInstanceDef(tables, module);
66 if (failed(main)) {
67 return failure();
68 }
69 if (!*main) {
70 return module.emitError("LLZK layout requires a concrete llzk.main");
71 }
72 if (main->viaInclude()) {
73 return module.emitError("inline includes before exporting an LLZK layout");
74 }
75
76 auto root = llzk::getTopRootModule(module);
77 if (failed(root)) {
78 return failure();
79 }
80 specializations.emplace(*root);
81
82 auto constrain = main->get().getConstrainFuncOp();
83 if (!constrain) {
84 return main->get().emitError("LLZK layout requires a main constrain function");
85 }
86
87 SmallVector<Attribute> path;
88 // Arguments to main's constrain come before the member defs in the layout.
89 layout.argumentNames.resize(constrain.getNumArguments());
90 for (unsigned index = 1; index < constrain.getNumArguments(); ++index) {
91 if (auto name = constrain.getArgNameAttr(index)) {
92 layout.argumentNames[index] = *name;
93 }
94 path.assign(
95 {builder.getStringAttr(llzk::LAYOUT_ARGUMENT_ROOT), builder.getI64IntegerAttr(index)}
96 );
97 if (failed(visitType(constrain.getFunctionType().getInput(index), constrain, path, true))) {
98 return failure();
99 }
100 }
101
102 path.assign({builder.getStringAttr(llzk::LAYOUT_MAIN_ROOT)});
103 if (failed(visitStruct(main->get(), path, true))) {
104 return failure();
105 }
106
107 return std::move(layout);
108 }
109
110private:
111 LogicalResult visitStruct(
112 llzk::component::StructDefOp definition, SmallVectorImpl<Attribute> &path, bool isMain = false
113 ) {
114 if (llzk::polymorphic::isInTemplate(definition)) {
115 return definition.emitError("LLZK layout requires monomorphized struct types");
116 }
117 if (!activeStructs.insert(definition).second) {
118 return definition.emitError("recursive struct storage has no finite LLZK layout");
119 }
120 for (auto member : definition.getMemberDefs()) {
121 path.push_back(member.getSymNameAttr());
122 bool isSignal = llzk::isLayoutSignalMember(member, isMain);
123 if (failed(visitType(member.getType(), member, path, isSignal))) {
124 return failure();
125 }
126 path.pop_back();
127 }
128 activeStructs.erase(definition);
129 return success();
130 }
131
134 LogicalResult loadFamily(Operation *site) {
135 if (families.contains(site)) {
136 return success();
137 }
138 auto &entries = families[site];
139 Attribute metadata = site->getAttr(FAMILY_ATTR);
140 if (!metadata) {
141 return success();
142 }
143 auto records = dyn_cast<ArrayAttr>(metadata);
144 if (!records) {
145 return site->emitError("invalid poly.family metadata for LLZK layout");
146 }
147 for (Attribute record : records) {
148 auto dict = dyn_cast<DictionaryAttr>(record);
149 auto type = dict ? dict.getAs<TypeAttr>(FAMILY_TYPE_KEY) : TypeAttr();
150 auto indices = dict ? dict.getAs<ArrayAttr>(FAMILY_INDICES_KEY) : ArrayAttr();
151 auto id = dict ? dict.getAs<IntegerAttr>(FAMILY_SPECIALIZATION_KEY) : IntegerAttr();
152 if (!isValidFamilyEntry(type, indices, id)) {
153 return site->emitError("invalid poly.family entry for LLZK layout");
154 }
155 if (!entries.try_emplace(std::make_pair(type.getValue(), indices), id.getInt()).second) {
156 return site->emitError("duplicate poly.family entry for LLZK layout");
157 }
158 }
159 return success();
160 }
161
162 LogicalResult visitType(
163 Type type, Operation *site, SmallVectorImpl<Attribute> &path, bool isSignal,
164 ArrayRef<int64_t> indices = {}
165 ) {
166 // Unmarked expressions are not signals, but component storage can contain
167 // members with their own signal annotations.
168 if (!isSignal && !walkContains<llzk::component::StructType>(type)) {
169 return success();
170 }
171 if (isa<llzk::felt::FeltType>(type)) {
172 auto signalPath = builder.getArrayAttr(path);
173 layout.signalIds.try_emplace(signalPath, layout.signalPaths.size());
174 layout.signalPaths.push_back(signalPath);
175 return success();
176 }
177 if (auto array = dyn_cast<llzk::array::ArrayType>(type)) {
178 for (int64_t size : array.getShape()) {
179 if (size < 0) {
180 return site->emitError("LLZK layout requires concrete array dimensions");
181 }
182 }
183 SmallVector<int64_t> elementIndices;
184 return visitArray(array, site, path, isSignal, elementIndices, 0);
185 }
186 if (auto pod = dyn_cast<llzk::pod::PodType>(type)) {
187 for (auto record : pod.getRecords()) {
188 path.push_back(record.getName());
189 if (failed(visitType(record.getType(), site, path, isSignal, indices))) {
190 return failure();
191 }
192 path.pop_back();
193 }
194 return success();
195 }
196 if (auto structure = dyn_cast<llzk::component::StructType>(type)) {
197 if (auto params = structure.getParams(); params && !params.empty()) {
198 SmallVector<Attribute> indexAttrs;
199 for (int64_t index : indices) {
200 indexAttrs.push_back(builder.getIndexAttr(index));
201 }
202 if (failed(loadFamily(site))) {
203 return failure();
204 }
205 auto found = families[site].find({structure, builder.getArrayAttr(indexAttrs)});
206 if (found == families[site].end()) {
207 return site->emitError("missing concrete struct family element for LLZK layout");
208 }
209 auto definition =
210 specializations->resolve(tables, structure, found->second, site, "LLZK layout");
211 if (failed(definition)) {
212 return failure();
213 }
214 return visitStruct(*definition, path);
215 }
216 auto definition = structure.getDefinition(tables, site);
217 if (failed(definition)) {
218 return failure();
219 }
220 if (!*definition) {
221 return site->emitError() << "cannot resolve struct type " << structure
222 << " for LLZK layout";
223 }
224 if (definition->viaInclude()) {
225 return site->emitError("inline includes before exporting an LLZK layout");
226 }
227 if (llzk::polymorphic::isInTemplate(definition->get())) {
228 return site->emitError("LLZK layout requires monomorphized struct types");
229 }
230 return visitStruct(definition->get(), path);
231 }
232 if (type.isIndex() || isa<IntegerType, llzk::string::StringType>(type)) {
233 return success();
234 }
235 return site->emitError() << "unsupported or unresolved type " << type << " in LLZK layout";
236 }
237
238 LogicalResult visitArray(
239 llzk::array::ArrayType array, Operation *site, SmallVectorImpl<Attribute> &path,
240 bool isSignal, SmallVectorImpl<int64_t> &indices, unsigned dimension
241 ) {
242 if (dimension == array.getRank()) {
243 return visitType(array.getElementType(), site, path, isSignal, indices);
244 }
245 for (int64_t index = 0; index < array.getShape()[dimension]; ++index) {
246 path.push_back(builder.getIndexAttr(index));
247 indices.push_back(index);
248 if (failed(visitArray(array, site, path, isSignal, indices, dimension + 1))) {
249 return failure();
250 }
251 path.pop_back();
252 indices.pop_back();
253 }
254 return success();
255 }

257 Builder builder;
258 SymbolTableCollection tables;
260 std::optional<StructSpecializationResolver> specializations;
261 llvm::DenseMap<Operation *, llvm::DenseMap<std::pair<Type, ArrayAttr>, int64_t>> families;
262 llvm::DenseSet<Operation *> activeStructs;
263 llzk::LLZKLayout layout;
264};

267void printPath(ArrayAttr path, ArrayRef<StringAttr> names, llvm::raw_ostream &output) {
268 // buildLLZKLayout prefixes paths with main, or arg followed by an argument position.
269 assert(path && !path.empty() && isa<StringAttr>(path[0]) && "expected a layout path root");
270 auto root = cast<StringAttr>(path[0]).getValue();
271 bool isMain = root == llzk::LAYOUT_MAIN_ROOT;
272 assert((isMain || root == llzk::LAYOUT_ARGUMENT_ROOT) && "unexpected layout path root");
273 if (isMain) {
274 output << llzk::LAYOUT_MAIN_ROOT;
275 } else {
276 assert(path.size() >= 2 && isa<IntegerAttr>(path[1]) && "expected an argument position");
277 auto argument = cast<IntegerAttr>(path[1]).getInt();
278 assert(argument > 0 && "self is excluded from the layout");
279 if (llzk::checkedCast<size_t>(argument) < names.size() && names[argument]) {
280 output << llzk::LAYOUT_ARGUMENT_ROOT << '[';
281 names[argument].print(output);
282 output << ']';
283 } else {
284 output << llzk::LAYOUT_ARGUMENT_ROOT << argument;
285 }
286 }
287 for (Attribute segment : path.getValue().drop_front(isMain ? 1 : 2)) {
288 output << '[';
289 if (auto name = dyn_cast<StringAttr>(segment)) {
290 name.print(output);
291 } else {
292 output << cast<IntegerAttr>(segment).getInt();
293 }
294 output << ']';
295 }
296}

308std::optional<uint64_t> llzk::LLZKLayout::getSignalId(ArrayAttr path) const {
309 if (!path) {
310 return std::nullopt;
311 }
312 auto found = signalIds.find(path);
313 if (found == signalIds.end()) {
314 return std::nullopt;
315 }
316 return found->second;
317}

319void llzk::printLLZKLayout(const LLZKLayout &layout, llvm::raw_ostream &output) {
320 output << "# LLZK layout map v1\n# signals\n";
321 for (auto [id, path] : llvm::enumerate(layout.signalPaths)) {
322 output << "signal " << id << '\t';
323 printPath(path, layout.argumentNames, output);
324 output << '\n';
325 }
