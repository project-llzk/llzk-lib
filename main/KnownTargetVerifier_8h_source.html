40 template <typename T>
41 mlir::LogicalResult verifyTypesMatch(
42 mlir::ValueTypeRange<T> originTypes, mlir::ArrayRef<mlir::Type> tgtTypes, const char *aspect
43 ) {
44 if (tgtTypes.size() != originTypes.size()) {
45 return origin->emitOpError()
46 .append("incorrect number of ", aspect, "s for callee, expected ", tgtTypes.size())
47 .attachNote(tgt.getLoc())
48 .append("callee defined here");
49 }
50 for (unsigned i = 0, e = tgtTypes.size(); i != e; ++i) {
51 if (!typesUnify(originTypes[i], tgtTypes[i], targetNamespace)) {
52 auto diag =
53 origin->emitOpError().append(aspect, " type mismatch: expected type ", tgtTypes[i]);
54 if (targetViaInclude) {
55 diag.append(" from included target \"", origin->getCalleeAttr(), '"');
56 }
57 return diag.append(", but found ", originTypes[i], " for ", aspect, " number ", i);
58 }
59 }
60 return mlir::success();
61 }
62
66 mlir::LogicalResult verifyTemplateInstantiation(polymorphic::TemplateOp tgtOpParent) {
67 auto realParams = tgtOpParent.getConstOps<polymorphic::TemplateParamOp>();
68 mlir::ArrayAttr callParams = origin->getTemplateParamsAttr();
69
70 // When there is no instantiation list, just ensure that it's not required.
71 if (isNullOrEmpty(callParams)) {
72 llvm::SmallDenseSet<mlir::SymbolRefAttr> referencedInSignature;
73 llzk::getSymbolsUsedIn(tgtType.getInputs(), referencedInSignature);
74 llzk::getSymbolsUsedIn(tgtType.getResults(), referencedInSignature);
75
76 bool allParamsReferenced = llvm::all_of(realParams, [&](polymorphic::TemplateParamOp p) {
77 return referencedInSignature.contains(mlir::FlatSymbolRefAttr::get(p.getNameAttr()));
78 });
79 if (allParamsReferenced) {
80 return mlir::success();
81 }
82 return origin->emitOpError().append(
83 "must provide template instantiation parameters when calling \"@", tgt.getSymName(),
84 "\" because not all template parameters of \"@", tgtOpParent.getSymName(),
85 "\" appear in the function type signature"
86 );
87 }
88
89 // Ensure `forceIntAttrTypes()` was successful on the call site's template parameters.
90 if (mlir::failed(llzk::forceIntAttrTypes(callParams.getValue(), [this] {
91 return llzk::InFlightDiagnosticWrapper(this->origin->emitOpError());
92 }))) {
93 return mlir::failure();
94 }
95
96 // The instantiation list is present. Check it has exactly one entry per template param.
97 size_t numTemplateParams = llvm::range_size(realParams);
98 if (callParams.size() != numTemplateParams) {
99 return origin->emitOpError().append(
100 "template instantiation has ", callParams.size(), " parameter(s) but \"@",
101 tgtOpParent.getSymName(), "\" expects ", numTemplateParams, " template parameter(s)"
102 );
103 }
104
105 // Check type compatibility of each provided value with the declared parameter type (if any).
106 if (mlir::failed(origin->verifyTemplateParamValuesCompatibility(realParams))) {
107 return mlir::failure();
108 }
109
110 // Check that the provided instantiation values are consistent with what type unification
111 // of the target function types against the call's operand and result types would determine.
112 mlir::FailureOr<UnificationMap> unifyResult =
113 origin->unifyTypeSignatureWithNamespace(tgtType, targetNamespace);
114 // Verification continues after input/output errors to aggregate diagnostics.
115 if (mlir::failed(unifyResult)) {
116 return mlir::failure();
117 }
118 return origin->verifyTemplateParamsMatchInferred(realParams, unifyResult.value());
119 }
120
121private:
122 OriginOp *origin;
123 TargetOp tgt;
124 mlir::FunctionType tgtType;
125 std::vector<llvm::StringRef> targetNamespace;
126 bool targetViaInclude;
127};
