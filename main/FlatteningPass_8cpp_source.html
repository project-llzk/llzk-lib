122static LogicalResult
123applyAndFoldGreedily(ModuleOp modOp, ConversionTracker &tracker, RewritePatternSet &&patterns) {
124 bool currStepModified = false;
125 MatchFailureListener failureListener;
126 auto greedyRewriteConfig =
127 GreedyRewriteConfig().enableFolding().setMaxIterations(20).setListener(&failureListener);
128 DenseSet<Operation *> preserved = tracker.getPreservedLoopOps(modOp);
129 LogicalResult result = success();
130 if (preserved.empty()) {
131 result = applyPatternsGreedily(
132 modOp->getRegion(0), std::move(patterns), greedyRewriteConfig, &currStepModified
133 );
134 } else {
135 FrozenRewritePatternSet frozenPatterns(std::move(patterns));
136 // Strictness prevents operand/user updates from re-enqueuing a preserved loop or
137 // admitting a new loop before its bounds have been checked in the next local sweep.
138 greedyRewriteConfig.setStrictness(GreedyRewriteStrictness::ExistingOps);
139 for (int64_t iteration = 0;; ++iteration) {
140 // The operation-list driver does not enforce the region driver's iteration limit.
141 if (iteration == greedyRewriteConfig.getMaxIterations()) {
142 result = modOp.emitError("greedy rewriting with preserved loops failed to converge after ")
143 << iteration << " iterations";
144 break;
145 }
146 if (iteration != 0) {
147 preserved = tracker.getPreservedLoopOps(modOp);
148 }
149 // The greedy worklist is LIFO: visit loop bodies before helpers that replace their loops.
150 auto candidates = walkCollect<Operation *, WalkOrder::PreOrder>(
151 modOp.getBodyRegion(), [&preserved](Operation *op) { return !preserved.contains(op); }
152 );
153 bool sweepModified = false;
154 result =
155 applyOpPatternsGreedily(candidates, frozenPatterns, greedyRewriteConfig, &sweepModified);
156 currStepModified |= sweepModified;
157 if (failed(result) || failureListener.hadFailure || !sweepModified) {
158 break;
159 }
160 }
161 }
162 tracker.updateModifiedFlag(currStepModified);
163 return failure(result.failed() || failureListener.hadFailure);
164}

179) {

184 Block &body = newTemplate.getBodyRegion().emplaceBlock();

188 assert(paramOp && "symbol must exist");
189 body.push_back(paramOp->clone());
190 }
191 return newTemplate;
192}

210static bool referencesMemberBinding(
211 Type type, const DenseMap<FlatSymbolRefAttr, Attribute> *bindings,
212 const ConversionTracker &tracker
213) {
214 return type
215 .walk([&](Type nestedType) {
216 if (auto variable = dyn_cast<TypeVarType>(nestedType);
217 variable && (!bindings || bindings->contains(variable.getNameRef()))) {
218 return WalkResult::interrupt();
219 }
220 if (auto arrayType = dyn_cast<ArrayType>(nestedType)) {
221 for (Attribute dimension : arrayType.getDimensionSizes()) {
222 if (auto name = dyn_cast<FlatSymbolRefAttr>(dimension);
223 name && (!bindings || bindings->contains(name))) {
224 return WalkResult::interrupt();
225 }
226 }
227 }
228 if (auto structType = dyn_cast<StructType>(nestedType)) {
229 if (ArrayAttr params = structType.getParams()) {
230 for (Attribute param : params) {
231 if (auto name = dyn_cast<FlatSymbolRefAttr>(param);
232 name && (!bindings || bindings->contains(name))) {
233 return WalkResult::interrupt();
234 }
235 }
236 }
237 if (auto preimage = tracker.getPreimage(structType);
238 preimage && referencesMemberBinding(*preimage, bindings, tracker)) {
239 return WalkResult::interrupt();
240 }
241 }
242 return WalkResult::advance();
243 }).wasInterrupted();
244}

249static bool appendMemberBinding(
250 FlatSymbolRefAttr variable, Attribute binding, const ConversionTracker &tracker,
251 SmallVectorImpl<std::pair<FlatSymbolRefAttr, Attribute>> &bindings,
252 const DenseMap<FlatSymbolRefAttr, Attribute> *relevantBindings
253) {
254 if (relevantBindings && !relevantBindings->contains(variable)) {
255 return true;
256 }
257 if (auto type = dyn_cast<TypeAttr>(binding)) {
258 if (auto identity = dyn_cast<TypeVarType>(type.getValue())) {
259 binding = identity.getNameRef();
260 }
261 }
262 bool mustResolve = relevantBindings || variable != binding;
263 if (mustResolve) {
264 if (isa<FlatSymbolRefAttr>(binding)) {
265 return false;
266 }
267 if (auto type = dyn_cast<TypeAttr>(binding);
268 type && referencesMemberBinding(type.getValue(), nullptr, tracker)) {
269 return false;
270 }
271 }
272 bindings.emplace_back(variable, binding);
273 return true;
274}

279/// When checking a sibling, skip subtrees unrelated to the candidate bindings.
280static bool collectMemberBindings(
281 Type declaredType, Type writtenType, const ConversionTracker &tracker,
282 SmallVectorImpl<std::pair<FlatSymbolRefAttr, Attribute>> &bindings,
283 const DenseMap<FlatSymbolRefAttr, Attribute> *relevantBindings = nullptr
284) {
285 if (!referencesMemberBinding(declaredType, relevantBindings, tracker)) {
286 return true;
287 }
288 if (auto variable = dyn_cast<TypeVarType>(declaredType)) {
289 return appendMemberBinding(
290 variable.getNameRef(), TypeAttr::get(writtenType), tracker, bindings, relevantBindings
291 );
292 }
293 if (auto declaredArray = dyn_cast<ArrayType>(declaredType)) {
294 auto writtenArray = dyn_cast<ArrayType>(writtenType);
295 if (!writtenArray ||
296 writtenArray.getDimensionSizes().size() < declaredArray.getDimensionSizes().size()) {
297 return false;
298 }
299 for (auto [declaredDimension, writtenDimension] : llvm::zip_equal(
300 declaredArray.getDimensionSizes(),
301 ArrayRef<Attribute>(writtenArray.getDimensionSizes())
302 .take_front(declaredArray.getDimensionSizes().size())
303 )) {
304 if (auto name = dyn_cast<FlatSymbolRefAttr>(declaredDimension);
305 name &&
306 !appendMemberBinding(name, writtenDimension, tracker, bindings, relevantBindings)) {
307 return false;
308 }
309 }
310 // Array-valued bindings flatten into the enclosing array's trailing dimensions.
311 return collectMemberBindings(
312 declaredArray.getElementType(),
313 writtenArray.getSelectionType(declaredArray.getDimensionSizes().size()), tracker, bindings,
314 relevantBindings
315 );
316 }
317 if (auto declaredPod = dyn_cast<pod::PodType>(declaredType)) {
318 auto writtenPod = dyn_cast<pod::PodType>(writtenType);
319 if (!writtenPod || declaredPod.getRecords().size() != writtenPod.getRecords().size()) {
320 return false;
321 }
322 for (auto [declaredRecord, writtenRecord] :
323 llvm::zip_equal(declaredPod.getRecords(), writtenPod.getRecords())) {
324 if (declaredRecord.getName() != writtenRecord.getName() ||
325 !collectMemberBindings(
326 declaredRecord.getType(), writtenRecord.getType(), tracker, bindings, relevantBindings
327 )) {
328 return false;
329 }
330 }
331 return true;
332 }
333 if (auto declaredStruct = dyn_cast<StructType>(declaredType)) {
334 auto writtenStruct = dyn_cast<StructType>(writtenType);
335 if (!writtenStruct) {
336 return false;
337 }
338 // Concrete and partially instantiated structs hide their original parameter lists.
339 while (auto preimage = tracker.getPreimage(declaredStruct)) {
340 declaredStruct = *preimage;
341 }
342 while (auto preimage = tracker.getPreimage(writtenStruct)) {
343 writtenStruct = *preimage;
344 }
345 if (declaredStruct.getNameRef() != writtenStruct.getNameRef()) {
346 return false;
347 }
348 ArrayAttr declaredParams = declaredStruct.getParams();
349 ArrayAttr writtenParams = writtenStruct.getParams();
350 if (!declaredParams || !writtenParams) {
351 return !declaredParams && !writtenParams;
352 }
353 if (declaredParams.size() != writtenParams.size()) {
354 return false;
355 }
356 for (auto [declaredParam, writtenParam] : llvm::zip_equal(declaredParams, writtenParams)) {
357 if (auto name = dyn_cast<FlatSymbolRefAttr>(declaredParam)) {
358 if (!appendMemberBinding(name, writtenParam, tracker, bindings, relevantBindings)) {
359 return false;
360 }
361 } else if (auto declaredTypeParam = dyn_cast<TypeAttr>(declaredParam)) {
362 auto writtenTypeParam = dyn_cast<TypeAttr>(writtenParam);
363 if (auto writtenName = dyn_cast<FlatSymbolRefAttr>(writtenParam)) {
364 writtenTypeParam = TypeAttr::get(TypeVarType::get(writtenName));
365 }
366 if (!writtenTypeParam || !collectMemberBindings(
367 declaredTypeParam.getValue(), writtenTypeParam.getValue(),
368 tracker, bindings, relevantBindings
369 )) {
370 return false;
371 }
372 }
373 }
374 }
375 return true;
376}

411static void appendCallBindingConflictLocation(
412 Diagnostic &diag, FunctionType actual, FunctionType declared,
413 ArrayRef<StringRef> calleeNamespace, const ConversionTracker &tracker,
414 bool allowUnresolvedStructs, Operation *scope
415) {
416 auto prefixUnifies = [&](unsigned inputs, unsigned results) {
417 return callSignatureUnifies(
418 FunctionType::get(
419 actual.getContext(), actual.getInputs().take_front(inputs),
420 actual.getResults().take_front(results)
421 ),
422 FunctionType::get(
423 declared.getContext(), declared.getInputs().take_front(inputs),
424 declared.getResults().take_front(results)
425 ),
426 calleeNamespace, tracker, allowUnresolvedStructs, scope
427 );
428 };
429 for (unsigned i = 0; i < actual.getNumInputs(); ++i) {
430 if (!prefixUnifies(i + 1, 0)) {
431 diag << " at operand number " << i;
432 return;
433 }
434 }
435 for (unsigned i = 0; i < actual.getNumResults(); ++i) {
436 if (!prefixUnifies(actual.getNumInputs(), i + 1)) {
437 diag << " at result number " << i;
438 return;
439 }
440 }
441}

445static LogicalResult verifyStructCallBindings(ModuleOp module, const ConversionTracker &tracker) {
446 SymbolTableCollection tables;
447 WalkResult result = module.walk([&](CallOp call) {
448 if (calleeReferencesTemplateParam(call) ||
449 (!call.calleeIsStructCompute() && !call.calleeIsStructConstrain())) {
450 return WalkResult::advance();
451 }
452 auto target = call.getCalleeTarget(tables);
453 if (failed(target)) {
454 return WalkResult::interrupt();
455 }
456 // Leave structural mismatches to the existing instantiation/verifier diagnostics.
457 if (!typesUnify(
458 call.getTypeSignature(), target->get().getFunctionType(), target->getNamespace()
459 )) {
460 return WalkResult::advance();
461 }
462 if (!callSignatureUnifies(
463 call.getTypeSignature(), target->get().getFunctionType(), target->getNamespace(),
464 tracker, /*allowUnresolvedStructs=*/true, call
465 )) {
466 auto diag = call.emitError(
467 "cannot instantiate struct call because the complete call signature has conflicting "
468 "template bindings"
469 );
470 appendCallBindingConflictLocation(
471 *diag.getUnderlyingDiagnostic(), call.getTypeSignature(), target->get().getFunctionType(),
472 target->getNamespace(), tracker, /*allowUnresolvedStructs=*/true, call
473 );
474 diag.attachNote(target->get().getLoc()) << "target function declared here";
475 return WalkResult::interrupt();
476 }
477 return WalkResult::advance();
478 });
479 return failure(result.wasInterrupted());
480}

485class NonDetMemberTypeVerifier {
487 ModuleOp module;
489 const ConversionTracker &tracker;
491 SymbolTableCollection tables;
493 DenseSet<Operation *> visited;
495 DenseSet<Operation *> indexedStructs;
497 DenseMap<Operation *, SmallVector<MemberWriteOp>> writesByMember;
499 DenseMap<Operation *, SymbolRefAttr> comparisonNames;
501 DenseMap<std::pair<Type, Operation *>, Type> normalizedTypes;
503 SmallVector<SymbolLookupResult<StructDefOp>> resolvedDefinitions;
504
506 Type normalize(Type type, Operation *scope) {
507 auto key = std::make_pair(type, scope);
508 if (auto cached = normalizedTypes.lookup(key)) {
509 return cached;
510 }
511 Type normalized = type;
512 if (auto structure = dyn_cast<StructType>(type)) {
513 if (auto preimage = tracker.getPreimage(structure, scope)) {
514 normalized = normalize(*preimage, scope);
515 } else {
516 auto definition = structure.getDefinition(tables, scope);
517 if (failed(definition)) {
518 return {};
519 }
520 Operation *family = definition->get();
521 auto [name, inserted] = comparisonNames.try_emplace(family);
522 if (inserted) {
523 // These private names are used only by typesUnify, never published in IR or diagnostics.
524 name->second = FlatSymbolRefAttr::get(
525 type.getContext(), "__nondet_family_" + std::to_string(comparisonNames.size())
526 );
527 resolvedDefinitions.push_back(std::move(*definition));
528 }
529 SymbolRefAttr comparisonName = name->second;
530 SmallVector<Attribute> params;
531 if (structure.getParams()) {
532 for (Attribute param : structure.getParams()) {
533 if (auto typed = dyn_cast<TypeAttr>(param)) {
534 Type nested = normalize(typed.getValue(), scope);
535 if (!nested) {
536 return {};
537 }
538 param = TypeAttr::get(nested);
539 }
540 params.push_back(param);
541 }
542 }
543 normalized =
544 params.empty()
545 ? StructType::get(comparisonName)
546 : StructType::get(comparisonName, ArrayAttr::get(type.getContext(), params));
547 }
548 } else if (auto array = dyn_cast<ArrayType>(type)) {
549 Type element = normalize(array.getElementType(), scope);
550 if (!element) {
551 return {};
552 }
553 normalized = array.cloneWith(element);
554 } else if (auto podType = dyn_cast<pod::PodType>(type)) {
555 SmallVector<pod::RecordAttr> records;
556 for (auto record : podType.getRecords()) {
557 Type nested = normalize(record.getType(), scope);
558 if (!nested) {
559 return {};
560 }
561 records.push_back(pod::RecordAttr::get(type.getContext(), record.getName(), nested));
562 }
563 normalized = pod::PodType::get(type.getContext(), records);
564 }
565 if (normalized) {
566 normalizedTypes.try_emplace(key, normalized);
567 }
568 return normalized;
569 }

572 LogicalResult indexWriters(MemberDefOp member) {
573 auto owner = member->getParentOfType<StructDefOp>();
574 if (!indexedStructs.insert(owner).second) {
575 return success();
576 }
577 WalkResult result = owner.walk([&](MemberWriteOp write) {
578 auto definition = write.getMemberDefOp(tables);
579 if (failed(definition)) {
580 return WalkResult::interrupt();
581 }
582 writesByMember[definition->get()].push_back(write);
583 return WalkResult::advance();
584 });
585 return failure(result.wasInterrupted());
586 }

595 LogicalResult verifyComponent(NonDetOp witness, Value extraValue = {}) {
596 // Module checking shares visited nodes; mutation-time callers use a fresh checker.
597 if (visited.contains(witness) && !extraValue) {
598 return success();
599 }
600 struct Requirement {
601 Type normalized;
602 Type original;
603 Operation *origin;
604 bool external;
605 };
606 SmallVector<Requirement> requirements;
607 DenseSet<Type> seenTypes;
608 bool hasExternalRequirement = false;
609 auto addRequirement = [&](Type type, Operation *origin,
610 Operation *declaration = nullptr) -> LogicalResult {
611 bool external = declaration != nullptr;
612 hasExternalRequirement |= external;
613 Type comparison = normalize(type, external ? declaration : origin);
614 if (!comparison) {
615 return failure();
616 }
617 if (!seenTypes.insert(comparison).second) {
618 return success();
619 }
620 requirements.push_back({comparison, type, origin, external});
621 return success();
622 };
623 if (extraValue && failed(addRequirement(extraValue.getType(), extraValue.getDefiningOp()))) {
624 return failure();
625 }
626 DenseSet<Operation *> componentVisited;
627 DenseSet<Operation *> memberWrites;
628 SmallVector<Operation *> pending {witness};
629 while (!pending.empty()) {
630 Operation *node = pending.pop_back_val();
631 if (!componentVisited.insert(node).second) {
632 continue;
633 }
634 visited.insert(node);
635 if (auto nondet = dyn_cast<NonDetOp>(node)) {
636 if (failed(addRequirement(nondet.getType(), nondet))) {
637 return failure();
638 }
639 for (OpOperand &use : nondet.getResult().getUses()) {
640 auto write = dyn_cast<MemberWriteOp>(use.getOwner());
641 if (!write || write.getVal() != nondet.getResult()) {
642 if (auto call = dyn_cast<CallOp>(use.getOwner())) {
643 // Affine-map operands instantiate result parameters; they have no callee argument
644 // requirement, even when the same witness also appears in the argument segment.
645 OperandRange arguments = call.getArgOperands();
646 if (arguments.empty()) {
647 continue;
648 }
649 unsigned argIdx = use.getOperandNumber() - arguments.getBeginOperandIndex();
650 if (argIdx >= arguments.size()) {
651 continue;
652 }
653 auto callee = call.getCalleeTarget(tables);
654 if (failed(callee)) {
655 return failure();
656 }
657 if (failed(addRequirement(
658 callee->get().getFunctionType().getInput(argIdx), call, callee->get()
659 ))) {
660 return failure();
661 }
662 }
663 continue;
664 }
665 auto member = write.getMemberDefOp(tables);
666 if (failed(member)) {
667 return failure();
668 }
669 pending.push_back(member->get());
670 }
671 continue;
672 }
673 auto member = llvm::cast<MemberDefOp>(node);
674 if (failed(addRequirement(member.getType(), member)) || failed(indexWriters(member))) {
675 return failure();
676 }
677 for (MemberWriteOp write : writesByMember[member]) {
678 memberWrites.insert(write);
679 if (failed(addRequirement(write.getVal().getType(), write))) {
680 return failure();
681 }
682 if (auto other = write.getVal().getDefiningOp<NonDetOp>()) {
683 pending.push_back(other);
684 }
685 }
686 }
687 // A single write also needs coordination when an external call observes the witness.
688 // Components without member writes are not refined here.
689 if (!extraValue &&
690 (memberWrites.empty() || (memberWrites.size() < 2 && !hasExternalRequirement))) {
691 return success();
692 }
693 for (size_t i = 0; i < requirements.size(); ++i) {
694 const Requirement &current = requirements[i];
695 for (const Requirement &previous : ArrayRef(requirements).take_front(i)) {
696 // Unification is not transitive: a generic type may unify with two conflicting types.
697 if (!typesUnify(previous.normalized, current.normalized)) {
698 auto diagnostic = witness.emitError(
699 previous.external || current.external
700 ? "cannot refine nondeterministic witness because an external call requires an "
701 "incompatible type"
702 : "cannot refine shared nondeterministic witness because member writes require "
703 "incompatible types"
704 );
705 diagnostic.attachNote(previous.origin->getLoc())
706 << "first required type: " << previous.original;
707 diagnostic.attachNote(current.origin->getLoc())
708 << "conflicting required type: " << current.original;
709 return diagnostic;
710 }
711 }
712 }
713 return success();
714 }

719 return failed(verifyComponent(witness)) ? WalkResult::interrupt() : WalkResult::advance();
720 });
721 return failure(result.wasInterrupted());
722 }

731 ModuleOp rootMod;
732 SymbolTableCollection symTables;
733 bool reportMissing = true;
734
735 FailureOr<StructType> genClone(StructType typeAtCaller, ArrayRef<Attribute> typeAtCallerParams) {
736 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] attempting clone of " << typeAtCaller << '\n');
737 // Find the StructDefOp for the original StructType
738 FailureOr<SymbolLookupResult<StructDefOp>> r =
739 typeAtCaller.getDefinition(symTables, rootMod, reportMissing);
740 if (failed(r)) {
741 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] skip: cannot find StructDefOp \n");
742 return failure(); // getDefinition() already emits a sufficient error message
743 }
744 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] found definition\n";);
745
746 StructDefOp origStruct = r->get();
747 StructType typeAtDef = origStruct.getType();
748 MLIRContext *ctx = origStruct.getContext();
749 TemplateOp parentTemplate = getParentOfType<TemplateOp>(origStruct);
750 assert(parentTemplate && "parameterized struct must be nested in a TemplateOp");
751 ModuleOp parentModule = getParentOfType<ModuleOp>(parentTemplate);
752 assert(parentModule && "TemplateOp must be nested in a ModuleOp");
753
754 auto destination = getRootModule(origStruct);
755 if (failed(destination)) {
756 return failure();
757 }
758 auto localParams = rebaseTemplateParams(
759 symTables, ArrayAttr::get(ctx, typeAtCallerParams), rootMod, *destination, origStruct
760 );
761 if (failed(localParams)) {
762 return failure();
763 }
764
765 // Map of StructDefOp parameter name to concrete Attribute at the current instantiation site.
766 DenseMap<Attribute, Attribute> paramNameToConcrete;
767 // Reduced from `typeAtCallerParams` to contain only the non-concrete Attributes.
768 ArrayAttr reducedCallerParams = nullptr;
769 SmallVector<Attribute> nonConcreteParams;
770 {
771 ArrayAttr paramNames = typeAtDef.getParams();
772
773 // pre-conditions
774 assert(!isNullOrEmpty(paramNames));
775 assert(paramNames.size() == typeAtCallerParams.size());
776
777 for (size_t i = 0, e = paramNames.size(); i < e; ++i) {
778 Attribute next = typeAtCallerParams[i];
779 if (isConcreteStructParamAttr(next, /*allowStructParams=*/false)) {
780 paramNameToConcrete[paramNames[i]] = (*localParams)[i];
781 } else {
782 nonConcreteParams.push_back(next);
783 }
784 }
785 // post-conditions
786 assert(nonConcreteParams.size() + paramNameToConcrete.size() == paramNames.size());
787
788 if (paramNameToConcrete.empty()) {
789 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] skip: no concrete params \n");
790 return failure();
791 }
792 if (!nonConcreteParams.empty()) {
793 reducedCallerParams = ArrayAttr::get(ctx, nonConcreteParams);
794 }
795 }
796
797 FailureOr<InstantiationLayout> layoutResult =
798 buildInstantiationLayout(parentTemplate, ArrayAttr(), paramNameToConcrete);
799 if (failed(layoutResult)) {
800 return failure();
801 }
802 InstantiationLayout layout = std::move(*layoutResult);
803 assert(layout.remainingNames.size() == nonConcreteParams.size());
804
805 // This list will be used to build the new remote/external type.
806 SmallVector<FlatSymbolRefAttr> typeAtCallerSymPieces = getPieces(typeAtCaller.getNameRef());
807 typeAtCallerSymPieces.pop_back(); // drop struct name
808
809 // Evaluate any poly.expr symbols whose param dependencies are now concrete; add them to the
810 // map so ClonedBodyConstReadOpPattern can replace uses of those symbols too.
811 evaluateTemplateExprs(parentTemplate, paramNameToConcrete);
812
813 // Clone the original struct.
814 StructDefOp newStruct = origStruct.clone();
815 convertCalleesInPlace(newStruct, paramNameToConcrete);
816 Operation *insertedCloneRoot = nullptr;
817 if (layout.remainingNames.empty()) { // FULL INSTANTIATION CASE
818 // Set name of the new struct by prepending its name with instantiated template name.
819 newStruct.setSymName(
820 (layout.templateNameWithAttrs + mlir::Twine('_') + newStruct.getSymName()).str()
821 );
822 // Insert 'newStruct' into the parent ModuleOp of the original TemplateOp. Use the
823 // `SymbolTable::insert()` function so that the name will be made unique if necessary.
824 symTables.getSymbolTable(parentModule).insert(newStruct, Block::iterator(parentTemplate));
825 insertedCloneRoot = newStruct.getOperation();
826 // Drop the old template name from the list.
827 typeAtCallerSymPieces.pop_back();
828 } else { // PARTIAL INSTANTIATION CASE
829 // Clone the template and set instantiated name.
830 TemplateOp newTemplate =
831 createPartialInstantiationTemplate(parentTemplate, layout, symTables);
832
833 // Insert the struct into the template and the template into the module. Use the
834 // `SymbolTable::insert()` function so that the name will be made unique if necessary.
835 symTables.getSymbolTable(newTemplate).insert(newStruct);
836 symTables.getSymbolTable(parentModule).insert(newTemplate, Block::iterator(parentTemplate));
837 insertedCloneRoot = newTemplate.getOperation();
838
839 // Replace the old template name in the list with the new one (get template name after
840 // symbol table insertion since it may be modified to make it unique).
841 typeAtCallerSymPieces.back() = FlatSymbolRefAttr::get(newTemplate.getSymNameAttr());
842 }
843
844 // Retrieve the new type AFTER inserting since the struct name may be appended to make
845 // it unique and use the remaining non-concrete parameters from the original type.
846 StructType newLocalType = newStruct.getType(reducedCallerParams);
847 typeAtCallerSymPieces.push_back(
848 FlatSymbolRefAttr::get(newLocalType.getNameRef().getLeafReference())
849 );
850 StructType newRemoteType =
851 StructType::get(asSymbolRefAttr(typeAtCallerSymPieces), newLocalType.getParams());
852 LLVM_DEBUG({
853 llvm::dbgs() << "[StructCloner] original def type: " << typeAtDef << '\n';
854 llvm::dbgs() << "[StructCloner] cloned def type: " << newStruct.getType() << '\n';
855 llvm::dbgs() << "[StructCloner] original remote type: " << typeAtCaller << '\n';
856 llvm::dbgs() << "[StructCloner] cloned local type: " << newLocalType << '\n';
857 llvm::dbgs() << "[StructCloner] cloned remote type: " << newRemoteType << '\n';
858 });
859
860 // Within the new struct, replace all references to the original StructType (i.e., the
861 // locally-parameterized version) with the new locally-parameterized StructType,
862 // and replace all uses of the removed struct parameters with the concrete values.
863 if (failed(substituteStructBody(
864 newStruct, typeAtDef, paramNameToConcrete,
865 tracker_.delayedDiagnosticSet(
866 llvm::cast<StructType>(tracker_.scopeType(newRemoteType, rootMod))
867 )
868 ))) {
869 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] instantiating body of struct failed \n");
870 assert(insertedCloneRoot && "clone root must have been inserted before body conversion");
871 insertedCloneRoot->erase();
872 return failure();
873 }
874 return newRemoteType;
875 }
876
877public:
879 StructCloner(ConversionTracker &tracker, ModuleOp root)
880 : tracker_(tracker), rootMod(root), symTables() {}
881
883 FailureOr<StructType> createInstantiatedClone(StructType orig) {
884 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] orig: " << orig << '\n');
885 if (ArrayAttr params = orig.getParams(); !isNullOrEmpty(params)) {
886 return genClone(orig, params.getValue());
887 }
888 LLVM_DEBUG(llvm::dbgs() << "[StructCloner] skip: nullptr for params \n");
889 return failure();
890 }
891
893 void enableReportMissing() { reportMissing = true; }
894
896 void disableReportMissing() { reportMissing = false; }

901class ParameterizedStructUseTypeConverter : public TypeConverter {
902 ConversionTracker &tracker_;
903 StructCloner cloner;
904 ModuleOp root;
905
907 StructType cacheType(StructType type, bool toCache) const {
908 return llvm::cast<StructType>(tracker_.scopeType(type, root, toCache));
909 }
910
911 friend DisableReportMissing;
912
913public:
915 ParameterizedStructUseTypeConverter(ConversionTracker &tracker, ModuleOp lookupRoot)
916 : tracker_(tracker), cloner(tracker, lookupRoot), root(lookupRoot) {
917
918 addConversion([](Type inputTy) { return inputTy; });
919
920 addConversion([this](StructType inputTy) -> StructType {
921 LLVM_DEBUG(
922 llvm::dbgs() << "[ParameterizedStructUseTypeConverter] attempting conversion of "
923 << inputTy << '\n';
924 );
925 // Normalize type-valued arguments in this lookup scope before choosing a specialization.
926 // Publish the structural refinement first so cache entries retain one exact preimage.
927 if (ArrayAttr params = inputTy.getParams()) {
928 AttrTypeReplacer arguments;
929 arguments.addReplacement([this](Type nested) -> std::optional<Type> {
930 return convertType(nested);
931 });
932 auto converted = llvm::cast<ArrayAttr>(arguments.replace(params));
933 if (converted != params) {
934 tracker_.updateModifiedFlag(true);
935 return StructType::get(inputTy.getNameRef(), converted);
936 }
937 }
938 // First check for a cached entry
939 if (auto opt = tracker_.getInstantiation(cacheType(inputTy, true))) {
940 return cacheType(*opt, false);
941 }
942
943 // Otherwise, try to create a clone of the struct with instantiated params. If that can't be
944 // done, return the original type to indicate that it's still legal (for this step at least).
945 FailureOr<StructType> cloneRes = cloner.createInstantiatedClone(inputTy);
946 if (failed(cloneRes)) {
947 return inputTy;
948 }
949 StructType newTy = cloneRes.value();
950 LLVM_DEBUG(
951 llvm::dbgs() << "[ParameterizedStructUseTypeConverter] instantiating " << inputTy
952 << " as " << newTy << '\n'
953 );
954 tracker_.recordInstantiation(cacheType(inputTy, true), cacheType(newTy, true));
955 return newTy;
956 });
957
958 addConversion([this](ArrayType inputTy) {
959 return inputTy.cloneWith(convertType(inputTy.getElementType()));
960 });
961
962 addConversion([this](pod::PodType inputTy) -> pod::PodType {
963 SmallVector<pod::RecordAttr> convertedRecords;
964 for (pod::RecordAttr record : inputTy.getRecords()) {
965 Type convertedType = convertType(record.getType());
966 if (!convertedType) {
967 return {};
968 }
969 convertedRecords.push_back(
970 pod::RecordAttr::get(inputTy.getContext(), record.getName(), convertedType)
971 );
972 }
973 return pod::PodType::get(inputTy.getContext(), convertedRecords);
974 });
975 }

985 CallStructFuncPattern(TypeConverter &converter, MLIRContext *ctx, ConversionTracker &tracker)

991 LogicalResult matchAndRewrite(
992 CallOp op, OpAdaptor adapter, ConversionPatternRewriter &rewriter
993 ) const override {
994 LLVM_DEBUG(llvm::dbgs() << "[CallStructFuncPattern] CallOp: " << op << '\n');
995
996 // Convert the result types of the CallOp
997 SmallVector<Type> newResultTypes;
998 if (failed(getTypeConverter()->convertTypes(op.getResultTypes(), newResultTypes))) {
999 return op->emitError("Could not convert Op result types.");
1000 }
1001 LLVM_DEBUG({

1003 << debug::toStringList(newResultTypes) << '\n';
1004 });
1005
1006 // Update the callee to reflect the new struct target if necessary. These checks are based on
1007 // `CallOp::calleeIsStructC*()` but the types must not come from the CallOp in this case.
1008 // Instead they must come from the converted versions.
1009 SymbolRefAttr calleeAttr = op.getCalleeAttr();
1010 SymbolTableCollection tables;
1011 FuncDefOp target;
1012 if (!calleeReferencesTemplateParam(op)) {
1013 auto found = op.getCalleeTarget(tables);
1014 if (failed(found)) {
1015 return failure();
1016 }
1017 target = found->get();
1018 }
1019 if (target && target.isStructCompute()) {
1020 if (StructType newStTy = getIfSingleton<StructType>(newResultTypes)) {
1021 LLVM_DEBUG(llvm::dbgs() << "[CallStructFuncPattern] newStTy: " << newStTy << '\n');
1022 calleeAttr = appendLeaf(newStTy.getNameRef(), calleeAttr.getLeafReference());
1023 tracker_.reportDelayedDiagnostics(newStTy, op);
1024 }
1025 } else if (target && target.isStructConstrain()) {
1026 if (StructType newStTy = getAtIndex<StructType>(adapter.getArgOperands().getTypes(), 0)) {
1027 LLVM_DEBUG(llvm::dbgs() << "[CallStructFuncPattern] newStTy: " << newStTy << '\n');
1028 calleeAttr = appendLeaf(newStTy.getNameRef(), calleeAttr.getLeafReference());
1029 }
1030 }
1031
1032 LLVM_DEBUG(llvm::dbgs() << "[CallStructFuncPattern] replaced " << op);
1033 CallOp newOp = replaceOpWithNewOp<CallOp>(
1034 rewriter, op, newResultTypes, calleeAttr, adapter.getMapOperands(),
1035 op.getNumDimsPerMapAttr(), adapter.getArgOperands()
1036 );
1037 (void)newOp; // tell compiler it's intentionally unused in release builds
1038 LLVM_DEBUG(llvm::dbgs() << " with " << newOp << '\n');
1039 return success();
1040 }

1071static std::optional<StructType> getComplementaryStructRefinement(
1072 StructType lhs, StructType rhs, const ConversionTracker &tracker, Operation *scope
1073) {
1074 lhs = tracker.getPreimage(lhs, scope).value_or(lhs);
1075 rhs = tracker.getPreimage(rhs, scope).value_or(rhs);
1076 if (lhs.getNameRef() != rhs.getNameRef()) {
1077 return std::nullopt;
1078 }
1079
1080 ArrayAttr lhsParams = lhs.getParams();
1081 ArrayAttr rhsParams = rhs.getParams();
1082 if (!lhsParams || !rhsParams || lhsParams.size() != rhsParams.size()) {
1083 return std::nullopt;
1084 }
1085
1086 SmallVector<Attribute> refinedParams;
1087 for (auto [lhsParam, rhsParam] : llvm::zip_equal(lhsParams, rhsParams)) {
1088 if (lhsParam == rhsParam) {
1089 refinedParams.push_back(lhsParam);
1090 continue;
1091 }
1092 auto lhsInt = llvm::dyn_cast<IntegerAttr>(lhsParam);
1093 auto rhsInt = llvm::dyn_cast<IntegerAttr>(rhsParam);
1094 if (lhsInt && !isDynamic(lhsInt) && llvm::isa<AffineMapAttr>(rhsParam)) {
1095 refinedParams.push_back(lhsParam);
1096 continue;
1097 }
1098 if (rhsInt && !isDynamic(rhsInt) && llvm::isa<AffineMapAttr>(lhsParam)) {
1099 refinedParams.push_back(rhsParam);
1100 continue;
1101 }
1102 return std::nullopt;
1103 }
1104 return StructType::get(lhs.getNameRef(), ArrayAttr::get(lhs.getContext(), refinedParams));
1105}

1108class NonDetOpPattern : public OpConversionPattern<NonDetOp> {
1109 ConversionTracker *tracker_;
1110 ModuleOp rootModule_;
1111 bool &hadConflict_;
1112
1113public:
1115 NonDetOpPattern(
1116 TypeConverter &converter, MLIRContext *ctx, ConversionTracker *tracker, ModuleOp rootModule,
1117 bool &hadConflict
1118 )
1119 : OpConversionPattern<NonDetOp>(converter, ctx, /*benefit=*/1), tracker_(tracker),
1120 rootModule_(rootModule), hadConflict_(hadConflict) {}
1121
1123 LogicalResult
1124 matchAndRewrite(NonDetOp op, OpAdaptor, ConversionPatternRewriter &rewriter) const override {
1125 if (hadConflict_) {
1126 return failure();
1127 }
1128 if (tracker_ && failed(NonDetMemberTypeVerifier(rootModule_, *tracker_).verifyComponent(op))) {
1129 hadConflict_ = true;
1130 return failure();
1131 }
1132 Type replacementInputType = op.getType();
1133 SmallVector<MemberDefOp> refinedMemberDefs;
1134 bool canRefineAllMembers = true;
1135 if (tracker_) {
1136 SymbolTableCollection tables;
1137 for (OpOperand &use : op.getResult().getUses()) {
1138 auto memberWrite = llvm::dyn_cast<MemberWriteOp>(use.getOwner());
1139 if (!memberWrite || memberWrite.getVal() != op.getResult()) {
1140 continue;
1141 }
1142 FailureOr<SymbolLookupResult<MemberDefOp>> memberDef = memberWrite.getMemberDefOp(tables);
1143 if (failed(memberDef)) {
1144 return failure();
1145 }
1146 auto witnessType = llvm::dyn_cast<StructType>(replacementInputType);
1147 auto memberType = llvm::dyn_cast<StructType>(memberDef->get().getType());
1148 if (!witnessType || !memberType) {
1149 continue;
1150 }
1151 std::optional<StructType> commonType =
1152 getComplementaryStructRefinement(witnessType, memberType, *tracker_, op);
1153 if (!commonType) {
1154 // A compatible but unsupported join must not commit a successful prefix of the uses.
1155 // Conflicting requirements are rejected by the shared-witness preflight.
1156 canRefineAllMembers = false;
1157 continue;
1158 }
1159 replacementInputType = *commonType;
1160 refinedMemberDefs.push_back(memberDef->get());
1161 }
1162 }
1163 if (!canRefineAllMembers) {
1164 replacementInputType = op.getType();
1165 refinedMemberDefs.clear();
1166 }
1167
1168 // Ordinary type conversion updates all users consistently. The additional member-driven
1169 // refinement below only repairs member writes; it cannot leave other users observing an
1170 // obsolete conversion cast or silently split their witness from the member writes.
1171 if (replacementInputType != op.getType()) {
1172 for (OpOperand &use : op.getResult().getUses()) {
1173 auto write = dyn_cast<MemberWriteOp>(use.getOwner());
1174 if (!write || write.getVal() != op.getResult()) {
1175 auto diagnostic = op.emitError(
1176 "cannot refine nondeterministic witness from member writes because it has an "
1177 "unsupported external use"
1178 );
1179 diagnostic.attachNote(use.getOwner()->getLoc()) << "source witness is also used here";
1180 hadConflict_ = true;
1181 return diagnostic;
1182 }
1183 }
1184 }
1185
1186 Type newType = getTypeConverter()->convertType(replacementInputType);
1187 if (!newType) {
1188 return op->emitError("Could not convert Op result type.");
1189 }
1190 bool witnessChanged = newType != op.getType();
1191 bool membersChanged = llvm::any_of(refinedMemberDefs, [newType](MemberDefOp member) {
1192 return member.getType() != newType;
1193 });
1194 if (!witnessChanged && !membersChanged) {
1195 return failure();
1196 }
1197 for (MemberDefOp memberDef : refinedMemberDefs) {
1198 if (memberDef.getType() != newType) {
1199 rewriter.modifyOpInPlace(memberDef, [&memberDef, &newType]() {
1200 memberDef.setType(newType);
1201 });
1202 }
1203 }
1204 if (witnessChanged) {
1205 replaceOpWithNewOp<NonDetOp>(rewriter, op, newType);
1206 }
1207 return success();
1208 }
1209};

1246class DisableReportMissing : public LegalityCheckCallback {
1247 ParameterizedStructUseTypeConverter &tyConv;
1248
1249public:
1251 explicit DisableReportMissing(ParameterizedStructUseTypeConverter &tc) : tyConv(tc) {}
1252
1254 void checkStarted() override { tyConv.cloner.disableReportMissing(); }
1255
1257 void checkEnded(bool) override { tyConv.cloner.enableReportMissing(); }

1261LogicalResult run(ModuleOp modOp, ConversionTracker &tracker) {
1262 MLIRContext *ctx = modOp.getContext();
1263 if (failed(NonDetMemberTypeVerifier(modOp, tracker).verifyModule())) {
1264 return failure();
1265 }
1266 if (failed(verifyStructCallBindings(modOp, tracker))) {
1267 return failure();
1268 }
1269 SmallVector<ModuleOp> roots {modOp};
1270 modOp.walk<WalkOrder::PreOrder>([&](ModuleOp nested) {
1271 if (nested != modOp && nested->hasAttr(LANG_ATTR_NAME)) {
1272 roots.push_back(nested);
1273 }
1274 });
1275 // Each converter resolves local names at its own language root. Canonical cache keys still
1276 // share clones with qualified outer uses, without conflating equal names in different roots.
1277 for (ModuleOp root : roots) {
1278 bool hadConflict = false;
1279 ParameterizedStructUseTypeConverter tyConv(tracker, root);
1280 // Module operations remain legal during conversion, so update each root's main type here.
1281 if (auto main = root->getAttrOfType<TypeAttr>(MAIN_ATTR_NAME)) {
1282 Type converted = tyConv.convertType(main.getValue());
1283 if (!converted) {
1284 return failure();
1285 }
1286 if (converted != main.getValue()) {
1287 root->setAttr(MAIN_ATTR_NAME, TypeAttr::get(converted));
1288 tracker.updateModifiedFlag(true);
1289 }
1290 }
1291 DisableReportMissing drm(tyConv);
1292 ConversionTarget target = newConverterDefinedTargetWithCallback<>(tyConv, ctx, drm);
1293 target.addLegalOp<ModuleOp>();
1294 target.markOpRecursivelyLegal<ModuleOp>([&](ModuleOp nested) {
1295 return nested != root && nested->hasAttr(LANG_ATTR_NAME);
1296 });
1297 target.addDynamicallyLegalOp<CallOp>([&](CallOp call) {
1298 drm.checkStarted();
1299 bool legal = defaultLegalityCheck(tyConv, call);
1300 drm.checkEnded(legal);
1301 if (!legal || calleeReferencesTemplateParam(call)) {
1302 return legal;
1303 }
1304 SymbolTableCollection tables;
1305 auto callee = call.getCalleeTarget(tables);
1306 if (failed(callee)) {
1307 return false;
1308 }
1309 StructType owner;
1310 if (callee->get().isStructCompute()) {
1311 owner = getIfSingleton<StructType>(call.getResultTypes());
1312 } else if (callee->get().isStructConstrain()) {
1313 owner = getAtIndex<StructType>(call.getArgOperands().getTypes(), 0);
1314 }
1315 // Nested type conversions can finish before the surrounding call is revisited.
1316 // Its result/receiver and callee must name the same specialization.
1317 return !owner || call.getCalleeAttr() ==
1318 appendLeaf(owner.getNameRef(), call.getCalleeAttr().getLeafReference());
1319 });
1320 target.addDynamicallyLegalOp<NonDetOp>([&tyConv](Operation *op) {
1321 return defaultLegalityCheck(tyConv, op);
1322 });
1323 RewritePatternSet patterns = newGeneralRewritePatternSet(tyConv, ctx, target);
1324 patterns.add<CallStructFuncPattern, MemberDefOpPattern>(tyConv, ctx, tracker);
1325 patterns.add<NonDetOpPattern>(tyConv, ctx, &tracker, modOp, hadConflict);
1326 patterns.add<MemberWriteRefinedNonDetPattern>(tyConv, ctx);
1327 if (failed(applyPartialConversion(root, target, std::move(patterns))) || hadConflict) {
1328 return failure();
1329 }
1330 }
1331 return success();
1332}

1336LogicalResult instantiateMainStruct(ModuleOp modOp, ConversionTracker &tracker) {
1337 FailureOr<StructType> mainTypeOpt = getMainInstanceType(modOp);
1338 if (failed(mainTypeOpt)) {
1339 return failure();
1340 }
1341
1342 StructType mainType = mainTypeOpt.value();
1343 if (!mainType || isNullOrEmpty(mainType.getParams()) || tracker.getInstantiation(mainType)) {
1344 return success();
1345 }
1346
1347 StructCloner cloner(tracker, modOp);
1348 FailureOr<StructType> cloneRes = cloner.createInstantiatedClone(mainType);
1349 if (failed(cloneRes)) {
1350 return failure();
1351 }
1352
1353 StructType instantiatedMainType = cloneRes.value();
1354 tracker.recordInstantiation(mainType, instantiatedMainType);
1355 modOp->setAttr(MAIN_ATTR_NAME, TypeAttr::get(instantiatedMainType));
1356 return success();
1357}

1384 SymbolTableCollection &symTables_;
1385 const DenseMap<Attribute, Attribute> &paramNameToConcrete_;
1386 SmallVector<std::pair<Operation *, FlatSymbolRefAttr>> activeInferences_;
1387
1388public:
1390 WildcardTypeBodyInferer(
1391 SymbolTableCollection &symTables, const DenseMap<Attribute, Attribute> &paramNameToConcrete
1392 )
1393 : symTables_(symTables), paramNameToConcrete_(paramNameToConcrete) {}

1400private:
1402 FailureOr<Attribute> rebaseEvidence(Attribute value, Operation *source, Operation *destination) {
1403 auto sourceRoot = getRootModule(source);
1404 auto destinationRoot = getRootModule(destination);
1405 if (failed(sourceRoot) || failed(destinationRoot)) {
1406 return failure();
1407 }
1408 if (*sourceRoot == *destinationRoot) {
1409 return value;
1410 }
1411 auto rebased = rebaseTemplateParams(
1412 symTables_, ArrayAttr::get(value.getContext(), {value}), source, *destinationRoot, source
1413 );
1414 if (failed(rebased)) {
1415 return failure();
1416 }
1417 return (*rebased)[0];
1418 }
1419
1421 FailureOr<Attribute> infer(
1422 FuncDefOp func, FlatSymbolRefAttr paramName,
1423 const DenseMap<Attribute, Attribute> &paramNameToConcrete
1424 ) {
1425 if (llvm::any_of(activeInferences_, [&](const auto &e) {
1426 return e.first == func.getOperation() && e.second == paramName;
1427 })) {
1428 return Attribute();
1429 }
1430 activeInferences_.emplace_back(func.getOperation(), paramName);
1431
1432 TemplateTypeConverter tyConv(paramNameToConcrete);
1433 std::optional<Attribute> inferred;
1434 bool ambiguous = false;
1435 bool invalidEvidence = false;
1436
1437 // Record a concrete candidate unless it conflicts with an earlier one, in which
1438 // case the wildcard is treated as ambiguous and left unresolved.
1439 auto noteCandidate = [&inferred, &ambiguous](Attribute candidate) {
1440 if (!candidate || !isConcreteStructParamAttr(candidate)) {
1441 return WalkResult::advance();
1442 }
1443 if (!inferred.has_value()) {
1444 inferred = candidate;
1445 return WalkResult::advance();
1446 }
1447 if (*inferred != candidate) {
1448 ambiguous = true;
1449 return WalkResult::interrupt();
1450 }
1451 return WalkResult::advance();
1452 };
1453
1454 func.walk([&](Operation *bodyOp) {
1455 if (auto castOp = llvm::dyn_cast<UnifiableCastOp>(bodyOp)) {
1456 Type inputTy = tyConv.convertType(castOp.getInput().getType());
1457 Type resultTy = tyConv.convertType(castOp.getResult().getType());
1458 if (auto inputTvar = llvm::dyn_cast<TypeVarType>(inputTy);
1459 inputTvar && inputTvar.getNameRef() == paramName && isConcreteType(resultTy)) {
1460 return noteCandidate(TypeAttr::get(resultTy));
1461 }
1462 if (auto resultTvar = llvm::dyn_cast<TypeVarType>(resultTy);
1463 resultTvar && resultTvar.getNameRef() == paramName && isConcreteType(inputTy)) {
1464 return noteCandidate(TypeAttr::get(inputTy));
1465 }
1466
1467 // A requested wildcard can occur beneath an aggregate type on either side of a local
1468 // cast. Recursively unify both converted types to retain the concrete binding recorded
1469 // at that nested position.
1470 UnificationMap unifications;
1471 if (!typesUnify(inputTy, resultTy, {}, &unifications)) {
1472 return WalkResult::advance();
1473 }
1474 for (Side side : {Side::LHS, Side::RHS}) {
1475 auto it = unifications.find({paramName, side});
1476 if (it != unifications.end()) {
1477 WalkResult candidateResult = noteCandidate(it->second);
1478 if (candidateResult.wasInterrupted()) {
1479 return candidateResult;
1480 }
1481 }
1482 }
1483 return WalkResult::advance();
1484 }
1485
1486 auto nestedCall = llvm::dyn_cast<CallOp>(bodyOp);
1487 if (!nestedCall) {
1488 return WalkResult::advance();
1489 }
1490
1491 FailureOr<SymbolLookupResult<FuncDefOp>> nestedTgtOpt =
1492 nestedCall.getCalleeTarget(symTables_);
1493 if (failed(nestedTgtOpt)) {
1494 return WalkResult::advance();
1495 }
1496 FuncDefOp nestedTgt = nestedTgtOpt->get();
1497 auto nestedTemplate = llvm::dyn_cast<TemplateOp>(nestedTgt->getParentOp());
1498 if (!nestedTemplate) {
1499 return WalkResult::advance();
1500 }
1501
1502 // Infer through either side of a nested call: an enclosing type variable can appear
1503 // in a call result or in a call operand. In both cases, the corresponding nested
1504 // callee type variable may be concretely bound by the nested call.
1505 auto inferFromNestedBinding = [&](Type enclosingTy, Type nestedTy) {
1506 auto nestedBindings = getNestedConcreteBindings(
1507 nestedCall, nestedTgt, nestedTemplate, nestedTgtOpt->getNamespace(), tyConv
1508 );
1509 if (failed(nestedBindings)) {
1510 invalidEvidence = true;
1511 return WalkResult::interrupt();
1512 }
1513 auto &nestedParamNameToConcrete = *nestedBindings;
1514
1515 // In addition to bindings from the call, type variables in the nested signature can
1516 // obtain their concrete values from the nested callee body. Add that evidence to the
1517 // nested scope before converting its full signature type below.
1518 nestedTy.walk([&](TypeVarType nestedTvar) {
1519 if (nestedParamNameToConcrete.contains(nestedTvar.getNameRef())) {
1520 return;
1521 }
1522 auto nestedCandidate =
1523 infer(nestedTgt, nestedTvar.getNameRef(), nestedParamNameToConcrete);
1524 if (failed(nestedCandidate)) {
1525 invalidEvidence = true;
1526 } else if (*nestedCandidate) {
1527 nestedParamNameToConcrete[nestedTvar.getNameRef()] = *nestedCandidate;
1528 }
1529 });
1530 if (invalidEvidence) {
1531 return WalkResult::interrupt();
1532 }
1533 TemplateTypeConverter nestedTyConv(nestedParamNameToConcrete);
1534 Type convertedNestedTy = nestedTyConv.convertType(nestedTy);
1535 if (!isConcreteType(convertedNestedTy)) {
1536 return WalkResult::advance();
1537 }
1538 auto enclosingEvidence = rebaseEvidence(TypeAttr::get(convertedNestedTy), nestedTgt, func);
1539 if (failed(enclosingEvidence)) {
1540 invalidEvidence = true;
1541 return WalkResult::interrupt();
1542 }
1543 convertedNestedTy = llvm::cast<TypeAttr>(*enclosingEvidence).getValue();
1544
1545 Type convertedEnclosingTy = tyConv.convertType(enclosingTy);
1546 // Apply the nested call's bindings recursively before unifying. In particular, this
1547 // preserves evidence when the nested callee's type variable occurs within an aggregate
1548 // signature rather than as the complete operand or result type.
1549 UnificationMap unifications;
1550 if (!typesUnify(convertedEnclosingTy, convertedNestedTy, {}, &unifications)) {
1551 return WalkResult::advance();
1552 }
1553 auto it = unifications.find({paramName, Side::LHS});
1554 if (it != unifications.end()) {
1555 return noteCandidate(it->second);
1556 }
1557 return WalkResult::advance();
1558 };
1559
1560 for (auto [result, nestedResultTy] :
1561 llvm::zip_equal(nestedCall.getResults(), nestedTgt.getFunctionType().getResults())) {
1562 WalkResult candidateResult = inferFromNestedBinding(result.getType(), nestedResultTy);
1563 if (candidateResult.wasInterrupted()) {
1564 return candidateResult;
1565 }
1566 }
1567 for (auto [operand, nestedInputTy] :
1568 llvm::zip_equal(nestedCall.getArgOperands(), nestedTgt.getFunctionType().getInputs())) {
1569 WalkResult candidateResult = inferFromNestedBinding(operand.getType(), nestedInputTy);
1570 if (candidateResult.wasInterrupted()) {
1571 return candidateResult;
1572 }
1573 }
1574 return WalkResult::advance();
1575 });
1576
1577 activeInferences_.pop_back();
1578 if (invalidEvidence) {
1579 return failure();
1580 }
1581 return ambiguous ? Attribute() : inferred.value_or(Attribute());
1582 }
1583
1588 FailureOr<DenseMap<Attribute, Attribute>> getNestedConcreteBindings(
1589 CallOp nestedCall, FuncDefOp nestedTgt, TemplateOp nestedTemplate,
1590 ArrayRef<StringRef> calleeNamespace, const TemplateTypeConverter &enclosingTyConv
1591 ) {
1592 DenseMap<Attribute, Attribute> bindings;
1593 auto nestedParams = nestedTemplate.getConstOps<TemplateParamOp>();
1594 ArrayAttr callParams = nestedCall.getTemplateParamsAttr();
1595
1596 if (!isNullOrEmpty(callParams)) {
1597 for (auto [paramOp, arg] : llvm::zip_equal(nestedParams, callParams.getValue())) {
1598 Attribute value = enclosingTyConv.convertAttr(arg);
1599 if (isConcreteStructParamAttr(value)) {
1600 bindings[FlatSymbolRefAttr::get(paramOp.getSymNameAttr())] = value;
1601 }
1602 }
1603 }
1604
1605 // Explicit wildcards and unresolved enclosing parameters can still acquire concrete
1606 // bindings from the nested call's signature. Preserve concrete explicit arguments as
1607 // the source of truth and fill only the remaining parameters from that evidence.
1608 auto unifyResult =
1609 nestedCall.unifyTypeSignatureWithNamespace(nestedTgt.getFunctionType(), calleeNamespace);
1610 if (succeeded(unifyResult)) {
1611 for (TemplateParamOp paramOp : nestedParams) {
1612 auto paramName = FlatSymbolRefAttr::get(paramOp.getSymNameAttr());
1613 if (bindings.contains(paramName)) {
1614 continue;
1615 }
1616 if (std::optional<Attribute> value = inferUnifiedParam(*unifyResult, paramName)) {
1617 Attribute convertedValue = enclosingTyConv.convertAttr(*value);
1618 if (isConcreteStructParamAttr(convertedValue)) {
1619 bindings[paramName] = convertedValue;
1620 }
1621 }
1622 }
1623 }
1624 // Call-site evidence uses the enclosing root; recursive body inference and substitution
1625 // must see declaration-local types, including struct types nested inside aggregates.
1626 for (auto &[name, value] : bindings) {
1627 auto localValue = rebaseEvidence(value, nestedCall, nestedTgt);
1628 if (failed(localValue)) {
1629 return failure();
1630 }
1631 value = *localValue;
1632 }
1633 return bindings;
1634 }
1635};

1661static LogicalResult copyReferencedTemplateExprs(
1662 TemplateOp parentTemplate, Block &newTemplateBody, ArrayRef<FuncDefOp> newFuncs,
1663 const DenseMap<Attribute, Attribute> &paramNameToConcrete,
1664 SmallVector<Diagnostic> &delayedDiagnostics
1665) {
1666 DenseSet<StringAttr> referencedExprNames;
1667 auto collectExprRef = [&referencedExprNames, &paramNameToConcrete](FlatSymbolRefAttr name) {
1668 if (!paramNameToConcrete.contains(name)) {
1669 referencedExprNames.insert(name.getAttr());
1670 }
1671 };
1672
1673 // Expression references can occur in a function signature, in type-bearing operands/results,
1674 // or in arbitrary attributes as well as in poly.read_const operations. In particular, a
1675 // signature-only reference must keep its defining expression in a partial template even if the
1676 // function body does not read that expression.
1677 for (FuncDefOp newFunc : newFuncs) {
1678 newFunc.walk([&](Operation *nestedOp) {
1679 auto collectTypeRefs = [&collectExprRef](Type type) { type.walk(collectExprRef); };
1680 for (Type type : nestedOp->getOperandTypes()) {
1681 collectTypeRefs(type);
1682 }
1683 for (Type type : nestedOp->getResultTypes()) {
1684 collectTypeRefs(type);
1685 }
1686 nestedOp->getAttrDictionary().walk(collectExprRef);
1687 });
1688 newFunc.walk([&collectExprRef](ConstReadOp readOp) {
1689 collectExprRef(readOp.getConstNameAttr());
1690 });
1691 }
1692
1693 for (TemplateExprOp expr : parentTemplate.getConstOps<TemplateExprOp>()) {
1694 if (!referencedExprNames.contains(expr.getSymNameAttr())) {
1695 continue;
1696 }
1697
1698 auto clonedExpr = llvm::cast<TemplateExprOp>(expr->clone());
1699 if (failed(substituteTemplateExprBody(clonedExpr, paramNameToConcrete, delayedDiagnostics))) {
1700 clonedExpr->erase();
1701 return failure();
1702 }
1703 newTemplateBody.push_back(clonedExpr);
1704 }
1705 return success();
1706}

1711static SmallVector<FuncDefOp> copyReferencedTemplateSiblingFuncs(
1712 TemplateOp parentTemplate, TemplateOp newTemplate, FuncDefOp originalFunc, FuncDefOp newFunc,
1713 SymbolTableCollection &symTables, const DenseMap<Attribute, Attribute> &paramNameToConcrete,
1714 SmallVectorImpl<CallOp> &copiedSiblingCalls
1715) {
1716 DenseMap<Operation *, FuncDefOp> cloned;
1717 DenseMap<Operation *, FuncDefOp> cloneSources;
1718 SmallVector<FuncDefOp> copiedFuncs {newFunc};
1719 cloned[originalFunc] = newFunc;
1720 cloneSources[newFunc] = originalFunc;
1721
1722 for (size_t i = 0; i < copiedFuncs.size(); ++i) {
1723 FuncDefOp current = copiedFuncs[i];
1724 SmallVector<CallOp> sourceCalls = walkCollect<CallOp>(*cloneSources.lookup(current));
1725 size_t callIndex = 0;
1726 current.walk([&](CallOp nestedCall) {
1727 assert(callIndex < sourceCalls.size() && "function clone must preserve calls");
1728 CallOp sourceCall = sourceCalls[callIndex++];
1729 FailureOr<SymbolLookupResult<FuncDefOp>> siblingTarget = lookupTopLevelSymbol<FuncDefOp>(
1730 symTables, sourceCall.getCalleeAttr(), sourceCall.getOperation(),
1731 /*reportMissing=*/false
1732 );
1733 if (failed(siblingTarget)) {
1734 return;
1735 }
1736 FuncDefOp sibling = siblingTarget->get();
1737 if (!sibling || sibling->getParentOp() != parentTemplate) {
1738 return;
1739 }
1740
1741 FuncDefOp clonedSibling;
1742 if (auto found = cloned.find(sibling); found != cloned.end()) {
1743 clonedSibling = found->second;
1744 } else {
1745 clonedSibling = cloneFunctionWithConvertedCallees(sibling, paramNameToConcrete);
1746 cloned[sibling] = clonedSibling;
1747 cloneSources[clonedSibling] = sibling;
1748 copiedFuncs.push_back(clonedSibling);
1749 }
1750
1751 // Use the resolved target's full path so calls through enclosing named modules do not
1752 // depend on a particular `@Template::@function` spelling.
1753 FailureOr<SymbolRefAttr> siblingPath = getPathFromTopRoot(sibling);
1754 if (failed(siblingPath)) {
1755 return;
1756 }
1757 SmallVector<FlatSymbolRefAttr> calleePieces = getPieces(*siblingPath);
1758 assert(
1759 calleePieces.size() >= 2 && "template function path must include template and function"
1760 );
1761 calleePieces.pop_back();
1762 calleePieces.pop_back();
1763 calleePieces.push_back(FlatSymbolRefAttr::get(newTemplate.getSymNameAttr()));
1764 calleePieces.push_back(FlatSymbolRefAttr::get(clonedSibling.getSymNameAttr()));
1765 nestedCall.setCalleeAttr(asSymbolRefAttr(calleePieces));
1766 copiedSiblingCalls.push_back(nestedCall);
1767 });
1768 }
1769 return copiedFuncs;

1772class InstantiateFuncAtCallOp final : public OpRewritePattern<CallOp> {
1773 ConversionTracker &tracker_;
1774
1775public:
1777 InstantiateFuncAtCallOp(MLIRContext *ctx, ConversionTracker &tracker)
1778 : OpRewritePattern<CallOp>(ctx), tracker_(tracker) {}
1779
1781 LogicalResult matchAndRewrite(CallOp op, PatternRewriter &rewriter) const override {
1782 LLVM_DEBUG(llvm::dbgs() << "[InstantiateFuncAtCallOp] op: " << op << '\n');
1783
1784 if (calleeReferencesTemplateParam(op)) {
1785 return failure();
1786 }
1787
1788 // Lookup callee target function
1789 SymbolTableCollection symTables;
1790 FailureOr<SymbolLookupResult<FuncDefOp>> callTgtOpt = op.getCalleeTarget(symTables);
1791 if (failed(callTgtOpt)) {
1792 return rewriter.notifyMatchFailure(op, [](Diagnostic &diag) {
1793 diag << "could not find target function for call";
1794 });
1795 }
1796 FuncDefOp callTgt = callTgtOpt->get();
1797
1798 // Check if callee is within a TemplateOp
1799 TemplateOp parentTemplate = llvm::dyn_cast<TemplateOp>(callTgt->getParentOp());
1800 if (!parentTemplate) {
1801 return failure(); // nothing to do if not parameterized
1802 }
1803 LLVM_DEBUG(
1804 llvm::dbgs() << "[InstantiateFuncAtCallOp] target function in template "
1805 << parentTemplate.getSymName() << '\n'
1806 );
1807
1808 // Perform type unification with tracking to infer the instantiated type(s). Even though
1809 // `CallOp` verification already checked that caller and callee types unify, the progress of
1810 // instantiation so far may have brought together a chain of calls across templates where each
1811 // individual unification check passed due to permissive type variables and/or symbols in the
1812 // middle but the overall chain does not unify. Hence, this unification may fail and should
1813 // produce a meaningful error message if it does.
1814 // See: `test/Transforms/Flattening/instantiate_funcs_fail.llzk`
1815 FailureOr<UnificationMap> unifyResult =
1816 unifyTypeSignature(op, callTgt, callTgtOpt->getNamespace(), rewriter);
1817 if (failed(unifyResult)) {
1818 return failure();
1819 }
1820 LLVM_DEBUG(
1821 llvm::dbgs() << "[InstantiateFuncAtCallOp] unifications of types: "
1822 << debug::toStringList(unifyResult.value()) << '\n'
1823 );
1824
1825 // Maps template parameter symbols to the instantiation value at the call site.
1826 DenseMap<Attribute, Attribute> paramNameToConcrete;
1827 auto collectRes = collectConcreteTemplateParams(
1828 op, rewriter, symTables, callTgt, parentTemplate, unifyResult.value(), paramNameToConcrete
1829 );
1830 if (failed(collectRes)) {
1831 return failure();
1832 }
1833
1834 if (paramNameToConcrete.empty()) {
1835 LLVM_DEBUG(llvm::dbgs() << "[InstantiateFuncAtCallOp] skip: no concrete params\n");
1836 return failure();
1837 }
1838
1839 evaluateTemplateExprs(parentTemplate, paramNameToConcrete);
1840
1841 FailureOr<InstantiationLayout> layoutResult =
1842 buildInstantiationLayout(parentTemplate, op.getTemplateParamsAttr(), paramNameToConcrete);
1843 if (failed(layoutResult)) {
1844 return failure();
1845 }
1846 InstantiationLayout layout = std::move(*layoutResult);
1847 ModuleOp parentModule = getParentOfType<ModuleOp>(parentTemplate);
1848 assert(parentModule && "TemplateOp must be nested in a ModuleOp");
1849
1850 FailureOr<SymbolRefAttr> newCalleeAttr =
1851 layout.remainingNames.empty() ? instantiateFully(
1852 op, rewriter, symTables, callTgt, parentTemplate,
1853 parentModule, layout.templateNameWithAttrs,
1854 layout.concreteParamKey, paramNameToConcrete, tracker_
1855 )
1856 : instantiatePartially(
1857 op, rewriter, symTables, callTgt, parentTemplate,
1858 parentModule, layout, paramNameToConcrete, tracker_
1859 );
1860 if (failed(newCalleeAttr)) {
1861 return failure();
1862 }
1863
1864 FailureOr<SymbolRefAttr> originalCalleePath = getPathFromTopRoot(callTgt);
1865 if (failed(originalCalleePath)) {
1866 return failure();
1867 }
1868 tracker_.recordInstantiation(*originalCalleePath);
1869
1870 // Update the CallOp to point to the instantiated function and mark the module as modified.
1871 rewriter.modifyOpInPlace(op, [&op, &newCalleeAttr, &layout]() {
1872 LLVM_DEBUG({
1873 llvm::dbgs() << "[InstantiateFuncAtCallOp] updating callee from " << op.getCalleeAttr()
1874 << " to " << *newCalleeAttr << '\n';
1875 });
1876 op.setCalleeAttr(*newCalleeAttr);
1877 op.setTemplateParamsAttr(layout.rewrittenCallParams);
1878 });
1879 tracker_.updateModifiedFlag(true);
1880 return success();
1881 }

1883private:
1886 FailureOr<UnificationMap> unifyTypeSignature(
1887 CallOp op, FuncDefOp callTgt, ArrayRef<StringRef> calleeNamespace, PatternRewriter &rewriter
1888 ) const {
1889 FailureOr<UnificationMap> unifyResult =
1890 op.unifyTypeSignatureWithNamespace(callTgt.getFunctionType(), calleeNamespace);
1891 if (succeeded(unifyResult)) {
1892 if (!callSignatureUnifies(
1893 op.getTypeSignature(), callTgt.getFunctionType(), calleeNamespace, tracker_,
1894 /*allowUnresolvedStructs=*/true, op
1895 )) {
1896 return rewriter.notifyMatchFailure(op, [&](Diagnostic &diag) {
1897 diag.append("cannot instantiate function because the complete call signature has ")
1898 .append("conflicting template bindings");
1899 appendCallBindingConflictLocation(
1900 diag, op.getTypeSignature(), callTgt.getFunctionType(), calleeNamespace, tracker_,
1901 /*allowUnresolvedStructs=*/true, op
1902 );
1903 diag.attachNote(callTgt.getLoc()).append("target function declared here");
1904 });
1905 }
1906 return unifyResult;
1907 }
1908 return rewriter.notifyMatchFailure(op, [&](Diagnostic &diag) {
1909 diag.append("target function type does not unify with call type ")
1910 .append(op.getTypeSignature())
1911 .attachNote(callTgt.getLoc())
1912 .append("target function declared here");
1913 });
1914 }

1918 static LogicalResult collectConcreteTemplateParams(
1919 CallOp op, PatternRewriter &rewriter, SymbolTableCollection &symTables, FuncDefOp callTgt,
1920 TemplateOp parentTemplate, const UnificationMap &unifyResult,
1921 DenseMap<Attribute, Attribute> &paramNameToConcrete
1922 ) {
1923 auto realParams = parentTemplate.getConstOps<TemplateParamOp>();
1924 ArrayAttr callParams = op.getTemplateParamsAttr();
1925 LLVM_DEBUG(
1926 llvm::dbgs() << "[InstantiateFuncAtCallOp] TemplateParamsAttr: " << callParams << '\n'
1927 );
1928
1929 auto recordConcreteParam = [&](FlatSymbolRefAttr paramName, TemplateParamOp paramOp,
1930 Attribute concreteValue, Operation *lookupFrom) {
1931 if (failed(op.verifyTemplateParamValueCompatibility(concreteValue, paramOp))) {
1932 return failIncompatibleInferredParam(op, rewriter, paramName, paramOp);
1933 }
1934 auto destination = getRootModule(callTgt);
1935 if (failed(destination)) {
1936 return failure();
1937 }
1938 auto localValue = rebaseTemplateParams(
1939 symTables, ArrayAttr::get(op.getContext(), {concreteValue}), lookupFrom, *destination, op
1940 );
1941 if (failed(localValue)) {
1942 return failure();
1943 }
1944 paramNameToConcrete[paramName] = (*localValue)[0];
1945 return success();
1946 };
1947
1948 // If there's no template instantiation list, must infer all template parameters.
1949 if (isNullOrEmpty(callParams)) {
1950 for (auto paramOp : realParams) {
1951 auto paramName = FlatSymbolRefAttr::get(paramOp.getSymNameAttr());
1952 auto inferredValOpt = inferUnifiedParam(unifyResult, paramName);
1953 if (!inferredValOpt.has_value()) {
1954 LLVM_DEBUG(
1955 llvm::dbgs() << "[InstantiateFuncAtCallOp] unification for param '" << paramName
1956 << "': not found\n"
1957 );
1958 continue;
1959 }
1960 Attribute inferredVal = *inferredValOpt;
1961 LLVM_DEBUG(
1962 llvm::dbgs() << "[InstantiateFuncAtCallOp] inferredVal: " << inferredVal << '\n'
1963 );
1964 if (!isConcreteStructParamAttr(inferredVal)) {
1965 LLVM_DEBUG(
1966 llvm::dbgs() << "[InstantiateFuncAtCallOp] unification for param '" << paramName
1967 << "': not concrete, " << inferredVal << '\n'
1968 );
1969 continue;
1970 }
1971 if (failed(recordConcreteParam(paramName, paramOp, inferredVal, op))) {
1972 return failure();
1973 }
1974 }
1975 return success();
1976 }
1977
1978 // As stated earlier, need to run the verification checks again to ensure the
1979 // instantiation is valid, except for the size check because that cannot change.
1980 assert((callParams.size() == llvm::range_size(realParams)) && "per CallOpVerifier");
1981 if (failed(op.verifyTemplateParamValuesCompatibility(realParams))) {
1982 return rewriter.notifyMatchFailure(op, [](Diagnostic &diag) {
1983 diag.append("incompatible with specified param type(s)");
1984 });
1985 }
1986 if (failed(op.verifyTemplateParamsMatchInferred(realParams, unifyResult))) {
1987 return rewriter.notifyMatchFailure(op, [](Diagnostic &diag) {
1988 diag.append("incompatible with inferred param value(s)");
1989 });
1990 }
1991
1992 // When template parameters are specified on the CallOp, use them as the source of truth
1993 // for concrete arguments, then infer wildcard parameters against the full explicit map.
1994 SmallVector<std::pair<TemplateParamOp, FlatSymbolRefAttr>> wildcardParams;
1995 for (auto [paramOp, attr] : llvm::zip_equal(realParams, callParams.getValue())) {
1996 auto paramName = FlatSymbolRefAttr::get(paramOp.getSymNameAttr());

2618 "result type params: "
2619 << debug::toStringList(out.paramsOfStructTy) << '\n';
2620 });
2621 }
2622

2626 return failure(); // nothing changed

2631 // prevents instantiation of the struct with the requested type.

2636 ", but found ", oldRetTy
2637 );
2638 });

2641 CallOp newOp = replaceOpWithNewOp<CallOp>(
2642 rewriter, op, TypeRange {newRetTy}, op.getCallee(),
2643 AffineMapFolder::getConvertedMapOpGroups(out), out.dimsPerGroup, op.getArgOperands()
2644 );

2653 inline LogicalResult instantiateViaTargetType(
2654 const AffineMapFolder::Input &in, AffineMapFolder::Output &out,
2655 OperandRange::type_range callArgTypes, FuncDefOp targetFunc,
2656 ArrayRef<StringRef> targetNamespace
2657 ) const {
2658 assert(targetFunc.isStructCompute()); // since `op.calleeIsStructCompute()`
2659 ArrayAttr targetResTyParams = targetFunc.getSingleResultTypeOfCompute().getParams();
2660 assert(!isNullOrEmpty(targetResTyParams)); // same cardinality as `in.paramsOfStructTy`
2661 assert(in.paramsOfStructTy.size() == targetResTyParams.size()); // verifier ensures this
2662
2663 if (llvm::all_of(in.paramsOfStructTy, [](Attribute attr) {
2664 return isConcreteStructParamAttr(attr);
2665 })) {
2666 // Nothing can change if everything is already concrete
2667 return failure();
2668 }
2669
2670 LLVM_DEBUG({
2671 llvm::dbgs() << '[' << __FUNCTION__ << ']'
2672 << " call arg types: " << debug::toStringList(callArgTypes) << '\n';
2673 llvm::dbgs() << '[' << __FUNCTION__ << ']' << " target func arg types: "
2674 << debug::toStringList(targetFunc.getArgumentTypes()) << '\n';
2675 llvm::dbgs() << '[' << __FUNCTION__ << ']'
2676 << " struct params @ call: " << debug::toStringList(in.paramsOfStructTy) << '\n';
2677 llvm::dbgs() << '[' << __FUNCTION__ << ']'
2678 << " target struct params: " << debug::toStringList(targetResTyParams) << '\n';
2679 });
2680
2681 UnificationMap unifications;
2682 bool unifies =
2683 typeListsUnify(callArgTypes, targetFunc.getArgumentTypes(), targetNamespace, &unifications);
2684 if (!unifies) {
2685 return failure();
2686 }
2687
2688 LLVM_DEBUG({
2689 llvm::dbgs() << '[' << __FUNCTION__ << ']'
2690 << " unifications of arg types: " << debug::toStringList(unifications) << '\n';
2691 });
2692
2693 // Check for RHS SymRef (i.e., from the target function) that have LHS concrete Attributes (i.e.
2694 // from the call argument types) without any struct parameters (because the type with concrete
2695 // struct parameters will be used to instantiate the target struct rather than the fully
2696 // flattened struct type resulting in type mismatch of the callee to target) and perform those
2697 // replacements in the `targetFunc` return type to produce the new result type for the CallOp.
2698 SmallVector<Attribute> newReturnStructParams = llvm::map_to_vector(
2699 llvm::zip_equal(targetResTyParams.getValue(), in.paramsOfStructTy),
2700 [&unifications](std::tuple<Attribute, Attribute> p) {
2701 Attribute fromCall = std::get<1>(p);
2702 // Preserve attributes that are already concrete at the call site. Otherwise attempt to lookup
2703 // non-parameterized concrete unification for the target struct parameter symbol.
2704 if (!isConcreteStructParamAttr(fromCall)) {
2705 Attribute fromTgt = std::get<0>(p);
2706 LLVM_DEBUG({
2707 llvm::dbgs() << "[instantiateViaTargetType] fromCall = " << fromCall << '\n';
2708 llvm::dbgs() << "[instantiateViaTargetType] fromTgt = " << fromTgt << '\n';
2709 });
2710 assert(llvm::isa<SymbolRefAttr>(fromTgt));
2711 auto it = unifications.find(std::make_pair(llvm::cast<SymbolRefAttr>(fromTgt), Side::RHS));
2712 if (it != unifications.end()) {
2713 Attribute unifiedAttr = it->second;
2714 LLVM_DEBUG({
2715 llvm::dbgs() << "[instantiateViaTargetType] unifiedAttr = " << unifiedAttr << '\n';
2716 });
2717 if (unifiedAttr && isConcreteStructParamAttr(unifiedAttr, /*allowStructParams=*/false)) {
2718 return unifiedAttr;
2719 }
2720 }
2721 }
2722 return fromCall;
2723 }
2724 );
2725
2726 out.paramsOfStructTy = std::move(newReturnStructParams);
2727 assert(out.paramsOfStructTy.size() == in.paramsOfStructTy.size() && "post-condition");
2728 assert(out.mapOpGroups.empty() && "post-condition");
2729 assert(out.dimsPerGroup.empty() && "post-condition");
2730 return success();
2731 }
2732};

2754static FailureOr<ArrayType> inferCreateArrayType(CreateArrayOp op, ConversionTracker &tracker) {
2755 Value createResult = op.getResult();
2756 ArrayType createResultType = dyn_cast<ArrayType>(createResult.getType());
2757 assert(createResultType && "CreateArrayOp must produce ArrayType");
2758 Type oldResultElemType = createResultType.getElementType();
2759
2760 Type newResultElemType = nullptr;
2761 // An initializer constrains array.new just as strongly as a later array.write: every element
2762 // must have the result array's element type. A member-read refinement can change an
2763 // initializer without creating a WriteArrayOp, so consider the explicit elements first.
2764 if (!op.getElements().empty()) {
2765 Type initializerType = op.getElements().front().getType();
2766 if (!llvm::all_of(op.getElements(), [initializerType](Value element) {
2767 return element.getType() == initializerType;
2768 })) {
2769 return failure();
2770 }
2771 if (initializerType != oldResultElemType) {
2772 newResultElemType = initializerType;
2773 }
2774 }
2775
2776 // Both stores and independently typed reads can constrain an allocation. All such evidence
2777 // belongs to this plan; accesses do not retag arrays owned by other producers.
2778 for (Operation *user : createResult.getUsers()) {
2779 Type evidence;
2780 if (auto write = dyn_cast<WriteArrayOp>(user); write && write.getArrRef() == createResult) {
2781 evidence = write.getRvalue().getType();
2782 } else if (auto read = dyn_cast<ReadArrayOp>(user); read && read.getArrRef() == createResult) {
2783 evidence = read.getResult().getType();
2784 }
2785 if (!evidence || evidence == oldResultElemType ||
2786 !tracker.isLegalConversion(oldResultElemType, evidence, "inferCreateArrayType", op)) {
2787 continue;
2788 }
2789 if (newResultElemType && newResultElemType != evidence) {
2790 return failure();
2791 }
2792 newResultElemType = evidence;
2793 }
2794 if (!newResultElemType) {
2795 // no replacement type found
2796 return failure();
2797 }
2798 if (!tracker.isLegalConversion(
2799 oldResultElemType, newResultElemType, "inferCreateArrayType", op
2800 )) {
2801 return failure();
2802 }
2803 ArrayType newType = createResultType.cloneWith(newResultElemType);
2804 return newType;
2805}

2825static FailureOr<Type>
2826inferMemberTypeFromWrites(MemberDefOp op, ConversionTracker &tracker, PatternRewriter &rewriter) {
2827 // Avoid a whole-struct symbol-use search when writes cannot refine the declaration.
2828 if (isFullySpecializedMemberType(op.getType())) {
2829 return failure();
2830 }
2831
2832 // Find all uses of the member symbol name within its parent struct.
2833 StructDefOp parentRes = getParentOfType<StructDefOp>(op);
2834 assert(parentRes && "MemberDefOp parent is always StructDefOp"); // per ODS def
2835
2836 // If the symbol is used by a MemberWriteOp with a different result type then change
2837 // the type of the MemberDefOp to match the MemberWriteOp result type.
2838 Type newType = nullptr;
2839 if (auto memberUsers = llzk::getSymbolUses(op, parentRes)) {
2840 std::optional<Location> newTypeLoc = std::nullopt;
2841 for (SymbolTable::SymbolUse symUse : memberUsers.value()) {
2842 if (MemberWriteOp writeOp = llvm::dyn_cast<MemberWriteOp>(symUse.getUser())) {
2843 Type writeToType = writeOp.getVal().getType();
2844 LLVM_DEBUG(llvm::dbgs() << "[inferMemberTypeFromWrites] checking " << writeOp << '\n');
2845 if (!newType) {
2846 // If a new type has not yet been discovered, store the new type.
2847 newType = writeToType;
2848 newTypeLoc = writeOp.getLoc();
2849 } else if (writeToType != newType) {
2850 // Typically, there will only be one write for each member of a struct but do not rely
2851 // on that assumption. If multiple writes with a different types A and B are found where
2852 // A->B is a legal conversion (i.e., more concrete unification), then it is safe to use
2853 // type B with the assumption that the write with type A will be updated by another
2854 // pattern to also use type B.
2855 auto commonType = getCommonRefinedType(newType, writeToType, tracker, op);
2856 if (succeeded(commonType)) {
2857 newType = *commonType;
2858 newTypeLoc = writeOp.getLoc();
2859 } else if (!tracker.isLegalConversion(
2860 writeToType, newType, "inferMemberTypeFromWrites", op
2861 )) {
2862 if (tracker.isLegalConversion(newType, writeToType, "inferMemberTypeFromWrites", op)) {
2863 // 'writeToType' is the more concrete type
2864 newType = writeToType;
2865 newTypeLoc = writeOp.getLoc();
2866 } else {
2867 // Give an error if the types are incompatible.
2868 return rewriter.notifyMatchFailure(op, [&](Diagnostic &diag) {
2869 diag.append(
2870 "Cannot update type of '", MemberDefOp::getOperationName(),
2871 "' because there are multiple '", MemberWriteOp::getOperationName(),
2872 "' with different value types"
2873 );
2874 if (newTypeLoc) {
2875 diag.attachNote(newTypeLoc).append("type written here is ", newType);
2876 }
2877 diag.attachNote(writeOp.getLoc()).append("type written here is ", writeToType);
2878 });
2879 }
2880 }
2881 }
2882 }
2883 }
2884 }
2885 if (!newType || newType == op.getType()) {
2886 return failure(); // nothing changed
2887 }
2888 if (!tracker.isLegalConversion(op.getType(), newType, "inferMemberTypeFromWrites", op)) {
2889 auto commonType = getCommonRefinedType(op.getType(), newType, tracker, op);
2890 if (failed(commonType) || *commonType != newType) {
2891 return failure();
2892 }
2893 }
2894
2895 // Infer shared template bindings from type and value parameter positions, including those
2896 // inside arrays, POD records and struct parameters. Comparing the complete member types
2897 // would incorrectly reject members with different surrounding shapes.
2898 SmallVector<std::pair<FlatSymbolRefAttr, Attribute>> candidateBindings;
2899 if (!collectMemberBindings(
2900 tracker.scopeType(op.getType(), op), tracker.scopeType(newType, op), tracker,
2901 candidateBindings
2902 )) {
2903 return failure();
2904 }
2905 // Later iterations can instantiate callers already present in this module. Keep repeated
2906 // bindings correlated until those callers have been checked.
2907 if (refinesRepeatedMemberBinding(candidateBindings)) {
2908 return failure();
2909 }
2910 DenseMap<FlatSymbolRefAttr, Attribute> sharedBindings;
2911 for (auto [variable, binding] : candidateBindings) {
2912 // Unchanged variables are safe here because mixed repeated bindings were deferred above.
2913 if (variable == binding) {
2914 continue;
2915 }
2916 auto [it, inserted] = sharedBindings.try_emplace(variable, binding);
2917 if (inserted) {
2918 continue;
2919 }
2920 auto commonType = getCommonRefinedParamAttr(it->second, binding, tracker);
2921 if (failed(commonType)) {
2922 return rewriter.notifyMatchFailure(op, [&](Diagnostic &diag) {
2923 diag.append("cannot refine ")
2924 .append(memberBindingDiagnosticName(variable, binding))
2925 .append(" because member '")
2926 .append(op.getName())
2927 .append("' is written with incompatible bindings ")
2928 .append(memberBindingDiagnosticValue(it->second))
2929 .append(" and ")
2930 .append(memberBindingDiagnosticValue(binding));
2931 });
2932 }
2933 it->second = *commonType;
2934 }
2935 if (!sharedBindings.empty()) {
2936 for (MemberDefOp sibling : parentRes.getOps<MemberDefOp>()) {
2937 if (sibling == op) {
2938 continue;
2939 }
2940 if (!referencesMemberBinding(
2941 tracker.scopeType(sibling.getType(), sibling), &sharedBindings, tracker
2942 )) {
2943 continue;
2944 }
2945 if (auto siblingUses = llzk::getSymbolUses(sibling, parentRes)) {
2946 for (SymbolTable::SymbolUse siblingUse : siblingUses.value()) {
2947 auto siblingWrite = llvm::dyn_cast<MemberWriteOp>(siblingUse.getUser());
2948 if (!siblingWrite) {
2949 continue;
2950 }
2951 SmallVector<std::pair<FlatSymbolRefAttr, Attribute>> siblingBindings;
2952 if (!collectMemberBindings(
2953 tracker.scopeType(sibling.getType(), sibling),
2954 tracker.scopeType(siblingWrite.getVal().getType(), siblingWrite), tracker,
2955 siblingBindings, &sharedBindings
2956 )) {
2957 return failure();
2958 }
2959 for (auto [variable, binding] : siblingBindings) {
2960 auto it = sharedBindings.find(variable);
2961 if (it == sharedBindings.end()) {
2962 continue;
2963 }
2964 auto commonType = getCommonRefinedParamAttr(it->second, binding, tracker);
2965 if (failed(commonType)) {
2966 return rewriter.notifyMatchFailure(op, [&](Diagnostic &diag) {
2967 diag.append("cannot refine ")
2968 .append(memberBindingDiagnosticName(variable, binding))
2969 .append(" to ")
2970 .append(memberBindingDiagnosticValue(it->second))
2971 .append(" because sibling member '")
2972 .append(sibling.getName())
2973 .append("' is written with incompatible type ")
2974 .append(memberBindingDiagnosticValue(binding));
2975 diag.attachNote(siblingWrite.getLoc()).append("incompatible write is here");
2976 });
2977 }
2978 it->second = *commonType;
2979 }
2980 }
2981 }
2982 }
2983 }
2984
2985 return newType;
2986}

2989static bool canRefineMemberDefinition(
2990 MemberDefOp op, Type newType, ConversionTracker &tracker, SymbolTableCollection &tables,
2991 const DenseMap<Value, Type> &pendingReads
2992) {
2993 StructDefOp parentRes = getParentOfType<StructDefOp>(op);
2994 // Do not commit a write-derived refinement that an existing read or one of its typed consumers
2995 // cannot adopt. In particular, two complementary writes can have a common refinement even
2996 // when a generic read is independently consumed as an incompatible concrete type.
2997 if (auto memberUsers = llzk::getSymbolUses(op, parentRes)) {
2998 for (SymbolTable::SymbolUse symUse : memberUsers.value()) {
2999 if (MemberWriteOp writeOp = llvm::dyn_cast<MemberWriteOp>(symUse.getUser())) {
3000 // A read of another member carries that source's independent type evidence. Wait
3001 // for its own propagation to establish the candidate type before narrowing this
3002 // destination; otherwise a generic read can hide an incompatible source write.
3003 if (writeOp.getVal().getDefiningOp<MemberReadOp>() &&
3004 writeOp.getVal().getType() != newType) {
3005 return false;
3006 }
3007 }
3008 if (MemberReadOp readOp = llvm::dyn_cast<MemberReadOp>(symUse.getUser())) {
3009 bool canUse = canUseScalarizedValueAsType(
3010 readOp.getVal().getType(), newType, tracker, "canRefineMemberDefinition", readOp
3011 );
3012 if (!canUse) {
3013 return false;
3014 }
3015 bool canRefine =
3016 canRefineMemberReadUsersToType(readOp, newType, tables, tracker, &pendingReads);
3017 if (!canRefine) {
3018 return false;
3019 }
3020 }
3021 }
3022 }
3023 return true;
3024}

3144static LogicalResult
3145propagateMemberTypes(ModuleOp module, ConversionTracker &tracker, bool &changed) {
3146 changed = false;
3147 MatchFailureListener listener;
3148 PatternRewriter rewriter(module.getContext());
3149 rewriter.setListener(&listener);
3150 SymbolTableCollection tables;
3151 DenseMap<MemberDefOp, Type> pendingMembers;
3152 module.walk([&](MemberDefOp member) {
3153 if (auto type = inferMemberTypeFromWrites(member, tracker, rewriter); succeeded(type)) {
3154 pendingMembers[member] = *type;
3155 }
3156 });
3157 if (listener.hadFailure) {
3158 return failure();
3159 }
3160
3161 DenseMap<Value, Type> pendingReads;
3162 DenseMap<Value, MemberDefOp> sourceByRead;
3163 SmallVector<MemberReadOp> reads = walkCollect<MemberReadOp>(module);
3164 for (MemberReadOp read : reads) {
3165 auto source = read.getMemberDefOp(tables);
3166 if (failed(source)) {
3167 return failure();
3168 }
3169 MemberDefOp member = source->get();
3170 Type type = pendingMembers.lookup(member);
3171 if (!type) {
3172 type = member.getType();
3173 }
3174 if (type != read.getVal().getType() &&
3175 canUseScalarizedValueAsType(
3176 read.getVal().getType(), type, tracker, "propagateMemberTypes", read
3177 )) {
3178 pendingReads[read.getVal()] = type;
3179 sourceByRead[read.getVal()] = member;
3180 }
3181 }
3182
3183 // Validate every jointly proposed call before rejecting individual candidates; otherwise
3184 // visitation order could retain one conflicting member at a generic type and conceal the issue.
3185 DenseMap<Value, Type> pendingValues = pendingReads;
3186 extendPendingRefinements(module, pendingValues, tracker);
3187 if (failed(verifyPendingCallBindings(module, pendingValues, tables, tracker))) {
3188 return failure();
3189 }
3190 if (failed(verifyPendingMemberReceivers(module, pendingValues, tables, tracker))) {
3191 return failure();
3192 }
3193 bool removed;
3194 do {
3195 removed = false;
3196 pendingValues = pendingReads;
3197 extendPendingRefinements(module, pendingValues, tracker);
3198 SmallVector<MemberDefOp> rejectedMembers;
3199 for (const auto &[member, type] : pendingMembers) {
3200 if (!canRefineMemberDefinition(member, type, tracker, tables, pendingValues)) {
3201 rejectedMembers.push_back(member);
3202 }
3203 }
3204 for (MemberDefOp member : rejectedMembers) {
3205 pendingMembers.erase(member);
3206 for (MemberReadOp read : reads) {
3207 if (sourceByRead.lookup(read.getVal()) == member) {
3208 pendingReads.erase(read.getVal());
3209 }
3210 }
3211 removed = true;
3212 }
3213 SmallVector<Value> rejectedReads;
3214 for (MemberReadOp read : reads) {
3215 if (Type type = pendingReads.lookup(read.getVal());
3216 type && !canRefineMemberReadUsersToType(read, type, tables, tracker, &pendingValues)) {
3217 rejectedReads.push_back(read.getVal());
3218 }
3219 }
3220 for (Value read : rejectedReads) {
3221 pendingReads.erase(read);
3222 removed = true;
3223 }
3224 } while (removed);
3225
3226 for (auto [member, type] : pendingMembers) {
3227 rewriter.modifyOpInPlace(member, [&]() { member.setType(type); });
3228 changed = true;
3229 }
3230 for (MemberReadOp read : reads) {
3231 if (Type type = pendingReads.lookup(read.getVal())) {
