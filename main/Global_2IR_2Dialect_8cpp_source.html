32public:
33 using Base::Base;

35 LogicalResult upgradeFromVersion(
36 Operation *root, const LLZKDialectVersion & /*current*/, const LLZKDialectVersion &requested
37 ) const final {
38 auto res = root->walk([](global::GlobalDefOp global) -> WalkResult {
39 if (Attribute initialValue = global.getInitialValueAttr()) {
40 auto errFn = [context = initialValue.getContext()] {
41 return InFlightDiagnosticWrapper::createSilent(context);
42 };
43 FailureOr<global::NormalizedGlobalInitializer> normalized =
44 global::normalizeGlobalInitializer(global.getType(), initialValue, errFn);
45 if (failed(normalized)) {
46 return global.emitError(
47 "contains a legacy initializer that is incompatible with its declared type"
48 );
49 }
50 global.setInitialValueAttr(normalized->value);
51 global.setTypeAttr(TypeAttr::get(normalized->type));
52 }
53 return WalkResult::advance();
54 });
55 if (res.wasInterrupted()) {
56 return failure();
57 }
58
59 // Before version 3, global reads did not carry a `constant` property.
60 // Reads of const definitions are equivalent to the new `global.read const`
61 // form, so upgrade them to preserve their effect-free semantics.
62 if (requested.majorVersion < 3) {
63 SymbolTableCollection tables;
64 res = root->walk([&tables](global::GlobalReadOp read) -> WalkResult {
65 auto globalRef = llvm::cast<global::GlobalRefOpInterface>(read.getOperation());
66 FailureOr<SymbolLookupResult<global::GlobalDefOp>> target =
67 globalRef.getGlobalDefOp(tables);
68 if (succeeded(target) && target->get().isConstant()) {
69 read.setConstant(true);
70 }
71 return WalkResult::advance();
72 });
73 }
74 return failure(res.wasInterrupted());
75 }
76};
