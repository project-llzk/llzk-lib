189 /*
190 Rule of three, needed for the mlir::SymbolTableCollection, which has no copy constructor.
191 Since the mlir::SymbolTableCollection is a caching mechanism, we simply allow default, empty
192 construction for copies.
193 */
194
195 ConstraintDependencyGraph(const ConstraintDependencyGraph &other)
196 : mod(other.mod), structDef(other.structDef), ctx(other.ctx), signalSets(other.signalSets),
197 constantSets(other.constantSets), ref2Val(other.ref2Val), tables() {}
198
199 ConstraintDependencyGraph &operator=(const ConstraintDependencyGraph &other) {
200 mod = other.mod;
201 structDef = other.structDef;
202 ctx = other.ctx;
203 signalSets = other.signalSets;
204 constantSets = other.constantSets;
205 ref2Val = other.ref2Val;
206 return *this;
207 }
208 virtual ~ConstraintDependencyGraph() = default;
209
210private:
211 mlir::ModuleOp mod;
212 // Using mutable because many operations are not const by default, even for "const"-like
213 // operations, like "getName()", and this reduces const_casts.
214 mutable component::StructDefOp structDef;
215 CDGAnalysisContext ctx;
216
217 // Transitive closure only over signals.
218 llvm::EquivalenceClasses<SourceRef> signalSets;
219 // A simple set mapping of constants, as we do not want to compute a transitive closure over
220 // constants.
221 std::unordered_map<SourceRef, SourceRefSet, SourceRef::Hash> constantSets;
222
223 // Maps references to the values where they are found.
224 SourceRefLattice::Ref2Val ref2Val;
225
226 // Also mutable for caching within otherwise const lookup operations.
227 mutable mlir::SymbolTableCollection tables;
228

280public:

286
287 ~ConstraintDependencyGraphModuleAnalysis() override = default;
288

291 }
292
293protected:
295 mlir::LogicalResult validateInput(mlir::ModuleOp module) const override;
296
297 void initializeSolver() override { (void)solver.load<SourceRefAnalysis>(); }
298
299 const CDGAnalysisContext &getContext() const override { return ctx; }
300
301private:
302 // This "intraprocedural" option is related to the CDG construction, not the dataflow analysis
303 // itself.
304 CDGAnalysisContext ctx = {.runIntraprocedural = false};
