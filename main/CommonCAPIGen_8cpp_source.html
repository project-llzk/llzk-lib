170struct ClangLexerContext::Impl {
172 LangOptions langOpts;
174 IntrusiveRefCntPtr<FileManager> fileMgr;
176 IntrusiveRefCntPtr<DiagnosticIDs> diagIDs;
178 std::unique_ptr<DiagnosticOptions> diagOpts;
180 std::unique_ptr<DiagnosticsEngine> diags;
182 std::unique_ptr<SourceManager> sourceMgr;
184 std::unique_ptr<Lexer> lexer;
185
186 Impl() : diagIDs(new DiagnosticIDs()), diagOpts(std::make_unique<DiagnosticOptions>()) {
187 // Enable C++ language features for lexing
188 langOpts.CPlusPlus = true;
189 langOpts.CPlusPlus11 = true;
190
191 FileSystemOptions fileSystemOpts;
192 fileMgr = new FileManager(fileSystemOpts);
193 diags = std::make_unique<DiagnosticsEngine>(diagIDs, *diagOpts);
194 sourceMgr = std::make_unique<SourceManager>(*diags, *fileMgr);
195 }
196};

198ClangLexerContext::ClangLexerContext(StringRef source, StringRef bufferName)
199 : impl(std::make_unique<Impl>()) {
200 if (source.empty()) {
201 llvm::errs() << "Warning: ClangLexerContext created with empty source\n";
202 return;
203 }
204
205 // Create a memory buffer for the input
206 std::unique_ptr<llvm::MemoryBuffer> buffer = llvm::MemoryBuffer::getMemBuffer(source, bufferName);
207 if (!buffer) {
208 llvm::errs() << "Error: Failed to create memory buffer for ClangLexerContext\n";
209 return;
210 }
211
212 FileID fileID = impl->sourceMgr->createFileID(std::move(buffer), SrcMgr::C_User);
213 llvm::MemoryBufferRef bufferRef = impl->sourceMgr->getBufferOrFake(fileID);
214
215 if (bufferRef.getBufferSize() == 0 && !source.empty()) {
216 llvm::errs() << "Error: Failed to get buffer from source manager in ClangLexerContext\n";
217 return;
218 }
219
220 // Create the lexer
221 impl->lexer = std::make_unique<Lexer>(fileID, bufferRef, *impl->sourceMgr, impl->langOpts);
222 // Enable comment parsing for extraClassDeclaration method extraction
223 impl->lexer->SetCommentRetentionState(true);
224 lexer = impl->lexer.get();
225}

242static inline bool isAccessModifier(StringRef tokenText) {
243 return tokenText == "private" || tokenText == "public" || tokenText == "protected";
244}
245
249static inline std::vector<Token> tokenize(const ClangLexerContext &lexerCtx) {
250 Lexer &lexer = lexerCtx.getLexer();
251 std::vector<Token> tokens;
252 tokens.reserve(128); // Reasonable default
253 for (Token tok; !lexer.LexFromRawLexer(tok);) {
254 if (tok.is(tok::eof)) {
255 break;
256 }
257 tokens.push_back(tok);
258 }
259 return tokens;
260}

263static inline std::string getDocumentation(
264 size_t returnTypeStart, const std::vector<Token> &tokens, const SourceManager &sourceMgr
265) {
266 std::string documentation;
267 for (size_t j = returnTypeStart; j > 0; --j) {
268 Token curr = tokens[j - 1];
269 if (curr.is(tok::comment)) {
270 StringRef comment(sourceMgr.getCharacterData(curr.getLocation()), curr.getLength());
271 comment.consume_front("///");
272 comment.consume_front("//");
273 if (comment.consume_front("/*")) {
274 comment.consume_back("*/");
275 }
276 comment = comment.trim();
277
278 // Trim whitespace
279 if (!comment.empty()) {
280 std::string newDoc("/// ");
281 newDoc += comment;
282 if (!documentation.empty()) {
283 newDoc += '\n';
284 newDoc += documentation;
285 }
286 documentation = std::move(newDoc);
287 }
288 } else if (!curr.is(tok::unknown)) {
289 // Stop looking backwards when we hit a non-comment, non-whitespace token
290 // that could be part of another declaration
291 if (curr.isOneOf(tok::semi, tok::r_brace, tok::l_brace)) {
292 break;
293 }
294 if (curr.is(tok::raw_identifier) && isAccessModifier(curr.getRawIdentifier())) {
295 break;
296 }
297 }
298 }
299 return documentation;
300}

309enum class AccessLevel : std::uint8_t { Public, Private, Protected };
310
318static bool
319updateAccessLevel(size_t &i, const std::vector<Token> &tokens, AccessLevel &currentAccess) {
320 if (i + 1 < tokens.size() && tokens[i + 1].is(tok::colon)) {
321 if (tokens[i].is(tok::raw_identifier)) {
322 StringRef name = tokens[i].getRawIdentifier();
323 if (name == "private") {
324 currentAccess = AccessLevel::Private;
325 i++; // extra skip for the colon
326 return true;
327 } else if (name == "public") {
328 currentAccess = AccessLevel::Public;
329 i++; // extra skip for the colon
330 return true;
331 } else if (name == "protected") {
332 currentAccess = AccessLevel::Protected;
333 i++; // extra skip for the colon
334 return true;
335 }
336 }
337 }
338 return false;
339}

351static std::string extractReturnType(
352 size_t i, const std::vector<Token> &tokens, const SourceManager &sourceMgr,
353 size_t &returnTypeStart
354) {
355 std::string returnType;
356 returnTypeStart = 0;
357 bool isStaticMethod = false;
358
359 // Look backwards for return type start, stopping at declaration boundaries
360 for (size_t j = i; j > 0; --j) {
361 Token curr = tokens[j - 1];
362 // Semicolon or right brace indicates lookback has reached the end of a prior declaration.
363 if (curr.isOneOf(tok::semi, tok::r_brace)) {
364 returnTypeStart = j;
365 break;
366 }
367 // Check for "static" or access modifiers in the return type lookback (both appear as
368 // `raw_identifier` in raw token stream).
369 if (curr.is(tok::raw_identifier)) {
370 StringRef text = curr.getRawIdentifier();
371 if (text == "static") {
372 isStaticMethod = true;
373 break;
374 }
375 if (tokens[j].is(tok::colon) && isAccessModifier(text)) {
376 // In this case, `returnTypeStart` must be after the colon.
377 returnTypeStart = j + 1;
378 assert(returnTypeStart < tokens.size());
379 break;
380 }
381 }
382 }
383
384 // Skip static methods (for now)
385 if (isStaticMethod) {
386 return "";
387 }
388
389 // Adjust `returnTypeStart` for potential comment tokens. Skip as many
390 // sequential comments as needed.
391 while (tokens[returnTypeStart].is(tok::comment)) {
392 returnTypeStart++;
393 assert(returnTypeStart <= i);
394 }
395
396 // Build return type from tokens, skipping modifiers and comments.
397 returnType.reserve(32); // Reasonable default for most type names
398 llvm::raw_string_ostream returnTypeStream(returnType);
399
400 for (size_t j = returnTypeStart; j < i; ++j) {
401 // Skip comments - they should be extracted as documentation, not part of the return type
402 if (tokens[j].is(tok::comment)) {
403 continue;
404 }
405
406 StringRef tokenText(sourceMgr.getCharacterData(tokens[j].getLocation()), tokens[j].getLength());
407
408 // Skip access specifiers (e.g., "private", "public", "protected").
409 if (tokens[j].is(tok::raw_identifier) && isAccessModifier(tokenText)) {
410 // If followed by a colon, skip that too
411 if (j + 1 < i && tokens[j + 1].is(tok::colon)) {
412 j++; // Skip the colon too
413 }
414 continue;
415 }
416
417 // Skip common implementation keywords that indicate we're in code, not a declaration.
418 if (tokens[j].is(tok::raw_identifier) && tokenText == "return") {
419 // This indicates we've hit implementation code, stop parsing
420 returnType.clear();
421 break;
422 }
423
424 // Skip modifiers and language keywords that shouldn't be in the return type
425 if (tokens[j].is(tok::raw_identifier) && isCppModifierKeyword(tokenText)) {
426 continue;
427 }
428
429 // Skip standalone colons (from lookback to access specifiers)
430 if (tokens[j].is(tok::colon)) {
431 // Only skip if it's not part of ::
432 if (j == 0 || j + 1 >= i || !tokens[j - 1].is(tok::colon)) {
433 continue;
434 }
435 }
436
437 // Add spacing between tokens (but not around ::)
438 if (!returnType.empty() && !returnType.ends_with("::") && tokenText != "::" &&
439 !tokenText.starts_with("::")) {
440 returnTypeStream << ' ';
441 }
442 returnTypeStream << tokenText;
443 }
444
445 // Trim possible whitespace
446 return StringRef(returnType).trim().str();
447}

459static bool parseMethodParameters(
460 size_t i, const std::vector<Token> &tokens, const SourceManager &sourceMgr,
461 size_t &closeParenIdx, bool &hasParameters, std::vector<MethodParameter> &parameters
462) {
463 const size_t tokenCount = tokens.size();
464 closeParenIdx = tokenCount;
465 hasParameters = false;
466 parameters.clear();
467
468 // Initialize parenDepth to 1 to account for the opening '(' at tokens[i+1]
469 // Start scanning from i+2 (the first token after the opening paren)
470 size_t parenDepth = 1;
471 for (size_t j = i + 2; j < tokenCount; ++j) {
472 if (tokens[j].is(tok::l_paren)) {
473 parenDepth++;
474 } else if (tokens[j].is(tok::r_paren)) {
475 parenDepth--;
476 if (parenDepth == 0) {
477 closeParenIdx = j;
478
479 // Parse parameters between '(' and ')'
480 // Parameters follow the pattern: type name [, type name ...]
481 std::vector<Token> paramTokens;
482 for (size_t k = i + 2; k < j; ++k) {
483 if (k >= tokenCount) {
484 break;
485 }
486 if (!tokens[k].is(tok::comment)) {
487 paramTokens.push_back(tokens[k]);
488 }
489 }
490 const size_t paramTokenCount = paramTokens.size();
491
492 // Check if we have actual parameters (excluding just "void")
493 if (paramTokenCount == 1) {
494 StringRef paramToken(
495 sourceMgr.getCharacterData(paramTokens[0].getLocation()), paramTokens[0].getLength()
496 );
497 if (paramToken != "void") {
498 hasParameters = true;
499 }
500 } else if (paramTokenCount > 1) {
501 hasParameters = true;
502 }
503
504 // Parse individual parameters
505 if (hasParameters) {
506 std::string currentParamType;
507 std::string currentParamName;
508 bool inDefaultValue = false;
509
510 for (size_t k = 0; k < paramTokenCount; ++k) {
511 // Check for end of current parameter
512 if (paramTokens[k].is(tok::comma)) {
513 // Add the current parameter if valid
514 if (!currentParamType.empty() && !currentParamName.empty()) {
515 parameters.push_back(MethodParameter(currentParamType, currentParamName));
516 }
517 currentParamType.clear();
518 currentParamName.clear();
519 inDefaultValue = false;
520 continue;
521 }
522 // Skip tokens that are part of the default value
523 if (inDefaultValue) {
524 continue;
525 }
526 // Check for '=' which indicates start of default value
527 if (paramTokens[k].is(tok::equal)) {
528 inDefaultValue = true;
529 continue;
530 }
531
532 StringRef tokenText(
533 sourceMgr.getCharacterData(paramTokens[k].getLocation()), paramTokens[k].getLength()
534 );
535
536 // Identifier token could be part of the type or the parameter name.
537 // Simple heuristic: last identifier before comma, equal, or end is the name
538 if (paramTokens[k].is(tok::raw_identifier)) {
539 if (k + 1 == paramTokenCount ||
540 (k + 1 < paramTokenCount && paramTokens[k + 1].isOneOf(tok::comma, tok::equal))) {
541 currentParamName = tokenText.str();
542 continue;
543 }
544 }
545
546 // Other identifiers and other tokens (keywords, ::, *, &, etc.) are part of type.
547 llvm::raw_string_ostream paramTypeStream(currentParamType);
548 if (!currentParamType.empty() && tokenText != "*" && tokenText != "&" &&
549 tokenText != "::" && !tokenText.starts_with("::") &&
550 !StringRef(currentParamType).ends_with("::")) {
551 paramTypeStream << ' ';
552 }
553 paramTypeStream << tokenText;
554 currentParamType = paramTypeStream.str();
555 }
556
557 // Add the last parameter if valid
558 if (!currentParamType.empty() && !currentParamName.empty()) {
559 parameters.push_back(MethodParameter(currentParamType, currentParamName));
560 }
561 }
562
563 return true;
564 }
565 }
566 }
567
568 // Couldn't find closing paren
569 return false;
570}

579static bool
580checkConstAndFindEnd(size_t closeParenIdx, const std::vector<Token> &tokens, size_t &endIdx) {
581 bool isConst = false;
582 endIdx = closeParenIdx + 1;
583
584 while (endIdx < tokens.size()) {
585 Token curr = tokens[endIdx];
586 if (curr.isOneOf(tok::semi, tok::l_brace)) {
587 break;
588 }
589 if (curr.is(tok::raw_identifier) && curr.getRawIdentifier() == "const") {
590 isConst = true;
591 }
592 endIdx++;
593 }
594
595 return isConst;
596}

616SmallVector<ExtraMethod> parseExtraMethods(StringRef extraDecl) {
617 if (extraDecl.empty()) {
618 return {};
619 }
620
621 // Use ClangLexerContext for simplified setup
622 const ClangLexerContext lexerCtx(extraDecl, "extraClassDecl");
623 if (!lexerCtx.isValid()) {
624 llvm::errs() << "Error: Failed to create lexer context for parseExtraMethods\n";
625 return {};
626 }
627
628 // Store methods uniqued by name to detect and skip overloads (duplicate method names).
629 llvm::StringMap<std::optional<ExtraMethod>> methods;
630
631 // Parse tokens to find method declarations
632 const std::vector<Token> tokens = tokenize(lexerCtx);
633 const size_t tokenCount = tokens.size();
634 const SourceManager &sourceMgr = lexerCtx.getSourceManager();
635
636 // Track current access level to avoid generating C API wrappers for private functions. Code
637 // generated by `mlir-tblgen` puts the extra declarations in the public section by default.
638 AccessLevel currentAccess = AccessLevel::Public;
639
640 for (size_t i = 0; i < tokenCount; ++i) {
641 // Skip comments (they'll be extracted separately)
642 if (tokens[i].is(tok::comment)) {
643 continue;
644 }
645
646 // Check for access specifier changes (e.g., "private:", "public:", "protected:").
647 if (updateAccessLevel(i, tokens, currentAccess)) {
648 continue;
649 }
650
651 // Skip private and protected methods - no need to generate C API wrappers
652 if (currentAccess != AccessLevel::Public) {
653 continue;
654 }
655
656 // Look for pattern: [modifiers] <return_type> <identifier> '(' [params] ')' [const] ';'
657 // Look for an identifier followed by '('
658 if (i + 1 < tokenCount && tokens[i + 1].is(tok::l_paren) && tokens[i].is(tok::raw_identifier)) {
659 StringRef methodName = tokens[i].getRawIdentifier();
660
661 // Skip control flow keywords and other language constructs that use parentheses
662 if (isCppLanguageConstruct(methodName)) {
663 continue;
664 }
665
666 // Extract return type (everything before method name)
667 size_t returnTypeStart = 0;
668 std::string returnType = extractReturnType(i, tokens, sourceMgr, returnTypeStart);
669
670 // Skip static methods (return type is empty if static)
671 if (returnType.empty()) {
672 continue;
673 }
674
675 // Parse method parameters
676 size_t closeParenIdx = tokenCount;
677 bool hasParameters = false;
678 std::vector<MethodParameter> parameters;
679 if (!parseMethodParameters(i, tokens, sourceMgr, closeParenIdx, hasParameters, parameters)) {
680 // Couldn't find closing paren, skip this method
681 continue;
682 }
683
684 // Check for 'const' and find declaration end
685 size_t endIdx;
686 bool isConst = checkConstAndFindEnd(closeParenIdx, tokens, endIdx);
687
688 // Create method struct
689 if (!returnType.empty() && !methodName.empty()) {
690 if (methods.contains(methodName)) {
691 warnSkipped(methodName, "C API does not support method overloading");
692 methods[methodName] = std::nullopt;
693 } else {
694 ExtraMethod method;
695 method.returnType = returnType;
696 method.methodName = methodName;
697 method.documentation = getDocumentation(returnTypeStart, tokens, sourceMgr);
698 method.isConst = isConst;
699 method.hasParameters = hasParameters;
700 method.parameters = parameters;
701 methods[methodName] = std::make_optional(method);
702 }
703 }
704
705 // Skip to end of this declaration for the next iteration.
706 i = endIdx;
707 }
708 }
709
710 // Return valid methods, skipping overloaded names (nullopt entries).
711 return llvm::to_vector(
712 llvm::map_range(
713 llvm::make_filter_range(methods, [](const auto &p) { return p.second.has_value(); }),
714 [](const auto &p) { return p.second.value(); }
715 )
716 );
717}

720bool matchesMLIRClass(StringRef cppType, StringRef typeName) {
721 if (cppType == typeName) {
722 return true;
723 }
724
725 // Check for "::mlir::" or "mlir::" prefix
726 StringRef prefix = cppType;
727 prefix.consume_front("::");
728 if (prefix.consume_front("mlir::")) {
729 return prefix == typeName;
730 }
731
732 return false;
733}

736std::optional<std::string> tryCppTypeToCapiType(StringRef cppType) {
737 cppType = cppType.trim();
738
739 // Primitive types are unchanged
740 if (isPrimitiveType(cppType)) {
741 return std::make_optional(cppType.str());
742 }
743
744 // APInt type is converted via llzk::fromAPInt()
745 if (isAPIntType(cppType)) {
746 return std::make_optional("int64_t");
747 }
748
749 // Pointer type conversions happen via the `unwrap()` function generated
750 // by `DEFINE_C_API_PTR_METHODS()` in `mlir/CAPI/IR.h`
751 if (cppType.ends_with(" *") || cppType.ends_with("*")) {
752 size_t starPos = cppType.rfind('*');
753 if (starPos != StringRef::npos) {
754 StringRef baseType = cppType.substr(0, starPos).trim();
755 if (matchesMLIRClass(baseType, "AsmState")) {
756 return std::make_optional("MlirAsmState");
757 }
758 if (matchesMLIRClass(baseType, "BytecodeWriterConfig")) {
759 return std::make_optional("MlirBytecodeWriterConfig");
760 }
761 if (matchesMLIRClass(baseType, "MLIRContext")) {
762 return std::make_optional("MlirContext");
763 }
764 if (matchesMLIRClass(baseType, "Dialect")) {
765 return std::make_optional("MlirDialect");
766 }
767 if (matchesMLIRClass(baseType, "DialectRegistry")) {
768 return std::make_optional("MlirDialectRegistry");
769 }
770 if (matchesMLIRClass(baseType, "Operation")) {
771 return std::make_optional("MlirOperation");
772 }
773 if (matchesMLIRClass(baseType, "Block")) {
774 return std::make_optional("MlirBlock");
775 }
776 if (matchesMLIRClass(baseType, "OpOperand")) {

853// Map C++ type to corresponding C API type

855 assert(!isArrayRefType(cppType) && "must check `isArrayRefType()` outside");

858 if (capiTypeOpt.has_value()) {
