47static int runMain(int argc, char **argv) {
48 llvm::cl::list<std::string> IncludeDirs(
49 "I", llvm::cl::desc("Directory of include files"), llvm::cl::value_desc("directory"),
50 llvm::cl::Prefix
51 );
52
53 llvm::cl::opt<bool> PrintAllOps(
54 "print-llzk-ops", llvm::cl::desc("Print a list of all ops registered in LLZK")
55 );
56
57 llvm::sys::PrintStackTraceOnErrorSignal(llvm::StringRef());
58 llvm::setBugReportMsg(
59 "PLEASE submit a bug report to " BUG_REPORT_URL
60 " and include the crash backtrace, relevant LLZK files,"
61 " and associated run script(s).\n"
62 );
63 llvm::cl::AddExtraVersionPrinter([](llvm::raw_ostream &os) {
64 os << "\nLLZK (" LLZK_URL "):\n LLZK version " LLZK_VERSION_STRING "\n";
65 });
66
67 // Register dialects and passes
68 mlir::DialectRegistry registry;
69 llzk::registerDialects(registry);
70 r1cs::registerDialects(registry);
71 zklean::registerDialects(registry);
72#if LLZK_WITH_PCL
73 pcl::registerDialects(registry);
74#endif // LLZK_WITH_PCL
75
76 llzk::registerPasses(registry);
77 r1cs::registerPasses(registry);
78 zklean::registerPasses(registry);
79#if LLZK_WITH_PCL
80 pcl::registerPasses(registry);
81#endif // LLZK_WITH_PCL
82
83 // Register and parse command line options.
84 std::string inputFilename, outputFilename;
85 std::tie(inputFilename, outputFilename) =
86 registerAndParseCLIOptions(argc, argv, "llzk-opt", registry);
87
88 if (PrintAllOps) {
89 mlir::MLIRContext context;
90 context.appendDialectRegistry(registry);
91 context.loadAllAvailableDialects();
92 llvm::outs() << "All ops registered in LLZK IR: {\n";
93 for (const auto &opName : context.getRegisteredOperations()) {
94 llvm::outs().indent(2) << opName.getStringRef() << '\n';
95 }
96 llvm::outs() << "}\n";
97 return EXIT_SUCCESS;
98 }
99
100 // Set the include directories from CL option
101 if (mlir::failed(llzk::GlobalSourceMgr::get().setup(IncludeDirs))) {
102 return EXIT_FAILURE;
103 }
104
105 // Run 'mlir-opt'
106 auto result = mlir::MlirOptMain(argc, argv, inputFilename, outputFilename, registry);
107 return mlir::asMainReturnCode(result);
108}
109
110int main(int argc, char **argv) noexcept {
111 try {
112 return runMain(argc, argv);
113 } catch (const std::exception &ex) {
114 llvm::errs() << "llzk-opt: unhandled exception: " << ex.what() << '\n';
115 } catch (...) {
116 llvm::errs() << "llzk-opt: unhandled non-standard exception\n";
117 }
118 return EXIT_FAILURE;
119}
