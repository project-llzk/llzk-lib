827 if (isVariadic[i]) ++prevVariadicCount;

829 // Calculate how many dynamic values a static variadic operand corresponds to.
830 // This assumes all static variadic operands have the same dynamic value count.

832 // `index` passed in as the parameter is the static index which counts each

834 // operand, we need to offset by (variadicSize - 1) to get where the dynamic
835 // value pack for this static operand starts.

1700::mlir::ParseResult MemberWriteOp::parse(::mlir::OpAsmParser &parser, ::mlir::OperationState &result) {
1701 ::mlir::OpAsmParser::UnresolvedOperand componentRawOperand{};
1702 ::llvm::ArrayRef<::mlir::OpAsmParser::UnresolvedOperand> componentOperands(&componentRawOperand, 1); ::llvm::SMLoc componentOperandsLoc;
1703 (void)componentOperandsLoc;
1704 ::mlir::FlatSymbolRefAttr member_nameAttr;
1705 ::mlir::OpAsmParser::UnresolvedOperand valRawOperand{};
1706 ::llvm::ArrayRef<::mlir::OpAsmParser::UnresolvedOperand> valOperands(&valRawOperand, 1); ::llvm::SMLoc valOperandsLoc;
1707 (void)valOperandsLoc;
1708 ::mlir::Type componentRawType{};
1709 ::llvm::ArrayRef<::mlir::Type> componentTypes(&componentRawType, 1);
1710 ::mlir::Type valRawType{};
1711 ::llvm::ArrayRef<::mlir::Type> valTypes(&valRawType, 1);
1712
1713 componentOperandsLoc = parser.getCurrentLocation();
1714 if (parser.parseOperand(componentRawOperand))
1715 return ::mlir::failure();
1716 if (parser.parseLSquare())
1717 return ::mlir::failure();
1718
1719 if (parser.parseCustomAttributeWithFallback(member_nameAttr, parser.getBuilder().getType<::mlir::NoneType>())) {
1720 return ::mlir::failure();
1721 }

1723 if (parser.parseRSquare())
1724 return ::mlir::failure();
1725 if (parser.parseEqual())
1726 return ::mlir::failure();
1727
1728 valOperandsLoc = parser.getCurrentLocation();
1729 if (parser.parseOperand(valRawOperand))
1730 return ::mlir::failure();
1731 if (parser.parseColon())
1732 return ::mlir::failure();
1733
1734 {
1735 ::llzk::component::StructType type;
1736 if (parser.parseCustomTypeWithFallback(type))
1737 return ::mlir::failure();
1738 componentRawType = type;
1739 }
1740 if (parser.parseComma())
1741 return ::mlir::failure();
1742

1744 ::mlir::Type type;
1745 if (parser.parseCustomTypeWithFallback(type))
1746 return ::mlir::failure();

1748 }

1750 auto loc = parser.getCurrentLocation();(void)loc;
1751 if (parser.parseOptionalAttrDict(result.attributes))
1752 return ::mlir::failure();
