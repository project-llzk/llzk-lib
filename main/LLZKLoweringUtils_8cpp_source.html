56bool collectStorageOrigins(Value value, DenseSet<Value> &origins, DenseSet<Value> &visited) {
57 if (!visited.insert(value).second) {
58 return true;
59 }
60 // Component values are identity-bearing handles, even when passed by value.
61 // Different arguments or aggregate paths may refer to the same component.
62 if (isa<StructType>(value.getType()) && !value.getDefiningOp<CreateStructOp>()) {
63 return false;
64 }
65 if (auto argument = dyn_cast<BlockArgument>(value)) {
66 if (!isa<FuncDefOp>(argument.getOwner()->getParentOp()) ||
67 argument.getOwner() != &argument.getOwner()->getParent()->front()) {
68 return false;
69 }
70 origins.insert(value);
71 return true;
72 }
73 Operation *def = value.getDefiningOp();
74 if (isa<array::CreateArrayOp, pod::NewPodOp, CreateStructOp>(def)) {
75 origins.insert(value);
76 return true;
77 }
78 bool isAccess = isa<MemberReadOp, array::ArrayAccessOpInterface, pod::PodAccessOpInterface>(def);
79 bool isAliasCast = isa<polymorphic::UnifiableCastOp, UnrealizedConversionCastOp>(def) &&
80 def->getNumOperands() == 1 && def->getNumResults() == 1;
81 if (!isAccess && !isAliasCast) {
82 return false;
83 }
84 bool hasSource = false;
85 bool known = true;
86 for (Value operand : def->getOperands()) {
87 if (isAggregateStorage(operand)) {
88 hasSource = true;
89 known &= collectStorageOrigins(operand, origins, visited);
90 }
91 }
92 return hasSource && known;
93}

97bool mayShareStorage(Value lhs, Value rhs) {
98 DenseSet<Value> lhsOrigins, rhsOrigins, lhsVisited, rhsVisited;
99 if (!collectStorageOrigins(lhs, lhsOrigins, lhsVisited) ||
100 !collectStorageOrigins(rhs, rhsOrigins, rhsVisited)) {
101 return true;
102 }
103 return llvm::any_of(lhsOrigins, [&](Value origin) { return rhsOrigins.contains(origin); });
104}

112bool callMayMutateStorage(CallOp call, Value storage, DenseSet<Operation *> &activeCallees) {
113 SmallVector<unsigned> overlappingArgs;
114 for (auto [index, argument] : llvm::enumerate(call.getArgOperands())) {
115 if (isAggregateStorage(argument) && mayShareStorage(argument, storage)) {
116 overlappingArgs.push_back(index);
117 }
118 }
119 if (overlappingArgs.empty()) {
120 return false;
121 }
122 SymbolTableCollection tables;
123 auto target = call.getCalleeTarget(tables);
124 if (failed(target)) {
125 return true;
126 }
127 FuncDefOp callee = target->get();
128 if (callee.getBody().empty() || !activeCallees.insert(callee).second) {
129 return true;
130 }
131 llvm::scope_exit cleanup([&]() { activeCallees.erase(callee); });
132 for (unsigned index : overlappingArgs) {
133 if (index >= callee.getNumArguments() ||
134 findStorageMutation(callee, callee.getArgument(index), activeCallees)) {
135 return true;
136 }
137 }
138 return false;
139}

145bool operationMayMutateStorage(Operation *op, Value storage, DenseSet<Operation *> &activeCallees) {
146 if (auto call = dyn_cast<CallOp>(op)) {
147 return callMayMutateStorage(call, storage, activeCallees);
148 }
149 if (auto access = dyn_cast<array::ArrayAccessOpInterface>(op)) {
150 return !access.isRead() && mayShareStorage(access.getArrRef(), storage);
151 }
152 if (auto access = dyn_cast<pod::PodAccessOpInterface>(op)) {
153 return !access.isRead() && mayShareStorage(access.getPodRef(), storage);
154 }
155 if (auto write = dyn_cast<MemberWriteOp>(op)) {
156 return mayShareStorage(write.getComponent(), storage);
157 }
158 // Constraint emission observes storage; terminators only transfer values.
159 if (isa<ConstraintOpInterface>(op) || op->hasTrait<OpTrait::IsTerminator>()) {
160 return false;
161 }
162 auto overlapsOperand = [&]() {
163 return llvm::any_of(op->getOperands(), [&](Value operand) {
164 return isAggregateStorage(operand) && mayShareStorage(operand, storage);
165 });
166 };
167 auto effects = getEffectsRecursively(op);
168 if (!effects) {
169 return overlapsOperand();
170 }
171 return llvm::any_of(*effects, [&](const MemoryEffects::EffectInstance &effect) {
172 if (!isa<MemoryEffects::Write, MemoryEffects::Free>(effect.getEffect())) {
173 return false;
174 }
175 if (Value target = effect.getValue()) {
176 return !isAggregateStorage(target) || mayShareStorage(target, storage);
177 }
178 // An unscoped write effect cannot establish which storage it modifies.
179 return true;
180 });
181}

185Operation *
186findStorageMutation(Operation *scope, Value storage, DenseSet<Operation *> &activeCallees) {
187 Operation *mutation = nullptr;
188 scope->walk([&](Operation *op) -> WalkResult {
189 if (operationMayMutateStorage(op, storage, activeCallees)) {
190 mutation = op;
191 return WalkResult::interrupt();
192 }
193 return WalkResult::advance();
194 });
195 return mutation;
196}

199Operation *findStorageMutation(Operation *scope, Value storage) {
200 DenseSet<Operation *> activeCallees;
201 return findStorageMutation(scope, storage, activeCallees);

204Value mapBlockArgumentInCompute(BlockArgument barg, FuncDefOp computeFunc) {
205 // Constrain entry arguments map onto compute inputs: constrain(%self, args...)
206 // corresponds to compute(args...), plus the compute-side `%self`.
207 if (barg.getArgNumber() == 0) {
208 return computeFunc.getSelfValueFromCompute();
209 }
210 Value argument = computeFunc.getArgument(barg.getArgNumber() - 1);
211 if (isAggregateStorage(argument) && findStorageMutation(computeFunc, argument)) {
212 emitError(
213 barg.getLoc()
214 ) << "cannot rebuild aggregate argument in compute-side auxiliary expression: "
215 "compute may mutate the input storage";
216 return nullptr;
217 }
218 return argument;
219}

221Value mapValueIntoCompute(
222 Value val, FuncDefOp computeFunc, OpBuilder &builder, DenseMap<Value, Value> &memo
223) {
224 if (auto it = memo.find(val); it != memo.end()) {
225 return it->second;
226 }
227 if (auto barg = llvm::dyn_cast<BlockArgument>(val)) {
228 return memo[val] = mapBlockArgumentInCompute(barg, computeFunc);
229 }
230 return rebuildExprInCompute(val, computeFunc, builder, memo);
231}

235Value rebuildExprInCompute(
236 Value val, FuncDefOp computeFunc, OpBuilder &builder, DenseMap<Value, Value> &memo
237) {
238 if (auto it = memo.find(val); it != memo.end()) {
239 return it->second;
240 }
241
242 if (auto barg = llvm::dyn_cast<BlockArgument>(val)) {
243 return memo[val] = mapBlockArgumentInCompute(barg, computeFunc);
244 }
245
246 if (Operation *readOp = val.getDefiningOp();
247 isa_and_nonnull<MemberReadOp, array::ReadArrayOp, array::ExtractArrayOp, pod::ReadPodOp>(
248 readOp
249 )) {
250 // Replay is safe only if preceding operations cannot change the read's storage.
251 // Use the same effect/provenance model as compute-side argument mappings.
252 for (Operation &preceding : *readOp->getBlock()) {
253 if (&preceding == readOp) {
254 break;
255 }
256 for (Value operand : readOp->getOperands()) {
257 if (isAggregateStorage(operand) && findStorageMutation(&preceding, operand)) {
258 readOp->emitError(
259 "cannot rebuild storage read in compute-side auxiliary expression: "
260 "storage may have been mutated; aggregate mutations cannot be replayed"
261 );
262 return nullptr;
263 }
264 }
265 }
266 IRMapping mapper;
267 for (Value operand : readOp->getOperands()) {
268 Value rebuiltOperand = mapValueIntoCompute(operand, computeFunc, builder, memo);
269 if (!rebuiltOperand) {
270 return nullptr;
271 }
272 mapper.map(operand, rebuiltOperand);
273 }
274
275 Operation *rebuiltOp = builder.clone(*readOp, mapper);
276 assert(rebuiltOp->getNumResults() == 1 && "storage reads have exactly one result");
277 return memo[val] = rebuiltOp->getResult(0);
278 }
279
280 if (auto callOp = val.getDefiningOp<CallOp>()) {
281 if (!callOp.getMapOperands().empty()) {
282 callOp
283 .emitError(
284 "cannot rebuild affine-instantiated function.call in compute-side auxiliary "
285 "expression"
286 )
287 .report();
288 return nullptr;
289 }
290
291 SymbolTableCollection tables;
292 FailureOr<SymbolLookupResult<FuncDefOp>> target = callOp.getCalleeTarget(tables);
293 if (failed(target)) {
294 return nullptr;
295 }
296 FuncDefOp targetFunc = target->get();
297 bool invalidTarget =
298 (targetFunc.hasAllowConstraintAttr() && !computeFunc.hasAllowConstraintAttr()) ||
299 (targetFunc.hasAllowWitnessAttr() && !computeFunc.hasAllowWitnessAttr()) ||
300 (targetFunc.hasAllowNonNativeFieldOpsAttr() &&
301 !computeFunc.hasAllowNonNativeFieldOpsAttr());
302 if (invalidTarget) {
303 callOp
304 .emitError(
305 "cannot rebuild function.call in compute-side auxiliary expression: callee "
306 "requires attributes not present on the compute function"
307 )
308 .report();
309 return nullptr;
310 }
311
312 SmallVector<Value> rebuiltArgs;
313 rebuiltArgs.reserve(callOp.getArgOperands().size());
314 for (Value arg : callOp.getArgOperands()) {
315 Value rebuiltArg = rebuildExprInCompute(arg, computeFunc, builder, memo);
316 if (!rebuiltArg) {
317 return nullptr;
318 }
319 rebuiltArgs.push_back(rebuiltArg);
320 }
321
322 ArrayRef<Attribute> templateParams;
323 if (ArrayAttr params = callOp.getTemplateParamsAttr()) {
324 templateParams = params.getValue();
325 }
326
327 CallOp rebuilt = CallOp::create(
328 builder, callOp.getLoc(), callOp.getResultTypes(), callOp.getCalleeAttr(), rebuiltArgs,
329 templateParams
330 );
331 for (auto [oldResult, newResult] : llvm::zip(callOp.getResults(), rebuilt.getResults())) {
332 memo[oldResult] = newResult;
333 }
334 return memo[val];
335 }
336
337 if (val.getType().isIndex()) {
338 // Preserve index producers used by member-read access operands so rebuilt reads
339 // keep the original access semantics.
340 Operation *defOp = val.getDefiningOp();
341 assert(defOp && "index block arguments should already be mapped");
342
343 IRMapping mapper;
344 for (Value operand : defOp->getOperands()) {
345 Value rebuiltOperand = mapValueIntoCompute(operand, computeFunc, builder, memo);
346 if (!rebuiltOperand) {
347 return nullptr;
348 }
349 mapper.map(operand, rebuiltOperand);
350 }
351
352 Operation *rebuiltOp = builder.clone(*defOp, mapper);
353 assert(
354 rebuiltOp->getNumResults() == defOp->getNumResults() &&
355 "cloned index op should preserve result count"
356 );
357 unsigned resultNumber = llvm::cast<OpResult>(val).getResultNumber();
358 return memo[val] = rebuiltOp->getResult(resultNumber);
359 }
360
361 if (auto add = val.getDefiningOp<AddFeltOp>()) {
362 Value lhs = rebuildExprInCompute(add.getLhs(), computeFunc, builder, memo);
363 Value rhs = rebuildExprInCompute(add.getRhs(), computeFunc, builder, memo);
364 if (!lhs || !rhs) {
365 return nullptr;
366 }
367 return memo[val] = AddFeltOp::create(builder, add.getLoc(), add.getType(), lhs, rhs);
368 }
369
370 if (auto sub = val.getDefiningOp<SubFeltOp>()) {
371 Value lhs = rebuildExprInCompute(sub.getLhs(), computeFunc, builder, memo);
372 Value rhs = rebuildExprInCompute(sub.getRhs(), computeFunc, builder, memo);
373 if (!lhs || !rhs) {
374 return nullptr;
375 }
376 return memo[val] = SubFeltOp::create(builder, sub.getLoc(), sub.getType(), lhs, rhs);
377 }
378
379 if (auto mul = val.getDefiningOp<MulFeltOp>()) {
380 Value lhs = rebuildExprInCompute(mul.getLhs(), computeFunc, builder, memo);
381 Value rhs = rebuildExprInCompute(mul.getRhs(), computeFunc, builder, memo);
382 if (!lhs || !rhs) {
383 return nullptr;
384 }
385 return memo[val] = MulFeltOp::create(builder, mul.getLoc(), mul.getType(), lhs, rhs);
386 }
387
388 if (auto neg = val.getDefiningOp<NegFeltOp>()) {
389 Value operand = rebuildExprInCompute(neg.getOperand(), computeFunc, builder, memo);
390 if (!operand) {
391 return nullptr;
392 }
393 return memo[val] = NegFeltOp::create(builder, neg.getLoc(), neg.getType(), operand);
394 }
395
396 if (auto div = val.getDefiningOp<DivFeltOp>()) {
397 Value lhs = rebuildExprInCompute(div.getLhs(), computeFunc, builder, memo);
398 Value rhs = rebuildExprInCompute(div.getRhs(), computeFunc, builder, memo);
399 if (!lhs || !rhs) {
400 return nullptr;
401 }
402 return memo[val] = DivFeltOp::create(builder, div.getLoc(), div.getType(), lhs, rhs);
403 }
404
405 if (auto c = val.getDefiningOp<FeltConstantOp>()) {
406 return memo[val] = FeltConstantOp::create(builder, c.getLoc(), c.getValueAttr());
407 }
408
409 if (Operation *op = val.getDefiningOp()) {
410 op->emitError("cannot rebuild unsupported operation in compute-side auxiliary expression")
411 .report();
412 }
413 return nullptr;
414}

416LogicalResult checkForAuxMemberConflicts(StructDefOp structDef, StringRef prefix) {
417 auto res = structDef.walk([&prefix](MemberDefOp memberDefOp) -> WalkResult {
418 if (memberDefOp.getName().starts_with(prefix)) {
419 return memberDefOp.emitOpError().append(
420 "name conflicts with reserved prefix '", prefix, '\''
421 );
422 }
423 return WalkResult::advance();
424 });
425 return failure(res.wasInterrupted());

428LogicalResult checkFuncBodyIsStraightLine(FuncDefOp func, StringRef passName) {
429 auto emitStraightLineError = [&func, &passName](Operation *op) -> LogicalResult {
430 StringRef funcName;
431 if (func.isStructCompute()) {
432 funcName = "compute";
433 } else if (func.isStructConstrain()) {
434 funcName = "constrain";
435 } else {
436 funcName = "function";
437 }
438 return op->emitError()
439 << passName << " expects a straight-line " << funcName
440 << " body; run `llzk-flatten` or another control-flow lowering pass first";
441 };
442
443 Region &body = func.getBody();
444 if (!body.hasOneBlock()) {
445 return emitStraightLineError(func.getOperation());
446 }
447
448 auto res = body.walk([&emitStraightLineError](Operation *op) -> WalkResult {
449 if (op->getNumRegions() != 0 || op->getNumSuccessors() != 0) {
450 return emitStraightLineError(op);
451 }
452 return WalkResult::advance();
453 });
454 return failure(res.wasInterrupted());
455}

457void replaceSubsequentUsesWith(Value oldVal, Value newVal, Operation *afterOp) {
458 assert(afterOp && "afterOp must be a valid Operation*");
459
460 for (auto &use : llvm::make_early_inc_range(oldVal.getUses())) {
461 Operation *user = use.getOwner();
462
463 // Skip uses that are:
464 // - Before afterOp in the same block.
465 // - Inside afterOp itself.
466 if ((user->getBlock() == afterOp->getBlock()) &&
467 (user == afterOp || user->isBeforeInBlock(afterOp))) {
468 continue;
469 }
470
471 // Replace this use of oldVal with newVal.
472 use.set(newVal);
473 }
474}

484unsigned getFeltDegree(Value val, DenseMap<Value, unsigned> &memo) {
485 if (auto it = memo.find(val); it != memo.end()) {
486 return it->second;
487 }
488
489 if (isa<FeltConstantOp>(val.getDefiningOp())) {
490 return memo[val] = 0;
491 }
492 if (isa<NonDetOp, MemberReadOp>(val.getDefiningOp()) || isa<BlockArgument>(val)) {
493 return memo[val] = 1;
494 }
495 if (auto add = val.getDefiningOp<AddFeltOp>()) {
496 return memo[val] =
497 std::max(getFeltDegree(add.getLhs(), memo), getFeltDegree(add.getRhs(), memo));
498 }
499 if (auto sub = val.getDefiningOp<SubFeltOp>()) {
500 return memo[val] =
501 std::max(getFeltDegree(sub.getLhs(), memo), getFeltDegree(sub.getRhs(), memo));
502 }
503 if (auto mul = val.getDefiningOp<MulFeltOp>()) {
504 return memo[val] = getFeltDegree(mul.getLhs(), memo) + getFeltDegree(mul.getRhs(), memo);
505 }
506 if (auto div = val.getDefiningOp<DivFeltOp>()) {
507 return memo[val] = getFeltDegree(div.getLhs(), memo) + getFeltDegree(div.getRhs(), memo);
508 }
509 if (auto neg = val.getDefiningOp<NegFeltOp>()) {
510 return memo[val] = getFeltDegree(neg.getOperand(), memo);
511 }
512
513 llvm::errs() << "Unhandled felt op in degree computation: " << val << '\n';
514 llvm_unreachable("Unhandled op in getFeltDegree");
515}
