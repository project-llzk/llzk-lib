19static DynamicAPInt positiveModulo(const DynamicAPInt &value, const DynamicAPInt &modulus) {
20 DynamicAPInt result = value % modulus;
21 return result < 0 ? result + modulus : result;

26std::optional<DynamicAPInt>
27applyIntToFeltOverflow(const DynamicAPInt &value, const Field &field, OverflowSemantics overflow) {
28 switch (overflow) {
29 case OverflowSemantics::ASSERT:
30 return value < 0 || value >= field.prime() ? std::nullopt : std::optional(value);
31 case OverflowSemantics::SATURATE:
32 return value < 0 ? field.zero() : (value < field.maxVal() ? value : field.maxVal());
33 case OverflowSemantics::WRAP:
34 return field.reduce(value);
35 case OverflowSemantics::TRUNCATE:
36 return field.reduce(positiveModulo(value, DynamicAPInt(1) << field.bitWidth()));
37 }
38 llvm_unreachable("unknown overflow semantics");
39}

41std::optional<DynamicAPInt> applyFeltToIndexOverflow(
42 const DynamicAPInt &value, unsigned indexBitWidth, OverflowSemantics overflow
43) {
44 const DynamicAPInt safeUpperBound = (DynamicAPInt(1) << (indexBitWidth - 1)) - 1;
45 switch (overflow) {
46 case OverflowSemantics::ASSERT:
47 return value < 0 || value > safeUpperBound ? std::nullopt : std::optional(value);
48 case OverflowSemantics::SATURATE:
49 return value < 0 ? DynamicAPInt(0) : (value < safeUpperBound ? value : safeUpperBound);
50 case OverflowSemantics::WRAP:
51 case OverflowSemantics::TRUNCATE:
52 return positiveModulo(value, safeUpperBound + 1);
53 }
54 llvm_unreachable("unknown overflow semantics");
55}
