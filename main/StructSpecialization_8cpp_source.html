20FailureOr<StructDefOp> StructSpecializationResolver::resolve(
21 SymbolTableCollection &tables, StructType type, int64_t id, Operation *site,
22 StringRef diagnosticContext
23) {
24 if (!definitions) {
25 definitions.emplace();
26 rootModule.walk([this](StructDefOp definition) {
27 auto metadata = definition->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR);
28 if (metadata && metadata.getValue().isSignedIntN(64) && metadata.getInt() >= 0) {
29 (*definitions)[metadata.getInt()].push_back(definition);
30 }
31 });
32 }
33 auto target = definitions->find(id);
34 if (target == definitions->end()) {
35 return site->emitError() << "unknown struct specialization in " << diagnosticContext
36 << " family";
37 }
38 if (target->second.size() != 1) {
39 return site->emitError() << "ambiguous struct specialization in " << diagnosticContext
40 << " family";
41 }
42 auto definition = target->second.front();
43 auto source = type.getDefinition(tables, site);
44 if (failed(source)) {
45 return failure();
46 }
47 if (!*source || source->viaInclude()) {
48 return site->emitError() << "cannot resolve " << diagnosticContext
49 << " family source; inline includes first";
50 }
51 auto origin = getPathRelativeToAncestor(source->get(), rootModule);
52 if (failed(origin) || definition->getAttr(SPECIALIZATION_ORIGIN_ATTR) != *origin) {
53 return site->emitError() << "struct specialization origin does not match " << diagnosticContext
54 << " family";
55 }
56 return definition;
57}
