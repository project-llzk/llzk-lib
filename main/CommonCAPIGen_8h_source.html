79inline std::string toPascalCase(mlir::StringRef str) {
80 if (str.empty()) {
81 return "";
82 }
83
84 std::string result;
85 result.reserve(str.size());
86 llvm::raw_string_ostream resultStream(result);
87 bool capitalizeNext = true;
88
89 for (char c : str) {
90 if (c == '_' || c == ':') {
91 capitalizeNext = true;
92 } else {
93 resultStream << (capitalizeNext ? llvm::toUpper(c) : c);
94 capitalizeNext = false;
95 }
96 }
97
98 return result;
99}

104inline bool isIntegerType(mlir::StringRef type) {
105 // Consume optional root namespace token
106 type.consume_front("::");
107 // Handle special names first
108 if (type == "signed" || type == "unsigned" || type == "size_t" || type == "char32_t" ||
109 type == "char16_t" || type == "char8_t" || type == "wchar_t") {
110 return true;
111 }
112 // Handle standard integer types with optional signed/unsigned prefix
113 type.consume_front("signed ") || type.consume_front("unsigned ");
114 if (type == "char" || type == "int" || type == "short" || type == "short int" || type == "long" ||
115 type == "long int" || type == "long long" || type == "long long int") {
116 return true;
117 }
118 // Handle fixed-width integer types (https://cppreference.com/w/cpp/types/integer.html)
119 type.consume_front("std::"); // optional
120 if (type.consume_back("_t") && (type.consume_front("int") || type.consume_front("uint"))) {
121 // intmax_t, intptr_t, uintmax_t, uintptr_t
122 if (type == "max" || type == "ptr") {
123 return true;
124 }
125 // Optional "_fast" or "_least" followed by bit width to cover the rest
126 type.consume_front("_fast") || type.consume_front("_least");
127 if (type == "8" || type == "16" || type == "32" || type == "64") {
128 return true;
129 }
130 }
131 return false;
132}

140inline bool isPrimitiveType(mlir::StringRef cppType) {
141 cppType.consume_front("::");
142 return cppType == "void" || cppType == "bool" || cppType == "float" || cppType == "double" ||
143 cppType == "long double" || isIntegerType(cppType);
144}

150 return llvm::StringSwitch<bool>(tokenText)

185inline bool isAPIntType(mlir::StringRef cppType) {
186 cppType.consume_front("::");
187 if (cppType == "llzk::APIntValue") {
188 return true;
189 }
190 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
191 return cppType == "APInt";
192}

197inline bool isArrayRefType(mlir::StringRef cppType) {
198 cppType.consume_front("::");
199 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
200 return cppType.starts_with("ArrayRef<");
201}

204inline mlir::StringRef extractArrayRefElementType(mlir::StringRef cppType) {
205 assert(isArrayRefType(cppType) && "must check `isArrayRefType()` outside");
206
207 // Remove "ArrayRef<" prefix and ">" suffix
208 cppType.consume_front("::");
209 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
210 cppType.consume_front("ArrayRef<") && cppType.consume_back(">");
211 return cppType;
212}

425 void genExtraMethod(const ExtraMethod &method) const override {
426 // Convert return type to C API type, skip if it can't be converted
427 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
428 if (!capiReturnTypeOpt.has_value()) {
429 warnSkippedNoConversion(method.methodName, method.returnType);
430 return;
431 }
432 std::string capiReturnType = capiReturnTypeOpt.value();
433
434 // Build parameter list
435 std::string paramList;
436 llvm::raw_string_ostream paramListStream(paramList);
437 paramListStream << llvm::formatv("Mlir{0} inp", kind);
438 for (const auto &param : method.parameters) {
439 // Convert C++ type to C API type for parameter, skip if it can't be converted
440 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
441 if (!capiParamTypeOpt.has_value()) {
442 warnSkippedNoConversion(method.methodName, param.type);
443 return;
444 }
445 const std::string &capiParamType = capiParamTypeOpt.value();
446 paramListStream << ", " << capiParamType << ' ' << param.name;
447 }
448
449 // Generate declaration
450 if (method.documentation.empty()) {
451 os << llvm::formatv("\n/// {0}\n", method.methodName);
452 } else {
453 os << llvm::formatv("\n{0}\n", method.documentation);
454 }
455 os << llvm::formatv(
456 "MLIR_CAPI_EXPORTED {0} {1}{2}_{3}{4}({5});\n",
457 capiReturnType, // {0}
458 FunctionPrefix, // {1}
459 dialectNameCapitalized, // {2}
460 className, // {3}
461 toPascalCase(method.methodName), // {4}
462 paramList // {5}
463 );
464 }

468struct ImplementationGenerator : public Generator {
469 using Generator::Generator;
470 ~ImplementationGenerator() override = default;
471
472 virtual void genPrologue() const {}
473
474 virtual void genIsAImpl() const {
475 static constexpr char fmt[] = R"(
476bool {0}{1}IsA_{2}_{3}(Mlir{1} inp) {{
477 return llvm::isa<{3}>(unwrap(inp));
478}
479)";
480 assert(!className.empty() && "className must be set");
481 os << llvm::formatv(fmt, FunctionPrefix, kind, dialectNameCapitalized, className);
482 }
483
485 void genExtraMethod(const ExtraMethod &method) const override {
486 // Convert return type to C API type, skip if it can't be converted
487 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
488 if (!capiReturnTypeOpt.has_value()) {
489 warnSkippedNoConversion(method.methodName, method.returnType);
490 return;
491 }
492 std::string capiReturnType = capiReturnTypeOpt.value();
493
494 // Build the return statement prefix and suffix
495 std::string returnPrefix;
496 std::string returnSuffix;
497 mlir::StringRef cppReturnType = method.returnType;
498
499 if (cppReturnType == "void") {
500 // "void" type doesn't even need "return"
501 returnPrefix = "";
502 returnSuffix = "";
503 } else {
504 // Check if return needs wrapping
505 if (isPrimitiveType(cppReturnType)) {
506 // Primitive types don't need wrapping
507 returnPrefix = "return ";
508 returnSuffix = "";
509 } else if (capiReturnType.starts_with("Mlir") || isAPIntType(cppReturnType)) {
510 // MLIR C API types and APInt type need wrapping
511 returnPrefix = "return wrap(";
512 returnSuffix = ")";
513 } else {
514 return;
515 }
516 }
517
518 // Build parameter list for C API function signature
519 std::string paramList;
520 llvm::raw_string_ostream paramListStream(paramList);
521 paramListStream << llvm::formatv("Mlir{0} inp", kind);
522 for (const auto &param : method.parameters) {
523 // Convert C++ type to C API type for parameter, skip if it can't be converted
524 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
525 if (!capiParamTypeOpt.has_value()) {
526 warnSkippedNoConversion(method.methodName, param.type);
527 return;
528 }
529 const std::string &capiParamType = capiParamTypeOpt.value();
530 paramListStream << ", " << capiParamType << ' ' << param.name;
531 }
532
533 // Build argument list for C++ method call
534 std::string argList;
535 llvm::raw_string_ostream argListStream(argList);
536 for (size_t i = 0; i < method.parameters.size(); ++i) {
537 if (i > 0) {
538 argListStream << ", ";
539 }
540 const auto &param = method.parameters[i];
541
542 // Check if parameter needs unwrapping
543 mlir::StringRef cppParamType = param.type;
544 if (isPrimitiveType(cppParamType)) {
545 // Primitive types don't need unwrapping
546 argListStream << param.name;
547 } else if (isAPIntType(cppParamType)) {
548 // APInt needs unwrapping
549 argListStream << "unwrap(" << param.name << ')';
550 } else {
551 // Convert C++ type to C API type for parameter, skip if it can't be converted
552 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(cppParamType);
553 if (capiParamTypeOpt.has_value() && capiParamTypeOpt->starts_with("Mlir")) {
554 // MLIR C API types need unwrapping
555 argListStream << "unwrap(" << param.name << ')';
556 } else {
557 warnSkippedNoConversion(method.methodName, cppParamType.str());
558 return;
559 }
560 }
561 }
562
563 // Generate implementation
564 os << '\n';
565 os << llvm::formatv(
566 "{0} {1}{2}_{3}{4}({5}) {{\n",
567 capiReturnType, // {0}
568 FunctionPrefix, // {1}
569 dialectNameCapitalized, // {2}
570 className, // {3}
571 toPascalCase(method.methodName), // {4}
572 paramList // {5}
573 );
574 os << llvm::formatv(
575 " {0}llvm::cast<{1}>(unwrap(inp)).{2}({3}){4};\n",
576 returnPrefix, // {0}
577 className, // {1}
578 method.methodName, // {2}
579 argList, // {3}
580 returnSuffix // {4}
581 );
582 os << "}\n";
583 }
584};

622 void genExtraMethod(const ExtraMethod &method) const override {
623 // Convert return type to C API type, skip if it can't be converted
624 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
625 if (!capiReturnTypeOpt.has_value()) {
626 warnSkippedNoConversion(method.methodName, method.returnType);
627 return;
628 }
629
630 // Build parameter list for dummy values
631 std::string dummyParams;
632 llvm::raw_string_ostream dummyParamsStream(dummyParams);
633 std::string paramList;
634 llvm::raw_string_ostream paramListStream(paramList);
635
636 for (const auto &param : method.parameters) {
637 // Convert C++ type to C API type for parameter, skip if it can't be converted
638 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
639 if (!capiParamTypeOpt.has_value()) {
640 warnSkippedNoConversion(method.methodName, param.type);
641 return;
642 }
643 const std::string &capiParamType = capiParamTypeOpt.value();
644 std::string name = param.name;
645
646 // Generate dummy value creation for each parameter
647 if (capiParamType == "bool") {
648 dummyParamsStream << " bool " << name << " = false;\n";
649 } else if (capiParamType == "MlirValue") {
650 dummyParamsStream << " auto " << name << " = mlirOperationGetResult(testOp, 0);\n";
651 } else if (capiParamType == "MlirType") {
652 dummyParamsStream << " auto " << name << " = createIndexType();\n";
653 } else if (capiParamType == "MlirAttribute") {
654 dummyParamsStream << " auto " << name << " = createIndexAttribute();\n";
655 } else if (capiParamType == "MlirStringRef") {
656 dummyParamsStream << " auto " << name << " = mlirStringRefCreateFromCString(\"\");\n";
657 } else if (isIntegerType(capiParamType)) {
658 dummyParamsStream << " " << capiParamType << ' ' << name << " = 0;\n";
659 } else {
660 // For unknown types, create a default-initialized variable
661 dummyParamsStream << " " << capiParamType << ' ' << name << " = {};\n";
662 }
663
664 paramListStream << ", " << name;
665 }
666
667 static constexpr char fmt[] = R"(
669TEST_F({2}{1}LinkTests, {0}_{3}_{4}) {{
670 auto test{1} = createIndex{1}();
671
672 if ({0}{1}IsA_{2}_{3}(test{1})) {{
673{5}
674 (void){0}{2}_{3}{4}(test{1}{6});
675 }
676
677 {7}(test{1});
678}
679)";
680 assert(!className.empty() && "className must be set");
681 os << llvm::formatv(
682 fmt,
683 FunctionPrefix, // {0}
684 kind, // {1}
685 dialectNameCapitalized, // {2}
686 className, // {3}
687 toPascalCase(method.methodName), // {4}
688 dummyParams, // {5}
689 paramList, // {6}
690 genCleanup() // {7}
691 );
692 }
