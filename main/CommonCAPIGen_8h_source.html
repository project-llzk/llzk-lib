100inline std::string toPascalCase(mlir::StringRef str) {
101 if (str.empty()) {
102 return "";
103 }
104
105 std::string result;
106 result.reserve(str.size());
107 llvm::raw_string_ostream resultStream(result);
108 bool capitalizeNext = true;
109
110 for (char c : str) {
111 if (c == '_' || c == ':') {
112 capitalizeNext = true;
113 } else {
114 resultStream << (capitalizeNext ? llvm::toUpper(c) : c);
115 capitalizeNext = false;
116 }
117 }
118
119 return result;
120}

125inline bool isIntegerType(mlir::StringRef type) {
126 // Consume optional root namespace token
127 type.consume_front("::");
128 // Handle special names first
129 if (type == "signed" || type == "unsigned" || type == "size_t" || type == "char32_t" ||
130 type == "char16_t" || type == "char8_t" || type == "wchar_t") {
131 return true;
132 }
133 // Handle standard integer types with optional signed/unsigned prefix
134 type.consume_front("signed ") || type.consume_front("unsigned ");
135 if (type == "char" || type == "int" || type == "short" || type == "short int" || type == "long" ||
136 type == "long int" || type == "long long" || type == "long long int") {
137 return true;
138 }
139 // Handle fixed-width integer types (https://cppreference.com/w/cpp/types/integer.html)
140 type.consume_front("std::"); // optional
141 if (type.consume_back("_t") && (type.consume_front("int") || type.consume_front("uint"))) {
142 // intmax_t, intptr_t, uintmax_t, uintptr_t
143 if (type == "max" || type == "ptr") {
144 return true;
145 }
146 // Optional "_fast" or "_least" followed by bit width to cover the rest
147 type.consume_front("_fast") || type.consume_front("_least");
148 if (type == "8" || type == "16" || type == "32" || type == "64") {
149 return true;
150 }
151 }
152 return false;
153}

161inline bool isPrimitiveType(mlir::StringRef cppType) {
162 cppType.consume_front("::");
163 return cppType == "void" || cppType == "bool" || cppType == "float" || cppType == "double" ||
164 cppType == "long double" || isIntegerType(cppType);
165}

206inline bool isAPIntType(mlir::StringRef cppType) {
207 cppType.consume_front("::");
208 if (cppType == "llzk::APIntValue") {
209 return true;
210 }
211 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
212 return cppType == "APInt";
213}

218inline bool isArrayRefType(mlir::StringRef cppType) {
219 cppType.consume_front("::");
220 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
221 return cppType.starts_with("ArrayRef<");
222}

225inline mlir::StringRef extractArrayRefElementType(mlir::StringRef cppType) {
226 assert(isArrayRefType(cppType) && "must check `isArrayRefType()` outside");
227
228 // Remove "ArrayRef<" prefix and ">" suffix
229 cppType.consume_front("::");
230 cppType.consume_front("llvm::") || cppType.consume_front("mlir::");
231 cppType.consume_front("ArrayRef<") && cppType.consume_back(">");
232 return cppType;
233}

447 void genExtraMethod(const ExtraMethod &method) const override {
448 // Convert return type to C API type, skip if it can't be converted
449 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
450 if (!capiReturnTypeOpt.has_value()) {
451 warnSkippedNoConversion(method.methodName, method.returnType);
452 return;
453 }
454 std::string capiReturnType = capiReturnTypeOpt.value();
455
456 // Build parameter list
457 std::string paramList;
458 llvm::raw_string_ostream paramListStream(paramList);
459 paramListStream << llvm::formatv("Mlir{0} inp", kind);
460 for (const auto &param : method.parameters) {
461 // Convert C++ type to C API type for parameter, skip if it can't be converted
462 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
463 if (!capiParamTypeOpt.has_value()) {
464 warnSkippedNoConversion(method.methodName, param.type);
465 return;
466 }
467 const std::string &capiParamType = capiParamTypeOpt.value();
468 paramListStream << ", " << capiParamType << ' ' << param.name;
469 }
470
471 // Generate declaration
472 if (method.documentation.empty()) {
473 os << llvm::formatv("\n/// {0}\n", method.methodName);
474 } else {
475 os << llvm::formatv("\n{0}\n", method.documentation);
476 }
477 os << llvm::formatv(
478 "MLIR_CAPI_EXPORTED {0} {1}{2}_{3}{4}({5});\n",
479 capiReturnType, // {0}
480 getFunctionPrefix(), // {1}
481 dialectNameCapitalized, // {2}
482 className, // {3}
483 toPascalCase(method.methodName), // {4}
484 paramList // {5}
485 );
486 }

505
507 void genExtraMethod(const ExtraMethod &method) const override {
508 // Convert return type to C API type, skip if it can't be converted
509 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
510 if (!capiReturnTypeOpt.has_value()) {
511 warnSkippedNoConversion(method.methodName, method.returnType);
512 return;
513 }
514 std::string capiReturnType = capiReturnTypeOpt.value();
515
516 // Build the return statement prefix and suffix
517 std::string returnPrefix;
518 std::string returnSuffix;
519 mlir::StringRef cppReturnType = method.returnType;
520
521 if (cppReturnType == "void") {
522 // "void" type doesn't even need "return"
523 returnPrefix = "";
524 returnSuffix = "";
525 } else {
526 // Check if return needs wrapping
527 if (isPrimitiveType(cppReturnType)) {
528 // Primitive types don't need wrapping
529 returnPrefix = "return ";
530 returnSuffix = "";
531 } else if (capiReturnType.starts_with("Mlir") || isAPIntType(cppReturnType)) {
532 // MLIR C API types and APInt type need wrapping
533 returnPrefix = "return wrap(";
534 returnSuffix = ")";
535 } else {
536 return;
537 }
538 }
539
540 // Build parameter list for C API function signature
541 std::string paramList;
542 llvm::raw_string_ostream paramListStream(paramList);
543 paramListStream << llvm::formatv("Mlir{0} inp", kind);
544 for (const auto &param : method.parameters) {
545 // Convert C++ type to C API type for parameter, skip if it can't be converted
546 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
547 if (!capiParamTypeOpt.has_value()) {
548 warnSkippedNoConversion(method.methodName, param.type);
549 return;
550 }
551 const std::string &capiParamType = capiParamTypeOpt.value();
552 paramListStream << ", " << capiParamType << ' ' << param.name;
553 }
554
555 // Build argument list for C++ method call
556 std::string argList;
557 llvm::raw_string_ostream argListStream(argList);
558 for (size_t i = 0; i < method.parameters.size(); ++i) {
559 if (i > 0) {
560 argListStream << ", ";
561 }
562 const auto &param = method.parameters[i];
563
564 // Check if parameter needs unwrapping
565 mlir::StringRef cppParamType = param.type;
566 if (isPrimitiveType(cppParamType)) {
567 // Primitive types don't need unwrapping
568 argListStream << param.name;
569 } else if (isAPIntType(cppParamType)) {
570 // APInt needs unwrapping
571 argListStream << "unwrap(" << param.name << ')';
572 } else {
573 // Convert C++ type to C API type for parameter, skip if it can't be converted
574 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(cppParamType);
575 if (capiParamTypeOpt.has_value() && capiParamTypeOpt->starts_with("Mlir")) {
576 // MLIR C API types need unwrapping
577 argListStream << "unwrap(" << param.name << ')';
578 } else {
579 warnSkippedNoConversion(method.methodName, cppParamType.str());
580 return;
581 }
582 }
583 }
584
585 // Generate implementation
586 os << '\n';
587 os << llvm::formatv(
588 "{0} {1}{2}_{3}{4}({5}) {{\n",
589 capiReturnType, // {0}
590 getFunctionPrefix(), // {1}
591 dialectNameCapitalized, // {2}
592 className, // {3}
593 toPascalCase(method.methodName), // {4}
594 paramList // {5}
595 );
596 os << llvm::formatv(
597 " {0}llvm::cast<{1}>(unwrap(inp)).{2}({3}){4};\n",
598 returnPrefix, // {0}
599 className, // {1}
600 method.methodName, // {2}
601 argList, // {3}
602 returnSuffix // {4}
603 );
604 os << "}\n";
605 }

644 void genExtraMethod(const ExtraMethod &method) const override {
645 // Convert return type to C API type, skip if it can't be converted
646 std::optional<std::string> capiReturnTypeOpt = tryCppTypeToCapiType(method.returnType);
647 if (!capiReturnTypeOpt.has_value()) {
648 warnSkippedNoConversion(method.methodName, method.returnType);
649 return;
650 }
651
652 // Build parameter list for dummy values
653 std::string dummyParams;
654 llvm::raw_string_ostream dummyParamsStream(dummyParams);
655 std::string paramList;
656 llvm::raw_string_ostream paramListStream(paramList);
657
658 for (const auto &param : method.parameters) {
659 // Convert C++ type to C API type for parameter, skip if it can't be converted
660 std::optional<std::string> capiParamTypeOpt = tryCppTypeToCapiType(param.type);
661 if (!capiParamTypeOpt.has_value()) {
662 warnSkippedNoConversion(method.methodName, param.type);
663 return;
664 }
665 const std::string &capiParamType = capiParamTypeOpt.value();
666 std::string name = param.name;
667
668 // Generate dummy value creation for each parameter
669 if (capiParamType == "bool") {
670 dummyParamsStream << " bool " << name << " = false;\n";
671 } else if (capiParamType == "MlirValue") {
672 dummyParamsStream << " auto " << name << " = mlirOperationGetResult(testOp, 0);\n";
673 } else if (capiParamType == "MlirType") {
674 dummyParamsStream << " auto " << name << " = createIndexType();\n";
675 } else if (capiParamType == "MlirAttribute") {
676 dummyParamsStream << " auto " << name << " = createIndexAttribute();\n";
677 } else if (capiParamType == "MlirStringRef") {
678 dummyParamsStream << " auto " << name << " = mlirStringRefCreateFromCString(\"\");\n";
679 } else if (isIntegerType(capiParamType)) {
680 dummyParamsStream << " " << capiParamType << ' ' << name << " = 0;\n";
681 } else {
682 // For unknown types, create a default-initialized variable
683 dummyParamsStream << " " << capiParamType << ' ' << name << " = {};\n";
684 }
685
686 paramListStream << ", " << name;
687 }
688
689 static constexpr char fmt[] = R"(
691TEST_F({2}{1}LinkTests, {0}_{3}_{4}) {{
692 auto test{1} = createIndex{1}();
693
694 if ({0}{1}IsA_{2}_{3}(test{1})) {{
695{5}
696 (void){0}{2}_{3}{4}(test{1}{6});
697 }
698
699 {7}(test{1});
700}
701)";
702 assert(!className.empty() && "className must be set");
703 os << llvm::formatv(
704 fmt,
705 getFunctionPrefix(), // {0}
706 kind, // {1}
707 dialectNameCapitalized, // {2}
708 className, // {3}
709 toPascalCase(method.methodName), // {4}
710 dummyParams, // {5}
711 paramList, // {6}
712 genCleanup() // {7}
713 );
714 }
