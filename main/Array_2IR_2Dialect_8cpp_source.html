29class ArrayValueCopyDialectInterface final : public llzk::ValueCopyDialectInterface {
30public:
31 explicit ArrayValueCopyDialectInterface(mlir::Dialect *owner)
32 : ValueCopyDialectInterface(owner) {}
33
34 bool canMaterializeValueCopy(mlir::Type type) const final {
35 auto arrayType = llvm::dyn_cast<llzk::array::ArrayType>(type);
36 return arrayType && (llvm::isa<mlir::NoneType>(arrayType.getElementType()) ||
37 (arrayType.hasStaticShape() &&
38 llzk::canMaterializeValueCopy(arrayType.getElementType())));
39 }
40
41 std::string getValueCopyFailureReason(mlir::Type type) const final {
42 auto arrayType = llvm::dyn_cast<llzk::array::ArrayType>(type);
43 if (!arrayType) {
44 return ValueCopyDialectInterface::getValueCopyFailureReason(type);
45 }
46 if (!arrayType.hasStaticShape()) {
47 return "dynamic or symbolic payload arrays cannot be copied independently; "
48 "flatten or resolve array shapes first";
49 }
50 return "array element copy is unsupported: " +
51 llzk::getValueCopyFailureReason(arrayType.getElementType());
52 }
53
54 mlir::FailureOr<mlir::Value> materializeValueCopy(
55 mlir::OpBuilder &builder, mlir::Location loc, mlir::Value source
56 ) const final {
57 auto arrayType = llvm::dyn_cast<llzk::array::ArrayType>(source.getType());
58 if (!arrayType || !canMaterializeValueCopy(arrayType)) {
59 return mlir::failure();
60 }
61
62 // Shape-only arrays have no element payload that can be independently mutated. Their shape is
63 // fixed by the value, so reusing the SSA value preserves copy semantics even for dynamic or
64 // symbolic shapes.
65 if (llvm::isa<mlir::NoneType>(arrayType.getElementType())) {
66 return source;
67 }
68
69 auto destination = llzk::array::CreateArrayOp::create(builder, loc, arrayType);
70 std::optional<mlir::SmallVector<mlir::ArrayAttr>> indices = arrayType.getSubelementIndices();
71 assert(indices.has_value() && "static arrays must provide concrete element indices");
72 for (mlir::ArrayAttr index : *indices) {
73 mlir::Value read = llzk::array::ArrayAccessOpInterface::genRead(builder, loc, source, index);
74 mlir::FailureOr<mlir::Value> copied = llzk::materializeValueCopy(builder, loc, read);
75 if (mlir::failed(copied)) {
76 return mlir::failure();
77 }
78 llzk::array::ArrayAccessOpInterface::genWrite(
79 builder, loc, destination.getResult(), index, *copied
80 );
81 }
82 return destination.getResult();
83 }
84};
