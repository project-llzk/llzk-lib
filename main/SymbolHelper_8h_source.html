170template <typename T>
171inline mlir::FailureOr<SymbolLookupResult<T>>
172resolveCallable(mlir::SymbolTableCollection &symbolTable, mlir::CallOpInterface call) {
173 mlir::CallInterfaceCallable callable = call.getCallableForCallee();
174 if (auto symbolVal = llvm::dyn_cast<mlir::Value>(callable)) {
175 return SymbolLookupResult<T>(symbolVal.getDefiningOp());
176 }
177
178 // If the callable isn't a value, lookup the symbol reference.
179 // We first try to resolve in the nearest symbol table, as per the default
180 // MLIR behavior. If the resulting operation is not found, we will then
181 // use the LLZK lookup helpers.
182 auto symbolRef = llvm::cast<mlir::SymbolRefAttr>(callable);
183 mlir::Operation *op = symbolTable.lookupNearestSymbolFrom(call.getOperation(), symbolRef);
184
185 if (op) {
186 return SymbolLookupResult<T>(std::move(op));
187 }
188 // Otherwise, use the top-level lookup.
189 return lookupTopLevelSymbol<T>(symbolTable, symbolRef, call.getOperation());
190}

196template <typename T>
197inline mlir::FailureOr<SymbolLookupResult<T>>
198resolveCallableSilently(mlir::SymbolTableCollection &symbolTable, mlir::CallOpInterface call) {
199 mlir::CallInterfaceCallable callable = call.getCallableForCallee();
200 if (auto symbolVal = llvm::dyn_cast<mlir::Value>(callable)) {
201 SymbolLookupResult<T> result(symbolVal.getDefiningOp());
202 if (!result) {
203 return mlir::failure();
204 }
205 return result;
206 }
207
208 auto symbolRef = llvm::cast<mlir::SymbolRefAttr>(callable);
209 if (mlir::Operation *op = symbolTable.lookupNearestSymbolFrom(call.getOperation(), symbolRef)) {
210 SymbolLookupResult<T> result(op);
211 if (!result) {
212 return mlir::failure();
213 }
214 return result;
215 }
216 return lookupTopLevelSymbol<T>(
217 symbolTable, symbolRef, call.getOperation(), /*reportMissing=*/false
218 );
219}

294template <std::ranges::input_range Range>
295mlir::LogicalResult verifyTypeResolution(
296 mlir::SymbolTableCollection &tables, mlir::Operation *origin, const Range &types
297) {
298 // Check all before returning to present all applicable type errors in one compilation.
299 bool failed = false;
300 for (const auto &t : types) {
301 failed |= mlir::failed(verifyTypeResolution(tables, origin, t));
302 }
303 return mlir::LogicalResult::failure(failed);
304}
