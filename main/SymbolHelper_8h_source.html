153template <typename T>
154inline mlir::FailureOr<SymbolLookupResult<T>>
155resolveCallable(mlir::SymbolTableCollection &symbolTable, mlir::CallOpInterface call) {
156 mlir::CallInterfaceCallable callable = call.getCallableForCallee();
157 if (auto symbolVal = llvm::dyn_cast<mlir::Value>(callable)) {
158 return SymbolLookupResult<T>(symbolVal.getDefiningOp());
159 }
160
161 // If the callable isn't a value, lookup the symbol reference.
162 // We first try to resolve in the nearest symbol table, as per the default
163 // MLIR behavior. If the resulting operation is not found, we will then
164 // use the LLZK lookup helpers.
165 auto symbolRef = llvm::cast<mlir::SymbolRefAttr>(callable);
166 mlir::Operation *op = symbolTable.lookupNearestSymbolFrom(call.getOperation(), symbolRef);
167
168 if (op) {
169 return SymbolLookupResult<T>(std::move(op));
170 }
171 // Otherwise, use the top-level lookup.
172 return lookupTopLevelSymbol<T>(symbolTable, symbolRef, call.getOperation());
173}

179template <typename T>
180inline mlir::FailureOr<SymbolLookupResult<T>>
181resolveCallableSilently(mlir::SymbolTableCollection &symbolTable, mlir::CallOpInterface call) {
182 mlir::CallInterfaceCallable callable = call.getCallableForCallee();
183 if (auto symbolVal = llvm::dyn_cast<mlir::Value>(callable)) {
184 SymbolLookupResult<T> result(symbolVal.getDefiningOp());
185 if (!result) {
186 return mlir::failure();
187 }
188 return result;
189 }
190
191 auto symbolRef = llvm::cast<mlir::SymbolRefAttr>(callable);
192 if (mlir::Operation *op = symbolTable.lookupNearestSymbolFrom(call.getOperation(), symbolRef)) {
193 SymbolLookupResult<T> result(op);
194 if (!result) {
195 return mlir::failure();
196 }
197 return result;
198 }
199 return lookupTopLevelSymbol<T>(
200 symbolTable, symbolRef, call.getOperation(), /*reportMissing=*/false
201 );
202}

277template <std::ranges::input_range Range>
278mlir::LogicalResult verifyTypeResolution(
279 mlir::SymbolTableCollection &tables, mlir::Operation *origin, const Range &types
280) {
281 // Check all before returning to present all applicable type errors in one compilation.
282 bool failed = false;
283 for (const auto &t : types) {
284 failed |= mlir::failed(verifyTypeResolution(tables, origin, t));
285 }
286 return mlir::LogicalResult::failure(failed);
287}
