54 Attribute resolveAttribute(Attribute attr) const {
55 if (auto bound = bindings.lookup(attr)) {
56 attr = bound;
57 }
58 if (auto type = dyn_cast<TypeAttr>(attr)) {
59 return TypeAttr::get(resolveType(type.getValue()));
60 }
61 return attr;
62 }
63
65 Type resolveType(Type type) const {
66 AttrTypeReplacer replacer;
67 replacer.addReplacement([this](TypeVarType variable) -> std::optional<Type> {
68 if (auto bound = dyn_cast_or_null<TypeAttr>(bindings.lookup(variable.getNameRef()))) {
69 return bound.getValue();
70 }
71 return std::nullopt;
72 });
73 replacer.addReplacement(
74 [this](StructType structure) -> std::optional<std::pair<Type, WalkResult>> {
75 SmallVector<Attribute> args;
76 if (!structure.getParams()) {
77 return std::make_pair(Type(structure), WalkResult::skip());
78 }
79 for (Attribute arg : structure.getParams()) {
80 args.push_back(resolveAttribute(arg));
81 }
82 return std::make_pair(
83 Type(
84 StructType::get(structure.getNameRef(), ArrayAttr::get(structure.getContext(), args))
85 ),
86 WalkResult::skip()
87 );
88 }
89 );
90 replacer.addReplacement(
91 [this](array::ArrayType array) -> std::optional<std::pair<Type, WalkResult>> {
92 SmallVector<Attribute> dimensions;
93 for (Attribute dimension : array.getDimensionSizes()) {
94 dimensions.push_back(resolveAttribute(dimension));
95 }
96 return std::make_pair(
97 Type(array::ArrayType::get(resolveType(array.getElementType()), dimensions)),
98 WalkResult::skip()
99 );
100 }
101 );
102 return replacer.replace(type);
103 }

106 Attribute resolveAffineArgument(AffineMapAttr attr, ArrayRef<Attribute> operands) const {
107 if (attr.getValue().getNumInputs() != operands.size() ||
108 llvm::any_of(operands, [](Attribute operand) {
109 return !llvm::isa_and_present<IntegerAttr>(operand);
110 })) {
111 return {};
112 }
113 SmallVector<Attribute> results;
114 bool poison = false;
115 if (failed(attr.getValue().constantFold(operands, results, &poison)) || poison ||
116 results.size() != 1) {
117 return {};
118 }
119 return results.front();
120 }

124 template <typename ResolveArgument>
125 FailureOr<StructType>
126 resolveAffineStructArguments(StructType structure, ResolveArgument &&resolveArgument) const {
127 SmallVector<Attribute> arguments;
128 for (Attribute argument : structure.getParams()) {
129 if (auto map = dyn_cast<AffineMapAttr>(argument)) {
130 argument = resolveArgument(map);
131 if (!argument) {
132 return failure();
133 }
134 }
135 arguments.push_back(argument);
136 }
137 return StructType::get(
138 structure.getNameRef(), ArrayAttr::get(structure.getContext(), arguments)
139 );
140 }

155class DiscoveryRun {
156 StructDefOp source;
157 StructSpecializationDiscovery::Bindings bindings;
158 SpecializationTypeResolver typeResolver;
159 SymbolTableCollection tables;
160 StructSpecializationDiscovery::Requests requests;
161 DenseSet<std::tuple<Operation *, Type, ArrayAttr, Type, unsigned>> seen;
162 DenseSet<std::pair<Operation *, Type>> enumeratedArrays;
163 using FamilyKey = std::pair<Operation *, ArrayAttr>;
164 DenseMap<FamilyKey, SmallVector<StructType>> familyCandidates;
165 DenseSet<FamilyKey> knownFamilies;
166 SmallVector<std::pair<function::CallOp, StructType>> rolledCalls;
167 DenseSet<std::pair<Operation *, Type>> seenRolledCalls;
168 uint64_t remaining;
169 SmallVectorImpl<Operation *> *visitedOperations;
170 StructSpecializationDiscovery::CallTargets *callTargets;

173 LogicalResult consumeStep(Operation *site) {
174 if (!remaining) {
175 return site->emitError("struct specialization discovery step limit exceeded");
176 }
177 --remaining;
178 return success();
179 }
180
182 LogicalResult recordRequest(
183 Operation *site, StructType type, ArrayRef<int64_t> indices = {}, StructType familyType = {},
184 StructSpecializationRequest::Kind kind = StructSpecializationRequest::Kind::Plain
185 ) {
186 if (type.getParams()) {
187 for (Attribute argument : type.getParams()) {
188 if (auto nested = dyn_cast<TypeAttr>(argument)) {
189 if (hasAffineMapAttr(nested.getValue())) {
190 return site->emitError("affine maps nested in struct type arguments are not supported");
191 }
192 }
193 if (!isConcreteStructParamAttr(argument)) {
194 return site->emitError("cannot resolve struct specialization argument ") << argument;
195 }
196 if (auto nested = dyn_cast<TypeAttr>(argument)) {
197 if (failed(collectTypes(site, nested.getValue()))) {
198 return failure();
199 }
200 }
201 }
202 }
203 if (type == llvm::cast<StructType>(typeResolver.resolveType(source.getType()))) {
204 return success();
205 }
206 SmallVector<Attribute> indexAttrs;
207 for (int64_t index : indices) {
208 indexAttrs.push_back(IntegerAttr::get(IndexType::get(source.getContext()), index));
209 }
210 auto requestKey = std::make_tuple(
211 site, Type(type), ArrayAttr::get(source.getContext(), indexAttrs), Type(familyType),
212 static_cast<unsigned>(kind)
213 );
214 if (!seen.insert(requestKey).second) {
215 return success();
216 }
217 requests.push_back({site, type, llvm::to_vector(indices), familyType, kind});
218 return success();
219 }
220
221 static bool hasDirectAffineArgument(StructType type) {
222 return type.getParams() && llvm::any_of(type.getParams(), [](Attribute argument) {
223 return isa<AffineMapAttr>(argument);
224 });
225 }
226
227 FailureOr<FamilyKey> getFamilyKey(Operation *site, StructType rolled) {
228 auto found = rolled.getDefinition(tables, site);
229 if (failed(found)) {
230 return failure();
231 }
232 return FamilyKey(found->get().getOperation(), rolled.getParams());
233 }
234
235 LogicalResult rememberFamily(Operation *site, StructType rolled) {
236 auto key = getFamilyKey(site, rolled);
237 if (failed(key)) {
238 return failure();
239 }
240 knownFamilies.insert(*key);
241 return success();
242 }
243
244 LogicalResult recordFamilyCandidate(Operation *site, StructType rolled, StructType concrete) {
245 auto key = getFamilyKey(site, rolled);
246 if (failed(key)) {
247 return failure();
248 }
249 knownFamilies.insert(*key);
250 auto &candidates = familyCandidates[*key];
251 if (!llvm::is_contained(candidates, concrete)) {
252 candidates.push_back(concrete);
253 }
254 return success();
255 }

259 LogicalResult
260 collectTypes(Operation *site, Type type, ArrayRef<int64_t> indices = {}, bool inArray = false) {
261 type = typeResolver.resolveType(type);
262 if (auto structure = dyn_cast<StructType>(type)) {
263 if (!hasDirectAffineArgument(structure)) {
264 return recordRequest(site, structure);
265 }
266 if (!inArray) {
267 if (isa<MemberDefOp>(site)) {
268 return site->emitError("affine struct type requires an enclosing array");
269 }
270 return success();
271 }
272 SmallVector<Attribute> operands;
273 for (int64_t index : indices) {
274 operands.push_back(IntegerAttr::get(IndexType::get(source.getContext()), index));
275 }
276 auto concrete = typeResolver.resolveAffineStructArguments(
277 structure, [this, site, &operands](AffineMapAttr map) -> Attribute {
278 if (map.getValue().getNumInputs() != operands.size()) {
279 site->emitError("affine struct argument requires one input per array dimension");
280 return {};
281 }
282 Attribute argument = typeResolver.resolveAffineArgument(map, operands);
283 if (!argument) {
284 site->emitError("cannot resolve affine array element specialization argument");
285 }
286 return argument;
287 }
288 );
289 if (failed(concrete)) {
290 return failure();
291 }
292 if (failed(recordFamilyCandidate(site, structure, *concrete))) {
293 return failure();
294 }
295 return recordRequest(
296 site, *concrete, indices, structure, StructSpecializationRequest::Kind::ArrayElement
297 );
298 }
299 if (auto array = dyn_cast<array::ArrayType>(type)) {
300 if (!hasAffineMapAttr(array.getElementType())) {
301 return collectTypes(site, array.getElementType());
302 }
303 if (!enumeratedArrays.insert({site, type}).second) {
304 return success();
305 }
306 SmallVector<int64_t> shape;
307 for (Attribute dimension : array.getDimensionSizes()) {
308 auto integer = dyn_cast<IntegerAttr>(dimension);
309 if (!integer || integer.getInt() < 0) {
310 return site->emitError("affine family discovery requires concrete array dimensions");
311 }
312 shape.push_back(integer.getInt());
313 }
314 if (llvm::is_contained(shape, 0)) {
315 bool invalid = false;
316 array.getElementType().walk([this, site, &invalid](Type nested) {
317 if (auto structure = dyn_cast<StructType>(nested);
318 structure && hasDirectAffineArgument(structure)) {
319 invalid |= failed(rememberFamily(site, structure));
320 }
321 });
322 return failure(invalid);
323 }
324 SmallVector<int64_t> elementIndices;
325 auto enumerate = [this, site, array, &shape,
326 &elementIndices](auto &&self, unsigned dimension) -> LogicalResult {
327 if (dimension < shape.size()) {
328 for (int64_t i = 0; i < shape[dimension]; ++i) {
329 elementIndices.push_back(i);
330 if (failed(self(self, dimension + 1))) {
331 return failure();
332 }
333 elementIndices.pop_back();
334 }
335 return success();
336 }
337 if (failed(consumeStep(site))) {
338 return failure();
339 }
340 return collectTypes(site, array.getElementType(), elementIndices, true);
341 };
342 return enumerate(enumerate, 0);
343 }
344 if (auto record = dyn_cast<pod::PodType>(type)) {
345 for (auto field : record.getRecords()) {
346 if (failed(collectTypes(site, field.getType(), indices, inArray))) {
347 return failure();
348 }
349 }
350 return success();
351 }
352 return failure(type.walk([this, site, indices, inArray](StructType structure) -> WalkResult {
353 return collectTypes(site, structure, indices, inArray);
354 }).wasInterrupted());
355 }

358 LogicalResult discoverCallTypes(function::CallOp call, const Environment &env) {
359 unsigned group = 0;
360 for (Type result : call.getResultTypes()) {
361 auto structure = dyn_cast<StructType>(typeResolver.resolveType(result));
362 if (!structure || !structure.getParams()) {
363 if (failed(collectTypes(call, result))) {
364 return failure();
365 }
366 continue;
367 }
368 auto concrete = typeResolver.resolveAffineStructArguments(
369 structure, [this, &call, &env, &group](AffineMapAttr map) -> Attribute {
370 if (group >= call.getMapOperands().size()) {
371 call.emitError("missing affine specialization operands");
372 return {};
373 }
374 SmallVector<Attribute> operands;
375 for (Value operand : call.getMapOperands()[group++]) {
376 operands.push_back(env.lookup(operand));
377 }
378 Attribute argument = typeResolver.resolveAffineArgument(map, operands);
379 if (!argument) {
380 call.emitError("cannot resolve affine struct specialization argument");
381 }
382 return argument;
383 }
384 );
385 if (failed(concrete)) {
386 return failure();
387 }
388 bool rolledResult = hasDirectAffineArgument(structure);
389 if (rolledResult) {
390 if (failed(recordFamilyCandidate(call, structure, *concrete))) {
391 return failure();
392 }
393 }
394 auto kind = rolledResult ? StructSpecializationRequest::Kind::RolledCallResult
395 : StructSpecializationRequest::Kind::Plain;
396 if (failed(recordRequest(call, *concrete, {}, {}, kind))) {
397 return failure();
398 }
399 }
400 for (Type argument : call.getArgOperands().getTypes()) {
401 if (auto structure = dyn_cast<StructType>(typeResolver.resolveType(argument));
402 structure && hasDirectAffineArgument(structure)) {
403 if (seenRolledCalls.insert({call, structure}).second) {
404 rolledCalls.emplace_back(call, structure);
405 }
406 }
407 }
408 return success();
409 }

412 static std::optional<int64_t> getKnownInteger(Attribute attr) {
413 auto value = dyn_cast_or_null<IntegerAttr>(attr);
414 if (!value || value.getValue().getSignificantBits() > 64) {
415 return std::nullopt;
416 }
417 return value.getInt();
418 }

421 static void bindResults(ValueRange results, ArrayRef<Attribute> values, Environment &env) {
422 for (auto [result, value] : llvm::zip_equal(results, values)) {
423 env[result] = value;
424 }

429 LogicalResult
430 evaluateIf(scf::IfOp branch, const Environment &env, SmallVector<Attribute> &results) {
431 auto condition = getKnownInteger(env.lookup(branch.getCondition()));
432 SmallVector<Attribute> left, right;
433 if (!condition || *condition) {
434 if (failed(evaluateBlock(branch.getThenRegion().front(), env, left))) {
435 return failure();
436 }
437 }
438 if ((!condition || !*condition) && !branch.getElseRegion().empty()) {
439 if (failed(evaluateBlock(branch.getElseRegion().front(), env, right))) {
440 return failure();
441 }
442 }
443 if (condition) {
444 results = *condition ? left : right;
445 } else {
446 for (unsigned i = 0; i < results.size(); ++i) {
447 results[i] = left[i] == right[i] ? left[i] : Attribute();
448 }
449 }
450 return success();
451 }
452
455 LogicalResult
456 evaluateFor(scf::ForOp loop, const Environment &env, SmallVector<Attribute> &results) {
457 auto lower = getKnownInteger(env.lookup(loop.getLowerBound()));
458 auto upper = getKnownInteger(env.lookup(loop.getUpperBound()));
459 auto stride = getKnownInteger(env.lookup(loop.getStep()));
460 if (!lower || !upper || !stride || *stride <= 0 || loop.getUnsignedCmp()) {
461 return loop.emitError(
462 "struct discovery requires known signed for bounds and a positive step"
463 );
464 }
465 results.clear();
466 for (Value initial : loop.getInitArgs()) {
467 results.push_back(env.lookup(initial));
468 }
469 for (int64_t i = *lower; i < *upper;) {
470 auto local = env;
471 local[loop.getInductionVar()] = IntegerAttr::get(loop.getInductionVar().getType(), i);
472 bindResults(loop.getRegionIterArgs(), results, local);
473 results.clear();
474 if (failed(evaluateBlock(*loop.getBody(), std::move(local), results))) {
475 return failure();
476 }
477 if (i > std::numeric_limits<int64_t>::max() - *stride) {
478 return loop.emitError("struct discovery loop induction overflow");
479 }
480 i += *stride;
481 }
482 return success();
483 }
484
487 LogicalResult
488 evaluateWhile(scf::WhileOp loop, const Environment &env, SmallVector<Attribute> &results) {
489 SmallVector<Attribute> carried;
490 for (Value initial : loop.getInits()) {
491 carried.push_back(env.lookup(initial));
492 }
493 while (true) {
494 auto before = env;
495 bindResults(loop.getBefore().front().getArguments(), carried, before);
496 SmallVector<Attribute> forwarded;
497 Attribute condition;
498 if (failed(
499 evaluateBlock(loop.getBefore().front(), std::move(before), forwarded, &condition)
500 )) {
501 return failure();
502 }
503 auto known = dyn_cast_or_null<IntegerAttr>(condition);
504 if (!known || !known.getType().isInteger(1)) {
505 return loop.emitError("struct discovery requires a known while condition");
506 }
507 if (known.getValue().isZero()) {
508 results = std::move(forwarded);
509 return success();
510 }
511 auto after = env;
512 bindResults(loop.getAfter().front().getArguments(), forwarded, after);
513 carried.clear();
514 if (failed(evaluateBlock(loop.getAfter().front(), std::move(after), carried))) {
515 return failure();
516 }
517 }
518 }

530 void evaluateFoldableOp(Operation &op, const Environment &env, SmallVector<Attribute> &results) {
531 SmallVector<Attribute> operands;
532 for (Value operand : op.getOperands()) {
533 operands.push_back(env.lookup(operand));
534 }
535 if (llvm::all_of(operands, [](Attribute attr) { return attr != nullptr; })) {
536 SmallVector<OpFoldResult> folded;
537 if (succeeded(op.fold(operands, folded)) && folded.size() == results.size()) {
538 for (auto [index, value] : llvm::enumerate(folded)) {
539 if (auto attr = dyn_cast<Attribute>(value)) {
540 results[index] = attr;
541 } else {
542 results[index] = env.lookup(cast<Value>(value));
543 }
544 }
545 }
546 }
547 }

551 LogicalResult
552 evaluateOperation(Operation &op, const Environment &env, SmallVector<Attribute> &results) {
553 if (visitedOperations) {
554 visitedOperations->push_back(&op);
555 }
556 if (auto branch = dyn_cast<scf::IfOp>(op)) {
557 if (failed(evaluateIf(branch, env, results))) {
558 return failure();
559 }
560 } else if (auto forLoop = dyn_cast<scf::ForOp>(op)) {
561 if (failed(evaluateFor(forLoop, env, results))) {
562 return failure();
563 }
564 } else if (auto whileLoop = dyn_cast<scf::WhileOp>(op)) {
565 if (failed(evaluateWhile(whileLoop, env, results))) {
566 return failure();
567 }
568 } else if (op.getNumRegions()) {
569 return op.emitError("unsupported control flow in struct specialization discovery");
570 } else {
571 // The environment tracks scalar constants produced by template reads and
572 // folding. Resolving global reads requires looking up global initializers;
573 // resolving POD, array and struct-member reads requires tracking stored values
574 // and writes. Those reads currently produce unknown values, including constant
575 // global reads, which have no fold hook. Type uses are still collected below.
576 // Unknown values are diagnosed when needed for a loop bound or specialization
577 // argument; unknown if conditions cause both branches to be explored.
578 if (auto read = dyn_cast<ConstReadOp>(op)) {
579 auto value = evaluateConstRead(read);
580 if (failed(value)) {
581 return failure();
582 }
583 results[0] = *value;
584 } else if (isMemoryEffectFree(&op)) {
585 evaluateFoldableOp(op, env, results);
586 }
587 if (auto call = dyn_cast<function::CallOp>(op)) {
588 return discoverCallTypes(call, env);
589 }
590 }
591 for (Type type : op.getResultTypes()) {
592 if (failed(collectTypes(&op, type))) {
593 return failure();
594 }
595 }
596 return success();
597 }
598
602 LogicalResult evaluateBlock(
603 Block &body, Environment env, SmallVectorImpl<Attribute> &yielded,
604 Attribute *condition = nullptr
605 ) {
606 for (Operation &op : body) {
607 if (failed(consumeStep(&op))) {
608 return failure();
609 }
610 if (auto terminator = dyn_cast<scf::ConditionOp>(op)) {
611 if (!condition) {
612 return op.emitError("scf.condition outside a while condition region");
613 }
614 *condition = env.lookup(terminator.getCondition());
615 for (Value operand : terminator.getArgs()) {
616 yielded.push_back(env.lookup(operand));
617 }
618 return success();
619 }
620 if (isa<YieldOp, scf::YieldOp, function::ReturnOp>(op)) {
621 for (Value operand : op.getOperands()) {
622 yielded.push_back(env.lookup(operand));
623 }
624 return success();
625 }
626 if (op.hasTrait<OpTrait::IsTerminator>()) {
627 return op.emitError("unsupported control flow in struct specialization discovery");
628 }
629 SmallVector<Attribute> results(op.getNumResults());
630 if (failed(evaluateOperation(op, env, results))) {
631 return failure();
632 }
633 bindResults(op.getResults(), results, env);
634 }
635 return success();
636 }

638public:
639 DiscoveryRun(
640 StructDefOp structure, const StructSpecializationDiscovery::Bindings &parameters,
641 uint64_t limit, SmallVectorImpl<Operation *> *visited,
642 StructSpecializationDiscovery::CallTargets *targets = nullptr
643 )
644 : source(structure), bindings(parameters), typeResolver(bindings), remaining(limit),
645 visitedOperations(visited), callTargets(targets) {}
646
647 uint64_t getRemainingSteps() const { return remaining; }
648
650 FailureOr<StructSpecializationDiscovery::Bindings> evaluateBindings() {
651 if (auto templ = source->getParentOfType<TemplateOp>()) {
652 for (auto expr : templ.getConstOps<TemplateExprOp>()) {
653 SmallVector<Attribute> values;
654 if (failed(evaluateBlock(expr.getInitializerRegion().front(), Environment(), values))) {
655 return failure();
656 }
657 if (values.size() == 1 && values[0]) {
658 bindings[FlatSymbolRefAttr::get(expr.getSymNameAttr())] = values[0];
659 }
660 }
661 }
662 return bindings;
663 }
664
666 FailureOr<StructSpecializationDiscovery::Requests> run() {
667 if (failed(evaluateBindings())) {
668 return failure();
669 }
670 for (auto member : source.getOps<MemberDefOp>()) {
671 if (failed(collectTypes(member, member.getType()))) {
672 return failure();
673 }
674 }
675 for (auto function : source.getOps<function::FuncDefOp>()) {
676 if (!function.getBody().hasOneBlock()) {
677 return function.emitError(
678 "struct discovery requires a defined single-block method; unstructured control flow "
679 "is unsupported"
680 );
681 }
682 for (Type type : function.getFunctionType().getInputs()) {
683 if (failed(collectTypes(function, type))) {
684 return failure();
685 }
686 }
687 for (Type type : function.getFunctionType().getResults()) {
688 if (failed(collectTypes(function, type))) {
689 return failure();
690 }
691 }
692 SmallVector<Attribute> values;
693 if (failed(evaluateBlock(function.getBody().front(), Environment(), values))) {
694 return failure();
695 }
696 }
697 for (auto [call, family] : rolledCalls) {
698 auto key = getFamilyKey(call, family);
699 if (failed(key)) {
700 return failure();
701 }
702 if (!knownFamilies.contains(*key)) {
703 return call.emitError("no known source for affine struct call argument ") << family;
704 }
705 auto candidates = familyCandidates.lookup(*key);
706 for (StructType candidate : candidates) {
707 if (failed(recordRequest(call, candidate))) {
708 return failure();
709 }
710 }
711 if (callTargets) {
712 callTargets->push_back({call, candidates});
713 }
714 }
715 return std::move(requests);
716 }
717};

721FailureOr<Attribute> llzk::polymorphic::detail::resolveConstReadBinding(
722 ConstReadOp read, Attribute binding, Type readType
723) {
724 if (!binding) {
725 return binding;
726 }
727 if (auto constant = dyn_cast<felt::FeltConstAttr>(binding)) {
728 if (constant.getType() != readType) {
729 return read.emitError("cannot read felt template binding ")
730 << read.getConstNameAttr() << " of type " << constant.getType() << " as " << readType
731 << (isa<felt::FeltType>(readType) ? "; read the binding using its declared felt type"
732 : "; use an explicit cast");
733 }
734 if (constant.getType().hasField()) {
735 const Field &field = constant.getType().getField();
736 return Attribute(
737 felt::FeltConstAttr::get(
738 read.getContext(), toAPInt(field.reduce(constant.getValue()), field.bitWidth()),
739 constant.getType()
740 )
741 );
742 }
743 return binding;
744 }
745 if (auto integer = dyn_cast<IntegerAttr>(binding)) {
746 const APInt &number = integer.getValue();
747 if (auto type = dyn_cast<felt::FeltType>(readType)) {
748 // An i1's set bit represents one; index values have signed semantics.
749 auto value = integer.getType().isSignlessInteger(1)
750 ? llvm::DynamicAPInt(number.getZExtValue())
751 : llvm::DynamicAPInt(number);
752 if (type.hasField()) {
753 const Field &field = type.getField();
754 return Attribute(
755 felt::FeltConstAttr::get(
756 read.getContext(), toAPInt(field.reduce(value), field.bitWidth()), type
757 )
758 );
759 }
760 if (value < 0) {
761 return read.emitError("cannot read negative template binding ")
762 << read.getConstNameAttr() << " as felt without a known field modulus";
763 }
764 return Attribute(felt::FeltConstAttr::get(read.getContext(), number, type));
765 }
766 if (readType.isSignlessInteger(1)) {
767 if (!number.isZero() && !number.isOne()) {
768 return read.emitError("cannot read template binding ")
769 << read.getConstNameAttr() << " with value " << binding
770 << " as i1: expected zero or one; use an explicit cast";
771 }
772 return Attribute(IntegerAttr::get(readType, number.isOne() ? 1 : 0));
773 }
774 if (readType.isIndex()) {
775 auto bits = integer.getType().isSignlessInteger(1) ? number.zext(64) : number.sextOrTrunc(64);
776 return Attribute(IntegerAttr::get(readType, bits));
777 }
778 }
779 return read.emitError("cannot read template binding ")
780 << read.getConstNameAttr() << " with value \"" << binding << "\" as type " << readType;
781}
