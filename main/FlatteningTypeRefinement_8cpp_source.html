43FailureOr<Attribute> getCommonRefinedParamAttr(
44 Attribute lhs, Attribute rhs, const ConversionTracker &tracker, bool unifyDynamicSize,
45 Operation *scope
46) {
47 assertValidAttrForParamOfType(lhs);
48 assertValidAttrForParamOfType(rhs);
49 if (lhs == rhs) {
50 return lhs;
51 }
52
53 if (auto lhsAffine = llvm::dyn_cast<AffineMapAttr>(lhs)) {
54 if (auto rhsInt = llvm::dyn_cast<IntegerAttr>(rhs)) {
55 if (!isDynamic(rhsInt)) {
56 return rhs;
57 }
58 }
59 }
60 if (auto rhsAffine = llvm::dyn_cast<AffineMapAttr>(rhs)) {
61 if (auto lhsInt = llvm::dyn_cast<IntegerAttr>(lhs)) {
62 if (!isDynamic(lhsInt)) {
63 return lhs;
64 }
65 }
66 }
67
68 if (unifyDynamicSize) {
69 auto dynamicInt = [](Attribute attr) -> IntegerAttr {
70 auto intAttr = llvm::dyn_cast<IntegerAttr>(attr);
71 return intAttr && isDynamic(intAttr) ? intAttr : nullptr;
72 };
73 if (dynamicInt(lhs) && llvm::isa_and_present<IntegerAttr, SymbolRefAttr, AffineMapAttr>(rhs)) {
74 return rhs;
75 }
76 if (dynamicInt(rhs) && llvm::isa_and_present<IntegerAttr, SymbolRefAttr, AffineMapAttr>(lhs)) {
77 return lhs;
78 }
79 }
80
81 auto lhsTy = llvm::dyn_cast<TypeAttr>(lhs);
82 auto rhsTy = llvm::dyn_cast<TypeAttr>(rhs);
83 if (lhsTy && rhsTy) {
84 auto commonType = getCommonRefinedType(lhsTy.getValue(), rhsTy.getValue(), tracker, scope);
85 if (succeeded(commonType)) {
86 return TypeAttr::get(*commonType);
87 }
88 }
89 return failure();
90}

93FailureOr<Type>
94getCommonRefinedType(Type lhs, Type rhs, const ConversionTracker &tracker, Operation *scope) {
95 if (scope) {
96 auto common =
97 getCommonRefinedType(tracker.scopeType(lhs, scope), tracker.scopeType(rhs, scope), tracker);
98 return failed(common) ? FailureOr<Type>(failure()) : tracker.scopeType(*common, scope, false);
99 }
100 if (lhs == rhs) {
101 return lhs;
102 }
103 if (tracker.isLegalConversion(lhs, rhs, "getCommonRefinedType")) {
104 return rhs;
105 }
106 if (tracker.isLegalConversion(rhs, lhs, "getCommonRefinedType")) {
107 return lhs;
108 }
109
110 if (auto lhsStruct = llvm::dyn_cast<StructType>(lhs)) {
111 auto rhsStruct = llvm::dyn_cast<StructType>(rhs);
112 if (!rhsStruct) {
113 return failure();
114 }
115 if (lhsStruct.getNameRef() != rhsStruct.getNameRef()) {
116 std::optional<StructType> lhsPreimage = tracker.getPreimage(lhsStruct);
117 std::optional<StructType> rhsPreimage = tracker.getPreimage(rhsStruct);
118 if (!lhsPreimage && !rhsPreimage) {
119 return failure();
120 }
121 auto commonPreimage = getCommonRefinedType(
122 lhsPreimage.value_or(lhsStruct), rhsPreimage.value_or(rhsStruct), tracker
123 );
124 if (failed(commonPreimage)) {
125 return failure();
126 }
127 if (auto commonStruct = llvm::dyn_cast<StructType>(*commonPreimage)) {
128 if (std::optional<StructType> instantiatedCommon = tracker.getInstantiation(commonStruct)) {
129 return *instantiatedCommon;
130 }
131 }
132 return *commonPreimage;
133 }
134
135 ArrayAttr lhsParams = lhsStruct.getParams();
136 ArrayAttr rhsParams = rhsStruct.getParams();
137 ArrayRef<Attribute> emptyParams;
138 ArrayRef<Attribute> lhsValues = lhsParams ? lhsParams.getValue() : emptyParams;
139 ArrayRef<Attribute> rhsValues = rhsParams ? rhsParams.getValue() : emptyParams;
140 if (lhsValues.size() != rhsValues.size()) {
141 return failure();
142 }
143
144 SmallVector<Attribute> commonParams;
145 for (auto [lhsParam, rhsParam] : llvm::zip_equal(lhsValues, rhsValues)) {
146 auto commonParam = getCommonRefinedParamAttr(lhsParam, rhsParam, tracker);
147 if (failed(commonParam)) {
148 return failure();
149 }
150 commonParams.push_back(*commonParam);
151 }
152 StructType commonStruct =
153 commonParams.empty()
154 ? StructType::get(lhsStruct.getNameRef())
155 : StructType::get(
156 lhsStruct.getNameRef(), ArrayAttr::get(lhs.getContext(), commonParams)
157 );
158 if (std::optional<StructType> instantiatedCommon = tracker.getInstantiation(commonStruct)) {
159 return *instantiatedCommon;
160 }
161 return commonStruct;
162 }
163
164 if (auto lhsPod = llvm::dyn_cast<pod::PodType>(lhs)) {
165 auto rhsPod = llvm::dyn_cast<pod::PodType>(rhs);
166 if (!rhsPod || lhsPod.getRecords().size() != rhsPod.getRecords().size()) {
167 return failure();
168 }
169 SmallVector<pod::RecordAttr> commonRecords;
170 for (auto [lhsRecord, rhsRecord] : llvm::zip_equal(lhsPod.getRecords(), rhsPod.getRecords())) {
171 if (lhsRecord.getName() != rhsRecord.getName()) {
172 return failure();
173 }
174 auto commonRecord = getCommonRefinedType(lhsRecord.getType(), rhsRecord.getType(), tracker);
175 if (failed(commonRecord)) {
176 return failure();
177 }
178 commonRecords.push_back(
179 pod::RecordAttr::get(lhs.getContext(), lhsRecord.getName(), *commonRecord)
180 );
181 }
182 return pod::PodType::get(lhs.getContext(), commonRecords);
183 }
184
185 if (auto lhsArray = llvm::dyn_cast<ArrayType>(lhs)) {
186 auto rhsArray = llvm::dyn_cast<ArrayType>(rhs);
187 if (!rhsArray) {
188 return failure();
189 }
190
191 auto commonElement =
192 getCommonRefinedType(lhsArray.getElementType(), rhsArray.getElementType(), tracker);
193 if (failed(commonElement)) {
194 return failure();
195 }
196 ArrayRef<Attribute> lhsDims = lhsArray.getDimensionSizes();
197 ArrayRef<Attribute> rhsDims = rhsArray.getDimensionSizes();
198 if (lhsDims.size() != rhsDims.size()) {
199 return failure();
200 }
201
202 SmallVector<Attribute> commonDims;
203 for (auto [lhsDim, rhsDim] : llvm::zip_equal(lhsDims, rhsDims)) {
204 auto commonDim =
205 getCommonRefinedParamAttr(lhsDim, rhsDim, tracker, /*unifyDynamicSize=*/true);
206 if (failed(commonDim)) {
207 return failure();
208 }
209 commonDims.push_back(*commonDim);
210 }
211 return ArrayType::get(*commonElement, commonDims);
212 }
213
214 return failure();
215}

218bool callSignatureUnifies(
219 FunctionType replacement, FunctionType callee, ArrayRef<StringRef> calleeNamespace,
220 const ConversionTracker &tracker, bool allowUnresolvedStructs, Operation *scope
221) {
222 // Qualify declaration-local names before consulting the caller's instantiation history.
223 AttrTypeReplacer qualify;
224 qualify.addReplacement([&](StructType type) -> std::optional<Type> {
225 SymbolRefAttr name = type.getNameRef();
226 for (StringRef part : calleeNamespace) {
227 name = asSymbolRefAttr(StringAttr::get(type.getContext(), part), name);
228 }
229 return StructType::get(name, type.getParams());
230 });
231 replacement = llvm::cast<FunctionType>(tracker.scopeType(replacement, scope));
232 callee = llvm::cast<FunctionType>(tracker.scopeType(qualify.replace(callee), scope));
233 calleeNamespace = {};
234
235 // Partial instantiations can already share a family even when only one signature has
236 // an exact recorded preimage. Preserve that agreement before restoring original types.
237 if (allowUnresolvedStructs &&
238 functionTypesUnifyConsistently(replacement, callee, calleeNamespace)) {
239 return true;
240 }
241 if (allowUnresolvedStructs) {
242 // Keep partial families aligned while restoring fully instantiated nested arguments.
243 AttrTypeReplacer concretePreimages;
244 concretePreimages.addReplacement([&](StructType type) -> std::optional<Type> {
245 return isNullOrEmpty(type.getParams()) ? tracker.getPreimage(type) : std::nullopt;
246 });
247 if (functionTypesUnifyConsistently(
248 llvm::cast<FunctionType>(concretePreimages.replace(replacement)),
249 llvm::cast<FunctionType>(concretePreimages.replace(callee)), calleeNamespace
250 )) {
251 return true;
252 }
253 }
254 // A proposed read refinement must agree with already-instantiated companions. Before function
255 // instantiation, however, a generic companion can still refine later: compare parameterized
256 // preimages to distinguish that case from two incompatible concrete specializations.
257 AttrTypeReplacer instantiations;
258 instantiations.addReplacement([&](StructType type) -> std::optional<Type> {
259 auto mapped =
260 allowUnresolvedStructs ? tracker.getPreimage(type) : tracker.getInstantiation(type);
261 if (mapped) {
262 return *mapped;
263 }
264 return std::nullopt;
265 });
266 replacement = llvm::cast<FunctionType>(instantiations.replace(replacement));
267 callee = llvm::cast<FunctionType>(instantiations.replace(callee));
268 return functionTypesUnifyConsistently(replacement, callee, calleeNamespace);
269}

272FunctionType getRefinedCallSignature(
273 CallOp call, Value refinedValue, Type refinedType,
274 const DenseMap<Value, Type> *pendingRefinements
275) {
276 SmallVector<Type> argumentTypes;
277 for (Value argument : call.getArgOperands()) {
278 Type type = argument.getType();
279 if (pendingRefinements) {
280 if (Type pending = pendingRefinements->lookup(argument)) {
281 type = pending;
282 }
283 }
284 if (argument == refinedValue) {
285 type = refinedType;
286 }
287 argumentTypes.push_back(type);
288 }
289 return FunctionType::get(call.getContext(), argumentTypes, call.getResultTypes());
290}

292FailureOr<Type> getRefinedMemberReadType(
293 MemberReadOp read, Type receiverType, SymbolTableCollection &tables,
294 const ConversionTracker &tracker, Type *projectedMemberType
295) {
296 auto structure = dyn_cast<StructType>(receiverType);
297 if (!structure) {
298 return failure();
299 }
300 while (auto instantiated = tracker.getInstantiation(structure, read)) {
301 structure = *instantiated;
302 }
303 auto receiver = structure.getDefinition(tables, read);
304 if (failed(receiver)) {
305 return failure();
306 }
307 StructType declaredReceiver = receiver->get().getType();
308 SmallVector<StringRef> receiverNamespace(receiver->getNamespace());
309 auto member = lookupSymbolIn<MemberDefOp>(
310 tables, SymbolRefAttr::get(read.getContext(), read.getMemberName()), std::move(*receiver),
311 read
312 );
313 if (failed(member)) {
314 return failure();
315 }
316 member->prependNamespace(receiverNamespace);
317 // Re-spell declaration-local struct names before inserting caller-local parameter bindings.
318 AttrTypeReplacer qualify;
319 qualify.addReplacement([&](StructType type) -> std::optional<Type> {
320 SymbolRefAttr name = type.getNameRef();
321 for (StringRef part : member->getNamespace()) {
322 name = asSymbolRefAttr(StringAttr::get(read.getContext(), part), name);
323 }
324 return StructType::get(name, type.getParams());
325 });
326 Type memberType = qualify.replace(member->get().getType());
327 if (ArrayAttr params = structure.getParams()) {
328 ArrayAttr names = declaredReceiver.getParams();
329 if (!names || names.size() != params.size()) {
330 return failure();
331 }
332 DenseMap<Attribute, Attribute> bindings;
333 for (auto [name, value] : llvm::zip_equal(names, params)) {
334 bindings[name] = value;
335 }
336 memberType = TemplateTypeConverter(std::move(bindings)).convertType(memberType);
337 }
338 if (projectedMemberType) {
339 *projectedMemberType = memberType;
340 }
341 return getCommonRefinedType(read.getVal().getType(), memberType, tracker, read);
342}

345static Type plannedType(Value value, const DenseMap<Value, Type> *pending) {
346 if (pending) {
347 if (Type type = pending->lookup(value)) {
348 return type;
349 }
350 }
351 return value.getType();

354void extendPendingRefinements(
355 ModuleOp module, DenseMap<Value, Type> &pending, const ConversionTracker &tracker
356) {
357 SymbolTableCollection tables;
358 bool changed;
359 do {
360 changed = false;
361 module.walk([&](Operation *op) {
362 Value result;
363 Type type;
364 if (auto create = dyn_cast<CreateArrayOp>(op)) {
365 if (create.getElements().empty() || !llvm::any_of(create.getElements(), [&](Value value) {
366 return pending.contains(value);
367 })) {
368 return;
369 }
370 Type element = plannedType(create.getElements().front(), &pending);
371 if (!llvm::all_of(create.getElements(), [&](Value value) {
372 return plannedType(value, &pending) == element;
373 })) {
374 return;
375 }
376 result = create.getResult();
377 type = create.getType().cloneWith(element);
378 } else if (auto read = dyn_cast<ReadArrayOp>(op)) {
379 auto array = dyn_cast_or_null<ArrayType>(pending.lookup(read.getArrRef()));
380 if (!array) {
381 return;
382 }
383 result = read.getResult();
384 type = array.getElementType();
385 } else if (auto extract = dyn_cast<ExtractArrayOp>(op)) {
386 auto array = dyn_cast_or_null<ArrayType>(pending.lookup(extract.getArrRef()));
387 if (!array) {
388 return;
389 }
390 result = extract.getResult();
391 type = array.getSelectionType(extract.getIndices().size());
392 } else if (auto memberRead = dyn_cast<MemberReadOp>(op)) {
393 Type receiver = pending.lookup(memberRead.getComponent());
394 if (!receiver) {
395 return;
396 }
397 auto refined = getRefinedMemberReadType(memberRead, receiver, tables, tracker);
398 if (failed(refined)) {
399 return;
400 }
401 result = memberRead.getVal();
402 type = *refined;
403 }
404 if (result && type != plannedType(result, &pending) &&
405 canUseScalarizedValueAsType(
406 plannedType(result, &pending), type, tracker, "extendPendingRefinements", op
407 )) {
408 pending[result] = type;
409 changed = true;
410 }
411 });
412 } while (changed);
413}

417bool canUseScalarizedValueAsType(
418 Type sourceType, Type targetType, const ConversionTracker &tracker, const char *patName,
419 Operation *scope
420) {
421 if (sourceType == targetType ||
422 tracker.isLegalConversion(sourceType, targetType, patName, scope)) {
423 return true;
424 }
425 auto commonType = getCommonRefinedType(sourceType, targetType, tracker, scope);
426 return succeeded(commonType) && *commonType == targetType;
427}

430static bool isFullyConcreteArrayConsumerType(Type type) {
431 auto arrayType = llvm::dyn_cast<ArrayType>(type);
432 if (!arrayType) {
433 return isConcreteType(type, /*allowStructParams=*/false);
434 }
435 return llvm::all_of(arrayType.getDimensionSizes(), [](Attribute attr) {
436 return isConcreteStructParamAttr(attr);
437 }) && isFullyConcreteArrayConsumerType(arrayType.getElementType());
438}

443bool canRefineConsumerUse(
444 OpOperand &use, Type refinedType, SymbolTableCollection &tables,
445 const ConversionTracker &tracker, ConsumerTypeCheck typeCheck,
446 const DenseMap<Value, Type> *pendingRefinements, Operation *diagnosticOp
447) {
448 Operation *user = use.getOwner();
449 Value value = use.get();
450 auto eventualType = [&](Value operand) {
451 return operand == value ? refinedType : plannedType(operand, pendingRefinements);
452 };
453 auto accepts = [&](Type requiredType) {
454 switch (typeCheck) {
455 case ConsumerTypeCheck::Exact:
456 return refinedType == requiredType;
457 case ConsumerTypeCheck::ConcreteExact:
458 return !isFullyConcreteArrayConsumerType(requiredType) || refinedType == requiredType;
459 case ConsumerTypeCheck::Refinement:
460 return canUseScalarizedValueAsType(
461 refinedType, requiredType, tracker, "ConsumerRefinement", user
462 );
463 case ConsumerTypeCheck::Unification:
464 return typesUnify(refinedType, requiredType);
465 }
466 llvm_unreachable("unknown consumer type check");
467 };
468 auto incompatible = [&](StringRef contract, Operation *observation,
469 function_ref<void(Diagnostic &)> explain) {
470 if (diagnosticOp) {
471 auto diag = diagnosticOp->emitError()
472 << "cannot scalarize heterogeneous array read because its index-specific value "
473 "type is incompatible with "
474 << contract;
475 explain(diag.attachNote(observation->getLoc()));
476 }
477 return false;
478 };
479 auto requiresType = [&](Type requiredType, StringRef contract, StringRef description,
480 Operation *observation) {
481 if (accepts(requiredType)) {
482 return true;
483 }
484 return incompatible(contract, observation, [&](Diagnostic &note) {
485 note << description << " requires " << requiredType << ", but this read is replaced with "
486 << refinedType;
487 });
488 };
489
490 // Independently generic requirements remain valid until their owning producer is refined.
491 // Exact propagation paths deliberately require agreement even with a generic declaration.
492 auto requiresDeclaredType = [&](Type requiredType, StringRef contract, StringRef description) {
493 return (typeCheck != ConsumerTypeCheck::Exact &&
494 !isFullyConcreteArrayConsumerType(requiredType)) ||
495 requiresType(requiredType, contract, description, user);
496 };
497
498 if (auto equality = dyn_cast<EmitEqualityOp>(user)) {
499 Type otherType =
500 eventualType(use.getOperandNumber() == 0 ? equality.getRhs() : equality.getLhs());
501 if (!isValidEmitEqType(refinedType) || !isValidEmitEqType(otherType)) {
502 return incompatible("an equality constraint", user, [&](Diagnostic &note) {
503 note << "equality operands must exclude struct and string types, but replacement operands "
504 "are "
505 << refinedType << " and " << otherType;
506 });
507 }
508 if ((typeCheck == ConsumerTypeCheck::Refinement &&
509 !isFullyConcreteArrayConsumerType(otherType)) ||
510 accepts(otherType)) {
511 return true;
512 }
513 return incompatible("an equality constraint", user, [&](Diagnostic &note) {
514 note << "other equality operand requires " << otherType << ", but this read is replaced with "
515 << refinedType;
516 });
517 }
518 if (auto containment = dyn_cast<EmitContainmentOp>(user)) {
519 Type lhsType = eventualType(containment.getLhs());
520 Type rhsType = eventualType(containment.getRhs());
521 if (areValidContainmentOperandTypes(lhsType, rhsType)) {
522 return true;
523 }
524 return incompatible("a containment constraint", user, [&](Diagnostic &note) {
525 note << "replacement containment operands " << lhsType << " and " << rhsType
526 << " must form an array and a compatible element or subarray";
527 });
528 }
529 if (auto call = dyn_cast<CallOp>(user)) {
530 unsigned argIdx = use.getOperandNumber() - call.getArgOperands().getBeginOperandIndex();
531 if (argIdx >= call.getArgOperands().size()) {
532 return false;
533 }
534 auto callee = call.getCalleeTarget(tables);
535 if (failed(callee)) {
536 return false;
537 }
538 FunctionType signature = getRefinedCallSignature(call, value, refinedType, pendingRefinements);
539 FunctionType declared = callee->get().getFunctionType();
540 if (callSignatureUnifies(signature, declared, callee->getNamespace(), tracker, false, call)) {
541 return true;
542 }
543 return incompatible("a function call", user, [&](Diagnostic &note) {
544 note << "call argument requires " << declared.getInput(argIdx)
545 << ", but this read is replaced with " << refinedType << "; replacement call signature "
546 << signature << " must unify consistently with " << declared;
547 });
548 }
549 if (auto returnOp = dyn_cast<ReturnOp>(user)) {
550 FuncDefOp function = returnOp->getParentOfType<FuncDefOp>();
551 TypeRange resultTypes = function.getFunctionType().getResults();
552 if (use.getOperandNumber() >= resultTypes.size()) {
553 return false;
554 }
555 SmallVector<Type> replacementTypes;
556 for (Value operand : returnOp.getOperands()) {
557 replacementTypes.push_back(eventualType(operand));
558 }
559 FunctionType signature = FunctionType::get(user->getContext(), {}, replacementTypes);
560 FunctionType declared = FunctionType::get(user->getContext(), {}, resultTypes);
561 if (callSignatureUnifies(signature, declared, {}, tracker, false, user) &&
562 (typeCheck == ConsumerTypeCheck::Unification ||
563 accepts(resultTypes[use.getOperandNumber()]))) {
564 return true;
565 }
566 return incompatible("a function return", user, [&](Diagnostic &note) {
567 note << "replacement return signature " << signature
568 << " does not unify with declared return signature " << declared;
569 });
570 }
571 if (auto memberWrite = dyn_cast<MemberWriteOp>(user)) {
572 if (memberWrite.getVal() != value) {
573 return false;
574 }
575 auto memberDef = memberWrite.getMemberDefOp(tables);
576 if (failed(memberDef)) {
577 return false;
578 }
579 Type memberType = memberDef->get().getType();
580 if (isFullyConcreteArrayConsumerType(memberType) || typeCheck == ConsumerTypeCheck::Exact) {
581 return requiresType(memberType, "a member write", "member write", user);
582 }
583 // A generic destination may be independently refined by its other writes. Check those
584 // writes with their pending types before accepting any one source's refinement.
585 auto parent = getParentOfType<StructDefOp>(memberDef->get());
586 if (!parent) {
587 return false;
588 }
589 auto uses = llzk::getSymbolUses(memberDef->get(), parent);
590 if (!uses) {
591 return false;
592 }
593 for (SymbolTable::SymbolUse memberUse : *uses) {
594 auto otherWrite = dyn_cast<MemberWriteOp>(memberUse.getUser());
595 if (otherWrite && otherWrite != memberWrite &&
596 !requiresType(
597 eventualType(otherWrite.getVal()), "another member write", "other member write",
598 otherWrite
599 )) {
600 return false;
601 }
602 }
603 return true;
604 }
605 if (auto write = dyn_cast<WriteArrayOp>(user)) {
606 if (write.getRvalue() != value) {
607 return false;
608 }
609 auto arrayType = dyn_cast<ArrayType>(eventualType(write.getArrRef()));
610 return arrayType &&
611 requiresDeclaredType(arrayType.getElementType(), "an array write", "array write");
612 }
613 if (auto insert = dyn_cast<InsertArrayOp>(user)) {
614 if (insert.getRvalue() != value) {
615 return false;
616 }
617 auto arrayType = dyn_cast<ArrayType>(eventualType(insert.getArrRef()));
618 return arrayType && requiresDeclaredType(
619 arrayType.getSelectionType(insert.getIndices().size()),
620 "an array insert", "array insert"
621 );
622 }
623 if (auto castOp = dyn_cast<UnifiableCastOp>(user)) {
624 return castOp.getInput() == value &&
625 requiresDeclaredType(
626 castOp.getResult().getType(), "a unifiable cast", "unifiable cast result"
627 );
628 }
629 if (auto create = dyn_cast<CreateArrayOp>(user)) {
630 if (use.getOperandNumber() >= create.getElements().size()) {
631 return false;
632 }
633 // The caller handles arrays whose result is also changed or removed by its plan.
634 Type elementType = create.getType().getElementType();
635 if (refinedType == elementType) {
636 return true;
637 }
638 return incompatible("an array initializer", user, [&](Diagnostic &note) {
639 note << "array initializer requires exactly " << elementType
640 << ", but this read is replaced with " << refinedType;
641 });
642 }
643 if (diagnosticOp) {
644 auto diag = diagnosticOp->emitError(
645 "cannot scalarize heterogeneous array read because it has an unsupported result user"
646 );
647 diag.attachNote(user->getLoc()) << "unsupported read result user is here";
648 }
649 return false;
650}

653static bool canRefineArrayAccessUsers(
654 Value result, Type type, SymbolTableCollection &tables, const ConversionTracker &tracker,
655 ConsumerTypeCheck typeCheck, const DenseMap<Value, Type> *pending
656) {
657 for (OpOperand &use : result.getUses()) {
658 if (auto read = dyn_cast<MemberReadOp>(use.getOwner())) {
659 if (read.getComponent() != result) {
660 return false;
661 }
662 auto refined = getRefinedMemberReadType(read, type, tables, tracker);
663 if (failed(refined) ||
664 (*refined != read.getVal().getType() &&
665 !canRefineMemberReadUsersToType(read, *refined, tables, tracker, pending))) {
666 return false;
667 }
668 } else if (!canRefineConsumerUse(use, type, tables, tracker, typeCheck, pending)) {
669 return false;
670 }
671 }
672 return true;
673}

679bool canRefineCreateArrayUsersToType(
680 CreateArrayOp createOp, Type refinedType, SymbolTableCollection &tables,
681 const ConversionTracker &tracker, const DenseMap<Value, Type> *pendingRefinements
682) {
683 // The allocation's result and initializer operands have one exact element type. A write
684 // can suggest a refinement, but cannot retag an initializer owned by another producer.
685 Type elementType = mlir::cast<ArrayType>(refinedType).getElementType();
686 for (Value initializer : createOp.getElements()) {
687 if (plannedType(initializer, pendingRefinements) != elementType) {
688 return false;
689 }
690 }
691 Value result = createOp.getResult();
692 for (OpOperand &use : result.getUses()) {
693 Operation *user = use.getOwner();
694 if (WriteArrayOp arrayWrite = llvm::dyn_cast<WriteArrayOp>(user)) {
695 if (arrayWrite.getArrRef() != result) {
696 return false;
697 }
698 // Retagging the allocation changes the element type observed by every direct write, but
699 // does not change the write's rvalue. In particular, concrete affine specializations can
700 // unify while still symbolic and become incompatible after instantiation, so require an
701 // exact element-type match instead of the usual unification check.
702 if (plannedType(arrayWrite.getRvalue(), pendingRefinements) !=
703 mlir::cast<ArrayType>(refinedType).getElementType()) {
704 return false;
705 }
706 continue;
707 } else if (InsertArrayOp arrayInsert = llvm::dyn_cast<InsertArrayOp>(user)) {
708 if (arrayInsert.getArrRef() != result) {
709 return false;
710 }
711 // Retagging the allocation changes the subarray type required by every insert, but does
712 // not change an insert's rvalue. As with array.write above, require an exact match rather
713 // than relying on pre-instantiation unification of concrete affine specializations.
714 ArrayType refinedArrayType = mlir::cast<ArrayType>(refinedType);
715 Type refinedRvalueType = refinedArrayType.getSelectionType(arrayInsert.getIndices().size());
716 if (plannedType(arrayInsert.getRvalue(), pendingRefinements) != refinedRvalueType) {
717 return false;
718 }
719 continue;
720 } else if (auto read = dyn_cast<ReadArrayOp>(user)) {
721 if (read.getArrRef() != result || !canRefineArrayAccessUsers(
722 read.getResult(), elementType, tables, tracker,
723 ConsumerTypeCheck::ConcreteExact, pendingRefinements
724 )) {
725 return false;
726 }
727 } else if (auto extract = dyn_cast<ExtractArrayOp>(user)) {
728 Type selected = cast<ArrayType>(refinedType).getSelectionType(extract.getIndices().size());
729 if (extract.getArrRef() != result || !canRefineArrayAccessUsers(
730 extract.getResult(), selected, tables, tracker,
731 ConsumerTypeCheck::Exact, pendingRefinements
732 )) {
733 return false;
734 }
735 } else if (!canRefineConsumerUse(
736 use, refinedType, tables, tracker, ConsumerTypeCheck::ConcreteExact,
737 pendingRefinements
738 )) {
739 return false;
740 }
741 }
742 return true;
743}

750bool canRefineMemberReadUsersToType(
751 MemberReadOp readOp, Type refinedType, SymbolTableCollection &tables,
752 const ConversionTracker &tracker, const DenseMap<Value, Type> *pendingRefinements
753) {
754 Value result = readOp.getVal();
755 for (OpOperand &use : result.getUses()) {
756 Operation *user = use.getOwner();
757 if (auto memberRead = dyn_cast<MemberReadOp>(user)) {
758 if (memberRead.getComponent() != result) {
759 return false;
760 }
761 auto type = getRefinedMemberReadType(memberRead, refinedType, tables, tracker);
762 if (failed(type) || (*type != memberRead.getVal().getType() &&
763 !canRefineMemberReadUsersToType(
764 memberRead, *type, tables, tracker, pendingRefinements
765 ))) {
766 return false;
767 }
768 continue;
769 }
770 if (CreateArrayOp createArray = llvm::dyn_cast<CreateArrayOp>(user)) {
771 if (use.getOperandNumber() >= createArray.getElements().size()) {
772 return false;
773 }
774 // array.new requires every initializer to exactly match its declared element type. Allow
775 // this refinement only when the array can follow it: every other initializer already has
776 // the refined type and the array element type can be refined accordingly.
777 bool canUse = canUseScalarizedValueAsType(
778 createArray.getType().getElementType(), refinedType, tracker,
779 "UpdateMemberDefTypeFromWrite", readOp
780 );
781 if (!canUse) {
782 return false;
783 }
784 for (OpOperand &initializer : createArray->getOpOperands()) {
785 if (&initializer != &use &&
786 plannedType(initializer.get(), pendingRefinements) != refinedType) {
787 return false;
788 }
789 }
790 ArrayType refinedArrayType = createArray.getType().cloneWith(refinedType);
791 if (!canRefineCreateArrayUsersToType(
792 createArray, refinedArrayType, tables, tracker, pendingRefinements
793 )) {
794 return false;
795 }
796 continue;
797 }
798 if (auto read = dyn_cast<ReadArrayOp>(user)) {
799 auto arrayType = dyn_cast<ArrayType>(refinedType);
800 if (read.getArrRef() != result || !arrayType ||
801 !canRefineArrayAccessUsers(
802 read.getResult(), arrayType.getElementType(), tables, tracker,
803 ConsumerTypeCheck::ConcreteExact, pendingRefinements
804 )) {
805 return false;
806 }
807 continue;
808 }
809 if (auto extract = dyn_cast<ExtractArrayOp>(user)) {
810 auto arrayType = dyn_cast<ArrayType>(refinedType);
811 if (extract.getArrRef() != result || !arrayType ||
812 !canRefineArrayAccessUsers(
813 extract.getResult(), arrayType.getSelectionType(extract.getIndices().size()), tables,
814 tracker, ConsumerTypeCheck::Exact, pendingRefinements
815 )) {
816 return false;
817 }
818 continue;
819 }
820 if (UnifiableCastOp castOp = llvm::dyn_cast<UnifiableCastOp>(user)) {
821 if (castOp.getInput() != result) {
822 return false;
823 }
824 Type castResultType = castOp.getResult().getType();
825 if (refinedType == castResultType) {
826 continue;
827 }
828 // Step 6 can repair a stale cast result only when every result use is a compatible member
829 // write. Any other consumer observes the independently declared result type, so leave the
830 // read generic rather than producing a cast whose input and output no longer unify.
831 if (castOp.getResult().use_empty() || typesUnify(refinedType, castResultType)) {
832 return false;
833 }
834 auto canUse = canUseScalarizedValueAsType(
835 castResultType, refinedType, tracker, "UpdateUnifiableCastResultFromInput", readOp
836 );
837 if (!canUse) {
838 return false;
839 }
840 for (OpOperand &castUse : castOp.getResult().getUses()) {
841 auto memberWrite = llvm::dyn_cast<MemberWriteOp>(castUse.getOwner());
842 if (!memberWrite || memberWrite.getVal() != castUse.get()) {
843 return false;
844 }
845 auto memberDef = memberWrite.getMemberDefOp(tables);
846 if (failed(memberDef) || !typesUnify(refinedType, memberDef->get().getType())) {
847 return false;
848 }
849 }
850 continue;
851 }
852 // Stores retain exact destination types; ordinary member consumers can also use a known
853 // specialization. Cast repair above remains specific to this propagation path.
854 ConsumerTypeCheck check = isa<WriteArrayOp, InsertArrayOp>(user)
855 ? ConsumerTypeCheck::Exact
856 : ConsumerTypeCheck::Refinement;
857 if (!canRefineConsumerUse(use, refinedType, tables, tracker, check, pendingRefinements)) {
858 return false;
859 }
860 }
861 return true;
862}

865static CallOp replaceStructConstraintCallWithSpecializedWitness(
866 PatternRewriter &rewriter, CallOp call, Value specializedWitness
867) {
868 SmallVector<Value> args(call.getArgOperands());
869 args.front() = specializedWitness;
870 StructType structType = llvm::cast<StructType>(specializedWitness.getType());
871 SymbolRefAttr callee =
872 appendLeaf(structType.getNameRef(), call.getCalleeAttr().getLeafReference());
873 // Argument and result dictionaries are inherent properties, so the replacement wrapper only
874 // preserves the discardable attributes. The operand/result ordering does not change here.
875 ArrayAttr argAttrs = call.getArgAttrsAttr(), resAttrs = call.getResAttrsAttr();
876 CallOp replacement = replaceOpWithNewOp<CallOp>(
877 rewriter, call, call.getResultTypes(), callee,
878 CallOp::toVectorOfValueRange(call.getMapOperands()), call.getNumDimsPerMapAttr(), args
879 );
880 replacement.setArgAttrsAttr(argAttrs);
881 replacement.setResAttrsAttr(resAttrs);
882 return replacement;
883}

888FailureOr<SmallVector<CallOp>> collectSpecializedWitnessConstraints(
889 Value source, Type type, SymbolTableCollection &tables,
890 llvm::function_ref<bool(OpOperand &)> isInternalUse, Operation *diagnosticOp
891) {
892 SmallVector<CallOp> calls;
893 for (OpOperand &use : source.getUses()) {
894 if (isInternalUse(use)) {
895 continue;
896 }
897 auto call = dyn_cast<CallOp>(use.getOwner());
898 if (!call || !call.calleeIsStructConstrain() || use.getOperandNumber() != 0 ||
899 !isa<StructType>(type)) {
900 if (diagnosticOp) {
901 auto diag = diagnosticOp->emitError(
902 "cannot scalarize array because a generic nondeterministic initializer has an "
903 "unsupported external use"
904 );
905 diag.attachNote(use.getOwner()->getLoc()) << "source witness is also used here";
906 }
907 return failure();
908 }
909 StructType structType = llvm::cast<StructType>(type);
910 SymbolRefAttr callee =
911 appendLeaf(structType.getNameRef(), call.getCalleeAttr().getLeafReference());
912 auto target = lookupTopLevelSymbol<FuncDefOp>(tables, callee, call);
913 if (failed(target)) {
914 return failure();
915 }
916 SmallVector<Type> argTypes(call.getArgOperands().getTypes());
917 argTypes.front() = type;
918 // Use the namespace-aware compatibility rule from CallOp verification, including generic
919 // companion arguments that need not exactly equal the specialized parameter types.
920 if (!typeListsUnify(argTypes, target->get().getArgumentTypes(), target->getNamespace())) {
921 if (diagnosticOp) {
922 auto diag = diagnosticOp->emitError(
923 "cannot scalarize array because specializing a generic nondeterministic initializer "
924 "would make an external constraint call incompatible with its retargeted callee"
925 );
926 diag.attachNote(call.getLoc())
927 << "retargeted constraint expects argument types " << target->get().getArgumentTypes();
928 }
929 return failure();
930 }
931 calls.push_back(call);
932 }
933 return calls;
934}

940Value materializeSpecializedNondet(
941 PatternRewriter &rewriter, NonDetOp sourceNondet, Location loc, Type type,
942 ArrayRef<CallOp> externalConstraints
943) {
944 OpBuilder::InsertionGuard guard(rewriter);
945 rewriter.setInsertionPointAfter(sourceNondet);
946 NonDetOp specializedNondet = NonDetOp::create(rewriter, loc, type);
947 specializedNondet->setDiscardableAttrs(sourceNondet->getDiscardableAttrDictionary());
948 Value specializedValue = specializedNondet.getResult();
949 for (CallOp call : externalConstraints) {
950 replaceStructConstraintCallWithSpecializedWitness(rewriter, call, specializedValue);
951 }
952 return specializedValue;
953}
