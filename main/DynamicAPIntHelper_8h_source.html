81template <std::unsigned_integral T>
82llvm::DynamicAPInt operator>>(const llvm::DynamicAPInt &lhs, T rhs) {
83 if (rhs > static_cast<uint64_t>(std::numeric_limits<int64_t>::max())) {
84 return llvm::DynamicAPInt(lhs < 0 ? -1 : 0);
85 }
86 return lhs >> static_cast<int64_t>(rhs);
87}
