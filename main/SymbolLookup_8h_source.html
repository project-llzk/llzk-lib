244template <typename T>
245inline mlir::FailureOr<SymbolLookupResult<T>> lookupSymbolIn(
246 mlir::SymbolTableCollection &tables, mlir::SymbolRefAttr symbol, Within &&lookupWithin,
247 mlir::Operation *origin, bool reportMissing = true
248) {
249 auto found = lookupSymbolIn(tables, symbol, std::move(lookupWithin), origin, reportMissing);
250 if (mlir::failed(found)) {
251 return mlir::failure(); // lookupSymbolIn() already emits a sufficient error message
252 }
253 // Keep a copy of the op ptr in case we need it for displaying diagnostics
254 mlir::Operation *op = found->get();
255 // ... since the untyped result gets moved here into a typed result.
256 SymbolLookupResult<T> ret(std::move(*found));
257 if (!ret) {
258 if (reportMissing) {
259 auto diag = origin->emitError() << "symbol \"" << symbol << "\" references a '"
260 << op->getName() << "' but expected ";
261 if constexpr (requires { T::getOperationName(); }) {
262 diag << "a '" << T::getOperationName() << '\'';
263 } else {
264 diag << "a symbol with the requested interface";
265 }
266 return diag;
267 } else {
268 return mlir::failure();
269 }
270 }
271 return ret;
272}
