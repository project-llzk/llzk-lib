68 static Value scalar(Value source, const Environment &frame) {
69 auto found = frame.find(source);
70 assert(found != frame.end() && "missing source binding");
71 const auto *value = std::get_if<Value>(&found->second);
72 assert(value && *value && "expected a scalar SSA binding");
73 return *value;
74 }

77 FailureOr<Value> readStorage(Value value, ArrayRef<Attribute> path, Type type, Operation *site) {
78 while (!path.empty()) {
79 if (auto arrayType = dyn_cast<array::ArrayType>(value.getType())) {
80 unsigned rank = arrayType.getShape().size();
81 if (path.size() < rank) {
82 return constraintError(site, "storage path does not identify a scalar leaf");
83 }
84 SmallVector<Value> indices;
85 for (Attribute step : path.take_front(rank)) {
86 auto index = dyn_cast<IntegerAttr>(step);
87 if (!index) {
88 return constraintError(site, "invalid array storage path");
89 }
90 indices.push_back(
91 arith::ConstantIndexOp::create(builder, site->getLoc(), index.getInt())
92 );
93 }
94 value = array::ReadArrayOp::create(
95 builder, site->getLoc(), arrayType.getElementType(), value, indices
96 );
97 path = path.drop_front(rank);
98 continue;
99 }
100 auto podType = dyn_cast<pod::PodType>(value.getType());
101 auto name = dyn_cast<StringAttr>(path.front());
102 if (!podType || !name) {
103 return constraintError(site, "invalid POD storage path");
104 }
105 auto recordType = podType.getRecord(name.getValue(), [site] {
106 return site->emitError(CONSTRAINT_DIAGNOSTIC_PREFIX);
107 });
108 if (failed(recordType)) {
109 return failure();
110 }
111 value = pod::ReadPodOp::create(builder, site->getLoc(), *recordType, value, name.getValue());
112 path = path.drop_front();
113 }
114 if (value.getType() != type) {
115 return constraintError(site, "storage leaf type mismatch");
116 }
117 auto info = storage->describe(value, site);
118 if (failed(info)) {
119 return failure();
120 }
121 if (!isa<StructType>(type) && !info->signal) {
122 nonSignalReads.insert(value.getDefiningOp());
123 }
124 return value;
125 }

128 return bindings.load(
129 path, site, [this](Value root, ArrayRef<Attribute> pathSteps, Type type, Operation *at) {
130 return readStorage(root, pathSteps, type, at);
131 }
132 );

136 FailureOr<BoundValue> copy(BoundValue source, Operation *site, Value input = {}) {
137 auto *path = std::get_if<AccessPath>(&source);
138 if (!path) {
139 return source;
140 }
141 auto target = bindings.create(path->type, input);
142 if (failed(transfer(target, *path, site))) {
143 return failure();
144 }
145 return BoundValue(target);
146 }

149 return bindings.transfer(
150 target, source, site,
151 [this](Value root, ArrayRef<Attribute> pathSteps, Type type, Operation *at) {
152 return readStorage(root, pathSteps, type, at);
153 }

158 LogicalResult assign(AccessPath target, BoundValue source, Operation *site) {
159 if (auto *value = std::get_if<Value>(&source)) {
160 bindings.store(target, *value);
161 return success();

167 FailureOr<AccessPath>
168 arrayPath(Value source, ValueRange indices, Environment &frame, Operation *site) {
169 auto path = std::get<AccessPath>(frame.lookup(source));
170 for (Value index : indices) {
171 auto coordinate = facts.integer(scalar(index, frame));
172 if (!coordinate) {
173 return constraintError(site, "array path requires compile-time indices");
174 }
175 auto selected = bindings.select(path, builder.getIndexAttr(*coordinate), site);
176 if (failed(selected)) {
177 return failure();
178 }
179 path = *selected;
180 }
181 return path;
182 }

185 FailureOr<AccessPath> podPath(Value source, StringRef name, Environment &frame, Operation *site) {
186 auto path = std::get<AccessPath>(frame.lookup(source));
187 return bindings.select(path, builder.getStringAttr(name), site);
188 }

191 LogicalResult readPath(Value result, AccessPath path, Environment &frame, Operation *site) {
192 if (aggregate(path.type)) {
193 auto copied = copy(BoundValue(path), site);
194 if (failed(copied)) {
195 return failure();
196 }
197 frame[result] = *copied;
198 return success();
199 }
200 auto value = load(path, site);
201 if (failed(value)) {
202 return failure();
203 }
204 frame[result] = *value;
205 return success();
206 }
207
209 LogicalResult writePath(AccessPath path, Value value, Environment &frame, Operation *site) {
210 if (failed(bindings.writable(path, site))) {
211 return failure();
212 }
213 return assign(path, frame.lookup(value), site);
214 }

217 LogicalResult arrayNew(array::CreateArrayOp op, Environment &frame) {
218 auto path = bindings.create(op.getType());
219 frame[op.getResult()] = path;
220 if (op.getElements().empty()) {
221 return success();
222 }
223 unsigned next = 0;
224 return bindings.elements(path, op, [this, &op, &frame, &next](AccessPath item) {
225 return assign(item, frame.lookup(op.getElements()[next++]), op);
226 });
227 }

230 LogicalResult podNew(pod::NewPodOp op, Environment &frame) {
231 auto path = bindings.create(op.getType());
232 frame[op.getResult()] = path;
233 for (auto record : op.getInitializedRecordValues()) {
234 auto field = podPath(op.getResult(), record.name, frame, op);
235 if (failed(field) || failed(assign(*field, frame.lookup(record.value), op))) {
236 return failure();
237 }
238 }
239 return success();
240 }

243 LogicalResult memberRead(MemberReadOp op, Environment &frame) {
244 if (op.getTableOffset()) {
245 return constraintError(op, "member read requires concrete non-column component storage");
246 }
247 IRMapping mapping;
248 mapping.map(op.getComponent(), scalar(op.getComponent(), frame));
249 auto emitted = llvm::cast<MemberReadOp>(builder.clone(*op, mapping));
250 auto info = storage->describe(emitted.getResult(), op);
251 if (failed(info)) {
252 return failure();
253 }
254 if (aggregate(op.getType())) {
255 frame[op.getResult()] = bindings.create(info->memberType, {}, emitted.getResult());
256 } else {
257 frame[op.getResult()] = emitted.getResult();
258 if (!isa<StructType>(op.getType()) && !info->signal) {
259 nonSignalReads.insert(emitted);
260 }
261 }
262 return success();
263 }

266 LogicalResult globalRead(global::GlobalReadOp op, Environment &frame) {
267 auto definition = lookupTopLevelSymbol<global::GlobalDefOp>(tables, op.getNameRefAttr(), op);
268 if (failed(definition) || !definition->get().isConstant() ||
269 !definition->get().getInitialValue()) {
270 return constraintError(op, "global read requires an initialized constant");
271 }
272 Attribute initial = definition->get().getInitialValue();
273 if (!aggregate(op.getType())) {
274 auto value = constant(initial, op);
275 if (failed(value)) {
276 return failure();
277 }
278 frame[op.getResult()] = *value;
279 return success();
280 }
281 auto values = dyn_cast<ArrayAttr>(initial);
282 if (!values || !isa<array::ArrayType>(op.getType())) {
283 return constraintError(op, "unsupported aggregate global initializer");
284 }
285 auto path = bindings.create(op.getType());
286 frame[op.getResult()] = path;
287 unsigned next = 0;
288 return bindings.walk(path, op, [this, &op, values, &next](AccessPath item) -> LogicalResult {
289 if (next >= values.size()) {
290 return constraintError(op, "global initializer does not match aggregate leaves");
291 }
292 auto value = constant(values[next++], op);
293 if (failed(value)) {
294 return failure();
295 }
296 bindings.store(item, *value);
297 return success();
298 });
299 }

302 FailureOr<Value> constant(Attribute attribute, Operation *site) {
303 if (auto felt = dyn_cast<felt::FeltConstAttr>(attribute)) {
304 return Value(felt::FeltConstantOp::create(builder, site->getLoc(), felt));
305 }
306 if (auto integer = dyn_cast<IntegerAttr>(attribute)) {
307 return Value(arith::ConstantOp::create(builder, site->getLoc(), integer));
308 }
309 return constraintError(site, "unsupported scalar global initializer");
310 }

313 FailureOr<IRMapping> scalarMapping(Operation &op, const Environment &frame) {
314 IRMapping mapping;
315 for (Value operand : op.getOperands()) {
316 if (aggregate(operand.getType())) {
317 return constraintError(&op, "unsupported aggregate operand");
318 }
319 mapping.map(operand, scalar(operand, frame));
320 }
321 return mapping;
322 }

335 static bool isIntegerExpression(Operation &op) {
336 return !op.getNumRegions() && op.getNumResults() && isMemoryEffectFree(&op) &&
337 llvm::all_of(op.getResultTypes(), [](Type type) {
338 return isa<IndexType, IntegerType>(type);
339 });
340 }

344 LogicalResult emitStructural(Operation &op, IRMapping &mapping, Environment &frame) {
345 Operation *emitted = builder.clone(op, mapping);
346 for (auto [source, result] : llvm::zip_equal(op.getResults(), emitted->getResults())) {
347 if (auto constant = dyn_cast_or_null<IntegerAttr>(facts.known(result))) {
348 frame[source] = Value(arith::ConstantOp::create(builder, op.getLoc(), constant));
349 } else {
350 if (auto conversion = dyn_cast<cast::FeltToIndexOp>(emitted);
351 conversion && !isMemoryEffectFree(conversion)) {
352 return constraintError(&op, "assert cast requires a statically valid index conversion");
353 }
354 frame[source] = result;
355 }
356 }
357 // Keep the temporary operation until final dead-code removal: other folded
358 // results may still depend on it, and the facts cache retains its SSA keys.
359 return success();
360 }

364 LogicalResult evaluateNonNative(Operation &op, Environment &frame) {
365 if (op.getNumRegions() || !op.getNumResults() || !isMemoryEffectFree(&op) ||
366 !llvm::all_of(op.getResultTypes(), [](Type type) { return isa<felt::FeltType>(type); })) {
367 return constraintError(&op, "lower non-native operations before constraint flattening");
368 }
369 auto mapping = scalarMapping(op, frame);
370 if (failed(mapping)) {
371 return failure();
372 }
373 Operation *emitted = builder.clone(op, *mapping);
374 for (auto [source, result] : llvm::zip_equal(op.getResults(), emitted->getResults())) {
375 auto constant = dyn_cast_or_null<felt::FeltConstAttr>(facts.known(result));
376 if (!constant || constant.getType() != result.getType()) {
377 return constraintError(&op, "lower non-native operations before constraint flattening");
378 }
379 frame[source] = Value(felt::FeltConstantOp::create(builder, op.getLoc(), constant));
380 }
381 // As with structural computations, keep cached SSA keys alive until DCE.
382 return success();
383 }

386 LogicalResult emit(Operation &op, Environment &frame) {
387 if (auto assertion = dyn_cast<boolean::AssertOp>(op)) {
388 auto condition = facts.integer(scalar(assertion.getCondition(), frame));
389 return condition && *condition ? success()
390 : constraintError(&op, "assertion must be statically true");
391 }
392 bool equality = isa<constrain::EmitEqualityOp>(op);
393 if (equality && !isa<felt::FeltType>(op.getOperand(0).getType())) {
394 return constraintError(&op, "equality constraints require scalar felt operands");
395 }
396 bool structural = isIntegerExpression(op) || isa<cast::FeltToIndexOp>(op);
397 if (!(equality || structural || isFieldExpression(op) || isa<cast::IntToFeltOp>(op))) {
398 if (op.hasTrait<function::NotFieldNative>() || isa<felt::DivFeltOp>(op)) {
399 return evaluateNonNative(op, frame);
400 }
401 if (isa<constrain::EmitContainmentOp>(op)) {
402 return constraintError(
403 &op, "unsupported operation 'constrain.in': containment constraints require lowering "
404 "before constraint flattening"
405 );
406 }
407 if (llvm::any_of(op.getResultTypes(), [](Type type) { return isa<FloatType>(type); })) {
408 return constraintError(
409 &op, "unsupported: floating-point results are not supported in flattened constraints"
410 );
411 }
412 return constraintError(
413 &op, "unsupported operation '" + op.getName().getStringRef() +
414 "': expected a supported field expression or pure integer computation"
415 );
416 }
417 auto mapping = scalarMapping(op, frame);
418 if (failed(mapping)) {
419 return failure();
420 }
421 if (structural) {
422 return emitStructural(op, *mapping, frame);
423 }
424 Operation *emitted = builder.clone(op, *mapping);
425 for (auto [source, result] : llvm::zip_equal(op.getResults(), emitted->getResults())) {
426 frame[source] = result;
427 }
428 return success();
429 }

439 LogicalResult invoke(
440 FuncDefOp callee, ArrayRef<BoundValue> arguments, Operation *site,
441 SmallVectorImpl<BoundValue> &results
442 ) {
443 if (!callee || !llvm::hasSingleElement(callee.getBody()) || isInTemplate(callee)) {
444 return constraintError(site, "callee must be concrete, defined, and single-block");
445 }
446 if (llvm::is_contained(activeCalls, callee)) {
447 return constraintError(site, "recursive calls are unsupported");
448 }
449 if (callee.getNumArguments() != arguments.size()) {
450 return constraintError(site, "callee argument arity mismatch");
451 }
452 if (activeCalls.size() >= 256) {
453 return constraintError(site, "call nesting limit exceeded");
454 }
455 Environment frame;
456 for (auto [parameter, argument] : llvm::zip_equal(callee.getArguments(), arguments)) {
457 frame[parameter] = argument;
458 if (const auto *path = std::get_if<AccessPath>(&argument)) {
459 bindings.protect(*path);
460 }
461 }
462 activeCalls.push_back(callee);
463 auto status = expand(callee.getBody().front(), frame, results);
464 activeCalls.pop_back();
465 for (const auto &argument : arguments) {
466 if (const auto *path = std::get_if<AccessPath>(&argument)) {
467 bindings.unprotect(*path);
468 }
469 }
470 return status;
471 }

474 LogicalResult call(CallOp op, Environment &frame) {
475 SmallVector<BoundValue> arguments;
476 for (Value argument : op.getArgOperands()) {
477 arguments.push_back(frame.lookup(argument));
478 }
479 FuncDefOp target;
480 if (!arguments.empty() && op.calleeIsStructConstrain()) {
481 auto info = storage->describe(std::get<Value>(arguments.front()), op);
482 if (failed(info)) {
483 return failure();
484 }
485 auto owner = info->definition;
486 target = owner ? owner.getConstrainFuncOp() : FuncDefOp();
487 } else {
488 auto found = lookupTopLevelSymbol<FuncDefOp>(tables, op.getCalleeAttr(), op);
489 if (failed(found)) {
490 return failure();
491 }
492 target = found->get();
493 }
494 SmallVector<BoundValue> results;
495 if (failed(invoke(target, arguments, op, results))) {
496 return failure();
497 }
498 if (results.size() != op.getNumResults()) {
499 return constraintError(op, "callee result arity mismatch");
500 }
501 bind(op.getResults(), results, frame);
502 return success();
503 }

506 LogicalResult branch(scf::IfOp op, Environment &frame) {
507 auto condition = facts.integer(scalar(op.getCondition(), frame));
508 if (!condition) {
509 return constraintError(
510 op, "symbolic i1 condition requires an explicit field selector representation"
511 );
512 }
513 auto &region = *condition ? op.getThenRegion() : op.getElseRegion();
514 SmallVector<BoundValue> results;
515 if (!region.empty() && failed(expand(region.front(), frame, results))) {
516 return failure();
517 }
518 bind(op.getResults(), results, frame);
519 return success();
520 }

523 LogicalResult forLoop(scf::ForOp op, Environment &frame) {
524 auto lower = facts.integer(scalar(op.getLowerBound(), frame));
525 auto upper = facts.integer(scalar(op.getUpperBound(), frame));
526 auto step = facts.integer(scalar(op.getStep(), frame));
527 if (!lower || !upper || !step || *step <= 0 || op.getUnsignedCmp()) {
528 return constraintError(op, "requires signed static for bounds and a positive step");
529 }
530 SmallVector<BoundValue> carried;
531 for (Value value : op.getInitArgs()) {
532 carried.push_back(frame.lookup(value));
533 }
534 for (int64_t i = *lower; i < *upper;) {
535 frame[op.getInductionVar()] = Value(
536 arith::ConstantOp::create(
537 builder, op.getLoc(), builder.getIntegerAttr(op.getInductionVar().getType(), i)
538 )
539 );
540 bind(op.getRegionIterArgs(), carried, frame);
541 carried.clear();
542 if (failed(expand(*op.getBody(), frame, carried))) {
543 return failure();
544 }
545 if (!hasNextStaticIteration(i, *upper, *step)) {
546 break;
547 }
548 i += *step;
549 }
550 bind(op.getResults(), carried, frame);
551 return success();
552 }
553
555 LogicalResult whileLoop(scf::WhileOp op, Environment &frame) {
556 SmallVector<BoundValue> carried;
557 for (Value value : op.getInits()) {
558 carried.push_back(frame.lookup(value));
559 }
560 while (true) {
561 bind(op.getBeforeArguments(), carried, frame);
562 SmallVector<BoundValue> condition;
563 if (failed(expand(op.getBefore().front(), frame, condition))) {
564 return failure();
565 }
566 auto test = facts.integer(std::get<Value>(condition.front()));
567 if (!test) {
568 return constraintError(op, "while condition must be statically known");
569 }
570 carried.assign(condition.begin() + 1, condition.end());
571 if (!*test) {
572 break;
573 }
574 bind(op.getAfterArguments(), carried, frame);
575 carried.clear();
576 if (failed(expand(op.getAfter().front(), frame, carried))) {
577 return failure();
578 }
579 }
580 bind(op.getResults(), carried, frame);
581 return success();
582 }

621 LogicalResult arrayLength(array::ArrayLengthOp op, Environment &frame) {
622 auto path = std::get<AccessPath>(frame.lookup(op.getArrRef()));
623 auto type = dyn_cast<array::ArrayType>(path.type);
624 auto dimension = facts.integer(scalar(op.getDim(), frame));
625 if (!type || !dimension || *dimension < 0 || uint64_t(*dimension) >= type.getShape().size()) {
626 return constraintError(op, "array dimension must be statically known and in bounds");
627 }
628 int64_t size = type.getShape()[*dimension];
629 if (size < 0) {
630 return constraintError(op, "aggregate shape must be static");
631 }
632 frame[op.getResult()] = Value(arith::ConstantIndexOp::create(builder, op.getLoc(), size));
633 return success();
634 }

650 LogicalResult dispatch(Operation &op, Environment &frame) {
651 return llvm::TypeSwitch<Operation *, LogicalResult>(&op)
652 .Case<scf::IfOp>([this, &frame](auto control) { return branch(control, frame); })
653 .Case<scf::ForOp>([this, &frame](auto control) { return forLoop(control, frame); })
654 .Case<scf::WhileOp>([this, &frame](auto control) { return whileLoop(control, frame); })
655 .Case<CallOp>([this, &frame](auto function) { return call(function, frame); })
656 .Case<arith::SelectOp>([this, &frame](auto choice) { return select(choice, frame); })
657 .Case<UnifiableCastOp>([&frame](auto conversion) {
658 frame[conversion.getResult()] = frame.lookup(conversion.getInput());
659 return success();
660 })
661 .Case<global::GlobalReadOp>([this, &frame](auto read) { return globalRead(read, frame); })
662 .Case<MemberReadOp>([this, &frame](auto read) { return memberRead(read, frame); })
663 .Case<MemberWriteOp>([](auto write) {
664 return constraintError(write, "cannot write to original component storage");
665 })
666 .Case<array::CreateArrayOp>([this, &frame](auto allocation) {
667 return arrayNew(allocation, frame);
668 })
669 .Case<pod::NewPodOp>([this, &frame](auto allocation) { return podNew(allocation, frame); })
670 .Case<array::ReadArrayOp, array::ExtractArrayOp>([this, &frame](auto read) {
671 return arrayRead(read, frame);
672 })
673 .Case<array::WriteArrayOp, array::InsertArrayOp>([this, &frame](auto write) {
674 return arrayWrite(write, frame);
675 })
676 .Case<pod::ReadPodOp>([this, &frame](auto read) { return podRead(read, frame); })
677 .Case<pod::WritePodOp>([this, &frame](auto write) { return podWrite(write, frame); })
678 .Case<array::ArrayLengthOp>([this, &frame](auto length) {
679 return arrayLength(length, frame);
680 }).Default([this, &frame](Operation *ordinary) { return emit(*ordinary, frame); });
681 }

684 LogicalResult expand(Block &block, Environment &frame, SmallVectorImpl<BoundValue> &results) {
685 for (Operation &op : block) {
686 if (isa<ReturnOp, scf::YieldOp, scf::ConditionOp>(op)) {
687 for (Value value : op.getOperands()) {
688 results.push_back(frame.lookup(value));
689 }
690 return success();
691 }
692 if (failed(dispatch(op, frame))) {
693 return failure();
694 }
695 }
696 return constraintError(block.getParentOp(), "missing supported terminator");
697 }

701 LogicalResult validateOperation(Operation &op) {
702 if (op.getNumRegions()) {
703 return constraintError(&op, "regions are unsupported in flattened constraints");
704 }
705 if (isFieldExpression(op) || isa<constrain::EmitEqualityOp>(op)) {
706 auto fieldType = [](Type type) { return isa<felt::FeltType>(type); };
707 if (llvm::all_of(op.getOperandTypes(), fieldType) &&
708 llvm::all_of(op.getResultTypes(), fieldType)) {
709 return success();
710 }
711 return constraintError(
712 &op, "constraint expressions require scalar felt operands and results"
713 );
714 }
715 if (auto constant = dyn_cast<arith::ConstantOp>(op)) {
716 Type type = constant.getType();
717 if (type.isIndex() || type.isInteger(1)) {
718 return success();
719 }
720 }
721 if (auto conversion = dyn_cast<cast::IntToFeltOp>(op)) {
722 Attribute literal;
723 if (matchPattern(conversion.getValue(), m_Constant(&literal)) && isa<IntegerAttr>(literal)) {
724 if (!isMemoryEffectFree(conversion) && !facts.known(conversion.getResult())) {
725 return constraintError(&op, "assert cast requires a statically valid field conversion");
726 }
727 return success();
728 }
729 return constraintError(&op, "field conversion requires a compile-time integer");
730 }
731 if (isa<array::ReadArrayOp, array::ExtractArrayOp>(op)) {
732 ValueRange indices = llvm::cast<array::ArrayAccessOpInterface>(op).getIndices();
733 for (Value index : indices) {
734 if (!index.getDefiningOp<arith::ConstantOp>()) {
735 return constraintError(&op, "array path requires compile-time indices");
736 }
737 }
738 return success();
739 }
740 if (auto read = dyn_cast<MemberReadOp>(op)) {
741 if (read.getTableOffset()) {
742 return constraintError(&op, "member read requires concrete non-column component storage");
743 }
744 return success();
745 }
746 if (isa<pod::ReadPodOp>(op)) {
747 return success();
748 }
749 if (isIntegerExpression(op)) {
750 return constraintError(&op, "structural operation requires a compile-time value");
751 }
752 return constraintError(
753 &op, "unsupported operation in flattened constraints: '" + op.getName().getStringRef() + "'"
754 );
755 }
756
758 LogicalResult validateOutput() {
759 for (Operation &op : llvm::make_early_inc_range(llvm::reverse(output.front()))) {
760 // A successful cast fold proves its range assertion, so the temporary
761 // cast can be removed despite its conservative effect interface.
762 bool foldedIndexCast = isa<cast::FeltToIndexOp>(op) && facts.known(op.getResult(0));
763 if (op.use_empty() &&
764 (foldedIndexCast || isMemoryEffectFree(&op) ||
765 isa<MemberReadOp, array::ReadArrayOp, array::ExtractArrayOp, pod::ReadPodOp>(op))) {
766 op.erase();
767 }
768 }
769 for (Operation &op : output.front()) {
770 if (failed(validateOperation(op))) {
771 return failure();
772 }
773 if (nonSignalReads.contains(&op)) {
774 return constraintError(
775 &op, "live unresolved non-signal scalar storage; algebraic elimination is required"
776 );
777 }
778 }
779 return success();
780 }
781
783 LogicalResult commit(FuncDefOp source) {
784 Region original;
785 original.takeBody(source.getBody());
786 source.getBody().takeBody(output);
787 if (failed(verify(source))) {
788 output.takeBody(source.getBody());
789 source.getBody().takeBody(original);
790 return failure();
791 }
792 return success();
793 }
794
795public:
796 explicit ConstraintElaborator(ModuleOp root)
797 : module(root), builder(root.getContext()), bindings(root.getContext()) {}
798
800 LogicalResult run() {
801 auto root = getRootStructDef(tables, module);
802 if (failed(root)) {
803 return failure();
804 }
805 if (!*root) {
806 return constraintError(module, "requires llzk.main");
807 }
808 main = root->get();
809 storage.emplace(module, main, builder, tables, facts);
810 auto source = main.getConstrainFuncOp();
811 if (isInTemplate(main) || !source || !llvm::hasSingleElement(source.getBody())) {
812 return constraintError(main, "main must have a concrete single-block constrain function");
813 }
814 output.push_back(new Block());
815 SmallVector<BoundValue> arguments;
816 for (auto [i, argument] : llvm::enumerate(source.getArguments())) {
817 Value emitted = output.front().addArgument(argument.getType(), argument.getLoc());
818 if (aggregate(argument.getType())) {
819 arguments.push_back(bindings.create(argument.getType(), argument, emitted));
820 } else {
821 arguments.push_back(emitted);
822 }
823 }
824 builder.setInsertionPointToEnd(&output.front());
825 SmallVector<BoundValue> results;
826 if (failed(invoke(source, arguments, source, results)) || failed(validateOutput())) {
827 return failure();
828 }
829 ReturnOp::create(builder, source.getLoc());
830 return commit(source);
831 }
832};

839 if (failed(ConstraintElaborator(getOperation()).run())) {
840 signalPassFailure();
841 }
842 }
