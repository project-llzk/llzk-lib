56static void walkAttrSymbolRefs(
57 Attribute attr, function_ref<void(SymbolRefAttr)> rootRef,
58 function_ref<void(SymbolRefAttr)> unsupportedRef
59) {
60 attr.walk<WalkOrder::PreOrder>([rootRef, unsupportedRef](TypeAttr typeAttr) {
61 walkTypeSymbolRefs(typeAttr.getValue(), rootRef, unsupportedRef);
62 return WalkResult::skip();
63 }, [unsupportedRef](SymbolRefAttr ref) { unsupportedRef(ref); });
64}

66static void walkTypeSymbolRefs(
67 Type type, function_ref<void(SymbolRefAttr)> rootRef,
68 function_ref<void(SymbolRefAttr)> unsupportedRef
69) {
70 type.walk<WalkOrder::PreOrder>([rootRef, unsupportedRef](component::StructType structType) {

75 return WalkResult::skip();

81static bool isInlinableFreeFunction(FuncDefOp func, ModuleOp root, SymbolTableCollection &tables) {
82 Operation *parent = func->getParentOp();
83 if (parent == root) {
84 // A normal MLIR call uses nearest-symbol lookup after cloning, so a struct
85 // method can capture a callee that originally resolved at the root. Keep
86 // such helpers in place until general symbol rebasing is implemented.
87 bool hasFuncCall = false;
88 func.walk([&hasFuncCall](func::CallOp) { hasFuncCall = true; });
89 return !hasFuncCall;
90 }
91
92 // Struct methods are handled by the struct inliner, not this pass. Also
93 // reject functions that are not owned by this module, such as definitions
94 // resolved through include.from.
95 if (!llvm::isa<ModuleOp, polymorphic::TemplateOp>(parent) ||
96 !root->isAncestor(func.getOperation())) {
97 return false;
98 }
99
100 // TODO: This is a temporary allowance for a common frontend pattern: a
101 // helper nested in a namespace-like template or module whose symbol
102 // references can all be resolved from the root module. General
103 // cross-symbol-table inlining must rebase symbol references in the cloned
104 // body so that they continue to resolve to the same definitions.
105 bool hasUnsupportedSymbolRef = false;
106 func.walk([&tables, &hasUnsupportedSymbolRef, root](Operation *op) {
107 auto detectUnsupportedSymbolRef = [&hasUnsupportedSymbolRef](SymbolRefAttr) {
108 hasUnsupportedSymbolRef = true;
109 };
110 auto detectMissingRootSymbolRef = [&tables, &hasUnsupportedSymbolRef, root](SymbolRefAttr ref) {
111 if (!tables.lookupSymbolIn(root, ref)) {
112 hasUnsupportedSymbolRef = true;
113 }
114 };
115 walkAttrSymbolRefs(
116 op->getDiscardableAttrDictionary(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef
117 );
118 if (Attribute properties = op->getPropertiesAsAttribute()) {
119 for (NamedAttribute property : llvm::cast<DictionaryAttr>(properties)) {
120 StringAttr name = property.getName();
121 // Member names resolve through the component's StructType, not the
122 // surrounding symbol table. The StructType itself is scanned below.
123 if (auto memberRef = llvm::dyn_cast<component::MemberRefOpInterface>(op)) {
124 if (name == memberRef.getMemberNameAttrName()) {
125 continue;
126 }
127 }
128 // LLZK calls and global references intentionally resolve these
129 // properties from the root module. Other properties, including
130 // template parameters, retain their ordinary scope-sensitive checks.
131 bool usesRootLookup = false;
132 if (auto call = llvm::dyn_cast<CallOp>(op)) {
133 usesRootLookup = name == call.getCalleeAttrName();
134 } else if (auto globalRef = llvm::dyn_cast<global::GlobalRefOpInterface>(op)) {
135 usesRootLookup = name == globalRef.getNameRefAttrName();
136 }
137 if (usesRootLookup) {
138 property.getValue().walk(detectMissingRootSymbolRef);
139 } else {
140 walkAttrSymbolRefs(
141 property.getValue(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef
142 );
143 }
144 }
145 }
146 for (Type type : llvm::concat<Type>(op->getOperandTypes(), op->getResultTypes())) {
147 walkTypeSymbolRefs(type, detectMissingRootSymbolRef, detectUnsupportedSymbolRef);
148 }
149 for (Region &region : op->getRegions()) {
150 for (Block &block : region) {
151 for (BlockArgument arg : block.getArguments()) {
152 walkTypeSymbolRefs(arg.getType(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef);
153 }
154 }
155 }
156 });
157 return !hasUnsupportedSymbolRef;
158}

171static llvm::DenseSet<Operation *> collectRecursiveFunctions(const llzk::CallGraph &cg) {
172 llvm::DenseSet<Operation *> recursive;
173 for (auto scc = llvm::scc_begin(&cg); !scc.isAtEnd(); ++scc) {
174 if (scc->size() == 1 && !scc.hasCycle()) {
175 continue;
176 }
177 for (const llzk::CallGraphNode *node : *scc) {
178 if (!node->isExternal()) {
179 recursive.insert(node->getCalledFunction().getOperation());
180 }
181 }
182 }
183 return recursive;
184}

188static FuncDefOp resolveFreeCallee(
189 CallOp call, ModuleOp root, SymbolTableCollection &tables,
190 const llvm::DenseSet<Operation *> &skippedCallees
191) {
192 auto tgtRes = call.getCalleeTarget(tables);
193 if (failed(tgtRes)) {
194 return nullptr;
195 }
196 FuncDefOp callee = tgtRes->get();
197 if (!isInlinableFreeFunction(callee, root, tables) || callee.isExternal() ||
198 skippedCallees.contains(callee)) {
199 return nullptr;
200 }
201 return callee;
202}

212static SmallVector<FreeFunctionCall> collectFreeFunctionCalls(
213 ModuleOp mod, SymbolTableCollection &tables, const llvm::DenseSet<Operation *> &skippedCallees
214) {
215 SmallVector<FreeFunctionCall> calls;
216 mod.walk([&](CallOp call) {
217 if (!isInlinableCallSite(call, mod)) {
218 return;
219 }
220 if (FuncDefOp callee = resolveFreeCallee(call, mod, tables, skippedCallees)) {
221 calls.push_back({call, callee});
222 }
223 });
224 return calls;
225}

230static SmallVector<FuncDefOp> collectUnusedHelpers(ModuleOp mod) {
231 SymbolUseGraph useGraph(mod.getOperation());
232 SmallVector<FuncDefOp> unusedFunctions;
233 for (FuncDefOp func : mod.getOps<FuncDefOp>()) {
234 if (func.isExternal()) {
235 continue;
236 }
237 const SymbolUseGraphNode *node = useGraph.lookupNode(func);
238 if (!node || !node->hasPredecessor()) {
239 unusedFunctions.push_back(func);
240 }
241 }
242 return unusedFunctions;
243}

249 void runOnOperation() override {
250 ModuleOp mod = getOperation();
251 SymbolTableCollection tables;
252 InlinerInterface inliner(&getContext());
253 // Seeded with recursive functions; grows with callees that cannot be
254 // inlined.
255 llvm::DenseSet<Operation *> skippedCallees =
256 collectRecursiveFunctions(getAnalysis<CallGraphAnalysis>().getCallGraph());
257
258 inlineCalls(mod, tables, inliner, skippedCallees);
259 removeUnusedFunctions(mod);
260 }
261
268 void inlineCalls(
269 ModuleOp mod, SymbolTableCollection &tables, InlinerInterface &inliner,
270 llvm::DenseSet<Operation *> &skippedCallees
271 ) {
272 SmallVector<FreeFunctionCall> callsToInline =
273 collectFreeFunctionCalls(mod, tables, skippedCallees);
274 while (!callsToInline.empty()) {
275 LLVM_DEBUG({
276 llvm::dbgs() << "[" DEBUG_TYPE "] round found " << callsToInline.size()
277 << " free-function call site(s) to inline\n";
278 });
279
280 for (auto [call, callee] : callsToInline) {
281 if (skippedCallees.contains(callee)) {
282 continue;
283 }
284 if (failed(inlineCall(inliner, call, callee, callee.getCallableRegion(), true))) {
285 call.emitWarning("failed to inline free function call; skipping this callee");
286 skippedCallees.insert(callee);
287 continue;
288 }
289 call.erase();
290 }
291
292 callsToInline = collectFreeFunctionCalls(mod, tables, skippedCallees);
293 }
294 }
295
298 void removeUnusedFunctions(ModuleOp mod) {
299 SmallVector<FuncDefOp> toErase = collectUnusedHelpers(mod);
300 while (!toErase.empty()) {
301 for (FuncDefOp func : toErase) {
302 func.erase();
303 }
304 toErase = collectUnusedHelpers(mod);
305 }
306 }
