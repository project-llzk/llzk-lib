58static void walkAttrSymbolRefs(
59 Attribute attr, function_ref<void(SymbolRefAttr)> rootRef,
60 function_ref<void(SymbolRefAttr)> unsupportedRef
61) {
62 attr.walk<WalkOrder::PreOrder>([rootRef, unsupportedRef](TypeAttr typeAttr) {
63 walkTypeSymbolRefs(typeAttr.getValue(), rootRef, unsupportedRef);
64 return WalkResult::skip();
65 }, [unsupportedRef](SymbolRefAttr ref) { unsupportedRef(ref); });
66}

68static void walkTypeSymbolRefs(
69 Type type, function_ref<void(SymbolRefAttr)> rootRef,
70 function_ref<void(SymbolRefAttr)> unsupportedRef
71) {
72 type.walk<WalkOrder::PreOrder>([rootRef, unsupportedRef](component::StructType structType) {
73 rootRef(structType.getNameRef());
74 if (ArrayAttr params = structType.getParams()) {
75 walkAttrSymbolRefs(params, rootRef, unsupportedRef);
76 }
77 return WalkResult::skip();
78 }, [unsupportedRef](SymbolRefAttr ref) { unsupportedRef(ref); });

83static bool isInlinableFreeFunction(FuncDefOp func, ModuleOp root, SymbolTableCollection &tables) {
84 Operation *parent = func->getParentOp();
85 if (parent == root) {
86 // A normal MLIR call uses nearest-symbol lookup after cloning, so a struct
87 // method can capture a callee that originally resolved at the root. Keep
88 // such helpers in place until general symbol rebasing is implemented.
89 bool hasFuncCall = walkContains<func::CallOp>(func);
90 LLVM_DEBUG({
91 if (hasFuncCall) {
92 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline " << func.getSymName() << " at "
93 << func.getLoc()
94 << ": body contains func.call whose callee could be captured after cloning\n";
95 }
96 });
97 return !hasFuncCall;
98 }
99
100 // Struct methods are handled by the struct inliner, not this pass. Also
101 // reject functions that are not owned by this module, such as definitions
102 // resolved through include.from.
103 if (!llvm::isa<ModuleOp, polymorphic::TemplateOp>(parent)) {
104 // Don't bother with debug message in this case. It's just checking the very
105 // basic case of whether the function is a free function.
106 return false;
107 }
108 if (!root->isAncestor(func.getOperation())) {
109 LLVM_DEBUG(
110 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline " << func.getSymName() << " at "
111 << func.getLoc() << ": function is defined outside the root module\n"
112 );
113 return false;
114 }
115
116 // TODO: This is a temporary allowance for a common frontend pattern: a
117 // helper nested in a namespace-like template or module whose symbol
118 // references can all be resolved from the root module. General
119 // cross-symbol-table inlining must rebase symbol references in the cloned
120 // body so that they continue to resolve to the same definitions.
121 bool hasUnsupportedSymbolRef = false;
122 func.walk([&](Operation *op) {
123 auto detectUnsupportedSymbolRef = [&](SymbolRefAttr ref) {
124 LLVM_DEBUG(
125 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline " << func.getSymName()
126 << ": unsupported scope-sensitive symbol reference " << ref << " in "
127 << op->getName() << " at " << op->getLoc() << '\n'
128 );
129 hasUnsupportedSymbolRef = true;
130 };
131 auto detectMissingRootSymbolRef = [&](SymbolRefAttr ref) {
132 if (!tables.lookupSymbolIn(root, ref)) {
133 LLVM_DEBUG(
134 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline " << func.getSymName()
135 << ": symbol reference " << ref << " in " << op->getName() << " at "
136 << op->getLoc() << " does not resolve in the root module\n"
137 );
138 hasUnsupportedSymbolRef = true;
139 }
140 };
141 walkAttrSymbolRefs(
142 op->getDiscardableAttrDictionary(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef
143 );
144 if (Attribute properties = op->getPropertiesAsAttribute()) {
145 for (NamedAttribute property : llvm::cast<DictionaryAttr>(properties)) {
146 StringAttr name = property.getName();
147 // Member names resolve through the component's StructType, not the
148 // surrounding symbol table. The StructType itself is scanned below.
149 if (auto memberRef = llvm::dyn_cast<component::MemberRefOpInterface>(op)) {
150 if (name == memberRef.getMemberNameAttrName()) {
151 continue;
152 }
153 }
154 // LLZK calls and global references intentionally resolve these
155 // properties from the root module. Other properties, including
156 // template parameters, retain their ordinary scope-sensitive checks.
157 bool usesRootLookup = false;
158 if (auto call = llvm::dyn_cast<CallOp>(op)) {
159 usesRootLookup = name == call.getCalleeAttrName();
160 } else if (auto globalRef = llvm::dyn_cast<global::GlobalRefOpInterface>(op)) {
161 usesRootLookup = name == globalRef.getNameRefAttrName();
162 }
163 if (usesRootLookup) {
164 // Check the complete path, not its nested FlatSymbolRefAttr components:
165 // those resolve within the preceding symbol, not from the root.
166 detectMissingRootSymbolRef(llvm::cast<SymbolRefAttr>(property.getValue()));
167 } else {
168 walkAttrSymbolRefs(
169 property.getValue(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef
170 );
171 }
172 }
173 }
174 for (Type type : llvm::concat<Type>(op->getOperandTypes(), op->getResultTypes())) {
175 walkTypeSymbolRefs(type, detectMissingRootSymbolRef, detectUnsupportedSymbolRef);
176 }
177 for (Region &region : op->getRegions()) {
178 for (Block &block : region) {
179 for (BlockArgument arg : block.getArguments()) {
180 walkTypeSymbolRefs(arg.getType(), detectMissingRootSymbolRef, detectUnsupportedSymbolRef);
181 }
182 }
183 }
184 });
185 return !hasUnsupportedSymbolRef;
186}

191static bool isInlinableCallSite(CallOp call, ModuleOp root) {
192 if (!call->getParentOfType<FuncDefOp>()) {
193 LLVM_DEBUG(
194 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee() << " at "
195 << call.getLoc() << ": call is outside a function.def body\n"
196 );
197 return false;
198 }
199 if (call->getParentOfType<ModuleOp>() != root) {
200 LLVM_DEBUG(
201 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee() << " at "
202 << call.getLoc() << ": call is in a nested module\n"
203 );
204 return false;
205 }
206 return true;
207}

213static llvm::DenseSet<Operation *> collectRecursiveFunctions(const llzk::CallGraph &cg) {
214 llvm::DenseSet<Operation *> recursive;
215 for (auto scc = llvm::scc_begin(&cg); !scc.isAtEnd(); ++scc) {
216 if (scc->size() == 1 && !scc.hasCycle()) {
217 continue;
218 }
219 for (const llzk::CallGraphNode *node : *scc) {
220 if (!node->isExternal()) {
221 LLVM_DEBUG(
222 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline "
223 << SymbolTable::getSymbolName(node->getCalledFunction().getOperation())
224 << " at " << node->getCalledFunction().getLoc()
225 << ": function participates in a call cycle\n"
226 );
227 recursive.insert(node->getCalledFunction().getOperation());
228 }
229 }
230 }
231 return recursive;
232}

236static FuncDefOp resolveFreeCallee(
237 CallOp call, ModuleOp root, SymbolTableCollection &tables,
238 const llvm::DenseSet<Operation *> &skippedCallees
239) {
240 auto tgtRes = call.getCalleeTarget(tables);
241 if (failed(tgtRes)) {
242 LLVM_DEBUG(
243 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee() << " at "
244 << call.getLoc() << ": callee target resolution failed\n"
245 );
246 return nullptr;
247 }
248 FuncDefOp callee = tgtRes->get();
249 if (!isInlinableFreeFunction(callee, root, tables)) {
250 return nullptr;
251 }
252 if (callee.isExternal()) {
253 LLVM_DEBUG(
254 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee() << " at "
255 << call.getLoc() << ": callee is external and has no body\n"
256 );
257 return nullptr;
258 }
259 if (skippedCallees.contains(callee)) {
260 LLVM_DEBUG(
261 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee() << " at "
262 << call.getLoc() << ": callee was skipped due to a call cycle or an earlier "
263 << "inlining failure\n"
264 );
265 return nullptr;
266 }
267 return callee;
268}

278static SmallVector<FreeFunctionCall> collectFreeFunctionCalls(
279 ModuleOp mod, SymbolTableCollection &tables, const llvm::DenseSet<Operation *> &skippedCallees
280) {
281 SmallVector<FreeFunctionCall> calls;
282 mod.walk([&](CallOp call) {
283 if (!isInlinableCallSite(call, mod)) {
284 return;
285 }
286 if (FuncDefOp callee = resolveFreeCallee(call, mod, tables, skippedCallees)) {
287 calls.push_back({call, callee});
288 }
289 });
290 return calls;
291}

295static SmallVector<FuncDefOp> collectUnusedHelpers(ModuleOp mod) {
296 SymbolUseGraph useGraph(mod.getOperation());
297 SmallVector<FuncDefOp> unusedFunctions;
298 mod.walk([&](FuncDefOp func) {
299 if (func.isExternal() || !llvm::isa<ModuleOp, polymorphic::TemplateOp>(func->getParentOp())) {
300 return;
301 }
302 const SymbolUseGraphNode *node = useGraph.lookupNode(func);
303 if (!node || !node->hasPredecessor()) {
304 unusedFunctions.push_back(func);
305 }
306 });
307 return unusedFunctions;
308}

314 void runOnOperation() override {
315 ModuleOp mod = getOperation();
316 SymbolTableCollection tables;
317 InlinerInterface inliner(&getContext());
318 // Seeded with recursive functions; grows with callees that cannot be
319 // inlined.
320 llvm::DenseSet<Operation *> skippedCallees =
321 collectRecursiveFunctions(getAnalysis<CallGraphAnalysis>().getCallGraph());
322
323 inlineCalls(mod, tables, inliner, skippedCallees);
324 removeUnusedFunctions(mod);
325 }
326
327 /// Collects the current free-function call sites, then inlines them.
328 /// Iterates until no inlinable calls remain: inlined bodies can expose new
329 /// calls, but skipped callees are excluded, so every exposed call chain is
330 /// finite. Best effort: a callee whose inlining fails is skipped with a
331 /// warning, leaving its remaining call sites in place while processing
332 /// continues with other callees.
333 void inlineCalls(
334 ModuleOp mod, SymbolTableCollection &tables, InlinerInterface &inliner,
335 llvm::DenseSet<Operation *> &skippedCallees
336 ) {
337 InlinerConfig inlinerConfig;
338 SmallVector<FreeFunctionCall> callsToInline =
339 collectFreeFunctionCalls(mod, tables, skippedCallees);
340 for (;;) {
341 LLVM_DEBUG({
342 llvm::dbgs() << "[" DEBUG_TYPE "] round found " << callsToInline.size()
343 << " free-function call site(s) to inline\n";
344 });
345
346 if (callsToInline.empty()) {
347 break;
348 }
349
350 for (auto [call, callee] : callsToInline) {
351 if (skippedCallees.contains(callee)) {
352 LLVM_DEBUG(
353 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee()
354 << " at " << call.getLoc()
355 << ": an earlier inlining attempt for this callee failed\n"
356 );
357 continue;
358 }
359 if (failed(inlineCall(
360 inliner, inlinerConfig.getCloneCallback(), call, callee, callee.getCallableRegion(),
361 true
362 ))) {
363 LLVM_DEBUG(
364 llvm::dbgs() << "[" DEBUG_TYPE "] cannot inline call to " << call.getCallee()
365 << " at " << call.getLoc() << ": MLIR inlineCall failed\n"
366 );
367 call.emitWarning("failed to inline free function call; skipping this callee");
368 skippedCallees.insert(callee);
369 continue;
370 }
371 call.erase();
372 }
373
374 callsToInline = collectFreeFunctionCalls(mod, tables, skippedCallees);
375 }
376 }

380 void removeUnusedFunctions(ModuleOp mod) {
381 SmallVector<FuncDefOp> toErase = collectUnusedHelpers(mod);
382 while (!toErase.empty()) {
383 for (FuncDefOp func : toErase) {
384 func.erase();
385 }
386 toErase = collectUnusedHelpers(mod);
387 }
388 }
