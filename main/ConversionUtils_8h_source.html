55template <typename OpTy>
56inline OpTy
57preserveDiscardableAttrsExcept(mlir::Operation *src, OpTy dst, mlir::StringRef excludedAttr) {
58 auto original = src->getDiscardableAttrDictionary();
59 mlir::SmallVector<mlir::NamedAttribute> attrs;
60 for (mlir::NamedAttribute attr : original.getValue()) {
61 if (attr.getName().getValue() != excludedAttr) {
62 attrs.push_back(attr);
63 }
64 }
65 dst->setDiscardableAttrs(mlir::DictionaryAttr::get(src->getContext(), attrs));
66 return dst;
67}

102inline std::string
103reserveUniqueAttrName(llvm::StringSet<> &usedNames, llvm::StringRef desiredName) {
104 if (!usedNames.contains(desiredName)) {
105 usedNames.insert(desiredName);
106 return desiredName.str();
107 }
108
109 for (unsigned suffix = 1;; ++suffix) {
110 std::string candidate = (desiredName + "#" + llvm::Twine(suffix)).str();
111 if (!usedNames.contains(candidate)) {
112 usedNames.insert(candidate);
113 return candidate;
114 }
115 }

140template <typename GetNameAttrFn, typename GetSplitSuffixesFn>
141inline SplitFunctionNameInfo collectSplitFunctionNameInfo(
142 mlir::ArrayRef<mlir::Type> origTypes, GetNameAttrFn &&getNameAttr,
143 GetSplitSuffixesFn &&getSplitSuffixes
144) {
145 SplitFunctionNameInfo info;
146 info.originalNames.reserve(origTypes.size());
147 info.splitNameSuffixes.reserve(origTypes.size());
148 for (auto [i, type] : llvm::enumerate(origTypes)) {
149 if (std::optional<mlir::StringAttr> nameAttr = getNameAttr(i)) {
150 info.originalNames.push_back(nameAttr->getValue());
151 info.existingNames.push_back(nameAttr->getValue());
152 } else {
153 info.originalNames.push_back(std::nullopt);
154 }
155 info.splitNameSuffixes.push_back(getSplitSuffixes(type));
156 }
157 return info;
158}

162inline mlir::ArrayAttr replicateFunctionNameAttrsAsNeeded(
163 mlir::ArrayAttr origAttrs, const llvm::SmallVector<size_t> &originalIdxToSize,
164 const llvm::SmallVector<mlir::Type> &newTypes, llvm::StringRef functionNameAttrName,
165 llvm::ArrayRef<std::optional<llvm::StringRef>> origNames = {},
166 llvm::ArrayRef<llvm::StringRef> existingNames = {},
167 llvm::ArrayRef<llvm::SmallVector<std::string>> splitNameSuffixes = {}
168) {
169 if (!origAttrs) {
170 return nullptr;
171 }
172 assert(originalIdxToSize.size() == origAttrs.size());
173 if (originalIdxToSize.size() == newTypes.size()) {
174 return nullptr;
175 }
176
177 llvm::SmallVector<mlir::Attribute> newAttrs;
178 llvm::StringSet<> usedNames;
179 if (!origNames.empty()) {
180 for (llvm::StringRef name : existingNames) {
181 usedNames.insert(name);
182 }
183 }
184
185 for (auto [i, s] : llvm::enumerate(originalIdxToSize)) {
186 mlir::Attribute attr = origAttrs[i];
187 if (!origNames.empty() && !splitNameSuffixes.empty() && s != 1 && origNames[i]) {
188 assert(i < splitNameSuffixes.size());
189 assert(splitNameSuffixes[i].size() == s);
190 auto dictAttr = llvm::cast<mlir::DictionaryAttr>(attr);
191 llvm::StringRef name = *origNames[i];
192 for (llvm::StringRef suffix : splitNameSuffixes[i]) {
193 std::string desiredName = (llvm::Twine(name) + suffix).str();
194 newAttrs.push_back(withFunctionNameAttr(
195 dictAttr, functionNameAttrName, reserveUniqueAttrName(usedNames, desiredName)
196 ));
197 }
198 continue;
199 }
200 newAttrs.append(s, attr);
201 }
202 return mlir::ArrayAttr::get(origAttrs.getContext(), newAttrs);
203}

211inline function::CallOp createCallPreservingInstantiationOperands(
212 mlir::Location loc, mlir::TypeRange newResultTypes, function::CallOp oldCall,
213 llvm::ArrayRef<mlir::ValueRange> mapOperands, mlir::ValueRange argOperands,
214 mlir::ConversionPatternRewriter &rewriter
215) {
216 llvm::SmallVector<mlir::Attribute> templateParams;
217 if (mlir::ArrayAttr templateParamsAttr = oldCall.getTemplateParamsAttr()) {
218 templateParams.append(templateParamsAttr.begin(), templateParamsAttr.end());
219 }
220
221 function::CallOp newCall;
222 if (oldCall.getMapOperands().empty()) {
223 newCall = rewriter.create<function::CallOp>(
224 loc, newResultTypes, oldCall.getCalleeAttr(), argOperands, templateParams
225 );
226 } else {
227 newCall = rewriter.create<function::CallOp>(
228 loc, newResultTypes, oldCall.getCalleeAttr(), mapOperands, oldCall.getNumDimsPerMapAttr(),
229 argOperands, templateParams
230 );
231 }
232 return preserveDiscardableAttrs(oldCall, newCall);
233}

236inline static mlir::Type replaceAffineMapArrayDimsWithWildcards(mlir::Type type) {
237 auto arrTy = llvm::dyn_cast<array::ArrayType>(type);
238 if (!arrTy) {
239 return type;
240 }
241
242 mlir::Builder builder(arrTy.getContext());
243 llvm::SmallVector<mlir::Attribute> dims;
244 dims.reserve(arrTy.getDimensionSizes().size());
245 for (mlir::Attribute dimSize : arrTy.getDimensionSizes()) {
246 if (llvm::isa<mlir::AffineMapAttr>(dimSize)) {
247 dims.push_back(builder.getIndexAttr(mlir::ShapedType::kDynamic));
248 } else {
249 dims.push_back(dimSize);
250 }
251 }
252
253 return arrTy.cloneWith(replaceAffineMapArrayDimsWithWildcards(arrTy.getElementType()), dims);
254}

273
274 void convert(function::FuncDefOp op, mlir::RewriterBase &rewriter) {
275 // Update in/out types of the function
276 mlir::FunctionType oldTy = op.getFunctionType();
277 llvm::SmallVector<mlir::Type> newInputs = convertInputs(oldTy.getInputs());
278 llvm::SmallVector<mlir::Type> newResults = convertResults(oldTy.getResults());
279 mlir::FunctionType newTy = mlir::FunctionType::get(
280 oldTy.getContext(), mlir::TypeRange(newInputs), mlir::TypeRange(newResults)
281 );
282 if (newTy == oldTy) {
283 return; // nothing to change
284 }
285
286 // Pre-condition: arg/result count equals corresponding attribute count
287 assert(!op.getResAttrsAttr() || op.getResAttrsAttr().size() == op.getNumResults());
288 assert(!op.getArgAttrsAttr() || op.getArgAttrsAttr().size() == op.getNumArguments());
289 rewriter.modifyOpInPlace(op, [&]() {
290 op.setFunctionType(newTy);
291
292 // If any input or result types were added, ensure the attributes are updated too.
293 if (mlir::ArrayAttr newArgAttrs = convertInputAttrs(op.getArgAttrsAttr(), newInputs)) {
294 op.setArgAttrsAttr(newArgAttrs);
295 }
296 if (mlir::ArrayAttr newResAttrs = convertResultAttrs(op.getResAttrsAttr(), newResults)) {
297 op.setResAttrsAttr(newResAttrs);
298 }
299 });
300 // Post-condition: arg/result count equals corresponding attribute count
301 assert(!op.getResAttrsAttr() || op.getResAttrsAttr().size() == op.getNumResults());
302 assert(!op.getArgAttrsAttr() || op.getArgAttrsAttr().size() == op.getNumArguments());
303
304 // If the function has a body, ensure the entry block arguments match the function inputs.
305 if (mlir::Region *body = op.getCallableRegion()) {
306 mlir::Block &entryBlock = body->front();
307 bool blockArgsNeedUpdate =
308 !std::cmp_equal(entryBlock.getNumArguments(), newInputs.size()) ||
309 llvm::any_of(llvm::zip_equal(entryBlock.getArgumentTypes(), newInputs), [](auto pair) {
310 return std::get<0>(pair) != std::get<1>(pair);
311 });
312 if (blockArgsNeedUpdate) {
313 processBlockArgs(entryBlock, rewriter);
314 // Post-condition: block args must match function inputs in both arity and type.
315 assert(std::cmp_equal(entryBlock.getNumArguments(), newInputs.size()));
316 for (unsigned i = 0, e = entryBlock.getNumArguments(); i < e; ++i) {
317 assert(entryBlock.getArgument(i).getType() == newInputs[i]);
318 }
319 }
320 }
321 }

331template <
332 typename ImplClass, HasInterface<component::MemberRefOpInterface> MemberRefOpClass,
333 typename GenHeaderType, typename IdType>
334class SplitAggregateInMemberRefOp : public mlir::OpConversionPattern<MemberRefOpClass> {
335public:
337 using MemberInfo = std::pair<mlir::StringAttr, mlir::Type>;
339 using LocalMemberReplacementMap = llvm::DenseMap<IdType, MemberInfo>;
341 using MemberReplacementMap = llvm::DenseMap<
342 component::StructDefOp, llvm::DenseMap<mlir::StringAttr, LocalMemberReplacementMap>>;
343
344private:
345 mlir::SymbolTableCollection &tables;
346 const MemberReplacementMap &repMapRef;
347
348 // Static check to ensure the methods are implemented in all subclasses.
349 inline static void ensureImplementedAtCompile() {
350 static_assert(
351 sizeof(MemberRefOpClass) == 0,
352 "SplitAggregateInMemberRefOp not implemented for requested type."
353 );
354 }
355
356protected:
357 using OpAdaptor = typename MemberRefOpClass::Adaptor;
358
361 static GenHeaderType genHeader(MemberRefOpClass, mlir::ConversionPatternRewriter &) {
362 ensureImplementedAtCompile();
363 llvm_unreachable("must have concrete instantiation");
364 }
365
368 static void forId(
369 mlir::Location, GenHeaderType &, IdType, MemberInfo, OpAdaptor,
370 mlir::ConversionPatternRewriter &
371 ) {
372 ensureImplementedAtCompile();
373 llvm_unreachable("must have concrete instantiation");
374 }
375
376public:
377 // Suppress false positive from `clang-tidy`
378 // NOLINTNEXTLINE(bugprone-crtp-constructor-accessibility)
379 SplitAggregateInMemberRefOp(
380 mlir::MLIRContext *ctx, mlir::SymbolTableCollection &symTables,
381 const MemberReplacementMap &memberRepMap
382 )
383 : mlir::OpConversionPattern<MemberRefOpClass>(ctx), tables(symTables),
384 repMapRef(memberRepMap) {}
385
386 static bool legal(MemberRefOpClass) {
387 ensureImplementedAtCompile();
388 llvm_unreachable("must have concrete instantiation");
389 return false;
390 }
391
392 mlir::LogicalResult matchAndRewrite(
393 MemberRefOpClass op, OpAdaptor adaptor, mlir::ConversionPatternRewriter &rewriter
394 ) const override {
395 if (ImplClass::legal(op)) {
396 return mlir::failure();
397 }
398 component::StructType tgtStructTy =
399 llvm::cast<component::MemberRefOpInterface>(op.getOperation()).getStructType();
400 assert(tgtStructTy);
401 auto tgtStructDef = tgtStructTy.getDefinition(tables, op);
402 assert(mlir::succeeded(tgtStructDef));
403
404 GenHeaderType prefixResult = ImplClass::genHeader(op, rewriter);
405
406 const LocalMemberReplacementMap &idToName =
407 repMapRef.at(tgtStructDef->get()).at(op.getMemberNameAttr().getAttr());
408 // Split the aggregate member into a series of scalar member ops.
409 for (const auto &[id, newMember] : idToName) {
410 ImplClass::forId(op.getLoc(), prefixResult, id, newMember, adaptor, rewriter);
411 }
412 if constexpr (requires { ImplClass::finalize(op, prefixResult, adaptor, rewriter); }) {
413 ImplClass::finalize(op, prefixResult, adaptor, rewriter);
414 }
415 rewriter.eraseOp(op);
416 return mlir::success();
417 }
418};
