22template <> struct mlir::FieldParser<llvm::APInt> {
23 static mlir::FailureOr<llvm::APInt> parse(mlir::AsmParser &parser) {
24 auto loc = parser.getCurrentLocation();
25 llvm::APInt val;
26 auto result = parser.parseOptionalInteger(val);
27 if (!result.has_value() || *result) {
28 return parser.emitError(loc, "expected integer value");
29 } else {
30 return val;
31 }
32 }
33};

68inline mlir::FailureOr<DynamicAPIntValue> parseDynamicAPIntValue(mlir::AsmParser &parser) {
69 auto value = mlir::FieldParser<llvm::APInt>::parse(parser);
70 if (mlir::failed(value)) {
71 return mlir::failure();
72 }
73 return DynamicAPIntValue(llvm::DynamicAPInt(*value));
74}

78inline llvm::DynamicAPInt integerAttrToDynamicAPInt(mlir::IntegerAttr attr) {
79 auto type = llvm::dyn_cast<mlir::IntegerType>(attr.getType());
80 bool isUnsigned = type && (type.isUnsigned() || (type.isSignless() && type.getWidth() == 1));
81 return isUnsigned ? toDynamicAPInt(attr.getValue()) : llvm::DynamicAPInt(attr.getValue());
82}

87 llvm::report_fatal_error("integer is not representable as an index");
