34 llvm::Error writeFieldElement(uint32_t size, const llvm::DynamicAPInt &value) {
35 if (size == 0 || size > std::numeric_limits<unsigned>::max() / CHAR_BIT) {
36 return llvm::createStringError(llvm::inconvertibleErrorCode(), "invalid field encoding size");
37 }
38 auto exact = checkedToAPInt(value, size * CHAR_BIT, false);
39 if (!exact) {
40 return exact.takeError();
41 }
42 for (uint32_t i = 0; i < size; ++i) {
43 buffer_.push_back(static_cast<char>(exact->extractBitsAsZExtValue(8, i * 8)));
44 }
45 return llvm::Error::success();
46 }
47
48 uint64_t size() const { return static_cast<uint64_t>(buffer_.size()); }
49 llvm::ArrayRef<char> bytes() const { return buffer_; }
50
51private:
52 template <typename T> void writeInteger(T value) {
53 char bytes[sizeof(T)];
54 llvm::support::endian::write<T, llvm::endianness::little>(bytes, value);
55 writeBytes(bytes);
56 }
57
58 llvm::SmallVector<char> buffer_;
59};
