25static bool isIdentityPreservingCopy(Type type) {
26 // Struct values are identity-bearing handles: copies continue to reference the same component.
27 // Keep concrete immutable types explicit so future mutable types cannot inherit this policy.
28 return llvm::isa<
29 IndexType, IntegerType, felt::FeltType, string::StringType, component::StructType>(type);
30}

32static const ValueCopyDialectInterface *getValueCopyInterface(Type type) {

38bool requiresValueCopy(Type type) { return !isIdentityPreservingCopy(type); }

40bool canMaterializeValueCopy(Type type) {
41 if (isIdentityPreservingCopy(type)) {
42 return true;
43 }
44 if (const auto *interface = getValueCopyInterface(type)) {
45 return interface->canMaterializeValueCopy(type);
46 }
47 return false;
48}

56std::string getValueCopyFailureReason(Type type) {
57 if (canMaterializeValueCopy(type)) {
58 return {};
59 }
60 if (const auto *interface = getValueCopyInterface(type)) {
61 return interface->getValueCopyFailureReason(type);
62 }
63 std::string reason;
64 llvm::raw_string_ostream(reason) << "unsupported value-copy type '" << type << '\'';
65 return reason;

68FailureOr<Value> materializeValueCopy(OpBuilder &builder, Location loc, Value source) {
69 Type type = source.getType();
70 if (isIdentityPreservingCopy(type)) {
71 return source;
72 }
73 if (const auto *interface = getValueCopyInterface(type)) {
74 return interface->materializeValueCopy(builder, loc, source);
75 }
76 return failure();
77}
