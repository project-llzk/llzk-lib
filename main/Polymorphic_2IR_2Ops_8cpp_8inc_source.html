162std::pair<unsigned, unsigned> ApplyMapOpGenericAdaptorBase::getODSOperandIndexAndLength(unsigned index, unsigned odsOperandsSize) {
163 bool isVariadic[] = {true};
164 int prevVariadicCount = 0;
165 for (unsigned i = 0; i < index; ++i)
166 if (isVariadic[i]) ++prevVariadicCount;
167
168 // Calculate how many dynamic values a static variadic operand corresponds to.
169 // This assumes all static variadic operands have the same dynamic value count.
170 int variadicSize = (odsOperandsSize - 0) / 1;
171 // `index` passed in as the parameter is the static index which counts each
172 // operand (variadic or not) as size 1. So here for each previous static variadic
173 // operand, we need to offset by (variadicSize - 1) to get where the dynamic
174 // value pack for this static operand starts.
175 int start = index + (variadicSize - 1) * prevVariadicCount;
176 int size = isVariadic[index] ? variadicSize : 1;
177 return {start, size};
178}
