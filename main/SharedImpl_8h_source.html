48// Default to true if the check is not for that particular operation type.
49template <typename Check> inline bool runCheck(mlir::Operation *op, Check check) {
50 if (auto specificOp =
51 llvm::dyn_cast_if_present<typename llvm::function_traits<Check>::template arg_t<0>>(op)) {
52 return check(specificOp);
53 }
54 return true;
55}

194template <typename... AdditionalOpClasses, typename... AdditionalChecks>
195mlir::ConversionTarget newConverterDefinedTargetWithCallback(
196 mlir::TypeConverter &tyConv, mlir::MLIRContext *ctx, LegalityCheckCallback &cb,
197 AdditionalChecks &&...checks
198) {
199 mlir::ConversionTarget target = newBaseTarget(ctx);
200 auto inserter = [&](auto... opClasses) {
201 target.addDynamicallyLegalOp<decltype(opClasses)...>([&cb, &tyConv,
202 &checks...](mlir::Operation *op) {
203 LLVM_DEBUG(if (op) {
204 llvm::dbgs() << "[newConverterDefinedTarget] checking legality of ";
205 op->dump();
206 });
207 cb.checkStarted();
208 auto legality =
209 defaultLegalityCheck(tyConv, op) && (runCheck<AdditionalChecks>(op, checks) && ...);
210
211 cb.checkEnded(legality);
212 LLVM_DEBUG(if (legality) { llvm::dbgs() << "[newConverterDefinedTarget] is legal\n"; } else {
213 llvm::dbgs() << "[newConverterDefinedTarget] is not legal\n";
214 });
215 return legality;
216 });
217 };
218 std::apply(inserter, OpClassesWithStructTypes.NoGeneralBuilder);
219 std::apply(inserter, OpClassesWithStructTypes.WithGeneralBuilder);
220 applyToMoreTypes<decltype(inserter), AdditionalOpClasses...>(inserter);
221 return target;
222}
