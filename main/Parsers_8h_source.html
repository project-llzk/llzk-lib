31struct NestedTextualOptions {
33 std::string str;
34
40 template <typename ValueT, typename CreateFnT, typename InitializeFnT>
41 std::unique_ptr<ValueT> createValidatedValue(
42 CreateFnT &&createValue, llvm::StringRef kind, InitializeFnT &&initializeValue
43 ) const {
44 auto value = createValue();
45 if (str.empty()) {
46 return value;
47 }
48
49 std::string error;
50 if (failed(initializeValue(*value, str, error))) {
51 llvm::report_fatal_error(
52 llvm::Twine("failed to initialize previously-validated nested ") + kind +
53 " options: " + error
54 );
55 }
56 return value;
57 }
58};
59
60} // namespace detail

64template <auto CreatePass> struct NestedPassOptions : detail::NestedTextualOptions {
66 std::unique_ptr<mlir::Pass> createPass() const {
67 return this->createValidatedValue<mlir::Pass>(CreatePass, "pass", initializePass);
68 }
69
70 static mlir::LogicalResult
71 initializePass(mlir::Pass &pass, llvm::StringRef options, std::string &error) {
72 return pass.initializeOptions(options, [&error](const llvm::Twine &message) {
73 error = message.str();
74 return mlir::failure();
75 });
76 }
77};

81template <typename PipelineOptionsT> struct NestedPipelineOptions : detail::NestedTextualOptions {
83 std::unique_ptr<PipelineOptionsT> createOptions() const {
84 return this->createValidatedValue<PipelineOptionsT>(
85 std::make_unique<PipelineOptionsT>, "pipeline", initializeOptions
86 );
87 }
88
89 static mlir::LogicalResult
90 initializeOptions(PipelineOptionsT &options, llvm::StringRef value, std::string &error) {
91 llvm::raw_string_ostream errorStream(error);
92 return options.parseFromString(value, errorStream);
93 }
94};

102template <typename OptionsT> class NestedOptionsParserBase : public basic_parser<OptionsT> {
103public:
104 NestedOptionsParserBase(Option &O) : basic_parser<OptionsT>(O) {}
105
106protected:
109 bool parseNestedOptions(Option &O, StringRef Arg, StringRef kind, StringRef &options) const {
110 options = Arg;
111 if (options.consume_front("{") && !options.consume_back("}")) {
112 return O.error(llvm::Twine("expected nested ") + kind + " options to end with '}'");
113 }
114 return false;
115 }
116
117public:
118 static void print(llvm::raw_ostream &OS, const OptionsT &Val) { OS << '{' << Val.str << '}'; }
119
120 void printOptionDiff(
121 const Option &O, const OptionsT &V, const OptionValue<OptionsT> &Default, size_t GlobalWidth
122 ) const {
123 this->printOptionName(O, GlobalWidth);
124 print(llvm::outs(), V);
125 llvm::outs() << " (default: ";
126 if (Default.hasValue()) {
127 print(llvm::outs(), Default.getValue());
128 } else {
129 llvm::outs() << "<unspecified>";
130 }
131 llvm::outs() << ")\n";
132 }
133};

136template <auto CreatePass>
137class parser<llzk::NestedPassOptions<CreatePass>>
138 : public NestedOptionsParserBase<llzk::NestedPassOptions<CreatePass>> {
139public:
140 using OptionsT = llzk::NestedPassOptions<CreatePass>;
141 using Base = NestedOptionsParserBase<OptionsT>;
142
143 parser(Option &O) : Base(O) {}
144
145 bool parse(Option &O, StringRef, StringRef Arg, OptionsT &Val) {
146 StringRef options;
147 if (this->parseNestedOptions(O, Arg, "pass", options)) {
148 return true;
149 }
150
151 auto pass = CreatePass();
152 std::string error;
153 if (failed(OptionsT::initializePass(*pass, options, error))) {
154 return O.error(error);
155 }
156
157 Val.str = options.str();
158 return false;
159 }
160};
