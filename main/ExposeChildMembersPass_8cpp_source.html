36class ChildStorageVisitor {
38 SymbolTableCollection tables;
40 StructDefOp storageRoot;
42 StructSpecializationResolver specializations;
44 llvm::DenseSet<Operation *> visited;
46 SmallVector<StructDefOp> children;
47
48public:
49 ChildStorageVisitor(ModuleOp module, StructDefOp root)
50 : storageRoot(root), specializations(module) {}
51
53 LogicalResult run() {
54 if (failed(visitStruct(storageRoot))) {
55 return failure();
56 }
57 for (auto child : children) {
58 for (auto member : child.getMemberDefs()) {
59 member.setPublicAttr();
60 }
61 }
62 return success();
63 }
64
65private:
67 LogicalResult visitStruct(StructDefOp definition) {
68 if (isInTemplate(definition)) {
69 return definition.emitError("llzk-expose-child-members requires monomorphized struct types");
70 }
71 if (!visited.insert(definition).second) {
72 return success();
73 }
74 if (definition != storageRoot) {
75 children.push_back(definition);
76 }
77 for (auto member : definition.getMemberDefs()) {
78 if (failed(visitType(member.getType(), member))) {
79 return failure();
80 }
81 }
82 return success();
83 }
84
85
87 LogicalResult visitFamily(StructType structure, Operation *site) {
88 Attribute metadata = site->getAttr(FAMILY_ATTR);
89 if (!metadata) {
90 return site->emitError(
91 "missing concrete struct family element for llzk-expose-child-members"
92 );
93 }
94 auto records = dyn_cast<ArrayAttr>(metadata);
95 if (!records) {
96 return site->emitError("invalid poly.family metadata for llzk-expose-child-members");
97 }
98 llvm::DenseSet<int64_t> seenIds;
99 for (Attribute record : records) {
100 auto dict = dyn_cast<DictionaryAttr>(record);
101 auto type = dict ? dict.getAs<TypeAttr>(FAMILY_TYPE_KEY) : TypeAttr();
102 if (!type || !isa<StructType>(type.getValue())) {
103 return site->emitError("invalid poly.family entry for llzk-expose-child-members");
104 }
105 if (type.getValue() != structure) {
106 continue;
107 }
108 auto id = dict.getAs<IntegerAttr>(FAMILY_SPECIALIZATION_KEY);
109 if (!id || !id.getValue().isSignedIntN(64) || id.getInt() < 0) {
110 return site->emitError("invalid poly.family entry for llzk-expose-child-members");
111 }
112 if (!seenIds.insert(id.getInt()).second) {
113 continue;
114 }
115 auto definition = specializations.resolve(
116 tables, structure, id.getInt(), site, "llzk-expose-child-members"
117 );
118 if (failed(definition) || failed(visitStruct(*definition))) {
119 return failure();
120 }
121 }
122 if (seenIds.empty()) {
123 return site->emitError(
124 "missing concrete struct family element for llzk-expose-child-members"
125 );
126 }
127 return success();
128 }
129
131 LogicalResult visitType(Type type, Operation *site) {
132 if (!walkContains<StructType>(type)) {
133 return success();
134 }
135 if (auto array = dyn_cast<array::ArrayType>(type)) {
136 if (!array.hasStaticShape()) {
137 return site->emitError("llzk-expose-child-members requires concrete array dimensions");
138 }
139 if (llvm::is_contained(array.getShape(), 0)) {
140 return success();
141 }
142 return visitType(array.getElementType(), site);
143 }
144 if (auto pod = dyn_cast<pod::PodType>(type)) {
145 for (auto record : pod.getRecords()) {
146 if (failed(visitType(record.getType(), site))) {
147 return failure();
148 }
149 }
150 return success();
151 }
152 auto structure = cast<StructType>(type);
153 if (auto params = structure.getParams(); params && !params.empty()) {
154 return visitFamily(structure, site);
155 }
156 auto definition = structure.getDefinition(tables, site);
157 if (failed(definition)) {
158 return failure();
159 }
160 if (!*definition || definition->viaInclude()) {
161 return site->emitError("cannot resolve child struct storage; inline includes first");
162 }
163 return visitStruct(definition->get());
164 }
165};

168class PassImpl : public llzk::component::impl::ExposeChildMembersPassBase<PassImpl> {
169 using Base = ExposeChildMembersPassBase<PassImpl>;
170 using Base::Base;
171
172 void runOnOperation() override {
173 SymbolTableCollection tables;
174 auto root = getRootStructDef(tables, getOperation(), rootStruct);
175 if (failed(root)) {
176 signalPassFailure();
177 return;
178 }
179 if (!*root) {
180 getOperation().emitError(
181 "llzk-expose-child-members requires llzk.main or an explicit root-struct"
182 );
183 signalPassFailure();
184 return;
185 }
186 if (root->viaInclude()) {
187 getOperation().emitError(
188 "llzk-expose-child-members requires a local root struct; inline includes first"
189 );
190 signalPassFailure();
191 return;
192 }
193 if (failed(exposeChildMembers(root->get()))) {
194 signalPassFailure();
195 }
196 }
197};

200LogicalResult llzk::component::exposeChildMembers(StructDefOp root) {
201 auto module = getTopRootModule(root);
202 if (failed(module)) {
203 return failure();
204 }
205 return ChildStorageVisitor(*module, root).run();
206}
