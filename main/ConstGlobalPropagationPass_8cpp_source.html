47static Value convertToConstantValue(GlobalReadOp readOp, Attribute attr) {
48 OpBuilder bldr(readOp);
49 Location loc = readOp.getLoc();
50 if (auto intAttr = llvm::dyn_cast<IntegerAttr>(attr)) {
51 return bldr.create<arith::ConstantOp>(loc, intAttr).getResult();
52 } else if (auto feltAttr = llvm::dyn_cast<felt::FeltConstAttr>(attr)) {
53 Value constant = bldr.create<felt::FeltConstantOp>(loc, feltAttr).getResult();
54 if (constant.getType() == readOp.getType()) {
55 return constant;
56 }
57 return bldr.create<polymorphic::UnifiableCastOp>(loc, readOp.getType(), constant).getResult();
58 } else if (auto strAttr = llvm::dyn_cast<StringAttr>(attr)) {
59 return bldr.create<string::LitStringOp>(loc, readOp.getType(), strAttr).getResult();
60 } else {
61 llvm::outs() << "Encountered: " << attr.getAbstractAttribute().getName() << '\n';
62 llvm_unreachable("Unsupported constant attribute type");
63 }
64}

66class PassImpl : public llzk::global::impl::ConstGlobalPropagationPassBase<PassImpl> {
67 using Base = ConstGlobalPropagationPassBase<PassImpl>;
68
69public:
70 using Base::Base;

72 void runOnOperation() override {
73 ModuleOp root = getOperation();
74 auto constGlobals = walkCollect<GlobalDefOp>(root, [](auto g) { return g.isConstant(); });
75 if (constGlobals.empty()) {
76 return;
77 }
78
79 // Keep track of erased operations to skip them during the inner loop iteration and avoid
80 // use-after-free errors. These MUST NOT be dereferenced because the ops have been erased.
81 llvm::DenseSet<void *> erased;
82 SymbolUseGraph &useGraph = getAnalysis<SymbolUseGraph>();
83 llvm::DenseMap<Operation *, SmallVector<const SymbolUseGraphNode *>> useNodesByDefinition;
84 SymbolTableCollection tables;
85 for (const SymbolUseGraphNode *candidate : useGraph.nodesIter()) {
86 auto target = candidate->lookupSymbol(tables, /*reportMissing=*/false);
87 if (succeeded(target)) {
88 useNodesByDefinition[target->get()].push_back(candidate);
89 }
90 }
91
92 for (GlobalDefOp globalDef : constGlobals) {
93 Attribute constValue = globalDef.getInitialValueAttr();
94
95 // Array constants cannot yet be materialized as operations. Leave both
96 // the global and all of its uses unchanged until they are supported.
97 if (llvm::isa<ArrayAttr>(constValue)) {
98 continue;
99 }
100
101 // A global definition and its uses can have different closest LLZK root
102 // modules after include inlining. Find every use-graph node that resolves
103 // to this definition instead of looking it up by a single symbol path.
104 for (const SymbolUseGraphNode *node : useNodesByDefinition[globalDef.getOperation()]) {
105 SymbolRefAttr symbolAttr = node->getSymbolPath();
106 for (Operation *userOp : node->getUserOps()) {
107 // Skip erased ops to avoid use-after-free errors
108 if (erased.contains(userOp)) {
109 continue;
110 }
111 LLVM_DEBUG(
112 llvm::dbgs() << "Replacing '" << symbolAttr << "' with '" << constValue << "' in "
113 << *userOp << '\n'
114 );
115
116 // Special handling to fully replace GlobalReadOp with the appropriate constant op.
117 if (auto readOp = llvm::dyn_cast<GlobalReadOp>(userOp);
118 readOp && readOp.getNameRef() == symbolAttr) {
119 Value constantResult = convertToConstantValue(readOp, constValue);
120 readOp.getResult().replaceAllUsesWith(constantResult);
121 readOp.erase();
122 erased.insert(readOp);
123 continue;
124 }
125
126 // Standard case that just replaces all uses of the symbol with the constant value.
127 AttrTypeReplacer replacer;
128 replacer.addReplacement([symbolAttr, constValue](SymbolRefAttr a) {
129 return (a == symbolAttr) ? std::make_optional(constValue) : std::nullopt;
130 });
131 replacer.replaceElementsIn(
132 userOp,
133 /*replaceAttrs=*/true,
134 /*replaceLocs=*/false,
135 /*replaceTypes=*/true
136 );
137 }
138 }
139 // Delete the const global since there are no (remaining) uses.
140 globalDef.erase();
141 erased.insert(globalDef);
142 }
143 }
144};
