25class ConstraintScalarFacts {
26 mlir::Builder builder;
29 llvm::DenseMap<mlir::Value, mlir::Attribute> constants;
30
32 void fold(mlir::Operation *producer, mlir::Value current) {
33 llvm::SmallVector<mlir::Attribute> operands;
34 for (mlir::Value operand : producer->getOperands()) {
35 operands.push_back(constants.lookup(operand));
36 }
37 // Casts expose canonicalization rather than fold hooks. Match the same
38 // nonnegative, signed-64 representability rule for structural indices.
39 if (llvm::isa<llzk::cast::FeltToIndexOp>(producer)) {
40 if (auto constant = llvm::dyn_cast_or_null<felt::FeltConstAttr>(operands.front())) {
41 auto reduced = constant.getReducedValue();
42 if (!reduced) {
43 return;
44 }
45 auto converted = checkedToInt64(*reduced);
46 if (converted && *converted >= 0) {
47 constants[current] = builder.getIndexAttr(*converted);
48 } else if (!converted) {
49 llvm::consumeError(converted.takeError());
50 }
51 }
52 return;
53 }
54 mlir::Operation *temporary = producer->clone();
55 llvm::SmallVector<mlir::OpFoldResult> results;
56 if (mlir::succeeded(temporary->fold(operands, results)) &&
57 results.size() == producer->getNumResults()) {
58 for (auto [result, folded] : llvm::zip_equal(producer->getResults(), results)) {
59 constants[result] = llvm::isa<mlir::Attribute>(folded)
60 ? llvm::cast<mlir::Attribute>(folded)
61 : constants.lookup(llvm::cast<mlir::Value>(folded));
62 }
63 }
64 temporary->destroy();
65 }

72 mlir::Attribute known(mlir::Value value) {
73 llvm::SmallVector<std::pair<mlir::Value, bool>> pending {{value, false}};
74 llvm::DenseSet<mlir::Value> examined;
75 while (!pending.empty()) {
76 auto [current, ready] = pending.pop_back_val();
77 if (constants.contains(current)) {
78 continue;
79 }
80 mlir::Attribute literal;
81 if (mlir::matchPattern(current, mlir::m_Constant(&literal))) {
82 constants[current] = literal;
83 continue;
84 }
85 mlir::Operation *producer = current.getDefiningOp();
86 if (!producer || producer->getNumRegions() || !mlir::isMemoryEffectFree(producer)) {
87 constants.try_emplace(current, mlir::Attribute());
88 continue;
89 }
90 if (!ready) {
91 if (!examined.insert(current).second) {
92 continue;
93 }
94 pending.push_back({current, true});
95 for (mlir::Value operand : producer->getOperands()) {
96 pending.push_back({operand, false});
97 }
98 continue;
99 }
100 fold(producer, current);
101 constants.try_emplace(current, mlir::Attribute());
102 }
103 return constants.lookup(value);
104 }
105
107 std::optional<int64_t> integer(mlir::Value value) {
108 auto attr = llvm::dyn_cast_or_null<mlir::IntegerAttr>(known(value));
109 if (!attr || !attr.getValue().isSignedIntN(64)) {
110 return std::nullopt;
111 }
112 return attr.getInt();
113 }
114};
