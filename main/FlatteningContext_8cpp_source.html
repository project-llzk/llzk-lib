47std::optional<APInt> getConstantTripCount(LoopLikeOpInterface loopOp) {
48 if (auto whileOp = dyn_cast<scf::WhileOp>(loopOp.getOperation());
49 whileOp && matchPattern(whileOp.getConditionOp().getCondition(), m_Zero())) {
50 return APInt(64, 0);
51 }
52 std::optional<OpFoldResult> lbVal = loopOp.getSingleLowerBound();
53 std::optional<OpFoldResult> ubVal = loopOp.getSingleUpperBound();
54 std::optional<OpFoldResult> stepVal = loopOp.getSingleStep();
55 if (!lbVal || !ubVal || !stepVal) {
56 return std::nullopt;
57 }
58 bool isSigned = true;
59 if (auto forOp = dyn_cast<scf::ForOp>(loopOp.getOperation())) {
60 isSigned = !forOp.getUnsignedCmp();
61 }
62 return constantTripCount(*lbVal, *ubVal, *stepVal, isSigned, scf::computeUbMinusLb);
63}

66static std::optional<std::string> getUnrollSkipReason(Operation *op, uint64_t limit) {
67 if (!isa<scf::ForOp, scf::WhileOp, affine::AffineForOp>(op)) {
68 return std::nullopt;
69 }
70 uint64_t tripCount;
71 auto bounds = op->getAttrOfType<LoopBoundsAttr>(LoopBoundsAttr::name);
72 if (bounds) {
73 if (bounds.getStep() == 0) {
74 return std::nullopt; // The annotated unroller diagnoses invalid steps separately.
75 }
76 llvm::DynamicAPInt count =
77 llvm::ceilDiv(bounds.getUpper() - bounds.getLower(), bounds.getStep());
78 auto checked = checkedToUInt64(std::max(count, llvm::DynamicAPInt(0)));
79 if (!checked) {
80 llvm::consumeError(checked.takeError());
81 return "llzk.loopbounds trip count does not fit in uint64_t; leaving loop unchanged";
82 }
83 tripCount = *checked;
84 } else {
85 if (isa<scf::WhileOp>(op)) {
86 return std::nullopt;
87 }
88 auto count = getConstantTripCount(llvm::cast<LoopLikeOpInterface>(op));
89 if (!count) {
90 return std::nullopt;
91 }
92 if (count->getActiveBits() > 64) {
93 return "computed loop trip count does not fit in uint64_t; leaving loop unchanged";
94 }
95 tripCount = count->getZExtValue();
96 }
97 if (tripCount > limit) {
98 return ("loop trip count " + Twine(tripCount) + " exceeds max-unroll limit of " + Twine(limit) +
99 "; leaving loop unchanged")
100 .str();
101 }
102 if (bounds && tripCount != 0) {
103 Value iv;
104 bool isSigned = true;
105 if (auto forOp = dyn_cast<scf::ForOp>(op)) {
106 iv = forOp.getInductionVar();
107 isSigned = !forOp.getUnsignedCmp();
108 } else if (auto affineForOp = dyn_cast<affine::AffineForOp>(op)) {
109 iv = affineForOp.getInductionVar();
110 }
111 if (iv && !iv.use_empty()) {
112 Type ivType = iv.getType();
113 unsigned width =
114 ivType.isIndex() ? IndexType::kInternalStorageBitWidth : ivType.getIntOrFloatBitWidth();
115 const llvm::DynamicAPInt &lower = bounds.getLower();
116 auto last = lower + toDynamicAPInt(tripCount - 1) * bounds.getStep();
117 // All intermediate induction values lie between these endpoints.
118 for (const auto &value : {lower, last}) {
119 auto encoded = checkedToAPInt(value, width, isSigned);
120 if (!encoded) {
121 llvm::consumeError(encoded.takeError());
122 return "llzk.loopbounds induction value does not fit the loop induction type; "
123 "leaving loop unchanged";
124 }
125 }
126 }
127 }
128 return std::nullopt;
129}

154bool ConversionTracker::shouldSkipUnroll(Operation *op) {
155 auto reason = getUnrollSkipReason(op, unrollIterationLimit);
156 if (!reason) {
157 return false;
158 }
159 if (unrollWarnings.insert({op->getLoc(), StringAttr::get(op->getContext(), *reason)}).second) {
160 op->emitWarning(*reason);
161 }
162 return true;

165bool ConversionTracker::mustPreserveLoop(Operation *op) {
166 if (!op) {
167 return false;
168 }
169 for (Operation *p = op; p && !isa<SymbolOpInterface>(p); p = p->getParentOp()) {
170 if (shouldSkipUnroll(p)) {
171 return true;
172 }
173 }
174 return walkContains<Operation *>(*op, [this](Operation *nested) {
175 return shouldSkipUnroll(nested);
176 });
177}

179DenseSet<Operation *> ConversionTracker::getPreservedLoopOps(ModuleOp modOp) {
180 DenseSet<Operation *> preserved;
181 modOp.walk<WalkOrder::PreOrder>([&](Operation *op) {
182 if (!shouldSkipUnroll(op)) {
183 return WalkResult::advance();
184 }
185 op->walk([&preserved](Operation *nested) {
186 if (nested->getNumRegions() != 0) {
187 preserved.insert(nested);
188 }
189 });
190 for (Operation *p = op->getParentOp(); p && !isa<SymbolOpInterface>(p); p = p->getParentOp()) {
191 preserved.insert(p);
192 }
193 return WalkResult::skip();
194 });
195 return preserved;
196}

198Type ConversionTracker::scopeType(Type type, Operation *scope, bool toCache) const {
199 if (!scope || !cacheRoot) {
200 return type;
201 }
202 auto root = getRootModule(scope);
203 assert(succeeded(root) && "verified operations have a lookup root");
204 if (*root == cacheRoot) {
205 return type;
206 }
207 auto path =
208 getPathRelativeToAncestor(llvm::cast<SymbolOpInterface>(root->getOperation()), cacheRoot);
209 assert(succeeded(path) && "nested lookup roots must have a symbol path");
210 auto prefix = getPieces(*path);
211 AttrTypeReplacer names;
212 names.addReplacement([&](StructType nested) -> std::optional<Type> {
213 auto pieces = getPieces(nested.getNameRef());
214 if (toCache) {
215 pieces.insert(pieces.begin(), prefix.begin(), prefix.end());
216 } else {
217 assert(llvm::equal(ArrayRef(pieces).take_front(prefix.size()), prefix));
218 pieces.erase(pieces.begin(), pieces.begin() + prefix.size());
219 }
220 return StructType::get(asSymbolRefAttr(pieces), nested.getParams());
221 });
222 return names.replace(type);
223}

230void ConversionTracker::recordInstantiation(StructType oldType, StructType newType) {
231 assert(!isNullOrEmpty(oldType.getParams()) && "cannot instantiate with no params");
232
233 auto forwardResult = structInstantiations.try_emplace(oldType, newType);
234 if (forwardResult.second) {
235 // Insertion was successful
236 // ASSERT: The reverse map does not contain this mapping either
237 assert(!reverseInstantiations.contains(newType));
238 reverseInstantiations[newType] = oldType;
239 // Set the modified flag
240 modified = true;
241 } else {
242 // ASSERT: If a mapping already existed for `oldType` it must be `newType`
243 assert(forwardResult.first->getSecond() == newType);
244 // ASSERT: The reverse mapping is already present as well
245 assert(reverseInstantiations.lookup(newType) == oldType);
246 }
247 assert(structInstantiations.size() == reverseInstantiations.size());
248}

296void ConversionTracker::reportDelayedDiagnostics(StructType newType, CallOp caller) {
297 newType = llvm::cast<StructType>(scopeType(newType, caller));
298 auto res = delayedDiagnostics.find(newType);
299 if (res != delayedDiagnostics.end()) {
300 llzk::polymorphic::detail::reportDelayedDiagnostics(caller, std::move(res->second));
301
302 // Emitting a Diagnostic consumes it (per DiagnosticEngine::emit) so remove them from the map.
303 // Unfortunately, this means if the key StructType is the result of instantiation at multiple
304 // `compute()` calls it will only be reported at one of those locations, not all.
305 delayedDiagnostics.erase(newType);
306 }
307}

321bool ConversionTracker::isLegalConversion(
322 Type oldType, Type newType, const char *patName, Operation *scope
323) const {
324 oldType = scopeType(oldType, scope);
325 newType = scopeType(newType, scope);
326 std::function<bool(Type, Type)> checkInstantiations = [&](Type oTy, Type nTy) {
327 // Check if `oTy` is a struct with a known instantiation to `nTy`
328 if (StructType oldStructType = llvm::dyn_cast<StructType>(oTy)) {
329 // Note: The values in `structInstantiations` must be no-parameter struct types
330 // so there is no need for recursive check, simple equality is sufficient.
331 if (this->structInstantiations.lookup(oldStructType) == nTy) {
332 return true;
333 }
334 }
335 // Check if `nTy` is the result of a struct instantiation and if the pre-image of
336 // that instantiation (i.e., the parameterized version of the instantiated struct)
337 // is a more concrete unification of `oTy`.
338 if (StructType newStructType = llvm::dyn_cast<StructType>(nTy)) {
339 if (auto preImage = this->reverseInstantiations.lookup(newStructType)) {
340 if (isMoreConcreteUnification(oTy, preImage, checkInstantiations)) {
341 return true;
342 }
343 }
344 }
345 return false;
346 };
347
348 if (isMoreConcreteUnification(oldType, newType, checkInstantiations)) {
349 return true;
350 }
351 LLVM_DEBUG(
352 llvm::dbgs() << "[" << patName << "] Cannot replace old type " << oldType << " with new type "
353 << newType
354 << " because it does not define a compatible and more concrete type.\n";
355 );
356 return false;
357}
