41class ConstraintBindings {
42 struct Root {
43 mlir::Value input;
44 mlir::Value storage;
45 unsigned protections = 0;
46 llvm::DenseMap<mlir::ArrayAttr, mlir::Value> leaves;
47 };
48
49 mlir::Builder builder;
50 llvm::SmallVector<Root> roots;
51 uint64_t &steps;
52 uint64_t limit;

59 ConstraintBindings(mlir::MLIRContext *context, uint64_t &work, uint64_t maximum)
60 : builder(context), steps(work), limit(maximum) {}

63 AccessPath create(mlir::Type type, mlir::Value input = {}, mlir::Value storage = {}) {
64 unsigned id = roots.size();
65 roots.emplace_back();
66 roots.back().input = input;
67 roots.back().storage = storage;
68 return {id, builder.getArrayAttr({}), type};
69 }

72 AccessPath child(AccessPath parent, mlir::Attribute step, mlir::Type type) {
73 llvm::SmallVector<mlir::Attribute> path(parent.steps.begin(), parent.steps.end());
74 path.push_back(step);
75 return {parent.root, builder.getArrayAttr(path), type};
76 }
77
79 AccessPath element(AccessPath parent, int64_t index) {
80 auto type = llvm::cast<array::ArrayType>(parent.type);
81 auto shape = type.getShape().drop_front();
82 mlir::Type selected =
83 shape.empty() ? type.getElementType() : type.cloneWith(shape, type.getElementType());
84 return child(parent, builder.getIndexAttr(index), selected);
85 }
86
88 mlir::FailureOr<AccessPath> select(AccessPath path, mlir::Attribute step, mlir::Operation *site) {
89 if (auto index = llvm::dyn_cast<mlir::IntegerAttr>(step)) {
90 auto type = llvm::dyn_cast<array::ArrayType>(path.type);
91 if (!type || type.getShape().front() < 0) {
92 return constraintError(site, "aggregate shape must be static");
93 }
94 int64_t coordinate = index.getInt();
95 if (coordinate < 0 || coordinate >= type.getShape().front()) {
96 return constraintError(site, "array path index is out of bounds");
97 }
98 return element(path, coordinate);
99 }
100 auto type = llvm::dyn_cast<pod::PodType>(path.type);
101 auto name = llvm::dyn_cast<mlir::StringAttr>(step);
102 if (!type || !name) {
103 return constraintError(site, "invalid POD path");
104 }
105 auto record = type.getRecord(name.getValue(), [site] {
106 return site->emitError(CONSTRAINT_DIAGNOSTIC_PREFIX);
107 });
108 if (mlir::failed(record)) {
109 return mlir::failure();
110 }
111 return child(path, name, *record);
112 }
113
115 mlir::LogicalResult elements(
116 AccessPath path, mlir::Operation *site,
117 llvm::function_ref<mlir::LogicalResult(AccessPath)> visit
118 ) {
119 auto type = llvm::dyn_cast<array::ArrayType>(path.type);
120 if (!type) {
121 return visit(path);
122 }
123 int64_t size = type.getShape().front();
124 if (size < 0) {
125 return constraintError(site, "aggregate shape must be static");
126 }
127 for (int64_t i = 0; i < size; ++i) {
128 if (++steps > limit) {
129 return constraintError(site, "evaluation step limit exceeded");
130 }
131 if (mlir::failed(elements(element(path, i), site, visit))) {
132 return mlir::failure();
133 }
134 }
135 return mlir::success();
136 }
137
139 mlir::LogicalResult walk(
140 AccessPath path, mlir::Operation *site,
141 llvm::function_ref<mlir::LogicalResult(AccessPath)> visit
142 ) {
143 if (llvm::isa<mlir::NoneType>(path.type)) {
144 return mlir::success();
145 }
146 if (llvm::isa<array::ArrayType>(path.type)) {
147 return elements(path, site, [this, site, visit](AccessPath item) {
148 return walk(item, site, visit);
149 });
150 }
151 if (auto type = llvm::dyn_cast<pod::PodType>(path.type)) {
152 for (auto record : type.getRecords()) {
153 if (++steps > limit) {
154 return constraintError(site, "evaluation step limit exceeded");
155 }
156 auto field = child(path, record.getName(), record.getType());
157 if (mlir::failed(walk(field, site, visit))) {
158 return mlir::failure();
159 }
160 }
161 return mlir::success();
162 }
163 return visit(path);
164 }
165
167 mlir::FailureOr<mlir::Value> load(AccessPath path, mlir::Operation *site, StorageReader read) {
168 auto &leaves = roots[path.root].leaves;
169 if (auto found = leaves.find(path.steps); found != leaves.end()) {
170 return found->second;
171 }
172 if (mlir::Value storage = roots[path.root].storage) {
173 auto value = read(storage, path.steps.getValue(), path.type, site);
174 if (mlir::failed(value)) {
175 return mlir::failure();
176 }
177 leaves[path.steps] = *value;
178 return *value;
179 }
180 return constraintError(
181 site, (roots[path.root].input || roots[path.root].protections)
182 ? "read of uninitialized function input storage"
183 : "read of uninitialized local storage"
184 );
185 }
186
188 void protect(AccessPath path) { ++roots[path.root].protections; }
189
191 void unprotect(AccessPath path) { --roots[path.root].protections; }
192
194 mlir::LogicalResult writable(AccessPath path, mlir::Operation *site) const {
195 const Root &root = roots[path.root];
196 if (root.input || root.storage || root.protections) {
197 return constraintError(site, "cannot write to function input or original storage");
198 }
199 return mlir::success();
200 }
201
203 void store(AccessPath path, mlir::Value value) { roots[path.root].leaves[path.steps] = value; }
204
207 mlir::LogicalResult
208 transfer(AccessPath destination, AccessPath source, mlir::Operation *site, StorageReader read) {
209 llvm::SmallVector<std::pair<mlir::ArrayAttr, mlir::Value>> snapshot;
210 bool original = bool(roots[source.root].storage);
211 auto collect = [this, original, site, read, destination, source,
212 &snapshot](AccessPath leaf) -> mlir::LogicalResult {
213 const auto &leaves = roots[leaf.root].leaves;
214 if (!leaves.contains(leaf.steps) && !original) {
215 return mlir::success();
216 }
217 auto value = load(leaf, site, read);
218 if (mlir::failed(value)) {
219 return mlir::failure();
220 }
221 AccessPath target = destination;
222 for (auto step : leaf.steps.getValue().drop_front(source.steps.size())) {
223 auto selected = select(target, step, site);
224 if (mlir::failed(selected)) {
225 return mlir::failure();
226 }
227 target = *selected;
228 }
229 snapshot.push_back({target.steps, *value});
230 return mlir::success();
231 };
232 if (mlir::failed(walk(source, site, collect))) {
233 return mlir::failure();
234 }
235 // Visit only the overwritten subtree, rather than scanning unrelated leaves
236 // in the root on every array.insert. The snapshot makes overlap safe.
237 if (mlir::failed(walk(destination, site, [this](AccessPath leaf) {
238 roots[leaf.root].leaves.erase(leaf.steps);
239 return mlir::success();
240 }))) {
241 return mlir::failure();
242 }
243 auto &targetLeaves = roots[destination.root].leaves;
244 for (auto [path, value] : snapshot) {
245 targetLeaves[path] = value;
246 }
247 return mlir::success();
248 }
249};
