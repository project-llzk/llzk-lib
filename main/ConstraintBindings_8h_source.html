41class ConstraintBindings {
42 struct Root {
43 mlir::Value input;
44 mlir::Value storage;
45 unsigned protections = 0;
46 llvm::DenseMap<mlir::ArrayAttr, mlir::Value> leaves;
47 };
48
49 mlir::Builder builder;
50 llvm::SmallVector<Root> roots;
51
52public:
53 using StorageReader = llvm::function_ref<mlir::FailureOr<mlir::Value>(
54 mlir::Value, llvm::ArrayRef<mlir::Attribute>, mlir::Type, mlir::Operation *
55 )>;

57 explicit ConstraintBindings(mlir::MLIRContext *context) : builder(context) {}

60 AccessPath create(mlir::Type type, mlir::Value input = {}, mlir::Value storage = {}) {
61 unsigned id = roots.size();
62 roots.emplace_back();
63 roots.back().input = input;
64 roots.back().storage = storage;
65 return {id, builder.getArrayAttr({}), type};
66 }
67
69 AccessPath child(AccessPath parent, mlir::Attribute step, mlir::Type type) {
70 llvm::SmallVector<mlir::Attribute> path(parent.steps.begin(), parent.steps.end());
71 path.push_back(step);
72 return {parent.root, builder.getArrayAttr(path), type};
73 }
74
76 AccessPath element(AccessPath parent, int64_t index) {
77 auto type = llvm::cast<array::ArrayType>(parent.type);
78 auto shape = type.getShape().drop_front();
79 mlir::Type selected =
80 shape.empty() ? type.getElementType() : type.cloneWith(shape, type.getElementType());
81 return child(parent, builder.getIndexAttr(index), selected);
82 }
83
85 mlir::FailureOr<AccessPath> select(AccessPath path, mlir::Attribute step, mlir::Operation *site) {
86 if (auto index = llvm::dyn_cast<mlir::IntegerAttr>(step)) {
87 auto type = llvm::dyn_cast<array::ArrayType>(path.type);
88 if (!type || type.getShape().front() < 0) {
89 return constraintError(site, "aggregate shape must be static");
90 }
91 int64_t coordinate = index.getInt();
92 if (coordinate < 0 || coordinate >= type.getShape().front()) {
93 return constraintError(site, "array path index is out of bounds");
94 }
95 return element(path, coordinate);
96 }
97 auto type = llvm::dyn_cast<pod::PodType>(path.type);
98 auto name = llvm::dyn_cast<mlir::StringAttr>(step);
99 if (!type || !name) {
100 return constraintError(site, "invalid POD path");
101 }
102 auto record = type.getRecord(name.getValue(), [site] {
103 return site->emitError(CONSTRAINT_DIAGNOSTIC_PREFIX);
104 });
105 if (mlir::failed(record)) {
106 return mlir::failure();
107 }
108 return child(path, name, *record);
109 }
110
112 mlir::LogicalResult elements(
113 AccessPath path, mlir::Operation *site,
114 llvm::function_ref<mlir::LogicalResult(AccessPath)> visit
115 ) {
116 auto type = llvm::dyn_cast<array::ArrayType>(path.type);
117 if (!type) {
118 return visit(path);
119 }
120 int64_t size = type.getShape().front();
121 if (size < 0) {
122 return constraintError(site, "aggregate shape must be static");
123 }
124 for (int64_t i = 0; i < size; ++i) {
125 if (mlir::failed(elements(element(path, i), site, visit))) {
126 return mlir::failure();
127 }
128 }
129 return mlir::success();
130 }
131
133 mlir::LogicalResult walk(
134 AccessPath path, mlir::Operation *site,
135 llvm::function_ref<mlir::LogicalResult(AccessPath)> visit
136 ) {
137 if (llvm::isa<mlir::NoneType>(path.type)) {
138 return mlir::success();
139 }
140 if (llvm::isa<array::ArrayType>(path.type)) {
141 return elements(path, site, [this, site, visit](AccessPath item) {
142 return walk(item, site, visit);
143 });
144 }
145 if (auto type = llvm::dyn_cast<pod::PodType>(path.type)) {
146 for (auto record : type.getRecords()) {
147 auto field = child(path, record.getName(), record.getType());
148 if (mlir::failed(walk(field, site, visit))) {
149 return mlir::failure();
150 }
151 }
152 return mlir::success();
153 }
154 return visit(path);
155 }
156
158 mlir::FailureOr<mlir::Value> load(AccessPath path, mlir::Operation *site, StorageReader read) {
159 auto &leaves = roots[path.root].leaves;
160 if (auto found = leaves.find(path.steps); found != leaves.end()) {
161 return found->second;
162 }
163 if (mlir::Value storage = roots[path.root].storage) {
164 auto value = read(storage, path.steps.getValue(), path.type, site);
165 if (mlir::failed(value)) {
166 return mlir::failure();
167 }
168 leaves[path.steps] = *value;
169 return *value;
170 }
171 return constraintError(
172 site, (roots[path.root].input || roots[path.root].protections)
173 ? "read of uninitialized function input storage"
174 : "read of uninitialized local storage"
175 );
176 }
177
179 void protect(AccessPath path) { ++roots[path.root].protections; }
180
182 void unprotect(AccessPath path) { --roots[path.root].protections; }
183
185 mlir::LogicalResult writable(AccessPath path, mlir::Operation *site) const {
186 const Root &root = roots[path.root];
187 if (root.input || root.storage || root.protections) {
188 return constraintError(site, "cannot write to function input or original storage");
189 }
190 return mlir::success();
191 }
192
194 void store(AccessPath path, mlir::Value value) { roots[path.root].leaves[path.steps] = value; }
195
198 mlir::LogicalResult
199 transfer(AccessPath destination, AccessPath source, mlir::Operation *site, StorageReader read) {
200 llvm::SmallVector<std::pair<mlir::ArrayAttr, mlir::Value>> snapshot;
201 bool original = bool(roots[source.root].storage);
202 auto collect = [this, original, site, read, destination, source,
203 &snapshot](AccessPath leaf) -> mlir::LogicalResult {
204 const auto &leaves = roots[leaf.root].leaves;
205 if (!leaves.contains(leaf.steps) && !original) {
206 return mlir::success();
207 }
208 auto value = load(leaf, site, read);
209 if (mlir::failed(value)) {
210 return mlir::failure();
211 }
212 AccessPath target = destination;
213 for (auto step : leaf.steps.getValue().drop_front(source.steps.size())) {
214 auto selected = select(target, step, site);
215 if (mlir::failed(selected)) {
216 return mlir::failure();
217 }
218 target = *selected;
219 }
220 snapshot.push_back({target.steps, *value});
221 return mlir::success();
222 };
223 if (mlir::failed(walk(source, site, collect))) {
224 return mlir::failure();
225 }
226 // Visit only the overwritten subtree, rather than scanning unrelated leaves
227 // in the root on every array.insert. The snapshot makes overlap safe.
228 if (mlir::failed(walk(destination, site, [this](AccessPath leaf) {
229 roots[leaf.root].leaves.erase(leaf.steps);
230 return mlir::success();
231 }))) {
232 return mlir::failure();
233 }
234 auto &targetLeaves = roots[destination.root].leaves;
235 for (auto [path, value] : snapshot) {
236 targetLeaves[path] = value;
237 }
238 return mlir::success();
239 }
240};
