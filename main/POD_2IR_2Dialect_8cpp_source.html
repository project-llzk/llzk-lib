39class PODValueCopyDialectInterface final : public ValueCopyDialectInterface {
40public:
41 explicit PODValueCopyDialectInterface(Dialect *owner) : ValueCopyDialectInterface(owner) {}
42
43 bool canMaterializeValueCopy(Type type) const final {
44 auto podType = llvm::dyn_cast<PodType>(type);
45 if (!podType) {
46 return false;
47 }
48 // Copy eligibility is type-based and must also hold for block arguments and read results.
49 // Until copies can recover their instantiation groups, do not synthesize invalid pod.new ops.
50 SmallVector<AffineMapAttr> maps;
51 collectPodMapAttrs(podType, maps);
52 if (!maps.empty()) {
53 return false;
54 }
55 return llvm::all_of(podType.getRecords(), [](RecordAttr record) {
56 return llzk::canMaterializeValueCopy(record.getType());
57 });
58 }
59
60 std::string getValueCopyFailureReason(Type type) const final {
61 auto podType = llvm::dyn_cast<PodType>(type);
62 if (!podType) {
63 return ValueCopyDialectInterface::getValueCopyFailureReason(type);
64 }
65 for (RecordAttr record : podType.getRecords()) {
66 if (!llzk::canMaterializeValueCopy(record.getType())) {
67 return "POD record '" + record.getName().getValue().str() +
68 "': " + llzk::getValueCopyFailureReason(record.getType());
69 }
70 }
71 return "POD copy requires affine-map instantiation operands that cannot be recovered";
72 }
73
74 FailureOr<Value>
75 materializeValueCopy(OpBuilder &builder, Location loc, Value source) const final {
76 auto podType = llvm::dyn_cast<PodType>(source.getType());
77 if (!podType || !canMaterializeValueCopy(podType)) {
78 return failure();
79 }
80
81 auto destination = NewPodOp::create(builder, loc, podType);
82 for (RecordAttr record : podType.getRecords()) {
83 auto read = ReadPodOp::create(builder, loc, record.getType(), source, record.getName());
84 FailureOr<Value> copied = llzk::materializeValueCopy(builder, loc, read.getResult());
85 if (failed(copied)) {
86 return failure();
87 }
88 WritePodOp::create(builder, loc, destination.getResult(), record.getName(), *copied);
89 }
90 return destination.getResult();
91 }
92};
