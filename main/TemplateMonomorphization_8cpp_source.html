52FailureOr<ArrayAttr> normalizeTypedArguments(
53 Operation *source, ArrayAttr arguments, Operation *site, StringRef kind = "struct"
54) {
55 auto templ = source->getParentOfType<TemplateOp>();
56 if (!templ) {
57 return arguments;
58 }
59 SmallVector<Attribute> normalized;
60 for (auto [parameter, argument] :
61 llvm::zip_equal(templ.getConstOps<TemplateParamOp>(), arguments)) {
62 Attribute value = argument;
63 if (std::optional<Type> expected = parameter.getTypeOpt()) {
64 if (auto indexType = dyn_cast<IndexType>(*expected)) {
65 if (auto felt = dyn_cast<FeltConstAttr>(argument)) {
66 APInt number = felt.getValue();
67 if (felt.getType().hasField()) {
68 const Field &field = felt.getType().getField();
69 number = toAPInt(field.reduce(number), field.bitWidth());
70 }
71 if (number.getActiveBits() > 63) {
72 return site->emitError("field element ")
73 << argument << " cannot be converted to index parameter @"
74 << parameter.getSymName()
75 << ": value exceeds the nonnegative signed 64-bit index range";
76 }
77 value = IntegerAttr::get(indexType, number.getZExtValue());
78 } else if (
79 auto integer = dyn_cast<IntegerAttr>(argument);
80 integer && integer.getType().isSignlessInteger(1)
81 ) {
82 value = IntegerAttr::get(indexType, integer.getValue().getZExtValue());
83 }
84 } else if (
85 auto integerType = dyn_cast<IntegerType>(*expected);
86 integerType && integerType.isSignlessInteger(1)
87 ) {
88 if (auto integer = dyn_cast<IntegerAttr>(argument);
89 integer && isa<IndexType>(integer.getType())) {
90 const APInt &number = integer.getValue();
91 if (!number.isZero() && !number.isOne()) {
92 return site->emitError("index argument ")
93 << argument << " cannot be converted to i1 parameter @" << parameter.getSymName()
94 << ": expected zero or one";
95 }
96 value = IntegerAttr::get(integerType, number.isOne() ? 1 : 0);
97 }
98 } else if (auto feltType = dyn_cast<FeltType>(*expected)) {
99 if (auto integer = dyn_cast<IntegerAttr>(argument);
100 integer && isa<IndexType>(integer.getType())) {
101 if (feltType.hasField()) {
102 const Field &field = feltType.getField();
103 auto reduced = field.reduce(llvm::DynamicAPInt(integer.getValue()));
104 value = FeltConstAttr::get(
105 source->getContext(), toAPInt(reduced, field.bitWidth()), feltType
106 );
107 } else {
108 if (integer.getValue().isNegative()) {
109 return site->emitError("index argument ")
110 << argument << " cannot be converted to felt parameter @"
111 << parameter.getSymName() << ": negative values require a known field modulus";
112 }
113 value = FeltConstAttr::get(source->getContext(), integer.getValue(), feltType);
114 }
115 } else if (
116 auto felt = dyn_cast<FeltConstAttr>(argument);
117 felt && (!felt.getType().getFieldName() || felt.getType() == feltType)
118 ) {
119 APInt number = felt.getValue();
120 if (feltType.hasField()) {
121 const Field &field = feltType.getField();
122 number = toAPInt(field.reduce(number), field.bitWidth());
123 }
124 value = FeltConstAttr::get(source->getContext(), number, feltType);
125 }
126 }
127 bool matches = false;
128 if (isa<TypeVarType>(*expected)) {
129 matches = isa<TypeAttr>(value);
130 } else if (auto typed = dyn_cast<TypedAttr>(value)) {
131 matches = typed.getType() == *expected;
132 }
133 if (!matches) {
134 return site->emitError() << kind << " argument " << argument
135 << " does not match parameter @" << parameter.getSymName()
136 << " of type " << *expected;
137 }
138 }
139 normalized.push_back(value);
140 }
141 return ArrayAttr::get(source->getContext(), normalized);
142}

186 FailureOr<ArrayAttr> canonicalizeCloneArguments(ArrayAttr arguments, Operation *site) {
187 AttrTypeReplacer replacer;
188 replacer.addReplacement(
189 [this, site](StructType type) -> std::optional<std::pair<Type, WalkResult>> {
190 auto found = type.getDefinition(tables, topRoot);
191 if (failed(found)) {
192 site->emitError("cannot resolve struct type argument ") << type;
193 return std::make_pair(Type(type), WalkResult::interrupt());
194 }
195 if (!structCloneToId.contains(found->get().getOperation())) {
196 return std::nullopt;
197 }
198 auto origin = found->get()->getAttrOfType<SymbolRefAttr>(SPECIALIZATION_ORIGIN_ATTR);
199 auto parameters = found->get()->getAttrOfType<ArrayAttr>(SPECIALIZATION_ARGUMENTS_ATTR);
200 if (!origin || !parameters) {
201 site->emitError("invalid struct specialization metadata for type argument ") << type;
202 return std::make_pair(Type(type), WalkResult::interrupt());
203 }
204 return std::make_pair(Type(getStructTypeWithParams(origin, parameters)), WalkResult::skip());
205 }
206 );
207 auto result = dyn_cast_if_present<ArrayAttr>(replacer.replace(arguments));
208 if (!result) {
209 return failure();
210 }
211 return result;
212 }

216 FailureOr<std::optional<SpecializationKey>>
217 getSpecializationKey(StructType type, Operation *lookupFrom, Operation *site, KeyMode mode) {
218 auto found = type.getDefinition(tables, lookupFrom);
219 if (failed(found)) {
220 return failure();
221 }
222 if (found->viaInclude()) {
223 return site->emitError("inline includes before template monomorphization");
224 }
225 ArrayAttr arguments = type.getParams();
226 if (!arguments) {
227 arguments = ArrayAttr::get(root.getContext(), {});
228 }
229 if (!llvm::all_of(arguments, [](Attribute argument) {
230 return isConcreteStructParamAttr(argument);
231 })) {
232 if (mode == KeyMode::Optional) {
233 return std::optional<SpecializationKey>();
234 }
235 return site->emitError("template monomorphization requires concrete struct arguments")
236 << type;
237 }
238 auto canonical = rebaseTemplateParams(tables, arguments, lookupFrom, topRoot, site);
239 if (failed(canonical)) {
240 return failure();
241 }
242 auto sourceArguments = canonicalizeCloneArguments(*canonical, site);
243 if (failed(sourceArguments)) {
244 return failure();
245 }
246 auto names = found->get().getType().getParams();
247 if ((names ? names.size() : 0) != sourceArguments->size()) {
248 if (mode == KeyMode::Optional) {
249 return std::optional<SpecializationKey>();
250 }
251 return site->emitError("struct specialization argument count mismatch");
252 }
253 auto normalized = normalizeTypedArguments(found->get().getOperation(), *sourceArguments, site);
254 if (failed(normalized)) {
255 return failure();
256 }
257 return std::make_optional<SpecializationKey>(found->get().getOperation(), *normalized);
258 }

262 LogicalResult loadExistingSpecializations() {
263 auto existing = walkCollect<StructDefOp>(root, [](StructDefOp structure) {
264 return structure->hasAttr(SPECIALIZATION_ID_ATTR);
265 });
266 for (auto structure : existing) {
267 if (!structure->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR)) {
268 return structure.emitError("expected integer specialization identity");
269 }
270 }
271 llvm::sort(existing, [](StructDefOp left, StructDefOp right) {
272 return left->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt() <
273 right->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt();
274 });
275 for (auto clone : existing) {
276 auto origin = clone->getAttrOfType<SymbolRefAttr>(SPECIALIZATION_ORIGIN_ATTR);
277 auto arguments = clone->getAttrOfType<ArrayAttr>(SPECIALIZATION_ARGUMENTS_ATTR);
278 if (!origin || !arguments ||
279 clone->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt() !=
280 checkedCast<int64_t>(structEntries.size())) {
281 return clone.emitError("invalid struct specialization metadata");
282 }
283 auto key = getSpecializationKey(
284 StructType::get(origin, arguments), topRoot, clone, KeyMode::Required
285 );
286 if (failed(key) || !*key) {
287 return failure();
288 }
289 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, (*key)->second);
290 if (!structCache.try_emplace(**key, structEntries.size()).second) {
291 return clone.emitError("duplicate struct specialization identity");
292 }
293 structCloneToId[clone.getOperation()] = structEntries.size();
294 structEntries.push_back({cast<StructDefOp>((*key)->first), clone, {}, evaluationLimit});
295 }
296 return success();
297 }
298
300 LogicalResult loadExistingFunctions() {
301 auto existing = walkCollect<function::FuncDefOp>(root, [](function::FuncDefOp function) {
302 return function->hasAttr(SPECIALIZATION_ORIGIN_ATTR);
303 });
304 for (auto clone : existing) {
305 auto origin = clone->getAttrOfType<SymbolRefAttr>(SPECIALIZATION_ORIGIN_ATTR);
306 auto arguments = clone->getAttrOfType<ArrayAttr>(SPECIALIZATION_ARGUMENTS_ATTR);
307 if (!origin || !arguments) {
308 return clone.emitError("invalid function specialization metadata");
309 }
310 auto source = tables.lookupSymbolIn<function::FuncDefOp>(topRoot, origin);
311 if (!source) {
312 return clone.emitError("cannot resolve function specialization origin");
313 }
314 auto templ = source->getParentOfType<TemplateOp>();
315 if (arguments.size() !=
316 (templ ? llvm::range_size(templ.getConstOps<TemplateParamOp>()) : 0) ||
317 !llvm::all_of(arguments, [](Attribute argument) {
318 return isConcreteStructParamAttr(argument);
319 })) {
320 return clone.emitError("invalid function specialization arguments");
321 }
322 auto key = SpecializationKey(source.getOperation(), arguments);
323 if (!functionCache.try_emplace(key, functionEntries.size()).second) {
324 return clone.emitError("duplicate function specialization identity");
325 }
326 functionCloneToId[clone.getOperation()] = functionEntries.size();
327 functionEntries.push_back({source, clone, {}, evaluationLimit, false, false, std::nullopt});
328 }
329 return success();
330 }

333 FailureOr<SpecializationKey> getFunctionKey(
334 function::CallOp call, function::FuncDefOp source, ArrayRef<StringRef> targetNamespace
335 ) {
336 auto templ = source->getParentOfType<TemplateOp>();
337 SmallVector<Attribute> arguments;
338 if (templ) {
339 auto parameters = templ.getConstOps<TemplateParamOp>();
340 auto explicitArguments = call.getTemplateParamsAttr();
341 if (explicitArguments && explicitArguments.size() != llvm::range_size(parameters)) {
342 return call.emitError("function specialization argument count mismatch");
343 }
344 std::optional<UnificationMap> unified;
345 bool needsInference =
346 !explicitArguments || llvm::any_of(explicitArguments, [](Attribute arg) {
347 return classifyAttrConcreteness(arg) == AttrConcreteness::Wildcard;
348 });
349 if (needsInference) {
350 auto result =
351 call.unifyTypeSignatureWithNamespace(source.getFunctionType(), targetNamespace);
352 if (failed(result)) {
353 return call.emitError("cannot infer concrete function specialization");
354 }
355 unified = *result;
356 }
357 for (auto [index, parameter] : llvm::enumerate(parameters)) {
358 Attribute value = explicitArguments ? explicitArguments[index] : Attribute();
359 if (!value || classifyAttrConcreteness(value) == AttrConcreteness::Wildcard) {
360 value = inferUnifiedParam(*unified, FlatSymbolRefAttr::get(parameter.getSymNameAttr()))
361 .value_or(Attribute());
362 }
363 if (!value || !isConcreteStructParamAttr(value)) {
364 return call.emitError("cannot resolve free-function template parameter @")
365 << parameter.getSymName();
366 }
367 if (failed(call.verifyTemplateParamValueCompatibility(value, parameter))) {
368 return failure();
369 }
370 arguments.push_back(value);
371 }
372 }
373 auto canonical = rebaseTemplateParams(
374 tables, ArrayAttr::get(root.getContext(), arguments), call, topRoot, call
375 );
376 if (failed(canonical)) {
377 return failure();
378 }
379 auto sourceArguments = canonicalizeCloneArguments(*canonical, call);
380 if (failed(sourceArguments)) {
381 return failure();
382 }
383 auto normalized = normalizeTypedArguments(source, *sourceArguments, call, "function");
384 if (failed(normalized)) {
385 return failure();
386 }
387 return SpecializationKey(source.getOperation(), *normalized);
388 }

391 static void
392 convertCallArguments(Operation *clone, const StructSpecializationDiscovery::Bindings &bindings) {
393 TemplateTypeConverter converter(bindings);
394 clone->walk([&converter](function::CallOp call) {
395 if (auto parameters = call.getTemplateParamsAttr()) {
396 SmallVector<Attribute> values;
397 for (Attribute parameter : parameters) {
398 values.push_back(converter.convertAttr(parameter));
399 }
400 call.setTemplateParamsAttr(ArrayAttr::get(call.getContext(), values));
401 }
402 });
403 }

407 FailureOr<StructSpecializationDiscovery::Bindings> bindArguments(
408 Operation *source, ArrayRef<Attribute> names, ArrayAttr arguments, uint64_t &remainingSteps
409 ) {
410 StructSpecializationDiscovery::Bindings bindings;
411 for (auto [name, argument] : llvm::zip_equal(names, arguments)) {
412 bindings[name] = argument;
413 }
414 remainingSteps = evaluationLimit;
415 if (!source->getParentOfType<TemplateOp>()) {
416 return bindings;
417 }
418 return StructSpecializationDiscovery(evaluationLimit)
419 .evaluateBindings(source, bindings, &remainingSteps);
420 }

424 static LogicalResult diagnoseUnresolvedBinding(
425 Operation *clone, TemplateOp templ, Operation *site,
426 llvm::function_ref<void(InFlightDiagnostic &)> describe
427 ) {
428 WalkResult result =
429 clone->walk([templ, site, describe](ConstReadOp read) mutable -> WalkResult {
430 if (templ && templ.getConstNamed<TemplateExprOp>(read.getConstNameAttr())) {
431 auto diagnostic = site->emitError("template expression ");
432 diagnostic << read.getConstNameAttr() << " could not be evaluated for ";
433 describe(diagnostic);
434 } else {
435 auto diagnostic = site->emitError("unresolved template binding ");
436 diagnostic << read.getConstNameAttr() << " while specializing ";
437 describe(diagnostic);
438 }
439 return WalkResult::interrupt();
440 });
441 return failure(result.wasInterrupted());
442 }

446 static LogicalResult verifyStructuredControlFlow(Operation *source) {
447 WalkResult result = source->walk([](BranchOpInterface branch) -> WalkResult {
448 return branch->emitError(
449 "unstructured control flow is unsupported in template monomorphization"
450 );
451 });
452 return failure(result.wasInterrupted());

457 FailureOr<bool> hasSpecializedSignature(function::CallOp call) {
458 bool required = false;
459 bool invalid = false;
460 auto inspect = [this, call, &required, &invalid](Type type) {
461 type.walk([this, call, &required, &invalid](StructType structure) {
462 auto key = getSpecializationKey(structure, call, call, KeyMode::Optional);
463 if (failed(key)) {
464 invalid = true;
465 return WalkResult::interrupt();
466 }
467 if (*key && structCache.contains(**key)) {
468 required = true;
469 return WalkResult::interrupt();
470 }
471 return WalkResult::advance();
472 });
473 };
474 for (Type type : call.getResultTypes()) {
475 inspect(type);
476 }
477 for (Type type : call.getArgOperands().getTypes()) {
478 inspect(type);
479 }
480 if (invalid) {
481 return failure();
482 }
483 return required;
484 }

487 LogicalResult instantiateFunction(function::CallOp call, bool active = true) {
488 auto target = call.getCalleeTarget(tables);
489 if (failed(target)) {
490 return failure();
491 }
492 if (target->viaInclude()) {
493 return call.emitError("inline includes before template monomorphization");
494 }
495 auto source = target->get();
496 if (auto clone = functionCloneToId.find(source.getOperation());
497 clone != functionCloneToId.end()) {
498 functionEntries[clone->second].active |= active;
499 return success();
500 }
501 if (source.isExternal()) {
502 return success();
503 }
504 auto key = getFunctionKey(call, source, target->getNamespace());
505 if (failed(key)) {
506 return failure();
507 }
508 unsigned id;
509 if (auto found = functionCache.find(*key); found != functionCache.end()) {
510 id = found->second;
511 functionEntries[id].active |= active;
512 } else {
513 if (structEntries.size() + functionEntries.size() >= limit) {
514 return call.emitError("template monomorphization specialization limit exceeded");
515 }
516 if (failed(verifyStructuredControlFlow(source))) {
517 return failure();
518 }
519 auto destination = getRootModule(source);
520 if (failed(destination)) {
521 return failure();
522 }
523 auto localArguments = rebaseTemplateParams(tables, key->second, topRoot, *destination, call);
524 if (failed(localArguments)) {
525 return failure();
526 }
527 auto origin = getPathRelativeToAncestor(source, topRoot, [call] {
528 return call->emitError("free-function specialization origin");
529 });
530 if (failed(origin)) {
531 return failure();
532 }
533 auto templ = source->getParentOfType<TemplateOp>();
534 SmallVector<Attribute> names;
535 if (templ) {
536 for (auto parameter : templ.getConstOps<TemplateParamOp>()) {
537 names.push_back(FlatSymbolRefAttr::get(parameter.getSymNameAttr()));
538 }
539 }
540 uint64_t remainingSteps;
541 auto evaluated = bindArguments(source, names, *localArguments, remainingSteps);
542 if (failed(evaluated)) {
543 return failure();
544 }
545 auto bindings = std::move(*evaluated);
546 auto clone = source.clone();
547 clone.setSymName((source.getSymName() + "__spec_" + Twine(functionEntries.size())).str());
548 Operation *parent = templ ? templ->getParentOp() : source->getParentOp();
549 tables.getSymbolTable(parent).insert(clone);
550 id = functionEntries.size();
551 functionCache[*key] = id;
552 functionCloneToId[clone.getOperation()] = id;
553 functionEntries.push_back({source, clone, {}, remainingSteps, active, false, call.getLoc()});
554 convertCalleesInPlace(clone, bindings);
555 convertCallArguments(clone, bindings);
556 SmallVector<Diagnostic> diagnostics;
557 if (failed(substituteFunctionBody(clone, bindings, diagnostics))) {
558 return failure();
559 }
560 if (failed(diagnoseUnresolvedBinding(
561 clone, templ, call,
562 [name = *origin, arguments = key->second](InFlightDiagnostic &diagnostic) {
563 diagnostic << "free function " << name << " with arguments " << arguments;
564 }
565 ))) {
566 return failure();
567 }
568 reportDelayedDiagnostics(call, std::move(diagnostics));
569 clone->setAttr(SPECIALIZATION_ORIGIN_ATTR, *origin);
570 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, key->second);
571 }
572 auto lookupRoot = getRootModule(call);
573 if (failed(lookupRoot)) {
574 return failure();
575 }
576 auto name = getPathRelativeToAncestor(functionEntries[id].clone, *lookupRoot, [call] {
577 return call->emitError("specialized free-function callee");
578 });
579 if (failed(name)) {
580 return failure();
581 }
582 call.setCalleeAttr(*name);
583 call.removeTemplateParamsAttr();
584 return success();
585 }

589 FailureOr<unsigned> instantiate(StructType type, Operation *site) {
590 auto resolved = getSpecializationKey(type, site, site, KeyMode::Required);
591 if (failed(resolved) || !*resolved) {
592 return failure();
593 }
594 auto key = **resolved;
595 StructDefOp source = cast<StructDefOp>(key.first);
596 if (auto existing = structCloneToId.find(source); existing != structCloneToId.end()) {
597 return existing->second;
598 }
599 auto destination = getRootModule(source);
600 if (failed(destination)) {
601 return failure();
602 }
603 auto names = source.getType().getParams();
604 auto localArguments = rebaseTemplateParams(tables, key.second, topRoot, *destination, site);
605 if (failed(localArguments)) {
606 return failure();
607 }
608 if (auto existing = structCache.find(key); existing != structCache.end()) {
609 return existing->second;
610 }
611 if (structEntries.size() + functionEntries.size() >= limit) {
612 return site->emitError("template monomorphization specialization limit exceeded");
613 }
614 if (failed(verifyStructuredControlFlow(source))) {
615 return failure();
616 }
617 auto templ = source->getParentOfType<TemplateOp>();
618 uint64_t remainingSteps;
619 auto evaluated = bindArguments(
620 source, names ? names.getValue() : ArrayRef<Attribute>(), *localArguments, remainingSteps
621 );
622 if (failed(evaluated)) {
623 return failure();
624 }
625 auto bindings = std::move(*evaluated);
626 TemplateTypeConverter parameterConverter(bindings);
627 auto reads = source.walk([&bindings, &parameterConverter](ConstReadOp read) -> WalkResult {
628 return WalkResult(resolveConstReadBinding(
629 read, bindings.lookup(read.getConstNameAttr()),
630 parameterConverter.convertType(read.getType())
631 ));
632 });
633 if (reads.wasInterrupted()) {
634 return failure();
635 }
636 auto clone = source.clone();
637 clone.setSymName((source.getSymName() + "__spec_" + Twine(structEntries.size())).str());
638 Operation *parent = templ ? templ->getParentOp() : source->getParentOp();
639 tables.getSymbolTable(parent).insert(clone);
640 unsigned id = structEntries.size();
641 structCache[key] = id;
642 structCloneToId[clone.getOperation()] = id;
643 structEntries.push_back({source, clone, {}, remainingSteps});
644 convertCalleesInPlace(clone, bindings);
645 clone.walk([&parameterConverter](function::CallOp call) {
646 if (auto parameters = call.getTemplateParamsAttr()) {
647 SmallVector<Attribute> values;
648 for (Attribute parameter : parameters) {
649 values.push_back(parameterConverter.convertAttr(parameter));
650 }
651 call.setTemplateParamsAttr(ArrayAttr::get(call.getContext(), values));
652 }
653 });
654 if (!names || names.empty()) {
655 StructType canonicalSelf = source.getType();
656 AttrTypeReplacer selfTypes;
657 selfTypes.addReplacement(
658 [canonicalSelf](StructType current) -> std::optional<std::pair<Type, WalkResult>> {
659 if (current.getNameRef() != canonicalSelf.getNameRef() ||
660 (current.getParams() && !current.getParams().empty())) {
661 return std::nullopt;
662 }
663 return std::make_pair(Type(canonicalSelf), WalkResult::skip());
664 }
665 );
666 selfTypes.recursivelyReplaceElementsIn(clone, true, false, true);
667 }
668 // Materialize reads with the same conversions used by expression discovery.
669 auto constants = clone.walk([&bindings, &parameterConverter](ConstReadOp read) -> WalkResult {
670 auto value = resolveConstReadBinding(
671 read, bindings.lookup(read.getConstNameAttr()),
672 parameterConverter.convertType(read.getType())
673 );
674 if (failed(value)) {
675 return WalkResult::interrupt();
676 }
677 if (!*value) {
678 return WalkResult::advance();
679 }
680 OpBuilder builder(read);
681 Value constant;
682 if (auto felt = dyn_cast<FeltConstAttr>(*value)) {
683 constant = FeltConstantOp::create(builder, read.getLoc(), felt);
684 } else {
685 constant = arith::ConstantOp::create(builder, read.getLoc(), cast<TypedAttr>(*value));
686 }
687 read.replaceAllUsesWith(constant);
688 read.erase();
689 return WalkResult::advance();
690 });
691 if (constants.wasInterrupted()) {
692 return failure();
693 }
694 SmallVector<Diagnostic> diagnostics;
695 if (failed(substituteStructBody(clone, source.getType(), bindings, diagnostics))) {
696 return failure();
697 }
698 if (failed(diagnoseUnresolvedBinding(
699 clone, templ, site, [type](InFlightDiagnostic &diagnostic) { diagnostic << type; }
700 ))) {
701 return failure();
702 }
703 reportDelayedDiagnostics(site, std::move(diagnostics));
704 Builder builder(root.getContext());
705 clone->setAttr(SPECIALIZATION_ID_ATTR, builder.getI64IntegerAttr(id));
706 auto origin = getPathRelativeToAncestor(source, topRoot, [source] {
707 return source->emitError("specialization origin");
708 });
709 if (failed(origin)) {

732 builder.getNamedAttr(FAMILY_TYPE_KEY, TypeAttr::get(request.familyType))}
733 );

735 name = FAMILY_SPECIALIZATIONS_ATTR;

742 llvm::append_range(records, existing);
743 }

745 records.push_back(record);

747 request.site->setAttr(name, builder.getArrayAttr(records));

791 LogicalResult retargetCall(function::CallOp call, const StructEntry *caller, bool visited) {
792 auto target = call.getCalleeTarget(tables);
793 if (failed(target)) {
794 return failure();
795 }
796 if (target->viaInclude()) {
797 return call.emitError("inline includes before template monomorphization");
798 }
799 auto owner = target->get()->getParentOfType<StructDefOp>();
800 if (!owner) {
801 return success();
802 }
803 if (call->hasAttr(FAMILY_SPECIALIZATIONS_ATTR)) {
804 return success();
805 }
806 SmallVector<Type> types(call.getResultTypes());
807 llvm::append_range(types, call.getArgOperands().getTypes());
808 for (Type type : types) {
809 auto structure = dyn_cast<StructType>(type);
810 if (!structure || hasAffineMapAttr(structure)) {
811 continue;
812 }
813 auto found = structure.getDefinition(tables, call);
814 if (failed(found)) {
815 return failure();
816 }
817 auto cloneId = structCloneToId.find(found->get().getOperation());
818 if (cloneId != structCloneToId.end() && structEntries[cloneId->second].source == owner) {
819 return setSpecializedCallee(
820 call, structEntries[cloneId->second].clone, target->get().getSymNameAttr()
821 );
822 }
823 if (found->get() == owner) {
824 return success();
825 }
826 }
827 if (caller && owner == caller->source) {
828 return setSpecializedCallee(call, caller->clone, target->get().getSymNameAttr());
829 }
830 if (visited) {
831 return call.emitError("cannot determine concrete struct owner for method call");
832 }
833 return success();
834 }

869 return failure();
870 }

875public:
876 TemplateInstantiationWorklist(ModuleOp module, unsigned maximum, uint64_t steps)
877 : root(module), limit(maximum), evaluationLimit(steps) {}
878
881 LogicalResult run() {
882 auto top = getTopRootModule(root);
883 if (failed(top)) {
884 return failure();
885 }
886 topRoot = *top;
887 if (failed(loadExistingSpecializations())) {
888 return failure();
889 }
890 if (failed(loadExistingFunctions())) {
891 return failure();
892 }
893 auto main = getMainInstanceType(root);
894 if (failed(main) || !*main) {
895 return root.emitError("template monomorphization requires a concrete llzk.main");
896 }
897 auto mainId = instantiate(*main, root);
898 if (failed(mainId)) {
899 return failure();
900 }
901 // Discovery appends to these worklists. Indices remain valid as they grow.
902 for (unsigned structIndex = 0;;) {
903 bool isStruct = structIndex < structEntries.size();
904 unsigned functionIndex = 0;
905 if (!isStruct) {
906 while (
907 functionIndex < functionEntries.size() &&
908 (!functionEntries[functionIndex].active || functionEntries[functionIndex].discovered)) {
909 ++functionIndex;
910 }
911 if (functionIndex == functionEntries.size()) {
912 break;
913 }
914 }
915 Operation *clone = isStruct ? structEntries[structIndex].clone.getOperation()
916 : functionEntries[functionIndex].clone.getOperation();
917 SmallVector<Operation *> visited;
918 uint64_t steps = isStruct ? structEntries[structIndex].remainingSteps
919 : functionEntries[functionIndex].remainingSteps;
920 std::optional<ScopedDiagnosticHandler> requestNote;
921 if (!isStruct && functionEntries[functionIndex].request) {
922 Location request = *functionEntries[functionIndex].request;
923 Attribute origin = clone->getAttr(SPECIALIZATION_ORIGIN_ATTR);
924 Attribute arguments = clone->getAttr(SPECIALIZATION_ARGUMENTS_ATTR);
925 requestNote.emplace(clone->getContext(), [request, origin, arguments](Diagnostic &diag) {
926 if (diag.getSeverity() == DiagnosticSeverity::Error) {
927 diag.attachNote(request)
928 << "while specializing free function " << origin << " with arguments " << arguments;
929 }
930 return failure();
931 });
932 }
933 StructSpecializationDiscovery::CallTargets rolledCalls;
934 StructSpecializationDiscovery discovery(steps);
935 auto requests = discovery.discover(
936 clone, StructSpecializationDiscovery::Bindings(), &visited, &rolledCalls
937 );
938 if (failed(requests)) {
939 return failure();
940 }
941 DenseSet<Operation *> visitedSet(visited.begin(), visited.end());
942 for (const auto &request : *requests) {
943 auto id = instantiate(request.type, request.site);
944 if (failed(id)) {
945 return failure();
946 }
947 recordFamily(request, *id);
948 }
949 for (const auto &targets : rolledCalls) {
950 if (failed(recordRolledCallTargets(targets))) {
951 return failure();
952 }
953 }
954 auto calls = clone->walk([this, &visitedSet](function::CallOp call) {
955 if (!visitedSet.contains(call)) {
956 return WalkResult::advance();
957 }
958 auto target = call.getCalleeTarget(tables);
959 if (failed(target)) {
960 return WalkResult::interrupt();
961 }
962 if (!target->get()->getParentOfType<StructDefOp>() && failed(instantiateFunction(call))) {
963 return WalkResult::interrupt();
964 }
965 return WalkResult::advance();
966 });
967 if (calls.wasInterrupted()) {
968 return failure();
969 }
970 if (isStruct) {
971 structEntries[structIndex].visited = std::move(visited);
972 ++structIndex;
973 } else {
974 functionEntries[functionIndex].visited = std::move(visited);
975 functionEntries[functionIndex].discovered = true;
976 }
977 }
978 // A call in an unselected branch is not a dependency to explore. Retain
979 // its template callee unless a known struct specialization changes its
980 // signature; in that case a concrete callee is needed for valid IR.
981 for (unsigned structIndex = 0, functionIndex = 0;
982 structIndex < structEntries.size() || functionIndex < functionEntries.size();) {
983 bool isStruct = structIndex < structEntries.size();
984 Operation *clone = isStruct ? structEntries[structIndex].clone.getOperation()
985 : functionEntries[functionIndex].clone.getOperation();
986 auto &visited =
987 isStruct ? structEntries[structIndex].visited : functionEntries[functionIndex].visited;
988 DenseSet<Operation *> visitedSet(visited.begin(), visited.end());
989 auto calls = clone->walk([this, &visitedSet](function::CallOp call) {
990 if (visitedSet.contains(call)) {
991 return WalkResult::advance();
992 }
993 auto target = call.getCalleeTarget(tables);
994 if (failed(target)) {
995 return WalkResult::interrupt();
996 }
997 if (target->get()->getParentOfType<StructDefOp>()) {
998 return WalkResult::advance();
999 }
1000 auto required = hasSpecializedSignature(call);
1001 if (failed(required) || (*required && failed(instantiateFunction(call, false)))) {
1002 return WalkResult::interrupt();
1003 }
1004 return WalkResult::advance();
1005 });
1006 if (calls.wasInterrupted()) {
1007 return failure();
1008 }
1009 if (isStruct) {
1010 ++structIndex;
1011 } else {
1012 ++functionIndex;
1013 }
1014 }
1015 if (failed(verifyExternalCalls())) {
1016 return failure();
1017 }
1018 // Each replacer resolves input names in one root. The cache identifies the
1019 // definition independently of its spelling, including uses in preserved paths
1020 // that discovery did not execute.
1021 llvm::MapVector<Operation *, SmallVector<Operation *>> clonesByRoot;
1022 for (StructEntry &entry : structEntries) {
1023 auto lookupRoot = getRootModule(entry.clone);
1024 if (failed(lookupRoot)) {
1025 return failure();
1026 }
1027 clonesByRoot[lookupRoot->getOperation()].push_back(entry.clone);
1028 }
1029 for (FunctionEntry &entry : functionEntries) {
1030 auto lookupRoot = getRootModule(entry.clone);
1031 if (failed(lookupRoot)) {
1032 return failure();
1033 }
1034 clonesByRoot[lookupRoot->getOperation()].push_back(entry.clone);
1035 }
1036 for (auto &[rootOp, clones] : clonesByRoot) {
1037 auto lookupRoot = cast<ModuleOp>(rootOp);
1038 // recursivelyReplaceElementsIn returns void and suppresses null replacements,
1039 // so the callback must retain a failure flag for this operation-level rewrite.
1040 bool invalid = false;
1041 // These checks validate names after substitution and clone insertion. The
1042 // callback receives only a type, so failures are reported at its lookup root.
1043 // For verified input, these failures indicate a bug in the pass.
1044 AttrTypeReplacer replacer;
1045 replacer.addReplacement(
1046 [this, lookupRoot,
1047 &invalid](StructType type) -> std::optional<std::pair<Type, WalkResult>> {
1048 auto key = getSpecializationKey(type, lookupRoot, lookupRoot, KeyMode::Optional);
1049 if (failed(key)) {
1050 invalid = true;
1051 return std::make_pair(Type(type), WalkResult::skip());
1052 }
1053 if (!*key) {
1054 return std::nullopt;
1055 }
1056 auto existing = structCache.find(**key);
1057 if (existing == structCache.end()) {
1058 return std::nullopt;
1059 }
1060 auto name = getPathRelativeToAncestor(
1061 structEntries[existing->second].clone, lookupRoot,
1062 [lookupRoot] { return lookupRoot->emitError("specialized struct"); }
1063 );
1064 if (failed(name)) {
1065 invalid = true;
1066 return std::make_pair(Type(type), WalkResult::skip());
1067 }
1068 return std::make_pair(Type(StructType::get(*name)), WalkResult::skip());
1069 }
1070 );
1071 for (auto clone : clones) {
1072 // Provenance uses top-root names, not the root of the rewritten body.
1073 auto origin = clone->removeAttr(SPECIALIZATION_ORIGIN_ATTR);
1074 auto args = clone->removeAttr(SPECIALIZATION_ARGUMENTS_ATTR);
1075 replacer.recursivelyReplaceElementsIn(clone, true, false, true);
1076 clone->setAttr(SPECIALIZATION_ORIGIN_ATTR, origin);
1077 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, args);
1078 }
1079 if (invalid) {
1080 return failure();
1081 }
1082 }
1083 for (StructEntry &entry : structEntries) {
1084 DenseSet<Operation *> visited(entry.visited.begin(), entry.visited.end());
1085 auto result = entry.clone.walk([this, &entry, &visited](function::CallOp call) {
1086 return WalkResult(retargetCall(call, &entry, visited.contains(call)));
1087 });
1088 if (result.wasInterrupted()) {
1089 return failure();
1090 }
1091 }
1092 for (FunctionEntry &entry : functionEntries) {
1093 DenseSet<Operation *> visited(entry.visited.begin(), entry.visited.end());
1094 auto result = entry.clone.walk([this, &visited](function::CallOp call) {
1095 return WalkResult(retargetCall(call, nullptr, visited.contains(call)));
1096 });
1097 if (result.wasInterrupted()) {
1098 return failure();
1099 }
1100 }
1101 auto mainRoot = getRootModule(root);
1102 if (failed(mainRoot)) {
1103 return failure();
1104 }
1105 auto mainName = getPathRelativeToAncestor(structEntries[*mainId].clone, *mainRoot, [this] {
1106 return root.emitError("specialized main");
1107 });
1108 if (failed(mainName)) {
1109 return failure();
1110 }
1111 root->setAttr(MAIN_ATTR_NAME, TypeAttr::get(StructType::get(*mainName)));
1112 return success();
1113 }
1114};
