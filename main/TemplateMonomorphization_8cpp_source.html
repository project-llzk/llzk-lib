50FailureOr<ArrayAttr>
51normalizeTypedArguments(StructDefOp source, ArrayAttr arguments, Operation *site) {
52 auto templ = source->getParentOfType<TemplateOp>();
53 if (!templ) {
54 return arguments;
55 }
56 SmallVector<Attribute> normalized;
57 for (auto [parameter, argument] :
58 llvm::zip_equal(templ.getConstOps<TemplateParamOp>(), arguments)) {
59 Attribute value = argument;
60 if (std::optional<Type> expected = parameter.getTypeOpt()) {
61 if (auto indexType = dyn_cast<IndexType>(*expected)) {
62 if (auto felt = dyn_cast<FeltConstAttr>(argument)) {
63 APInt number = felt.getValue();
64 if (felt.getType().hasField()) {
65 const Field &field = felt.getType().getField();
66 number = toAPInt(field.reduce(number), field.bitWidth());
67 }
68 if (number.getActiveBits() > 63) {
69 return site->emitError("field element ")
70 << argument << " cannot be converted to index parameter @"
71 << parameter.getSymName()
72 << ": value exceeds the nonnegative signed 64-bit index range";
73 }
74 value = IntegerAttr::get(indexType, number.getZExtValue());
75 } else if (
76 auto integer = dyn_cast<IntegerAttr>(argument);
77 integer && integer.getType().isSignlessInteger(1)
78 ) {
79 value = IntegerAttr::get(indexType, integer.getValue().getZExtValue());
80 }
81 } else if (
82 auto integerType = dyn_cast<IntegerType>(*expected);
83 integerType && integerType.isSignlessInteger(1)
84 ) {
85 if (auto integer = dyn_cast<IntegerAttr>(argument);
86 integer && isa<IndexType>(integer.getType())) {
87 const APInt &number = integer.getValue();
88 if (!number.isZero() && !number.isOne()) {
89 return site->emitError("index argument ")
90 << argument << " cannot be converted to i1 parameter @" << parameter.getSymName()
91 << ": expected zero or one";
92 }
93 value = IntegerAttr::get(integerType, number.isOne() ? 1 : 0);
94 }
95 } else if (auto feltType = dyn_cast<FeltType>(*expected)) {
96 if (auto integer = dyn_cast<IntegerAttr>(argument);
97 integer && isa<IndexType>(integer.getType())) {
98 if (feltType.hasField()) {
99 const Field &field = feltType.getField();
100 auto reduced = field.reduce(llvm::DynamicAPInt(integer.getValue()));
101 value = FeltConstAttr::get(
102 source.getContext(), toAPInt(reduced, field.bitWidth()), feltType
103 );
104 } else {
105 if (integer.getValue().isNegative()) {
106 return site->emitError("index argument ")
107 << argument << " cannot be converted to felt parameter @"
108 << parameter.getSymName() << ": negative values require a known field modulus";
109 }
110 value = FeltConstAttr::get(source.getContext(), integer.getValue(), feltType);
111 }
112 } else if (
113 auto felt = dyn_cast<FeltConstAttr>(argument);
114 felt && (!felt.getType().getFieldName() || felt.getType() == feltType)
115 ) {
116 APInt number = felt.getValue();
117 if (feltType.hasField()) {
118 const Field &field = feltType.getField();
119 number = toAPInt(field.reduce(number), field.bitWidth());
120 }
121 value = FeltConstAttr::get(source.getContext(), number, feltType);
122 }
123 }
124 bool matches = false;
125 if (isa<TypeVarType>(*expected)) {
126 matches = isa<TypeAttr>(value);
127 } else if (auto typed = dyn_cast<TypedAttr>(value)) {
128 matches = typed.getType() == *expected;
129 }
130 if (!matches) {
131 return site->emitError("struct argument ")
132 << argument << " does not match parameter @" << parameter.getSymName() << " of type "
133 << *expected;
134 }
135 }
136 normalized.push_back(value);
137 }
138 return ArrayAttr::get(source.getContext(), normalized);
139}

166 FailureOr<ArrayAttr> canonicalizeCloneArguments(ArrayAttr arguments, Operation *site) {
167 AttrTypeReplacer replacer;
168 replacer.addReplacement(
169 [this, site](StructType type) -> std::optional<std::pair<Type, WalkResult>> {
170 auto found = type.getDefinition(tables, topRoot);
171 if (failed(found)) {
172 site->emitError("cannot resolve struct type argument ") << type;
173 return std::make_pair(Type(type), WalkResult::interrupt());
174 }
175 if (!cloneToId.contains(found->get().getOperation())) {
176 return std::nullopt;
177 }
178 auto origin = found->get()->getAttrOfType<SymbolRefAttr>(SPECIALIZATION_ORIGIN_ATTR);
179 auto parameters = found->get()->getAttrOfType<ArrayAttr>(SPECIALIZATION_ARGUMENTS_ATTR);
180 if (!origin || !parameters) {
181 site->emitError("invalid struct specialization metadata for type argument ") << type;
182 return std::make_pair(Type(type), WalkResult::interrupt());
183 }
184 return std::make_pair(Type(getStructTypeWithParams(origin, parameters)), WalkResult::skip());
185 }
186 );
187 auto result = dyn_cast_if_present<ArrayAttr>(replacer.replace(arguments));
188 if (!result) {
189 return failure();
190 }
191 return result;
192 }

196 FailureOr<std::optional<SpecializationKey>>
197 getSpecializationKey(StructType type, Operation *lookupFrom, Operation *site, KeyMode mode) {
198 auto found = type.getDefinition(tables, lookupFrom);
199 if (failed(found)) {
200 return failure();
201 }
202 if (found->viaInclude()) {
203 return site->emitError("inline includes before template monomorphization");
204 }
205 ArrayAttr arguments = type.getParams();
206 if (!arguments) {
207 arguments = ArrayAttr::get(root.getContext(), {});
208 }
209 if (!llvm::all_of(arguments, [](Attribute argument) {
210 return isConcreteStructParamAttr(argument);
211 })) {
212 if (mode == KeyMode::Optional) {
213 return std::optional<SpecializationKey>();
214 }
215 return site->emitError("template monomorphization requires concrete struct arguments")
216 << type;
217 }
218 auto canonical = rebaseTemplateParams(tables, arguments, lookupFrom, topRoot, site);
219 if (failed(canonical)) {
220 return failure();
221 }
222 auto sourceArguments = canonicalizeCloneArguments(*canonical, site);
223 if (failed(sourceArguments)) {
224 return failure();
225 }
226 auto names = found->get().getType().getParams();
227 if ((names ? names.size() : 0) != sourceArguments->size()) {
228 if (mode == KeyMode::Optional) {
229 return std::optional<SpecializationKey>();
230 }
231 return site->emitError("struct specialization argument count mismatch");
232 }
233 auto normalized = normalizeTypedArguments(found->get(), *sourceArguments, site);
234 if (failed(normalized)) {
235 return failure();
236 }
237 return std::make_optional<SpecializationKey>(found->get().getOperation(), *normalized);
238 }

242 LogicalResult loadExistingSpecializations() {
243 auto existing = walkCollect<StructDefOp>(root, [](StructDefOp structure) {
244 return structure->hasAttr(SPECIALIZATION_ID_ATTR);
245 });
246 for (auto structure : existing) {
247 if (!structure->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR)) {
248 return structure.emitError("expected integer specialization identity");
249 }
250 }
251 llvm::sort(existing, [](StructDefOp left, StructDefOp right) {
252 return left->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt() <
253 right->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt();
254 });
255 for (auto clone : existing) {
256 auto origin = clone->getAttrOfType<SymbolRefAttr>(SPECIALIZATION_ORIGIN_ATTR);
257 auto arguments = clone->getAttrOfType<ArrayAttr>(SPECIALIZATION_ARGUMENTS_ATTR);
258 if (!origin || !arguments ||
259 clone->getAttrOfType<IntegerAttr>(SPECIALIZATION_ID_ATTR).getInt() !=
260 checkedCast<int64_t>(entries.size())) {
261 return clone.emitError("invalid struct specialization metadata");
262 }
263 auto key = getSpecializationKey(
264 StructType::get(origin, arguments), topRoot, clone, KeyMode::Required
265 );
266 if (failed(key) || !*key) {
267 return failure();
268 }
269 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, (*key)->second);
270 if (!cache.try_emplace(**key, entries.size()).second) {
271 return clone.emitError("duplicate struct specialization identity");
272 }
273 cloneToId[clone.getOperation()] = entries.size();
274 entries.push_back({cast<StructDefOp>((*key)->first), clone, {}, evaluationLimit});
275 }
276 return success();
277 }

281 FailureOr<unsigned> instantiate(StructType type, Operation *site) {
282 auto resolved = getSpecializationKey(type, site, site, KeyMode::Required);
283 if (failed(resolved) || !*resolved) {
284 return failure();
285 }
286 auto key = **resolved;
287 StructDefOp source = cast<StructDefOp>(key.first);
288 if (auto existing = cloneToId.find(source); existing != cloneToId.end()) {
289 return existing->second;
290 }
291 auto destination = getRootModule(source);
292 if (failed(destination)) {
293 return failure();
294 }
295 auto names = source.getType().getParams();
296 auto localArguments = rebaseTemplateParams(tables, key.second, topRoot, *destination, site);
297 if (failed(localArguments)) {
298 return failure();
299 }
300 if (auto existing = cache.find(key); existing != cache.end()) {
301 return existing->second;
302 }
303 if (entries.size() >= limit) {
304 return site->emitError("template monomorphization specialization limit exceeded");
305 }
306 auto branches = source.walk([](BranchOpInterface op) -> WalkResult {
307 return op.emitError("unstructured control flow is unsupported in template monomorphization");
308 });
309 if (branches.wasInterrupted()) {
310 return failure();
311 }
312 StructSpecializationDiscovery::Bindings bindings;
313 if (names) {
314 for (auto [name, value] : llvm::zip_equal(names, *localArguments)) {
315 bindings[name] = value;
316 }
317 }
318 auto templ = source->getParentOfType<TemplateOp>();
319 uint64_t remainingSteps = evaluationLimit;
320 if (templ) {
321 auto evaluated = StructSpecializationDiscovery(evaluationLimit)
322 .evaluateBindings(source, bindings, &remainingSteps);
323 if (failed(evaluated)) {
324 return failure();
325 }
326 bindings = std::move(*evaluated);
327 }
328 TemplateTypeConverter parameterConverter(bindings);
329 auto reads = source.walk([&bindings, &parameterConverter](ConstReadOp read) -> WalkResult {
330 return WalkResult(resolveConstReadBinding(
331 read, bindings.lookup(read.getConstNameAttr()),
332 parameterConverter.convertType(read.getType())
333 ));
334 });
335 if (reads.wasInterrupted()) {
336 return failure();
337 }
338 auto clone = source.clone();
339 clone.setSymName((source.getSymName() + "__spec_" + Twine(entries.size())).str());
340 Operation *parent = templ ? templ->getParentOp() : source->getParentOp();
341 tables.getSymbolTable(parent).insert(clone);
342 unsigned id = entries.size();
343 cache[key] = id;
344 cloneToId[clone.getOperation()] = id;
345 entries.push_back({source, clone, {}, remainingSteps});
346 convertCalleesInPlace(clone, bindings);
347 clone.walk([&parameterConverter](function::CallOp call) {
348 if (auto parameters = call.getTemplateParamsAttr()) {
349 SmallVector<Attribute> values;
350 for (Attribute parameter : parameters) {
351 values.push_back(parameterConverter.convertAttr(parameter));
352 }
353 call.setTemplateParamsAttr(ArrayAttr::get(call.getContext(), values));
354 }
355 });
356 if (!names || names.empty()) {
357 StructType canonicalSelf = source.getType();
358 AttrTypeReplacer selfTypes;
359 selfTypes.addReplacement(
360 [canonicalSelf](StructType current) -> std::optional<std::pair<Type, WalkResult>> {
361 if (current.getNameRef() != canonicalSelf.getNameRef() ||
362 (current.getParams() && !current.getParams().empty())) {
363 return std::nullopt;
364 }
365 return std::make_pair(Type(canonicalSelf), WalkResult::skip());
366 }
367 );
368 selfTypes.recursivelyReplaceElementsIn(clone, true, false, true);
369 }
370 // Materialize reads with the same conversions used by expression discovery.
371 auto constants = clone.walk([&bindings, &parameterConverter](ConstReadOp read) -> WalkResult {
372 auto value = resolveConstReadBinding(
373 read, bindings.lookup(read.getConstNameAttr()),
374 parameterConverter.convertType(read.getType())
375 );
376 if (failed(value)) {
377 return WalkResult::interrupt();
378 }
379 if (!*value) {
380 return WalkResult::advance();
381 }
382 OpBuilder builder(read);
383 Value constant;
384 if (auto felt = dyn_cast<FeltConstAttr>(*value)) {
385 constant = FeltConstantOp::create(builder, read.getLoc(), felt);
386 } else {
387 constant = arith::ConstantOp::create(builder, read.getLoc(), cast<TypedAttr>(*value));
388 }
389 read.replaceAllUsesWith(constant);
390 read.erase();
391 return WalkResult::advance();
392 });
393 if (constants.wasInterrupted()) {
394 return failure();
395 }
396 SmallVector<Diagnostic> diagnostics;
397 if (failed(substituteStructBody(clone, source.getType(), bindings, diagnostics))) {
398 return failure();
399 }
400 auto unresolved = clone.walk([site, &templ, type](ConstReadOp read) -> WalkResult {
401 if (templ && templ.getConstNamed<TemplateExprOp>(read.getConstNameAttr())) {
402 return site->emitError("template expression ")
403 << read.getConstNameAttr() << " could not be evaluated for " << type;
404 }
405 return site->emitError("unresolved template binding ")
406 << read.getConstNameAttr() << " while specializing " << type;
407 });
408 if (unresolved.wasInterrupted()) {
409 return failure();
410 }
411 reportDelayedDiagnostics(site, std::move(diagnostics));
412 Builder builder(root.getContext());
413 clone->setAttr(SPECIALIZATION_ID_ATTR, builder.getI64IntegerAttr(id));
414 auto origin = getPathRelativeToAncestor(source, topRoot, [source] {
415 return source->emitError("specialization origin");
416 });
417 if (failed(origin)) {
418 return failure();
419 }
420 clone->setAttr(SPECIALIZATION_ORIGIN_ATTR, *origin);
421 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, key.second);
422 return id;
423 }

427 void recordFamily(const StructSpecializationRequest &request, unsigned id) {
428 Builder builder(root.getContext());
429 StringRef name;
430 Attribute record;
431 if (request.kind == StructSpecializationRequest::Kind::ArrayElement) {
432 name = FAMILY_ATTR;
433 SmallVector<Attribute> indices;
434 for (int64_t index : request.arrayIndices) {
435 indices.push_back(builder.getIndexAttr(index));
436 }
437 record = builder.getDictionaryAttr(
438 {builder.getNamedAttr(FAMILY_INDICES_KEY, builder.getArrayAttr(indices)),
439 builder.getNamedAttr(FAMILY_SPECIALIZATION_KEY, builder.getI64IntegerAttr(id)),
440 builder.getNamedAttr(FAMILY_TYPE_KEY, TypeAttr::get(request.familyType))}
441 );
442 } else if (request.kind == StructSpecializationRequest::Kind::RolledCallResult) {
443 name = FAMILY_SPECIALIZATIONS_ATTR;
444 record = builder.getI64IntegerAttr(id);
445 } else {
446 return;
447 }
448 SmallVector<Attribute> records;
449 if (auto existing = request.site->getAttrOfType<ArrayAttr>(name)) {
450 llvm::append_range(records, existing);
451 }
452 if (!llvm::is_contained(records, record)) {
453 records.push_back(record);
454 }
455 request.site->setAttr(name, builder.getArrayAttr(records));
456 }

460 LogicalResult recordRolledCallTargets(const RolledCallTargets &targets) {
461 auto call = cast<function::CallOp>(targets.site);
462 SmallVector<Attribute> ids;
463 if (auto existing = call->getAttrOfType<ArrayAttr>(FAMILY_SPECIALIZATIONS_ATTR)) {
464 llvm::append_range(ids, existing);
465 }
466 for (StructType candidate : targets.candidates) {
467 auto id = instantiate(candidate, call);
468 if (failed(id)) {
469 return failure();
470 }
471 auto attr = Builder(root.getContext()).getI64IntegerAttr(*id);
472 if (!llvm::is_contained(ids, attr)) {
473 ids.push_back(attr);
474 }
475 }
476 call->setAttr(FAMILY_SPECIALIZATIONS_ATTR, ArrayAttr::get(root.getContext(), ids));
477 return success();
478 }

482 LogicalResult setSpecializedCallee(function::CallOp call, StructDefOp clone, StringAttr method) {
483 auto lookupRoot = getRootModule(call);
484 if (failed(lookupRoot)) {
485 return failure();
486 }
487 auto name = getPathRelativeToAncestor(clone, *lookupRoot, [call] {
488 return call->emitError("specialized method owner");
489 });
490 if (failed(name)) {
491 return failure();
492 }
493 call.setCalleeAttr(appendLeaf(*name, method));
494 return success();
495 }

499 LogicalResult retargetCall(function::CallOp call, const Entry &caller, bool visited) {
500 auto target = call.getCalleeTarget(tables);
501 if (failed(target)) {
502 return failure();
503 }
504 if (target->viaInclude()) {
505 return call.emitError("inline includes before template monomorphization");
506 }
507 auto owner = target->get()->getParentOfType<StructDefOp>();
508 if (!owner) {
509 if (visited) {
510 return call.emitError(
511 "free-function calls are not yet supported by template monomorphization"
512 );
513 }
514 return success();
515 }
516 if (call->hasAttr(FAMILY_SPECIALIZATIONS_ATTR)) {
517 return success();
518 }
519 SmallVector<Type> types(call.getResultTypes());
520 llvm::append_range(types, call.getArgOperands().getTypes());
521 for (Type type : types) {
522 auto structure = dyn_cast<StructType>(type);
523 if (!structure || hasAffineMapAttr(structure)) {
524 continue;
525 }
526 auto found = structure.getDefinition(tables, call);
527 if (failed(found)) {
528 return failure();
529 }
530 auto cloneId = cloneToId.find(found->get().getOperation());
531 if (cloneId != cloneToId.end() && entries[cloneId->second].source == owner) {
532 return setSpecializedCallee(
533 call, entries[cloneId->second].clone, target->get().getSymNameAttr()
534 );
535 }
536 if (found->get() == owner) {
537 return success();
538 }
539 }
540 if (owner == caller.source) {
541 return setSpecializedCallee(call, caller.clone, target->get().getSymNameAttr());
542 }
543 if (visited) {
544 return call.emitError("cannot determine concrete struct owner for method call");
545 }
546 return success();
547 }

549public:
550 StructInstantiationWorklist(ModuleOp module, unsigned maximum, uint64_t steps)
551 : root(module), limit(maximum), evaluationLimit(steps) {}
552
555 LogicalResult run() {
556 auto top = getTopRootModule(root);
557 if (failed(top)) {
558 return failure();
559 }
560 topRoot = *top;
561 if (failed(loadExistingSpecializations())) {
562 return failure();
563 }
564 auto main = getMainInstanceType(root);
565 if (failed(main) || !*main) {
566 return root.emitError("template monomorphization requires a concrete llzk.main");
567 }
568 auto mainId = instantiate(*main, root);
569 if (failed(mainId)) {
570 return failure();
571 }
572 // Discovery appends to entries while processing dependencies. Indexing both
573 // visits those new entries and avoids retaining iterators across reallocation.
574 for (unsigned i = 0; i < entries.size(); ++i) {
575 auto clone = entries[i].clone;
576 SmallVector<Operation *> visited;
577 StructSpecializationDiscovery::CallTargets rolledCalls;
578 auto requests =
579 StructSpecializationDiscovery(entries[i].remainingSteps)
580 .discover(clone, StructSpecializationDiscovery::Bindings(), &visited, &rolledCalls);
581 if (failed(requests)) {
582 return failure();
583 }
584 entries[i].visited = std::move(visited);
585 for (const auto &request : *requests) {
586 auto id = instantiate(request.type, request.site);
587 if (failed(id)) {
588 return failure();
589 }
590 recordFamily(request, *id);
591 }
592 for (const auto &targets : rolledCalls) {
593 if (failed(recordRolledCallTargets(targets))) {
594 return failure();
595 }
596 }
597 }
598 // Each replacer resolves input names in one root. The cache identifies the
599 // definition independently of its spelling, including uses in preserved paths
600 // that discovery did not execute.
601 llvm::MapVector<Operation *, SmallVector<StructDefOp>> clonesByRoot;
602 for (Entry &entry : entries) {
603 auto lookupRoot = getRootModule(entry.clone);
604 if (failed(lookupRoot)) {
605 return failure();
606 }
607 clonesByRoot[lookupRoot->getOperation()].push_back(entry.clone);
608 }
609 for (auto &[rootOp, clones] : clonesByRoot) {
610 auto lookupRoot = cast<ModuleOp>(rootOp);
611 // recursivelyReplaceElementsIn returns void and suppresses null replacements,
612 // so the callback must retain a failure flag for this operation-level rewrite.
613 bool invalid = false;
614 // These checks validate names after substitution and clone insertion. The
615 // callback receives only a type, so failures are reported at its lookup root.
616 // For verified input, these failures indicate a bug in the pass.
617 AttrTypeReplacer replacer;
618 replacer.addReplacement(
619 [this, lookupRoot,
620 &invalid](StructType type) -> std::optional<std::pair<Type, WalkResult>> {
621 auto key = getSpecializationKey(type, lookupRoot, lookupRoot, KeyMode::Optional);
622 if (failed(key)) {
623 invalid = true;
624 return std::make_pair(Type(type), WalkResult::skip());
625 }
626 if (!*key) {
627 return std::nullopt;
628 }
629 auto existing = cache.find(**key);
630 if (existing == cache.end()) {
631 return std::nullopt;
632 }
633 auto name =
634 getPathRelativeToAncestor(entries[existing->second].clone, lookupRoot, [lookupRoot] {
635 return lookupRoot->emitError("specialized struct");
636 });
637 if (failed(name)) {
638 invalid = true;
639 return std::make_pair(Type(type), WalkResult::skip());
640 }
641 return std::make_pair(Type(StructType::get(*name)), WalkResult::skip());
642 }
643 );
644 for (auto clone : clones) {
645 // Provenance uses top-root names, not the root of the rewritten body.
646 auto origin = clone->removeAttr(SPECIALIZATION_ORIGIN_ATTR);
647 auto args = clone->removeAttr(SPECIALIZATION_ARGUMENTS_ATTR);
648 replacer.recursivelyReplaceElementsIn(clone, true, false, true);
649 clone->setAttr(SPECIALIZATION_ORIGIN_ATTR, origin);
650 clone->setAttr(SPECIALIZATION_ARGUMENTS_ATTR, args);
651 }
652 if (invalid) {
653 return failure();
654 }
655 }
656 for (Entry &entry : entries) {
657 DenseSet<Operation *> visited(entry.visited.begin(), entry.visited.end());
658 auto result = entry.clone.walk([this, &entry, &visited](function::CallOp call) {
659 return WalkResult(retargetCall(call, entry, visited.contains(call)));
660 });
661 if (result.wasInterrupted()) {
662 return failure();
663 }
664 }
665 auto mainRoot = getRootModule(root);
666 if (failed(mainRoot)) {
667 return failure();
668 }
669 auto mainName = getPathRelativeToAncestor(entries[*mainId].clone, *mainRoot, [this] {
670 return root.emitError("specialized main");
671 });
672 if (failed(mainName)) {
673 return failure();
674 }
675 root->setAttr(MAIN_ATTR_NAME, TypeAttr::get(StructType::get(*mainName)));
676 return success();
677 }
