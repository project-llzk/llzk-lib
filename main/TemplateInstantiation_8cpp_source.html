47template <typename Impl, typename Op, typename... HandledAttrs>
48class SymbolUserHelper : public OpConversionPattern<Op> {
49private:
50 const DenseMap<Attribute, Attribute> &paramNameToValue;
51
52 SymbolUserHelper(
53 TypeConverter &converter, MLIRContext *ctx, unsigned patternBenefit,
54 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue
55 )
56 : OpConversionPattern<Op>(converter, ctx, patternBenefit),
57 paramNameToValue(paramNameToInstantiatedValue) {}
58
59public:
60 using OpAdaptor = typename mlir::OpConversionPattern<Op>::OpAdaptor;
61
62 virtual Attribute getNameAttr(Op) const = 0;
63
64 virtual LogicalResult handleDefaultRewrite(
65 Attribute, Op op, OpAdaptor, ConversionPatternRewriter &, Attribute a
66 ) const {
67 return op->emitOpError().append("expected value with type ", op.getType(), " but found ", a);
68 }
69
70 LogicalResult
71 matchAndRewrite(Op op, OpAdaptor adaptor, ConversionPatternRewriter &rewriter) const override {
72 LLVM_DEBUG(llvm::dbgs() << "[SymbolUserHelper] op: " << op << '\n');
73 auto res = this->paramNameToValue.find(getNameAttr(op));
74 if (res == this->paramNameToValue.end()) {
75 LLVM_DEBUG(llvm::dbgs() << "[SymbolUserHelper] no instantiation for " << op << '\n');
76 return failure();
77 }
78 llvm::TypeSwitch<Attribute, LogicalResult> TS(res->second);
79 llvm::TypeSwitch<Attribute, LogicalResult> *ptr = &TS;
80
81 ((ptr = &(ptr->template Case<HandledAttrs>([&](HandledAttrs a) {
82 return static_cast<const Impl *>(this)->handleRewrite(res->first, op, adaptor, rewriter, a);
83 }))),
84 ...);
85
86 return TS.Default([&](Attribute a) {
87 return handleDefaultRewrite(res->first, op, adaptor, rewriter, a);
88 });
89 }
90 friend Impl;
91};

93class ClonedBodyConstReadOpPattern
94 : public SymbolUserHelper<
95 ClonedBodyConstReadOpPattern, ConstReadOp, IntegerAttr, FeltConstAttr> {
96 SmallVector<Diagnostic> &diagnostics;
97
98 using super =
99 SymbolUserHelper<ClonedBodyConstReadOpPattern, ConstReadOp, IntegerAttr, FeltConstAttr>;
100
101public:
102 ClonedBodyConstReadOpPattern(
103 TypeConverter &converter, MLIRContext *ctx,
104 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue,
105 SmallVector<Diagnostic> &instantiationDiagnostics
106 )
107 // benefit>0 so this applies instead of GeneralTypeReplacePattern<ConstReadOp>
108 : super(converter, ctx, /*patternBenefit=*/1, paramNameToInstantiatedValue),
109 diagnostics(instantiationDiagnostics) {}
110
111 Attribute getNameAttr(ConstReadOp op) const override { return op.getConstNameAttr(); }
112
113 LogicalResult handleRewrite(
114 Attribute sym, ConstReadOp op, OpAdaptor, ConversionPatternRewriter &rewriter, IntegerAttr a
115 ) const {
116 const APInt &attrValue = a.getValue();
117 Type origResTy = op.getType();
118 Type newResTy = getTypeConverter()->convertType(origResTy);
119 if (!newResTy) {
120 return op->emitOpError().append("could not convert result type ", origResTy);
121 }
122
123 if (FeltType ty = llvm::dyn_cast<FeltType>(newResTy)) {
124 replaceOpWithNewOp<FeltConstantOp>(
125 rewriter, op, FeltConstAttr::get(getContext(), attrValue, ty)
126 );
127 return success();
128 }
129
130 if (llvm::isa<IndexType>(newResTy)) {
131 replaceOpWithNewOp<arith::ConstantIndexOp>(rewriter, op, fromAPInt(attrValue));
132 return success();
133 }
134
135 if (newResTy.isSignlessInteger(1)) {
136 // Treat 0 as false and any other value as true (but give a warning if it's not 1)
137 if (attrValue.isZero()) {
138 replaceOpWithNewOp<arith::ConstantIntOp>(rewriter, op, newResTy, false);
139 return success();
140 }
141 if (!attrValue.isOne()) {
142 Location opLoc = op.getLoc();
143 Diagnostic diag(opLoc, DiagnosticSeverity::Warning);
144 diag << "Interpreting non-zero value " << stringWithoutType(a) << " as true";
145 if (getContext()->shouldPrintOpOnDiagnostic()) {
146 diag.attachNote(opLoc) << "see current operation: " << *op;
147 }
148 diag.attachNote(UnknownLoc::get(getContext()))
149 << "when instantiating '" << StructDefOp::getOperationName() << "' parameter \"" << sym
150 << "\" for this specialization";
151 diagnostics.push_back(std::move(diag));
152 }
153 replaceOpWithNewOp<arith::ConstantIntOp>(rewriter, op, newResTy, true);
154 return success();
155 }
156 return op->emitOpError().append("unexpected result type ", newResTy);
157 }
158
159 LogicalResult handleRewrite(
160 Attribute, ConstReadOp op, OpAdaptor, ConversionPatternRewriter &rewriter, FeltConstAttr a
161 ) const {
162 replaceOpWithNewOp<FeltConstantOp>(rewriter, op, a);
163 return success();
164 }
165};

219class MappedTypeConverter : public TypeConverter {
220 StructType origTy;
221 StructType newTy;
222 const DenseMap<Attribute, Attribute> &paramNameToValue;
223
224 inline Attribute convertIfPossible(Attribute a) const {
225 auto res = this->paramNameToValue.find(a);
226 return (res != this->paramNameToValue.end()) ? res->second : a;
227 }
228
229public:
230 MappedTypeConverter(
231 StructType originalType, StructType newType,
233 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue
234 )
235 : TypeConverter(), origTy(originalType), newTy(newType),
236 paramNameToValue(paramNameToInstantiatedValue) {
237
238 addConversion([](Type inputTy) { return inputTy; });
239
240 addConversion([this](StructType inputTy) {
241 LLVM_DEBUG(llvm::dbgs() << "[MappedTypeConverter] convert " << inputTy << '\n');
242
243 // Check for replacement of the full type
244 if (inputTy == this->origTy) {
245 return this->newTy;
246 }
247 // Check for replacement of parameter symbol names with concrete values
248 if (ArrayAttr inputTyParams = inputTy.getParams()) {
249 SmallVector<Attribute> updated;
250 for (Attribute a : inputTyParams) {
251 if (TypeAttr ta = dyn_cast<TypeAttr>(a)) {
252 updated.push_back(TypeAttr::get(this->convertType(ta.getValue())));
253 } else {
254 updated.push_back(convertIfPossible(a));
255 }
256 }
257 return getStructTypeWithParams(inputTy.getNameRef(), inputTy.getContext(), updated);
258 }
259 // Otherwise, return the type unchanged
260 return inputTy;
261 });
262
263 addConversion([this](ArrayType inputTy) {
264 // Check for replacement of parameter symbol names with concrete values
265 ArrayRef<Attribute> dimSizes = inputTy.getDimensionSizes();
266 if (!dimSizes.empty()) {
267 SmallVector<Attribute> updated;
268 for (Attribute a : dimSizes) {
269 updated.push_back(convertIfPossible(a));
270 }
271 return ArrayType::get(this->convertType(inputTy.getElementType()), updated);
272 }
273 // Otherwise, return the type unchanged
274 return inputTy;
275 });
276
277 addConversion([this](TypeVarType inputTy) -> Type {
278 // Check for replacement of parameter symbol name with a concrete type
279 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(convertIfPossible(inputTy.getNameRef()))) {
280 Type convertedType = tyAttr.getValue();
281 // Use the new type unless it contains a TypeVarType because a TypeVarType from a
282 // different struct references a parameter name from that other struct, not from the
283 // current struct so the reference would be invalid.
284 if (isConcreteType(convertedType)) {
285 return convertedType;
286 }
287 }
288 return inputTy;
289 });
290 }
291};

332static SymbolRefAttr
333convertCalleeSymRefs(SymbolRefAttr callee, const DenseMap<Attribute, Attribute> &paramNameToValue) {
334 auto it = paramNameToValue.find(FlatSymbolRefAttr::get(callee.getRootReference()));
335 if (it == paramNameToValue.end()) {
336 return callee;
337 }
338
339 auto tyAttr = llvm::dyn_cast<TypeAttr>(it->second);
340 if (!tyAttr) {
341 return callee;
342 }
343
344 auto structTy = llvm::dyn_cast<StructType>(tyAttr.getValue());
345 if (!structTy) {
346 return callee;
347 }
348
349 SmallVector<FlatSymbolRefAttr> newPieces = getPieces(structTy.getNameRef());
350 llvm::append_range(newPieces, callee.getNestedReferences());
351 return asSymbolRefAttr(newPieces);
352}

358static std::optional<Attribute>
359evaluateExpr(TemplateExprOp exprOp, const DenseMap<Attribute, Attribute> &paramNameToConcrete) {
360 // Map from SSA value in the expr body to its concrete Attribute.
361 DenseMap<Value, Attribute> valueMap;
362 for (Operation &bodyOp : exprOp.getInitializerRegion().front()) {
363 if (auto yieldOp = llvm::dyn_cast<YieldOp>(bodyOp)) {
364 auto it = valueMap.find(yieldOp.getVal());
365 return it != valueMap.end() ? std::make_optional(it->second) : std::nullopt;
366 }
367
368 if (auto constReadOp = llvm::dyn_cast<ConstReadOp>(bodyOp)) {
369 auto it = paramNameToConcrete.find(constReadOp.getConstNameAttr());
370 if (it == paramNameToConcrete.end()) {
371 return std::nullopt; // a referenced param is not concrete
372 }
373 // If the attribute type is `FeltType` but it's stored as an IntegerAttr, promote to
374 // a `FeltConstAttr`.
375 Attribute val = it->second;
376 if (auto intAttr = llvm::dyn_cast<IntegerAttr>(val)) {
377 if (auto feltTy = llvm::dyn_cast<FeltType>(constReadOp.getResult().getType())) {
378 val = FeltConstAttr::get(bodyOp.getContext(), intAttr.getValue(), feltTy);
379 }
380 }
381 valueMap[constReadOp.getResult()] = val;
382 continue;
383 }
384
385 // Gather constant attributes for all operands.
386 SmallVector<Attribute> operandAttrs;
387 operandAttrs.reserve(bodyOp.getNumOperands());
388 for (Value operand : bodyOp.getOperands()) {
389 auto it = valueMap.find(operand);
390 if (it == valueMap.end()) {
391 return std::nullopt; // operand not known as a constant
392 }
393 operandAttrs.push_back(it->second);
394 }
395
396 // Try constant folding.
397 SmallVector<OpFoldResult> foldResults;
398 if (succeeded(bodyOp.fold(operandAttrs, foldResults)) &&
399 foldResults.size() == bodyOp.getNumResults()) {
400 for (auto [result, fr] : llvm::zip_equal(bodyOp.getResults(), foldResults)) {
401 if (Attribute a = llvm::dyn_cast<Attribute>(fr)) {
402 valueMap[result] = a;
403 } else {
404 return std::nullopt;
405 }
406 }
407 }
408 }
409 return std::nullopt; // no YieldOp found (shouldn't happen in a valid expr)
410}

416void reportDelayedDiagnostics(Operation *site, SmallVector<Diagnostic> &&diagnostics) {
417 DiagnosticEngine &engine = site->getContext()->getDiagEngine();
418 for (Diagnostic &diagnostic : diagnostics) {
419 for (Diagnostic &note : diagnostic.getNotes()) {
420 assert(note.getNotes().empty() && "notes cannot have notes attached");
421 if (isa<UnknownLoc>(note.getLocation())) {
422 note = std::move(Diagnostic(site->getLoc(), note.getSeverity()).append(note.str()));
423 }
424 }
425 engine.emit(std::move(diagnostic));
426 }
427}

429FailureOr<ArrayAttr> rebaseTemplateParams(
430 SymbolTableCollection &tables, ArrayAttr templateParams, Operation *lookupFrom,
431 ModuleOp destinationRoot, Operation *requestSite
432) {
433 if (!templateParams) {
434 return ArrayAttr::get(requestSite->getContext(), {});
435 }
436 AttrTypeReplacer rebaser;
437 rebaser.addReplacement(
438 [&tables, lookupFrom, destinationRoot,
439 requestSite](StructType type) -> std::optional<std::pair<Type, WalkResult>> {
440 auto found = type.getDefinition(tables, lookupFrom);
441 if (failed(found)) {
442 return std::make_pair(Type(type), WalkResult::interrupt());
443 }
444 if (found->viaInclude()) {
445 requestSite->emitError("inline includes before rebasing template parameters");
446 return std::make_pair(Type(type), WalkResult::interrupt());
447 }
448 auto name = getPathRelativeToAncestor(found->get(), destinationRoot, [requestSite, type] {
449 return requestSite->emitError("struct type argument ") << type;
450 });
451 if (failed(name)) {
452 return std::make_pair(Type(type), WalkResult::interrupt());
453 }
454 auto nested =
455 rebaseTemplateParams(tables, type.getParams(), lookupFrom, destinationRoot, requestSite);
456 if (failed(nested)) {
457 return std::make_pair(Type(type), WalkResult::interrupt());
458 }
459 auto rebased = getStructTypeWithParams(*name, type.getParams() ? *nested : ArrayAttr());
460 return std::make_pair(Type(rebased), WalkResult::skip());
461 }
462 );
463 auto result = dyn_cast_if_present<ArrayAttr>(rebaser.replace(templateParams));
464 if (!result) {
465 return failure();
466 }
467 return result;
468}

475TemplateTypeConverter::TemplateTypeConverter(DenseMap<Attribute, Attribute> bindings)
476 : paramNameToValue(std::move(bindings)) {
477 addConversion([](Type t) { return t; });
478
479 addConversion([this](TypeVarType inputTy) -> Type {
480 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(convertIfPossible(inputTy.getNameRef()))) {
481 Type convertedType = tyAttr.getValue();
482 if (isConcreteType(convertedType)) {
483 return convertedType;
484 }
485 }
486 return inputTy;
487 });
488
489 addConversion([this](ArrayType inputTy) {
490 SmallVector<Attribute> updated;
491 bool changed = false;
492 for (Attribute a : inputTy.getDimensionSizes()) {
493 Attribute converted = convertIfPossible(a);
494 updated.push_back(converted);
495 if (converted != a) {
496 changed = true;
497 }
498 }
499 Type newElemTy = this->convertType(inputTy.getElementType());
500 if (!changed && newElemTy == inputTy.getElementType()) {
501 return inputTy;
502 }
503 return flattenArrayElementType(inputTy.cloneWith(inputTy.getElementType(), updated), newElemTy);
504 });
505
506 addConversion([this](StructType inputTy) -> StructType {
507 if (ArrayAttr params = inputTy.getParams()) {
508 SmallVector<Attribute> updated;
509 bool changed = false;
510 for (Attribute a : params) {
511 if (TypeAttr ta = dyn_cast<TypeAttr>(a)) {
512 Type newTy = this->convertType(ta.getValue());
513 if (newTy != ta.getValue()) {
514 updated.push_back(TypeAttr::get(newTy));
515 changed = true;
516 continue;
517 }
518 } else {
519 Attribute converted = convertIfPossible(a);
520 if (converted != a) {
521 updated.push_back(converted);
522 changed = true;
523 continue;
524 }
525 }
526 updated.push_back(a);
527 }
528 if (changed) {
529 return getStructTypeWithParams(inputTy.getNameRef(), inputTy.getContext(), updated);
530 }
531 }
532 return inputTy;
533 });
534}

536Attribute TemplateTypeConverter::convertAttr(Attribute attr) const {
537 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(attr)) {
538 Type convertedTy = convertType(tyAttr.getValue());
539 if (convertedTy != tyAttr.getValue()) {
540 return TypeAttr::get(convertedTy);
541 }
542 }
543 return convertIfPossible(attr);

555void evaluateTemplateExprs(
556 TemplateOp templateOp, DenseMap<Attribute, Attribute> &paramNameToConcrete
557) {
558 LLVM_DEBUG(
559 llvm::dbgs() << "[evaluateTemplateExprs] before: " << debug::toStringList(paramNameToConcrete)
560 << '\n'
561 );
562 for (TemplateExprOp exprOp : templateOp.getConstOps<TemplateExprOp>()) {
563 std::optional<Attribute> result = evaluateExpr(exprOp, paramNameToConcrete);
564 if (result.has_value()) {
565 auto exprNameAttr = FlatSymbolRefAttr::get(exprOp.getSymNameAttr());
566 paramNameToConcrete.try_emplace(exprNameAttr, *result);
567 LLVM_DEBUG(
568 llvm::dbgs() << "[evaluateTemplateExprs] expr @" << exprOp.getSymName()
569 << " evaluated to " << *result << '\n'
570 );
571 }
572 }
573 LLVM_DEBUG(
574 llvm::dbgs() << "[evaluateTemplateExprs] after: " << debug::toStringList(paramNameToConcrete)
575 << '\n'
576 );
577}

586LogicalResult substituteStructBody(
587 StructDefOp newStruct, StructType typeAtDef,
588 const DenseMap<Attribute, Attribute> &paramNameToConcrete, SmallVector<Diagnostic> &diagnostics
589) {
590 MLIRContext *ctx = newStruct.getContext();
591 MappedTypeConverter tyConv(typeAtDef, newStruct.getType(), paramNameToConcrete);
592 ConversionTarget target =
593 newConverterDefinedTarget<EmitEqualityOp>(tyConv, ctx, tableOffsetIsntSymbol);
594 target.addDynamicallyLegalOp<ConstReadOp>([&paramNameToConcrete](ConstReadOp op) {
595 // Legal if it's not in the map of concrete attribute instantiations
596 return !paramNameToConcrete.contains(op.getConstNameAttr());
597 });
598
599 RewritePatternSet patterns = newGeneralRewritePatternSet<EmitEqualityOp>(tyConv, ctx, target);
600 patterns.add<ClonedBodyConstReadOpPattern>(tyConv, ctx, paramNameToConcrete, diagnostics);
601 patterns.add<ClonedMemberReadOpPattern>(tyConv, ctx, paramNameToConcrete);
602 return applyFullConversion(newStruct, target, std::move(patterns));
603}
604
605LogicalResult substituteFunctionBody(
606 FuncDefOp newFunc, const DenseMap<Attribute, Attribute> &paramNameToConcrete,
607 SmallVector<Diagnostic> &delayedDiagnostics
608) {
609 MLIRContext *ctx = newFunc.getContext();
610 TemplateTypeConverter tyConv(paramNameToConcrete);
611 ConversionTarget target = newConverterDefinedTarget<>(tyConv, ctx, tableOffsetIsntSymbol);
612 target.addDynamicallyLegalOp<ConstReadOp>([&tyConv](ConstReadOp p) {
613 // Legal if it's not in the map of concrete attribute instantiations
614 return !tyConv.containsParam(p.getConstNameAttr());
615 });
616 RewritePatternSet bodyPatterns = newGeneralRewritePatternSet(tyConv, ctx, target);
617 bodyPatterns.add<ClonedBodyConstReadOpPattern>(
618 tyConv, ctx, tyConv.getParamMap(), delayedDiagnostics
619 );
620 bodyPatterns.add<ClonedBodyArrayReadOpPattern, ClonedBodyArrayWriteOpPattern>(tyConv, ctx);
621 bodyPatterns.add<ClonedMemberReadOpPattern>(tyConv, ctx, paramNameToConcrete);
622 return applyFullConversion(newFunc, target, std::move(bodyPatterns));
623}
