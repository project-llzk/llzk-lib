45template <typename Impl, typename Op, typename... HandledAttrs>
46class SymbolUserHelper : public OpConversionPattern<Op> {
47private:
48 const DenseMap<Attribute, Attribute> &paramNameToValue;
49
50 SymbolUserHelper(
51 TypeConverter &converter, MLIRContext *ctx, unsigned patternBenefit,
52 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue
53 )
54 : OpConversionPattern<Op>(converter, ctx, patternBenefit),
55 paramNameToValue(paramNameToInstantiatedValue) {}
56
57public:
58 using OpAdaptor = typename mlir::OpConversionPattern<Op>::OpAdaptor;
59
60 virtual Attribute getNameAttr(Op) const = 0;
61
62 virtual LogicalResult handleDefaultRewrite(
63 Attribute, Op op, OpAdaptor, ConversionPatternRewriter &, Attribute a
64 ) const {
65 return op->emitOpError().append("expected value with type ", op.getType(), " but found ", a);
66 }
67
68 LogicalResult
69 matchAndRewrite(Op op, OpAdaptor adaptor, ConversionPatternRewriter &rewriter) const override {
70 LLVM_DEBUG(llvm::dbgs() << "[SymbolUserHelper] op: " << op << '\n');
71 auto res = this->paramNameToValue.find(getNameAttr(op));
72 if (res == this->paramNameToValue.end()) {
73 LLVM_DEBUG(llvm::dbgs() << "[SymbolUserHelper] no instantiation for " << op << '\n');
74 return failure();
75 }
76 llvm::TypeSwitch<Attribute, LogicalResult> TS(res->second);
77 llvm::TypeSwitch<Attribute, LogicalResult> *ptr = &TS;
78
79 ((ptr = &(ptr->template Case<HandledAttrs>([&](HandledAttrs a) {
80 return static_cast<const Impl *>(this)->handleRewrite(res->first, op, adaptor, rewriter, a);
81 }))),
82 ...);
83
84 return TS.Default([&](Attribute a) {
85 return handleDefaultRewrite(res->first, op, adaptor, rewriter, a);
86 });
87 }
88 friend Impl;
89};

91class ClonedBodyConstReadOpPattern
92 : public SymbolUserHelper<
93 ClonedBodyConstReadOpPattern, ConstReadOp, IntegerAttr, FeltConstAttr> {
94 SmallVector<Diagnostic> &diagnostics;
95
96 using super =
97 SymbolUserHelper<ClonedBodyConstReadOpPattern, ConstReadOp, IntegerAttr, FeltConstAttr>;
98
99public:
100 ClonedBodyConstReadOpPattern(
101 TypeConverter &converter, MLIRContext *ctx,
102 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue,
103 SmallVector<Diagnostic> &instantiationDiagnostics
104 )
105 // benefit>0 so this applies instead of GeneralTypeReplacePattern<ConstReadOp>
106 : super(converter, ctx, /*patternBenefit=*/1, paramNameToInstantiatedValue),
107 diagnostics(instantiationDiagnostics) {}
108
109 Attribute getNameAttr(ConstReadOp op) const override { return op.getConstNameAttr(); }
110
111 LogicalResult handleRewrite(
112 Attribute sym, ConstReadOp op, OpAdaptor, ConversionPatternRewriter &rewriter, IntegerAttr a
113 ) const {
114 const APInt &attrValue = a.getValue();
115 Type origResTy = op.getType();
116 Type newResTy = getTypeConverter()->convertType(origResTy);
117 if (!newResTy) {
118 return op->emitOpError().append("could not convert result type ", origResTy);
119 }
120
121 if (FeltType ty = llvm::dyn_cast<FeltType>(newResTy)) {
122 replaceOpWithNewOp<FeltConstantOp>(
123 rewriter, op, FeltConstAttr::get(getContext(), attrValue, ty)
124 );
125 return success();
126 }
127
128 if (llvm::isa<IndexType>(newResTy)) {
129 replaceOpWithNewOp<arith::ConstantIndexOp>(rewriter, op, fromAPInt(attrValue));
130 return success();
131 }
132
133 if (newResTy.isSignlessInteger(1)) {
134 // Treat 0 as false and any other value as true (but give a warning if it's not 1)
135 if (attrValue.isZero()) {
136 replaceOpWithNewOp<arith::ConstantIntOp>(rewriter, op, newResTy, false);
137 return success();
138 }
139 if (!attrValue.isOne()) {
140 Location opLoc = op.getLoc();
141 Diagnostic diag(opLoc, DiagnosticSeverity::Warning);
142 diag << "Interpreting non-zero value " << stringWithoutType(a) << " as true";
143 if (getContext()->shouldPrintOpOnDiagnostic()) {
144 diag.attachNote(opLoc) << "see current operation: " << *op;
145 }
146 diag.attachNote(UnknownLoc::get(getContext()))
147 << "when instantiating '" << StructDefOp::getOperationName() << "' parameter \"" << sym
148 << "\" for this call";
149 diagnostics.push_back(std::move(diag));
150 }
151 replaceOpWithNewOp<arith::ConstantIntOp>(rewriter, op, newResTy, true);
152 return success();
153 }
154 return op->emitOpError().append("unexpected result type ", newResTy);
155 }
156
157 LogicalResult handleRewrite(
158 Attribute, ConstReadOp op, OpAdaptor, ConversionPatternRewriter &rewriter, FeltConstAttr a
159 ) const {
160 replaceOpWithNewOp<FeltConstantOp>(rewriter, op, a);
161 return success();
162 }
163};

217class MappedTypeConverter : public TypeConverter {
218 StructType origTy;
219 StructType newTy;
220 const DenseMap<Attribute, Attribute> &paramNameToValue;
221
222 inline Attribute convertIfPossible(Attribute a) const {
223 auto res = this->paramNameToValue.find(a);
224 return (res != this->paramNameToValue.end()) ? res->second : a;
225 }
226
227public:
228 MappedTypeConverter(
229 StructType originalType, StructType newType,
231 const DenseMap<Attribute, Attribute> &paramNameToInstantiatedValue
232 )
233 : TypeConverter(), origTy(originalType), newTy(newType),
234 paramNameToValue(paramNameToInstantiatedValue) {
235
236 addConversion([](Type inputTy) { return inputTy; });
237
238 addConversion([this](StructType inputTy) {
239 LLVM_DEBUG(llvm::dbgs() << "[MappedTypeConverter] convert " << inputTy << '\n');
240
241 // Check for replacement of the full type
242 if (inputTy == this->origTy) {
243 return this->newTy;
244 }
245 // Check for replacement of parameter symbol names with concrete values
246 if (ArrayAttr inputTyParams = inputTy.getParams()) {
247 SmallVector<Attribute> updated;
248 for (Attribute a : inputTyParams) {
249 if (TypeAttr ta = dyn_cast<TypeAttr>(a)) {
250 updated.push_back(TypeAttr::get(this->convertType(ta.getValue())));
251 } else {
252 updated.push_back(convertIfPossible(a));
253 }
254 }
255 return getStructTypeWithParams(inputTy.getNameRef(), inputTy.getContext(), updated);
256 }
257 // Otherwise, return the type unchanged
258 return inputTy;
259 });
260
261 addConversion([this](ArrayType inputTy) {
262 // Check for replacement of parameter symbol names with concrete values
263 ArrayRef<Attribute> dimSizes = inputTy.getDimensionSizes();
264 if (!dimSizes.empty()) {
265 SmallVector<Attribute> updated;
266 for (Attribute a : dimSizes) {
267 updated.push_back(convertIfPossible(a));
268 }
269 return ArrayType::get(this->convertType(inputTy.getElementType()), updated);
270 }
271 // Otherwise, return the type unchanged
272 return inputTy;
273 });
274
275 addConversion([this](TypeVarType inputTy) -> Type {
276 // Check for replacement of parameter symbol name with a concrete type
277 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(convertIfPossible(inputTy.getNameRef()))) {
278 Type convertedType = tyAttr.getValue();
279 // Use the new type unless it contains a TypeVarType because a TypeVarType from a
280 // different struct references a parameter name from that other struct, not from the
281 // current struct so the reference would be invalid.
282 if (isConcreteType(convertedType)) {
283 return convertedType;
284 }
285 }
286 return inputTy;
287 });
288 }

330static SymbolRefAttr
331convertCalleeSymRefs(SymbolRefAttr callee, const DenseMap<Attribute, Attribute> &paramNameToValue) {
332 auto it = paramNameToValue.find(FlatSymbolRefAttr::get(callee.getRootReference()));
333 if (it == paramNameToValue.end()) {
334 return callee;
335 }
336
337 auto tyAttr = llvm::dyn_cast<TypeAttr>(it->second);
338 if (!tyAttr) {
339 return callee;
340 }
341
342 auto structTy = llvm::dyn_cast<StructType>(tyAttr.getValue());
343 if (!structTy) {
344 return callee;
345 }
346
347 SmallVector<FlatSymbolRefAttr> newPieces = getPieces(structTy.getNameRef());
348 llvm::append_range(newPieces, callee.getNestedReferences());
349 return asSymbolRefAttr(newPieces);
350}

356static std::optional<Attribute>
357evaluateExpr(TemplateExprOp exprOp, const DenseMap<Attribute, Attribute> &paramNameToConcrete) {
358 // Map from SSA value in the expr body to its concrete Attribute.
359 DenseMap<Value, Attribute> valueMap;
360 for (Operation &bodyOp : exprOp.getInitializerRegion().front()) {
361 if (auto yieldOp = llvm::dyn_cast<YieldOp>(bodyOp)) {
362 auto it = valueMap.find(yieldOp.getVal());
363 return it != valueMap.end() ? std::make_optional(it->second) : std::nullopt;
364 }
365
366 if (auto constReadOp = llvm::dyn_cast<ConstReadOp>(bodyOp)) {
367 auto it = paramNameToConcrete.find(constReadOp.getConstNameAttr());
368 if (it == paramNameToConcrete.end()) {
369 return std::nullopt; // a referenced param is not concrete
370 }
371 // If the attribute type is `FeltType` but it's stored as an IntegerAttr, promote to
372 // a `FeltConstAttr`.
373 Attribute val = it->second;
374 if (auto intAttr = llvm::dyn_cast<IntegerAttr>(val)) {
375 if (auto feltTy = llvm::dyn_cast<FeltType>(constReadOp.getResult().getType())) {
376 val = FeltConstAttr::get(bodyOp.getContext(), intAttr.getValue(), feltTy);
377 }
378 }
379 valueMap[constReadOp.getResult()] = val;
380 continue;
381 }
382
383 // Gather constant attributes for all operands.
384 SmallVector<Attribute> operandAttrs;
385 operandAttrs.reserve(bodyOp.getNumOperands());
386 for (Value operand : bodyOp.getOperands()) {
387 auto it = valueMap.find(operand);
388 if (it == valueMap.end()) {
389 return std::nullopt; // operand not known as a constant
390 }
391 operandAttrs.push_back(it->second);
392 }
393
394 // Try constant folding.
395 SmallVector<OpFoldResult> foldResults;
396 if (succeeded(bodyOp.fold(operandAttrs, foldResults)) &&
397 foldResults.size() == bodyOp.getNumResults()) {
398 for (auto [result, fr] : llvm::zip_equal(bodyOp.getResults(), foldResults)) {
399 if (Attribute a = llvm::dyn_cast<Attribute>(fr)) {
400 valueMap[result] = a;
401 } else {
402 return std::nullopt;
403 }
404 }
405 }
406 }
407 return std::nullopt; // no YieldOp found (shouldn't happen in a valid expr)
408}

414Attribute TemplateTypeConverter::convertIfPossible(Attribute attr) const {
415 auto it = paramNameToValue.find(attr);
416 return it == paramNameToValue.end() ? attr : it->second;
417}
418
419TemplateTypeConverter::TemplateTypeConverter(DenseMap<Attribute, Attribute> bindings)
420 : paramNameToValue(std::move(bindings)) {
421 addConversion([](Type t) { return t; });
422
423 addConversion([this](TypeVarType inputTy) -> Type {
424 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(convertIfPossible(inputTy.getNameRef()))) {
425 Type convertedType = tyAttr.getValue();
426 if (isConcreteType(convertedType)) {
427 return convertedType;
428 }
429 }
430 return inputTy;
431 });
432
433 addConversion([this](ArrayType inputTy) {
434 SmallVector<Attribute> updated;
435 bool changed = false;
436 for (Attribute a : inputTy.getDimensionSizes()) {
437 Attribute converted = convertIfPossible(a);
438 updated.push_back(converted);
439 if (converted != a) {
440 changed = true;
441 }
442 }
443 Type newElemTy = this->convertType(inputTy.getElementType());
444 if (!changed && newElemTy == inputTy.getElementType()) {
445 return inputTy;
446 }
447 return flattenArrayElementType(inputTy.cloneWith(inputTy.getElementType(), updated), newElemTy);
448 });
449
450 addConversion([this](StructType inputTy) -> StructType {
451 if (ArrayAttr params = inputTy.getParams()) {
452 SmallVector<Attribute> updated;
453 bool changed = false;
454 for (Attribute a : params) {
455 if (TypeAttr ta = dyn_cast<TypeAttr>(a)) {
456 Type newTy = this->convertType(ta.getValue());
457 if (newTy != ta.getValue()) {
458 updated.push_back(TypeAttr::get(newTy));
459 changed = true;
460 continue;
461 }
462 } else {
463 Attribute converted = convertIfPossible(a);
464 if (converted != a) {
465 updated.push_back(converted);
466 changed = true;
467 continue;
468 }
469 }
470 updated.push_back(a);
471 }
472 if (changed) {
473 return getStructTypeWithParams(inputTy.getNameRef(), inputTy.getContext(), updated);
474 }
475 }
476 return inputTy;
477 });
478}

480Attribute TemplateTypeConverter::convertAttr(Attribute attr) const {
481 if (TypeAttr tyAttr = llvm::dyn_cast<TypeAttr>(attr)) {
482 Type convertedTy = convertType(tyAttr.getValue());
483 if (convertedTy != tyAttr.getValue()) {
484 return TypeAttr::get(convertedTy);
485 }
486 }
487 return convertIfPossible(attr);
488}

499void evaluateTemplateExprs(
500 TemplateOp templateOp, DenseMap<Attribute, Attribute> &paramNameToConcrete
501) {
502 LLVM_DEBUG(
503 llvm::dbgs() << "[evaluateTemplateExprs] before: " << debug::toStringList(paramNameToConcrete)
504 << '\n'
505 );
506 for (TemplateExprOp exprOp : templateOp.getConstOps<TemplateExprOp>()) {
507 std::optional<Attribute> result = evaluateExpr(exprOp, paramNameToConcrete);
508 if (result.has_value()) {
509 auto exprNameAttr = FlatSymbolRefAttr::get(exprOp.getSymNameAttr());
510 paramNameToConcrete.try_emplace(exprNameAttr, *result);
511 LLVM_DEBUG(
512 llvm::dbgs() << "[evaluateTemplateExprs] expr @" << exprOp.getSymName()
513 << " evaluated to " << *result << '\n'
514 );
515 }
516 }
517 LLVM_DEBUG(
518 llvm::dbgs() << "[evaluateTemplateExprs] after: " << debug::toStringList(paramNameToConcrete)
519 << '\n'
520 );
521}

530LogicalResult substituteStructBody(
531 StructDefOp newStruct, StructType typeAtDef,
532 const DenseMap<Attribute, Attribute> &paramNameToConcrete, SmallVector<Diagnostic> &diagnostics
533) {
534 MLIRContext *ctx = newStruct.getContext();
535 MappedTypeConverter tyConv(typeAtDef, newStruct.getType(), paramNameToConcrete);
536 ConversionTarget target =
537 newConverterDefinedTarget<EmitEqualityOp>(tyConv, ctx, tableOffsetIsntSymbol);
538 target.addDynamicallyLegalOp<ConstReadOp>([&paramNameToConcrete](ConstReadOp op) {
539 // Legal if it's not in the map of concrete attribute instantiations
540 return !paramNameToConcrete.contains(op.getConstNameAttr());
541 });
542
543 RewritePatternSet patterns = newGeneralRewritePatternSet<EmitEqualityOp>(tyConv, ctx, target);
544 patterns.add<ClonedBodyConstReadOpPattern>(tyConv, ctx, paramNameToConcrete, diagnostics);
545 patterns.add<ClonedMemberReadOpPattern>(tyConv, ctx, paramNameToConcrete);
546 return applyFullConversion(newStruct, target, std::move(patterns));
547}
548
549LogicalResult substituteFunctionBody(
550 FuncDefOp newFunc, const DenseMap<Attribute, Attribute> &paramNameToConcrete,
551 SmallVector<Diagnostic> &delayedDiagnostics
552) {
553 MLIRContext *ctx = newFunc.getContext();
554 TemplateTypeConverter tyConv(paramNameToConcrete);
555 ConversionTarget target = newConverterDefinedTarget<>(tyConv, ctx, tableOffsetIsntSymbol);
556 target.addDynamicallyLegalOp<ConstReadOp>([&tyConv](ConstReadOp p) {
557 // Legal if it's not in the map of concrete attribute instantiations
558 return !tyConv.containsParam(p.getConstNameAttr());
559 });
560 RewritePatternSet bodyPatterns = newGeneralRewritePatternSet(tyConv, ctx, target);
561 bodyPatterns.add<ClonedBodyConstReadOpPattern>(
562 tyConv, ctx, tyConv.getParamMap(), delayedDiagnostics
563 );
564 bodyPatterns.add<ClonedBodyArrayReadOpPattern, ClonedBodyArrayWriteOpPattern>(tyConv, ctx);
565 bodyPatterns.add<ClonedMemberReadOpPattern>(tyConv, ctx, paramNameToConcrete);
566 return applyFullConversion(newFunc, target, std::move(bodyPatterns));
567}
